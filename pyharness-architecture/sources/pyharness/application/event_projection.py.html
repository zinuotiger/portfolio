54def _redact_dict(d: dict, *, depth: int = 0) -> dict:
55 """参数脱敏(INV-09):敏感键打码、长串截断、容器限深限长——渲染永不因单事件炸。"""
56 out: dict[str, Any] = {}
57 for k, v in d.items():
58 key = str(k)
59 if any(hint in key.lower() for hint in _SENSITIVE_HINTS):
60 out[key] = "[redacted]"
61 elif isinstance(v, dict):
62 out[key] = _redact_dict(v, depth=depth + 1) if depth < 2 else "<dict>"
63 elif isinstance(v, (list, tuple)):
64 items = [_redact_value(x, depth + 1) for x in v[:8]]
65 out[key] = items + [f"…共{len(v)}项"] if len(v) > 8 else items
66 elif isinstance(v, str):
67 out[key] = _trunc(v, 120)
68 elif isinstance(v, (int, float, bool)) or v is None:
69 out[key] = v
70 else:
71 out[key] = f"<{type(v).__name__}>"
72 return out

75def _redact_value(v: Any, depth: int = 0) -> Any:
76 """单值脱敏(容器内递归用)。"""
77 if isinstance(v, dict):
78 return _redact_dict(v, depth=depth)
79 if isinstance(v, (list, tuple)):
80 return [_redact_value(x, depth) for x in v[:8]]
81 if isinstance(v, str):
82 return _trunc(v, 120)
83 if isinstance(v, (int, float, bool)) or v is None:
84 return v
85 return f"<{type(v).__name__}>"

88def redact_args(payload: dict) -> dict:
89 """tool.call 参数脱敏:工具名 + 校验后 args 脱敏 + raw_args 打码对照(INV-06 审计友好)。"""
90 p = payload or {}
91 name = p.get("name") or p.get("tool") or "?"
92 out: dict[str, Any] = {"tool": str(name)}
93 args = p.get("args")
94 if isinstance(args, dict):
95 out["args"] = _redact_dict(args)
96 elif args is not None:
97 out["args"] = _trunc(str(args), 200)
98 raw = p.get("raw_args")
99 if isinstance(raw, dict):
100 out["raw_args_masked"] = _redact_dict(raw)
101 elif raw is not None:
102 out["raw_args_masked"] = _trunc(str(raw), 200)
103 return out

111def approval_node(ev: Envelope, base: dict) -> TimelineNode:
112 """approval.* 事件 → 轨迹帧(granted/denied/timeout 同构;approval_id=requested seq)。"""
113 p = ev.payload or {}
114 verdict = ev.type.rsplit(".", 1)[-1] # requested|granted|denied|timeout
115 if verdict == "requested":
116 title, sev = "审批请求", "info"
117 elif verdict == "granted":
118 title, sev = "审批通过", "success"
119 elif verdict == "denied":
120 title, sev = "审批拒绝", "danger"
121 else:
122 title, sev = "审批超时(自动拒绝)", "warn"
123 detail: dict[str, Any] = {}
124 if verdict == "requested":
125 detail = {"tool": p.get("tool"), "args_summary": p.get("args_summary"),
126 "risk": p.get("risk"), "ttl_ms": p.get("ttl_ms")}
127 else:
128 detail = {"approval_id": p.get("approval_id"), "by": p.get("by")}
129 return TimelineNode(kind="approval", title=title, severity=sev,
130 detail={k: v for k, v in detail.items() if v is not None},
131 **base)

134def render_timeline_node(ev: Envelope) -> TimelineNode:
135 """事件 → 时间线一帧(纯函数;payload 字段缺失按 get 兜底,渲染永不中断整条线)。
136
137 类型映射(轨迹 = 日志投影 INV-01):user.message→user;agent.message→message;
138 tool.call→tool(脱敏参数);guard.rejected→guard(danger);approval.*→approval;
139 llm.usage→budget;session.recovered→recovery;session.created/finished→recovery;
140 其余持久事件 fallback→info(瞬时 llm.chunk 被 append 拒写,天然不在日志)。
141 """
142 p = ev.payload if hasattr(ev, "payload") else {}
143 p = dict(p or {})
144 base = {"seq": ev.seq, "ts": str(getattr(ev, "ts", "")),
145 "actor": getattr(ev, "actor", "system")}
146 t = getattr(ev, "type", "")
147 if t == "user.message":
148 return TimelineNode(kind="user", title="用户",
149 detail={"text": _trunc(p.get("content", ""), 400)}, **base)
150 if t == "agent.message":
151 return TimelineNode(kind="message", title="AI 回复", severity="success",
152 detail={"text": p.get("content", "")}, **base)
153 if t == "tool.call":
154 name = p.get("name") or p.get("tool") or "?"
155 return TimelineNode(kind="tool", title=f"调用 {name}",
156 detail=redact_args(p), **base)
157 if t == "guard.rejected":
158 return TimelineNode(kind="guard", title="guard 拦截", severity="danger",
159 detail={"guard": p.get("guard_id") or p.get("guard"),
160 "reason": p.get("reason"),
161 "policy_ref": p.get("policy_ref")}, **base)
162 if t.startswith("approval."):
163 return approval_node(ev, base)
164 if t == "llm.usage":
165 return TimelineNode(
166 kind="budget", title="用量",
167 severity="warn" if p.get("out_tokens") else "info",
168 detail={"model": p.get("model"), "in_tokens": p.get("in_tokens"),
169 "out_tokens": p.get("out_tokens"),
170 "cost_est": p.get("cost_est")}, **base)
171 if t == "session.recovered":
172 return TimelineNode(kind="recovery", title="崩溃修复声明", severity="warn",
173 detail={"fixed": p.get("fixed", []),
174 "lost": p.get("lost") or p.get("quarantined"),
175 "backup": p.get("backup")}, **base)
176 if t in ("session.created", "session.finished"):
177 detail: dict[str, Any] = {}
178 if t == "session.finished":
179 detail["reason"] = p.get("reason")
180 return TimelineNode(kind="recovery", title=t, detail=detail, **base)
181 # 其余持久事件兜底渲染(redact 防敏感原文进面板)
182 return TimelineNode(kind="info", title=str(t), detail=redact(p), **base)

185def derive_timeline(events: Iterable[Envelope], kinds: str = "") -> list[TimelineNode]:
186 """事件流(seq 升序)→ 时间线节点序列;kinds = 逗号过滤(空 = 全部)。
187
188 纯函数:输入只读事件迭代器,无副作用、不读存储;轨迹面板逐帧回放数据源的唯一派生口。
189 """
190 want = {k for k in kinds.split(",") if k} if kinds else None
191 nodes: list[TimelineNode] = []
192 for ev in events:
193 node = render_timeline_node(ev)
194 if want is None or node.kind in want:
195 nodes.append(node)
196 return nodes
