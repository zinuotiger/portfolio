65def _needs_two_args(handler: Callable) -> bool:
66 """handler 形如 (type_, payload) → True;仅 (payload) → False。
67

97class EventBus:
98 """事件总线(specs/bus.py.md EventBus 全量契约)。
99
100 构造不失败;backpressure_limit 默认 1000 在途(F005,与 PARAMETER-ANCHOR
101 锁定一致;可配上限通常来自 config plugins.backpressure_limit,由调用方
102 读取后传入)。
103 """
104
105 # 总线内部瞬时广播类型(仅内存、无 payload 模型——EVENT-SCHEMA §8.1
106 # 落盘矩阵:registry.updated 仅总线,不进 77 词表 EVENT_TYPES)。
107 # 预置入 _types 使 Registry 的 F003 留痕 emit 不被自身 EVT-102 闸拦截。
108 _INTERNAL_TYPES: tuple[str, ...] = ("registry.updated",)
109
110 def __init__(self, backpressure_limit: int = 1000,
111 deadletter_samples: int = 0):
112 self._by_type: dict[str, list[Subscription]] = {} # 精确索引
113 self._wild: list[Subscription] = [] # 通配索引(tool.*)
114 # 事件 schema 注册表(EVT-102 判定):register_type 写入 + 预置内部瞬时
115 # 广播;其余 77 词表类型经 events.vocab 联动放行(见 _type_registered)
116 self._types: dict[str, Optional[type]] = {t: None for t in self._INTERNAL_TYPES}
117 self._queues: dict[str, deque] = {} # per-sender FIFO
118 self.dropped: dict[str, int] = {} # 背压丢弃计数(可观测)
119 self._inflight: set[str] = set() # 投递中 owner(F004 busy 判定)
120 self.backpressure_limit = backpressure_limit # F005 阈值(public)
121 # F005 **死信样本**(R24 接线 `plugins.deadletter_samples`):背压丢弃时
122 # 留最近 N 条**摘要**供诊断(0 = 不采样)。此前该键零读取者 —— 丢弃只有
123 # 计数与 `bus.backpressure` 事件,拿不到"丢的是什么"。
124 self.deadletter_samples = max(0, int(deadletter_samples or 0))
125 self.deadletters: deque = deque(maxlen=self.deadletter_samples or 0)
126 self._sub_seq = itertools.count() # 订阅注册序发号器
127 self._draining: set[str] = set() # 正在排空队列的 sender
128 self._qlocks: dict[str, asyncio.Lock] = {} # per-sender 队列锁

172 def unsubscribe_all(self, owner: str) -> int:
173 """按属主精确摘除全部订阅(精确+通配);返回实际摘除条数,幂等。"""
174 removed = 0
175
176 def _drop(lst: list[Subscription]) -> list[Subscription]:
177 nonlocal removed
178 keep: list[Subscription] = []
179 for s in lst:
180 if s.owner == owner:
181 removed += 1
182 else:
183 keep.append(s)
184 return keep
185
186 self._by_type = {t: _drop(ss) for t, ss in self._by_type.items()}
187 self._by_type = {t: ss for t, ss in self._by_type.items() if ss}
188 self._wild = _drop(self._wild)
189 return removed

191 def _match(self, type_: str) -> list[Subscription]:
192 """匹配:精确先、通配后,各按注册序(spec emit 顺序契约)。"""
193 subs = list(self._by_type.get(type_, ()))
194 for w in self._wild:
195 if _wild_match(w.pattern, type_):
196 subs.append(w)
197 return subs

200 def emit(self, type_: str, payload: dict, mode: str = "sequential"):
201 """总线入口:类型校验(EVT-102)→ 返回分发协程(await 后得统计 dict)。

206 """

212 async def _dispatch(self, type_: str, payload: dict,
213 mode: str = "sequential") -> dict:
214 """三模式分发主体(注册序逐条/前值后入/只读投影 gather)。"""
215 if type_ in SYNC_TYPES: # 强同步事件恒 sequential(§3.7)
216 mode = "sequential"
217 subs = self._match(type_)
218 stats: dict = {"delivered": 0, "errored": 0}
219 if mode == "sequential":
220 for s in subs:
221 if s.when and not self._safe_when(s, payload):
222 continue
223 r = await self._run(s, type_, payload)
224 if r is _FAILED:
225 stats["errored"] += 1
226 else:
227 stats["delivered"] += 1
228 elif mode == "waterfall": # 前返回值 = 后入参;STOP 短路
229 acc = payload
230 for s in subs:
231 if s.when and not self._safe_when(s, acc):
232 continue
233 r = await self._run(s, type_, acc)
234 if r is _FAILED:
235 stats["errored"] += 1
236 continue
237 stats["delivered"] += 1 # 成功执行即计(含 void/STOP 件)
238 if r is STOP:
239 break
240 if r is not None:
241 acc = r
242 return {"result": acc, **stats}
243 else: # parallel:只读投影 gather
244 todo = [s for s in subs
245 if not (s.when and not self._safe_when(s, payload))]
246 if todo:
247 results = await asyncio.gather(
248 *(self._run(s, type_, payload) for s in todo),
249 return_exceptions=True)
250 for r in results:
251 if r is _FAILED or isinstance(r, Exception):
252 stats["errored"] += 1
253 else:
254 stats["delivered"] += 1
255 return stats

257 def emit_sync(self, type_: str, payload: dict) -> dict:
258 """同步投递口:仅供总线内部瞬时事件(如 registry.updated)由同步调用方
259 (Registry.register/unregister 等)使用,保证 F003 留痕确定可见。
260
261 订阅者须为同步函数(registry.updated 仅内存,订阅者即同步刷新器);
262 异步订阅者:有事件循环时 create_task 调度,无循环时跳过并记日志。
263 订阅者异常一律 EVT-103 隔离,不外抛。
264 """
265 if not self._type_registered(type_):
266 raise_code("EVT-102", type_=type_, hint="emit_sync 仅限已注册(含总线内部瞬时)类型")
267 stats: dict = {"delivered": 0, "errored": 0}
268 for s in self._match(type_):
269 if s.when and not self._safe_when(s, payload):
270 continue
271 self._inflight.add(s.owner)
272 try:
273 args: tuple = (type_, payload) if _needs_two_args(s.handler) else (payload,)
274 fn = s.handler(*args)
275 if inspect.isawaitable(fn):
276 try:
277 loop = asyncio.get_running_loop()
278 except RuntimeError:
279 log.warning("emit_sync 无事件循环,跳过异步订阅者 owner=%s pattern=%s",
280 s.owner, s.pattern)
281 continue
282 task = loop.create_task(self._consume_task(fn, s, type_))
283 # 默认参数快照循环变量 s/type_:lambda 闭包延迟绑定会共享循环
284 # 末值,多异步订阅者匹配时 EVT-103 会归属到错误订阅者(B023)。
285 task.add_done_callback(
286 lambda t, s=s, type_=type_: self._task_error(t, s, type_))
287 stats["delivered"] += 1
288 except Exception as exc: # 订阅者崩溃 → EVT-103 隔离
289 log.error("%s", struct_error("EVT-103", type_=type_, owner=s.owner,
290 pattern=s.pattern,
291 exc=f"{type(exc).__name__}: {exc}"))
292 stats["errored"] += 1
293 finally:
294 self._inflight.discard(s.owner)
295 return stats

319 # ------------------------------------------------------------ 背压通道
320 async def emit_ordered(self, sender: str, type_: str, payload: dict) -> dict:
321 """同 sender 事件严格 FIFO 串行(F005);队列满=拒新不丢旧。
322
323 在途 ≥ backpressure_limit → 丢弃计数 +1 并广播 bus.backpressure 瞬时
324 事件(可观测,禁静默),返回 {"delivered":0,"errored":0,"dropped":True};
325 队内旧件按序投递完。首个入队者兼任 drainer 逐件顺序分发;其余调用
326 方入队即返回 {"queued":True}(事件由 drainer 保证 FIFO 送达)。
327 """
328 if not self._type_registered(type_): # 前置类型校验:拒投脏件,队列不动
329 raise_code("EVT-102", type_=type_, hint="emit_ordered 前置类型校验:先 register_type")
330 lock = self._qlocks.setdefault(sender, asyncio.Lock())
331 drop: Optional[int] = None
332 drainer = False
333 async with lock:
334 q = self._queues.setdefault(sender, deque())
335 if len(q) >= self.backpressure_limit: # 满 = 拒新(不丢旧)
336 self.dropped[sender] = self.dropped.get(sender, 0) + 1
337 drop = self.dropped[sender]
338 if self.deadletter_samples > 0: # F005 死信样本(有界)
339 self.deadletters.append({"sender": sender, "type": type_,
340 "dropped_n": drop})
341 else:
342 q.append((type_, payload)) # 入队 = 已承诺,绝不丢
343 if sender not in self._draining:
344 self._draining.add(sender)
345 drainer = True
346 if drop is not None:
347 await self.emit("bus.backpressure", # 背压瞬时事件(可观测)
348 {"sender": sender, "dropped": drop})
349 return {"delivered": 0, "errored": 0, "dropped": True}
350 if not drainer:
351 return {"delivered": 0, "errored": 0, "queued": True}
352 total: dict = {"delivered": 0, "errored": 0}
353 try:
354 while True:
355 async with lock: # 退出判定与清标记同锁原子
356 if not self._queues[sender]:
357 self._draining.discard(sender)
358 return total
359 t, p = self._queues[sender].popleft()
360 st = await self.emit(t, p, mode="sequential") # 未完成不取下一件
361 total["delivered"] += st["delivered"]
362 total["errored"] += st["errored"]
363 finally: # 异常兜底:防队列悬挂
364 self._draining.discard(sender)

377 """单订阅执行器:async 直接 await;同步 handler 经 to_thread 执行

394 exc=f"{type(exc).__name__}: {exc}"))

415def schedule_emit(bus: EventBus, type_: str, payload: dict, *,
416 mode: str = "sequential") -> Optional[asyncio.Task]:
417 """同步调用方安全调度 emit:类型校验同步抛 EVT-102,分发协程排入运行中循环。
418
419 emit() 返回协程,裸调用会丢弃;本助手保留 task 引用并统一收口异常,供
420 config/approval/scope/guard 等 fire-and-forget 出口使用。无运行中事件循环
421 时关闭协程并记警告(事件不投递但不泄漏协程)。
422 """
423 result = (bus.emit(type_, payload, mode=mode)
424 if mode != "sequential" else bus.emit(type_, payload))
425 if not asyncio.iscoroutine(result):
426 return result
427 try:
428 loop = asyncio.get_running_loop()
429 except RuntimeError:
430 log.warning("schedule_emit: 无运行中事件循环,%s 未投递", type_)
431 result.close()
432 return None
433 task = loop.create_task(_await_dispatch(result))
434 _PENDING_EMITS.add(task)
435 task.add_done_callback(_PENDING_EMITS.discard)
436 return task
