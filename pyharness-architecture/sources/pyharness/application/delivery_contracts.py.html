19class DeliveryOutputReference(StrictModel):
20 kind: Literal['artifact', 'message']
21 execution_id: ExecutionId
22 session_id: str = Field(pattern=r'^s-[A-Za-z0-9_-]{1,120}$')
23 task_id: str = Field(pattern=r'^(?:p-[0-9a-f]{32}|subagent:s-[A-Za-z0-9_-]{1,120})$')
24 source_seq: int = Field(ge=1)
25 record_digest: Digest
26 sha256: Digest
27 size_bytes: int = Field(ge=0, le=32*1024*1024)
28 artifact_id: str | None = Field(default=None, pattern=r'^f-[0-9a-f]{32}$')
29
30 @model_validator(mode='after')
31 def coherent(self):
32 if (self.kind == 'artifact') != (self.artifact_id is not None):
33 raise ValueError('artifact handle applies only to generated files')
34 if self.kind == 'message' and not self.size_bytes:
35 raise ValueError('empty public message is not a delivery output')
36 return self

39def output_identity(value):
40 """Public immutable identity; changing a digest cannot create another event."""
41 return value.execution_id, value.session_id, value.task_id, value.source_seq, value.kind

44def distinct_outputs(values):
45 if len({output_identity(v) for v in values}) != len(values):
46 raise ValueError('duplicate output event')
47 return values

60class DeliveryPackageCommand(VersionOne):
61 schema_version: Literal[1] = Field(...)
62 package_id: PackageId
63 project_revision: int = Field(ge=1)
64 execution_ref: ExecutionReference
65 delivery_ref: DeliveryReference

67
68 @model_validator(mode='after')
69 def bounded(self):
70 if len({b.item_id for b in self.bindings}) != len(self.bindings):
71 raise ValueError('duplicate delivery item')
72 outputs = [o for b in self.bindings for o in b.outputs]
73 if len(outputs)>60 or len({output_identity(o) for o in outputs})>40:
74 raise ValueError('delivery output limit')
75 return self
