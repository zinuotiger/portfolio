25FRONT_RE = re.compile(r"^---\s*\n(.*?)^---\s*$", re.S | re.M)
26_NAME_RE = re.compile(r"^\s*name:\s*([A-Za-z0-9._-]+)\s*$", re.M)
27_DESC_RE = re.compile(r"^\s*description:\s*(.+?)\s*$", re.M)
28_BODY_CAP = 6000 # 单次装载正文上限(超出截断,防上下文爆炸)

31def _parse_frontmatter(text: str) -> Optional[dict]:
32 """取 SKILL.md 头 frontmatter → {name, description};无/坏 → None。"""
33 m = FRONT_RE.search(text)
34 if not m:
35 return None
36 head = m.group(1)
37 nm = _NAME_RE.search(head)
38 if not nm:
39 return None
40 desc = _DESC_RE.search(head)
41 return {"name": nm.group(1),
42 "description": (desc.group(1).strip().strip('"\'')
43 if desc else "")}

46class SkillManager:
47 """技能库索引/装载器:scan() → list() 目录 / load(name) 正文 / render_catalog()。
48
49 纯只读文本操作;会话事件(skill.used)由工具面经 session.append 落盘。
50 """

52 def __init__(self, roots, *, session: Any = None, bus: Any = None) -> None:
53 self.roots: list[Path] = [Path(r) for r in roots]
54 self._session = session
55 self._bus = bus
56 self._index: dict[str, dict] = {}
57 self._source_hashes: dict[str, str] = {}
58 self._frozen = False
59 self._loaded_views: dict[str, dict] = {}
60 self.scan()

62 # ------------------------------------------------------------ 索引
63 def scan(self) -> int:
64 """重建索引(启动/热重载调用);格式坏或重复名的技能跳过并记日志。"""
65 if self._frozen:
66 return len(self._index)
67 from pyharness.core.skill_registry import _safe_under
68 from pyharness.errors import PyHError
69 found: dict[str, dict] = {}
70 hashes: dict[str, str] = {}
71 for root in self.roots:
72 if not root.exists():
73 continue
74 for md in sorted(root.glob("*/SKILL.md")):
75 try:
76 target = _safe_under(root, md.parent.name, 'SKILL.md')
77 raw = target.read_bytes()
78 if _safe_under(root, md.parent.name, 'SKILL.md') != target:
79 continue
80 # Preserve the original read_text universal-newline view;
81 # retain the physical bytes separately for source identity.
82 text = raw.decode('utf-8').replace('\r\n', '\n').replace('\r', '\n')
83 except (OSError, UnicodeError, PyHError) as e:
84 log.warning("skill read failed %s: %s", md, e)
85 continue
86 meta = _parse_frontmatter(text)
87 if not meta:
88 log.warning("skill skip(坏 frontmatter): %s", md)
89 continue
90 name = meta["name"]
91 if name in found:
92 log.warning("skill 重名跳过: %s (%s)", name, md)
93 continue
94 found[name] = {
95 "name": name,
96 "description": meta["description"],
97 "dir": str(md.parent),
98 "body": self._strip_front(text),
99 }
100 hashes[name] = hashlib.sha256(raw).hexdigest()
101 self._index = found
102 self._source_hashes = hashes
103 return len(found)

105 def frozen_copy(self, names=None):
106 """Private loaded index copy; explicit classic scans remain available."""
107 clone = SkillManager([], session=self._session, bus=self._bus)
108 chosen = list(self._index) if names is None else list(names)
109 for name in chosen:
110 self.load(name) # Original unknown-name contract.
111 clone._index = {name: copy.deepcopy(self._index[name]) for name in chosen}
112 clone._source_hashes = {name: self._source_hashes[name] for name in chosen}
113 clone._loaded_views = {name: copy.deepcopy(self._loaded_views[name])
114 for name in chosen if name in self._loaded_views}
115 clone._frozen = True
116 return clone

118 @classmethod
119 def from_loaded_snapshot(cls, snapshot):
120 """Restore a server-owned bounded load view, without rereading its path."""
121 clone = cls([])
122 name = snapshot['name']
123 view = {k: snapshot[k] for k in ('name', 'description', 'dir', 'body')}
124 clone._index[name] = copy.deepcopy(view)
125 clone._source_hashes[name] = snapshot['source_sha256']
126 clone._loaded_views[name] = copy.deepcopy(view)
127 clone._frozen = True
128 if clone.snapshot(name)['loaded_sha256'] != snapshot['loaded_sha256']:
129 raise_code('SKL-901', name=name, advice='固定技能视图校验失败')
130 return clone

132 def snapshot(self, name: str) -> dict:
133 """Describe exact loaded text separately from complete source identity."""
134 loaded = self.load(name)
135 raw = json.dumps({k: loaded[k] for k in ('name', 'description', 'body')},
136 ensure_ascii=False, sort_keys=True, separators=(',', ':')).encode('utf-8')
137 return {**loaded, 'source_sha256': self._source_hashes[name],
138 'loaded_sha256': hashlib.sha256(raw).hexdigest(),
139 'truncated': len(self._index[name]['body']) > _BODY_CAP,
140 'frozen': self._frozen}

142 @staticmethod
143 def _strip_front(text: str) -> str:
144 m = FRONT_RE.match(text)
145 body = text[m.end():] if m else text
146 return body.strip()

148 # ------------------------------------------------------------ 读面
149 def list(self) -> list[dict]:
150 """技能目录(名字+一句话),按名排序;目录注入/工具面共用。"""
151 return [{"name": m["name"], "description": m["description"]}
152 for m in sorted(self._index.values(), key=lambda x: x["name"])]

154 def get(self, name: str) -> Optional[dict]:
155 return self._index.get(name)

157 def load(self, name: str) -> dict:
158 """装载技能全文;未知名 → SKL-901(advice 指向 skill.list)。"""
159 if name in self._loaded_views:
160 return copy.deepcopy(self._loaded_views[name])
161 m = self._index.get(name)
162 if m is None:
163 raise_code("SKL-901", name=name,
164 advice="技能未找到;先 skill.list 看可用技能清单")
165 body = m["body"]
166 if len(body) > _BODY_CAP:
167 body = body[:_BODY_CAP] + "\n…(正文超长截断)"
168 return {"name": m["name"], "description": m["description"],
169 "dir": m["dir"], "body": body}

171 def render_catalog(self) -> str:
172 """目录段文本(system prompt 注入;空技能库返回空串零开销)。"""
173 rows = [f"- {m['name']}: {m['description']}"
174 for m in self.list()]

176 return ""
177 return ("可用技能目录(任务匹配某技能时,调 skill.load 装载后按技能执行):\n"
178 + "\n".join(rows))
