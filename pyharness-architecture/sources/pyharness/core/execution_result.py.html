75def _short_value(v: Any, width: int = 80, key: Optional[str] = None) -> str:
76 """摘要值短化:dict/list 只报结构;字符串截断;None/bool/int 直写。
77
78 内容承载键(key ∈ _SUMMARY_TAIL_KEYS)超宽时头尾双展示(保留尾部 _TAIL_KEEP
79 字符)——审批视图不得把危险命令段藏在省略号后;普通键单侧截断。
80 """
81 if isinstance(v, dict):
82 return f"<dict {len(v)}键>"
83 if isinstance(v, (list, tuple)):
84 return f"<list {len(v)}项>"
85 if v is None:
86 return "null"
87 s = str(v)
88 if len(s) <= width:
89 return s
90 if key is not None and key in _SUMMARY_TAIL_KEYS:
91 head = s[: max(0, width - _TAIL_KEEP - 2)]
92 return f"{head} … {s[-_TAIL_KEEP:]}"
93 return s[: width - 1] + "…"

95def summarize(args: Any, *, max_chars: int = 400) -> str:
96 """Consumer 层参数摘要化(审批展示用;spec 速览表 def summarize(args))。
97
98 不由 Provider/LLM 生成(防注入操纵摘要,SECURITY §5):只做键值短化 + 动作
99 键优先排序;超长截断。content/command 类键头尾双展示(尾部可见,防截断隐藏)。
100 """
101 if not isinstance(args, dict) or not args:
102 return "(无参数)"
103 ordered = sorted(args, key=lambda k: (_SUMMARY_PRIORITY.index(k)
104 if k in _SUMMARY_PRIORITY else 99, k))
105 parts: list[str] = []
106 for k in ordered:
107 seg = f"{k}={_short_value(args[k], key=k)}"
108 if sum(len(p) for p in parts) + len(seg) > max_chars:
109 parts.append("…")
110 break
111 parts.append(seg)
112 return ", ".join(parts)

114def _approval_binding(call: Any, args: dict) -> str:
115 """审批绑定指纹:sha1(tool + 规范化参数序列化)。
116
117 唯一用途 = 把"人类批准"与"实际执行"绑定到同一调用:granted 后重入 guard
118 得到 approval 决策时,校验本指纹 == 审批提供器为该 call_id 记录的绑定,
119 不一致 = 批准针对的是不同参数(注入/篡改),拒绝执行(GRD-403)。参数序列化
120 用 sort_keys 定序(同一调用内同 args 确定性一致;跨会话信任仍走 approval 层
121 自己的 canonical_json)。
122 """
123 canon = json.dumps(args, sort_keys=True, ensure_ascii=False, default=str)
124 return hashlib.sha1(f"{call.name}\n{canon}".encode("utf-8")).hexdigest()
