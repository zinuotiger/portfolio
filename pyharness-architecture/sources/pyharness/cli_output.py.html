73 # ---------------------------------------------------- 瞬时增量(llm.chunk)
74 def delta(self, payload: dict) -> None:
75 text = payload.get("delta", "")
76 if self.mode == "json":
77 self.event("llm.chunk", payload) # 机器流:逐事件一行
78 return
79 if text:
80 sys.stdout.write(text)
81 sys.stdout.flush()
82 self.carriage = True

84 # ---------------------------------------------------- 卡片/结构化事件
85 def event(self, type_: str, payload: Any) -> None:
86 """单事件出口:text 模式按类型画卡片;json 模式逐事件一行 JSONL。"""
87 payload = _normalize_payload(payload)
88 if self.mode == "json":

93 self.carriage = False

104def _normalize_payload(payload: Any) -> dict:
105 """归一化到**事件 payload**(Envelope/模型/裸 dict 三形态,鸭子类型兼容)。
106
107 总线投递的 canonical 形状是 Envelope(SessionLog._dispatch →
108 ``bus.emit(type, env)``),渲染面只消费其内层 ``payload``;故 dump 出信封形状
109 (含 session_id/seq/payload 三键)时取内层——否则渲染面按信封字段取值会全部落空
110 (工具名/结果/终态理由退化为 ``?``/``{}``/``None``)。瞬时类型(llm.chunk)不经
111 信封,仍走裸 dict。
112 """
113 if isinstance(payload, dict):
114 return payload
115 dump = getattr(payload, "model_dump", None)
116 if callable(dump):
117 try:
118 data = dump(mode="json")
119 except TypeError: # 老版 pydantic 无 mode 参
120 data = dump()
121 inner = data.get("payload") if isinstance(data, dict) else None
122 if isinstance(inner, dict) and "session_id" in data and "seq" in data:
123 return inner # Envelope → 内层 payload
124 return data
125 return {"raw": str(payload)}

131def _text_card(type_: str, payload: dict) -> Optional[str]:
132 """单事件文本卡片;未列类型返回 None(text 模式静默)。"""
133 if type_ not in _TEXT_CARD_TYPES:
134 return None
135 if type_ == "agent.message":
136 return payload.get("content", "") or ""
137 if type_ == "user.message":
138 return ""
139 if type_ == "tool.call":
140 name = payload.get("tool") or payload.get("name") or "?"
141 args = payload.get("args_summary") or payload.get("args") or {}
142 return f"[工具] {name} {_trunc(json.dumps(args, ensure_ascii=False))}"
143 if type_ == "tool.result":
144 data = payload.get("content") or payload.get("summary") or payload.get("result")
145 return f"[结果] {_trunc(str(data))}"
146 if type_ == "tool.error":
147 return f"[工具错误] {_trunc(payload.get('error') or str(payload))}"
148 if type_ == "guard.rejected":
149 return f"[拒绝] {_trunc(json.dumps(payload, ensure_ascii=False))}"
150 if type_ == "session.finished":
151 return f"[会话结束] {payload.get('reason', '')}"
152 if type_ == "task.completed":
153 return (f"[任务完成] {payload.get('task_id', '')} "
154 f"{payload.get('reason', '')}").strip()
155 if type_ == "task.failed":
156 return (f"[任务失败] {payload.get('task_id', '')} "
157 f"{payload.get('error', '')} {payload.get('reason', '')}".strip())
158 return None

217async def _wait_and_render(sh: ShellCtx, task_id: str) -> Any:
218 """wait_for 终态 + 并行总线渲染:渲染子任务随 wait 返回强制取消(终态事件或
219 取消兜底双通道,杜绝悬挂订阅)。"""
220 render = None
221 bus = getattr(sh.ctx, "bus", None)
222 if bus is not None:
223 render = asyncio.create_task(_render_events(sh, task_id))
224 try:
225 return await sh.ctx.task_queue.wait_for(task_id)
226 finally:
227 if render is not None:
228 render.cancel()
229 try:
230 await render
231 except (asyncio.CancelledError, Exception): # noqa: BLE001 渲染清理
232 pass

234async def _render_events(sh: ShellCtx, task_id: str) -> None:
235 """事件流渲染:订阅 llm.chunk(瞬时)/agent.message/tool.call/guard.rejected/
236 approval.* 等;text 模式增量打字机 + 卡片,approval 触发交互提示;json 模式
237 逐事件一行 JSONL 到 stdout(机器可读事件流)。终态事件或外部取消即收尾。"""
238 renderer = StreamRenderer(mode="json" if sh.flags.get("json") else "text")
239 q: asyncio.Queue = asyncio.Queue()
240 bus = getattr(sh.ctx, "bus", None)
241 if bus is None:
242 return # 无总线:渲染退化为摘要兜底
243 subscriptions = []
244 try:
245 for pattern in _RENDER_PATTERNS:
246 sub = bus.subscribe(pattern, lambda t, p: q.put_nowait((t, p)),
247 owner="cli")
248 subscriptions.append(sub)
249 while True:
250 type_, payload = await q.get()
251 if type_.startswith("approval."):
252 await prompt_approval(sh, type_, payload) # 审批交互(F015)
253 continue
254 if type_ == "llm.chunk":
255 renderer.delta(_normalize_payload(payload))
256 continue
257 renderer.event(type_, payload) # 卡片/json 行
258 if type_ in _END_EVENT_TYPES:
259 break
260 finally:
261 bus.unsubscribe_all("cli")
262 renderer.newline()

264async def prompt_approval(sh: ShellCtx, type_: str, payload: Any) -> None:
265 """审批交互(F015 人类裁决):**仅** approval.requested 且通道非 None → 打印工具/
266 参数摘要与风险级,键盘 [y/N] 裁决 → approve/deny;headless(channel=None)不提示
267 ——裁决侧 APR-501 自动拒(零等待,勿悬挂)。
268
269 身份契约(D1b):approval_id = 该 approval.requested 事件的 **Envelope.seq**
270 (events/payload.py:189 载荷契约 / approval.py:292 生产赋值 / acp.py cmd_approve
271 入口校验 三处同源;ApprovalRequestedPayload 四字段不含 approval_id)。结果事件
272 (granted/denied/timeout)载荷虽含 approval_id,但不是人类动作请求 → 一律不提问。
273 """
274 if type_ != "approval.requested":
275 return # 结果事件:不提问(防误裁决)
276 if sh.headless or sh.ctx.channel is None:
277 return # 无通道即拒:裁决侧已处理
278 aid = getattr(payload, "seq", None) # canonical identity(信封 seq)
279 if not isinstance(aid, int):
280 return # 非信封来源:fail-safe 不提问

282 approval = getattr(sh.ctx, "approval", None)
283 ttl = payload.get("ttl_ms", "")
284 risk = payload.get("risk", "?")
285 tool = payload.get("tool", type_)
286 print(f"\n[审批] {tool} {_trunc(str(payload.get('args_summary', '')), 160)}"
287 f" (risk={risk}, ttl={ttl}ms)", file=sys.stderr)
288 try:
289 ans = (await asyncio.to_thread(input, "")).strip().lower()
290 except (EOFError, KeyboardInterrupt): # 用户中止 = 拒绝(等价 deny)
291 ans = "n"
292 verdict_fn = (approval.approve if ans in {"y", "yes"} else approval.deny)

294 out = verdict_fn(aid, by="cli")
295 if hasattr(out, "__await__"):
296 await out # 裁决入日志(approval.granted/denied)
