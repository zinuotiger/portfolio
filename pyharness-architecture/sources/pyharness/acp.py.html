90 """ACP 会话队列惰性装配:轻量 ctx(无队列/异会话队列)时接真实引擎。
91
92 原 cli.acp 只注入轻量门面,队列/runner 从未接线;这里把 engine 装配
93 原语搬到 ACP chat 前,使 initialize → chat 也能走真实 AgentLoop。
94 """
95 cfg = getattr(ctx, "settings", None) or getattr(ctx, "config", None)
96 runtimes = getattr(ctx, '_acp_runtimes', None)
97 if runtimes is None:
98 runtimes = ctx._acp_runtimes = {}
99 if sid in runtimes:
100 runtime = runtimes[sid]
101 for name in ('engine_spine', 'task_queue', 'approval', 'guard', 'scope', 'llm', 'tools', 'governance'):
102 setattr(ctx, name, getattr(runtime, name))
103 return runtime.task_queue
104 q = getattr(ctx, "task_queue", None)
105 if q is not None and (not hasattr(q, "_session")
106 or getattr(q, "_session", None) is log_):
107 return q
108 if cfg is None:
109 return q
110 from pathlib import Path
111 from pyharness import engine as _eng
112 storage = getattr(ctx, "storage", None)
113 sessions_dir = getattr(storage, "sessions_dir", None)
114 if not sessions_dir:
115 try:
116 sessions_dir = cfg.storage.sessions_dir
117 except Exception: # noqa: BLE001 配置缺键:不装配
118 return q
119 import copy
120 runtime = copy.copy(ctx)
121 runtime.session = log_
122 runtime.engine_spine = None
123 runtime.task_queue = None
124 if getattr(ctx, 'llm_runtime', None) is None:
125 from pyharness.core.llm import LLMRuntime
126 ctx.llm_runtime = LLMRuntime()
127 ctx._acp_owns_llm = True
128 runtime.llm_runtime = ctx.llm_runtime
129 runtimes[sid] = runtime
130 session_cfg = cfg.model_copy(deep=True) if hasattr(cfg, 'model_copy') else cfg
131 try:
132 await _eng.attach_engine_to_ctx(
133 runtime, session_cfg, log_=log_,
134 sessions_dir=Path(str(sessions_dir)).expanduser(),
135 bus=getattr(ctx, "bus", None),
136 store=getattr(log_, "_persistence", None),
137 # GAP-11:ACP 桥即人类通道。``_set_channel`` 已在 bootstrap 期写入
138 # ``ctx.channel = "acp:<client_id>"``;此处把它**声明给引擎**(spine →
139 # agent ctx),否则治理决策会 APR-503 fail-closed。字段缺失时传 None =
140 # 显式"无人类通道"语义(与"从未声明"是两件事)。
141 channel=getattr(ctx, "channel", None))
142 except BaseException as primary:
143 try:
144 if runtime.engine_spine is not None: await runtime.engine_spine.close()
145 runtimes.pop(sid, None)
146 except BaseException as secondary:
147 primary.add_note(f'ACP initialization cleanup failed: {type(secondary).__name__}')
148 raise
149 for name in ('engine_spine', 'task_queue', 'approval', 'guard', 'scope', 'llm', 'tools', 'governance'):
150 setattr(ctx, name, getattr(runtime, name))
151 return runtime.task_queue

154async def cmd_initialize(ctx: Any, st: AcpState, params: dict) -> dict:
155 """能力协商(initialize):能力清单 + 会话绑定/新建 + 通道确立。
156
157 sessionId 参数 → 绑定既有会话(repair 前置 → open → 终态 EVT-104 拒);
158 缺省 → 新建会话(session.created 由门面 create 落,created=True)。
159 st.cursor 复位 0(新连接游标从 0 起);ctx.channel = "acp:<client>"。
160 """
161 params = params or {}
162 sid = params.get("sessionId")
163 if sid is not None and (not isinstance(sid, str) or not sid.strip()):
164 raise_code("EVT-100", field="sessionId", session_id=sid,
165 hint="sessionId 须非空字符串(缺省则新建会话)")
166 if sid:
167 await _ensure_repaired(ctx, sid) # 损坏先 repair(F060 前置)
168 log_ = await _open_session(ctx, sid) # 绑定既有(重放重建)
169 if _closed_of(log_): # 终态会话不可续 chat
170 raise_code("EVT-104", session_id=sid,
171 advice="会话已终态(session.finished);请另开新 sessionId")

173 else:
174 sid = await _create_session(ctx) # 新建会话(created seq=1)
175 created = True
176 st.session_id = sid
177 st.cursor = 0
178 st._bound_log = None # 绑定柄下次操作惰性解析
179 _set_channel(ctx, st) # 桥 = 审批通道(非 headless)
180 return {"protocolVersion": PROTOCOL_VERSION,
181 "methods": list(BRIDGE_METHODS),
182 "events": {"cursor": True},
183 "session": {"id": sid, "created": created},
184 "engine": {"model": _engine_model(ctx),
185 "headless_channel": False}}

188def _chat_summary(log_: Any, task_id: str, enq_seq: Optional[int],
189 res: Any, reason: str) -> dict:
190 """阻塞 chat 终局摘要(偏离 3:从会话日志派生,无第二份状态)。
191
192 final:TaskResult.summary 优先 → 任务段内最后 agent.message/llm.response
193 内容兜底 → 终态码文本;rejected = 段内 guard.rejected payload 清单;
194 approvals = 段内 approval.* payload 清单(事件带 task_id 或窗口内无主事件)。
195 """
196 after = enq_seq if enq_seq is not None else 0
197 final = res.summary if getattr(res, "summary", None) else None
198 rejected: list[dict] = []
199 approvals: list[dict] = []
200 for env in _iter_log(log_):
201 if env.seq <= after:
202 continue
203 tid = env.task_id
204 if tid is not None and tid != task_id:
205 continue # 异任务事件隔离(串行队列下少见)
206 if env.type == "guard.rejected":
207 rejected.append(dict(env.payload))
208 elif env.type.startswith("approval."):
209 approvals.append(dict(env.payload))
210 elif final is None and env.type in ("agent.message", "llm.response"):
211 content = (env.payload or {}).get("content")
212 if content:
213 final = content # 最后一条非空展示文本即终局
214 if final is None:
215 final = str(getattr(res, "code", None) or reason)
216 return {"final": final, "rejected": rejected, "approvals": approvals}

219async def cmd_chat(ctx: Any, st: AcpState, params: dict) -> dict:
220 """结构化驱动主方法(chat,F066):wait=true 阻塞至终态,wait=false 秒回 task_id。
221
222 忙锁:st.busy 期间再来 chat → PyHError("BUSY")(dispatch 特判 -32000,偏离 4);
223 空 text → EVT-100;会话经 sessionId 显式指定或 initialize 绑定;user.message
224 强同步落盘(origin="acp:<client>",审计同 CLI)后 submit → wait_for。
225 """
226 params = params or {}
227 if st.busy: # 防御性忙拒(偏离 9;并发入口生效)
228 raise PyHError("BUSY", ctx={"advice":
229 "上一条 chat 未完成;等 read_events 就绪或 wait=false"})
230 text = str(params.get("text") or "").strip()
231 if not text:
232 raise_code("EVT-100", field="text",
233 advice="text 必填(非空字符串)")
234 log_, sid = await _ensure_bound(ctx, st, params.get("sessionId"))
235 append = getattr(log_, "append", None)
236 if not callable(append):
237 raise_code("EVT-106", session_id=sid,
238 hint="会话无 append 写面(只读投影);无法发起 chat")
239 queue = await _ensure_engine_queue(ctx, log_, sid)
240 if queue is None or not callable(getattr(queue, "submit", None)):
241 raise_code("CYC-999", hint="桥未装配任务队列(ctx.task_queue.submit)")
242 st.busy = True
243 try:
244 # 强同步 user.message(审计同 CLI);origin 记桥通道溯源
245 await _maybe_await(append("user.message", {"content": text},
246 actor="user", origin=f"acp:{st.client_id}",
247 sync=True))
248 task_id = await _maybe_await(queue.submit(
249 text, meta={"channel": f"acp:{st.client_id}", "session_id": sid}))
250 enq_seq = _task_enqueued_seq(log_, task_id)
251 if not params.get("wait", True): # 异步:read_events 跟进结果
252 return {"task_id": task_id, "reason": "queued", "summary": {},
253 "seq_range": [enq_seq, enq_seq] if enq_seq is not None else None}
254 wait_for = getattr(queue, "wait_for", None)
255 if not callable(wait_for):
256 raise_code("CYC-999", hint="队列未装配 wait_for;任务已入队"
257 "(任务仍会执行,结果经 read_events 消费)")
258 res = await _maybe_await(wait_for(task_id)) # 阻塞至终态(串行语义)
259 reason = ("complete" if res.ok else
260 ("cancelled" if getattr(res, "code", None) == "cancelled"
261 else "error"))
262 summary = _chat_summary(log_, task_id, enq_seq, res, reason)
263 hi = _last_seq_of(log_)
264 seq_range = [enq_seq, hi] if enq_seq is not None else None
265 return {"task_id": task_id, "reason": reason,
266 "summary": summary, "seq_range": seq_range}
267 finally:
268 st.busy = False

305async def cmd_read_events(ctx: Any, st: AcpState, params: dict) -> dict:
306 """事件游标订阅(read_events):从游标读持久事件批,返回 to_seq 供断线续拉。
307
308 after_seq/cursor 显式或取 st.cursor(单调推进);limit ≤500(锁死);空批
309 from_seq=to_seq=当前游标;瞬时事件(如 llm.chunk)天然不在日志(append 拒写),
310 本方法只给持久事件——真源派生,只追加流按 seq 序。
311 """
312 params = params or {}
313 log_, sid = await _ensure_bound(ctx, st, params.get("sessionId"))
314 after_raw = params.get("after_seq", params.get("cursor", st.cursor))
315 if not isinstance(after_raw, int) or isinstance(after_raw, bool):
316 raise_code("EVT-100", field="after_seq", value=after_raw,
317 hint="after_seq/cursor 须 int(seq 游标)")
318 after = max(0, after_raw)
319 try:
320 limit = int(params.get("limit", DEFAULT_EVENT_LIMIT))
321 except (TypeError, ValueError):
322 raise_code("EVT-100", field="limit", value=params.get("limit"),
323 hint=f"limit 须 int(≤{MAX_EVENT_LIMIT})")
324 limit = max(0, min(limit, MAX_EVENT_LIMIT))
325 batch: list[dict] = []
326 last = after
327 ea = getattr(log_, "events_after", None)
328 if limit > 0 and callable(ea):
329 for env in ea(after_seq=after): # 只追加流按 seq 序
330 batch.append(_env_dict(env))
331 last = env.seq
332 if len(batch) >= limit:
333 break
334 st.cursor = max(st.cursor, last) # 游标只进不退(单连接单调)
335 return {"from_seq": after, "to_seq": last, "events": batch,
336 "has_more": len(batch) == limit and limit > 0}

359 elif m == "approve":
360 r = await cmd_approve(ctx, st, p)
361 elif m == "read_events":
362 r = await cmd_read_events(ctx, st, p)
363 elif m == "shutdown":
364 st.notify_shutdown = True # 回响应后净退 0
365 return {"jsonrpc": "2.0", "id": rid, "result": {"shutdown": True}}
366 else:
367 return error_resp(rid, -32601, "Method not found", {"method": m})
368 return {"jsonrpc": "2.0", "id": rid, "result": r}
369 except PyHError as e:
370 if e.code == "BUSY": # 偏离 4:忙拒 → 标准码 -32000
371 return error_resp(rid, -32000, "Busy",
372 {"code": "BUSY", "advice": e.ctx.get("advice")})
373 return engine_error(rid, e) # F019 码进 data.code
374 except asyncio.TimeoutError:
375 return error_resp(rid, -32000, "Busy/Timeout", {"code": "BUSY"})
376 except Exception as e: # noqa: BLE001 未预期兜底
377 log.error("acp dispatch 未预期异常 method=%s", m, exc_info=True)
378 return error_resp(rid, -32603, "Internal error",
379 {"code": "CYC-999", "detail": f"{type(e).__name__}: {e}"[:300]})

385# ------------------------------------------------------------ 主循环
386async def serve(ctx: Any, *, client_id: Optional[str] = None,
387 stdin: Any = None, stdout: Any = None) -> int:
388 """桥主循环(F066 验收直译):逐行读 stdin,一次一请求串行,净退 0。
389
390 client_id 缺省 "acp:<pid>";stdin/stdout 缺省 sys.stdin/sys.stdout(单测注入
391 StringIO 替身);EOF/空行 → 净退 0;shutdown(请求回响应后/通知)净退 0;
392 OSError(stdin/stdout 断)→ 记日志退 1;空行容忍(编辑器/echo 尾随换行)。
393 """
394 st = AcpState(client_id=client_id or f"acp:{os.getpid()}")
395 inp = stdin if stdin is not None else sys.stdin
396 out = stdout if stdout is not None else sys.stdout
397 try:
398 while True:
399 line = await _readline_async(inp) # EOF → None → 净退 0
400 if line is None:
401 return 0
402 line = line.strip()
403 if not line:
404 continue # 空行容忍
405 req = parse_request(line, write=out)
406 if req is None:
407 continue # 协议错已回标准码
408 if req.id is None: # 通知:处理不响应(推进净退旗标)
409 await notify(ctx, st, req)
410 if st.notify_shutdown:
411 return 0
412 continue
413 resp = await dispatch(ctx, st, req) # 一次一请求串行
414 _write_line(out, resp)
415 if st.notify_shutdown: # shutdown 请求已回响应 → 净退 0
416 return 0
417 except OSError as e:
418 log.error("acp stdin/stdout 断(OSError): %s", e)
419 return 1
420 finally:
421 errors = []
422 try: await close_owned_runtimes(ctx)
423 except BaseException as exc: errors.append(exc)
424 shutdown = getattr(getattr(ctx, 'session', None), 'shutdown_all', None)
425 if callable(shutdown):
426 try: await _maybe_await(shutdown())
427 except BaseException as exc: errors.append(exc)
428 if errors: raise BaseExceptionGroup('ACP shutdown failed', errors)

431class AcpBridge:
432 """外壳装配面(cli.py.md:AcpBridge(ctx, channel="acp:cli").serve())。
433
434 channel 形如 "acp:cli"/"acp:<client>":serve 时剥 "acp:" 前缀作 client_id
435 (缺省 client_id 为 "acp:<pid>" 语义保持)。
436 """
437
438 def __init__(self, ctx: Any, *, channel: Optional[str] = None) -> None:
439 self.ctx = ctx
440 self.channel = channel
441
442 async def serve(self, *, client_id: Optional[str] = None,
443 stdin: Any = None, stdout: Any = None) -> int:
444 """委托模块 serve;未显式 client_id 时取 channel 后缀。"""
445 cid = client_id
446 if cid is None and self.channel:
447 cid = (self.channel[4:] if str(self.channel).startswith("acp:")
448 else str(self.channel))
449 return await serve(self.ctx, client_id=cid,
450 stdin=stdin, stdout=stdout)
