63 def _default_child_scope(self, parent_scope: Any, sub_id: str,
64 child_session: Any, limits: Any,
65 deny_extra: Optional[list[str]] = None) -> Any:
66 """默认子 scope 工厂:父策略快照值拷贝 + deny_extra 只收窄 + 独立根。
67
68 workspace 根 = storage.workspaces_dir/{sub_id}(F055 每会话独立根,派生见
69 `_child_workspace_root`)且**立即建目录**(否则子代理 fs.* 必然 TLB-802);
70 预算 limits 用缩放后的子预算(父×ratio,下限 1);spec.deny_extra 并入子策略
71 deny_tools(只紧不松,无父级担保通道)。Scope 直接构造不注册全局表——子 scope
72 生命周期随 handle(偏离 1)。
73 """
74 if parent_scope is None:
75 return None
76 snap = getattr(parent_scope, "snapshot", None)
77 if not callable(snap):
78 return None
79 try:
80 from pathlib import Path
81 from pyharness.core.scope import Scope, ScopePolicy
82 s = snap()
83 pol = ScopePolicy(**s.policy.model_dump()) # 值拷贝(COW 起点)
84 root = self._child_workspace_root(s.policy.workspace_root, sub_id)
85 pol.workspace_root = root # 独立根(F055)
86 try:
87 Path(root).mkdir(parents=True, exist_ok=True) # 根必须存在
88 except OSError as exc: # 权限/磁盘:不降级 scope
89 log.warning("子 workspace 根创建失败 sub=%s path=%s: %s"
90 "(文件类工具将报 TLB-802)", sub_id, root, exc)
91 pol.deny_tools |= set(deny_extra or ()) # deny 只收窄(单调)
92 return Scope(policy=pol, limits=limits,
93 session_id=sub_id,
94 window_tokens=getattr(s, "window_tokens", 64000),
95 window_ratio=getattr(s, "window_ratio", 0.75),
96 session=child_session,
97 # 继承父预算计数器:子 agent 用量并入同一预算闸(P2 修复,
98 # 此前 None → 子 scope 预算恒 ok,烧钱无感)
99 counters=self._child_counters(parent_scope, limits),
100 bus=getattr(self, "_bus", None))
101 except Exception as exc: # noqa: BLE001
102 raise PyHError('GRD-403', ctx={'reason': 'child_scope_unavailable'}) from exc
