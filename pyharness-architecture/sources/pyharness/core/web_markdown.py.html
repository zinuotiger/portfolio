25class _TextExtractor(html.parser.HTMLParser):
26 """手写 HTMLParser 状态机正文提取器(spec 实现要点权威)。
27
28 状态:skip(命中丢弃块后的后代深度计数)/parts(输出流,块界以 "\\n" 标记)/
29 links(a 标签栈,出标签拼 [text](url));script/style/noscript/svg/head/
30 iframe/nav/footer 标签与 class/id 噪声命中块整体跳过(其后代不产任何输出),
31 HTML 标签/脚本永不泄漏进结果。feed 全程容错(标准库 parser 容错模式)。
32 """
33
34 def __init__(self, base_url: str = "") -> None:
35 super().__init__(convert_charrefs=True) # &amp; 等实体并入文本
36 self.base_url = base_url or ""
37 self.parts: list[str] = [] # 输出流(文本 + "\n" 行断标记)
38 self.skip: int = 0 # 丢弃块深度(skip>0 后代全跳)
39 self.links: list[dict] = [] # a 标签栈 {"href", "parts"}
40
41 # ---------------------------------------------------------- 工具方法
42 def _break(self) -> None:
43 """行断标记(连续块界折叠为单个)。"""
44 if not self.parts or self.parts[-1] != "\n":
45 self.parts.append("\n")
46
47 def _push_text(self, s: str) -> None:
48 """文本入流:a 内收进栈顶链接缓冲,否则直出(相邻块空隙去重空格)。"""
49 if not s:
50 return
51 if self.links:
52 self.links[-1]["parts"].append(s)
53 return
54 if (self.parts and self.parts[-1].endswith(" ")
55 and s.startswith(" ")):
56 s = s.lstrip(" ") # 前段尾空格 + 本段首空格去重
57 self.parts.append(s)
58
59 # ---------------------------------------------------------- parser 回调
60 def handle_starttag(self, tag: str, attrs: list) -> None:
61 tag = tag.lower()
62 if self.skip: # 丢弃块内:每开一标签深度 +1,
63 self.skip += 1 # 直至块闭合(后代整体跳过)
64 return
65 if tag in _SKIP_TAGS or _class_hit(attrs):
66 self.skip = 1 # 命中丢弃块:跳过其后代
67 return
68 if tag == "br": # 单行断(非块栈成员)
69 self._break()
70 return
71 if tag in _BLOCK_TAGS:
72 self._break() # 块级开:前导换行
73 level = _HEADING_LEVELS.get(tag)
74 if level: # 标题:写入 "#"*level + " "
75 self.parts.append("#" * level + " ")
76 return
77 if tag == "a":
78 href = _attr(attrs, "href")
79 self.links.append({"href": href or "", "parts": []})
80 return
81 if tag == "img": # 图片取 alt(spec: 图片取 alt)
82 alt = _attr(attrs, "alt")
83 if alt:
84 self._push_text(str(alt))
85 return
86 # 其余行内标签(span/b/strong/code…):忽略,文本由 handle_data 流入
87
88 def handle_endtag(self, tag: str) -> None:
89 tag = tag.lower()
90 if self.skip: # 丢弃块收口(每个闭合标签 -1)
91 self.skip = max(0, self.skip - 1)
92 return
93 if tag == "a" and self.links:
94 info = self.links.pop()
95 text = _WS_RE.sub(" ", "".join(info["parts"])).strip()
96 href = (info.get("href") or "").strip()
97 if text and href:
98 self._push_text(f"[{text}]({urljoin(self.base_url, href)})")
99 elif text: # 无 href 锚:仅保留文本
100 self._push_text(text)
101 return
102 if tag in _BLOCK_TAGS:
103 self._break() # 块级闭:尾随换行
104
105 def handle_data(self, data: str) -> None:
106 if self.skip:
107 return # 丢弃块内文本不产出
108 s = _WS_RE.sub(" ", data) # 空白折叠(HTML 空白 → 空格)
109 if s:
110 self._push_text(s)

112def _attr(attrs: list, name: str) -> Optional[str]:
113 """属性大小写不敏感取值(HTMLParser 属性名大小写行为跨版本不一,统一收敛)。"""
114 target = name.lower()
115 for k, v in attrs or ():
116 if k is not None and str(k).lower() == target:
117 return v
118 return None
119
120def _class_hit(attrs: list) -> bool:
121 """class/id 噪声黑名单命中(class/id 正则,spec 实现要点;re.I 偏离 6)。
122
123 命中 class="nav"/"ad-banner"/"footer x" 等即整块跳过(去导航/广告噪声)。
124 """
125 for attr_name in ("class", "id"):
126 v = _attr(attrs, attr_name)
127 if v and _NOISE_RE.search(str(v)):
128 return True
129 return False

131def html_to_markdown(html: str, base_url: str) -> str:
132 """正文提取转 Markdown(spec 伪码权威;标准库自含,禁第三方解析依赖)。
133
134 流式解析(不建 DOM):去 script/style/noscript/svg/head/nav/footer 与 class/id
135 噪声块,块级标签换行、标题加 #、链接转 [text](url)、图片取 alt、空白折叠;
136 输出为纯文本 Markdown,HTML 标签/脚本永不泄漏进结果。坏 HTML 容错不中断
137 (parser 默认容错,无异常上抛;feed 层兜底记 debug)。输出 ≤ _MAX_MD(2MB)。
138 """
139 if not isinstance(html, str):
140 html = str(html or "")
141 parser = _TextExtractor(str(base_url or ""))
142 try:
143 parser.feed(html)
144 parser.close()
145 except Exception as exc: # noqa: BLE001 坏 HTML 不中断
146 log.debug("html_to_markdown feed 异常(容错继续): %s", exc)
147 text = "".join(parser.parts)
148 lines = [ln.strip() for ln in text.splitlines()] # 逐行清理
149 md = "\n".join(ln for ln in lines if ln) # 去空行
150 return md[:_MAX_MD] # 防御性上限
