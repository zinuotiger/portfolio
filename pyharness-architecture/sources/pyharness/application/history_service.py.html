113 async def governance_audit(self, sid: str, *, decision_id: str = "",
114 since_seq: int = 0, reconcile: bool = False,
115 limit: int = 100) -> dict:
116 """治理审计查询面:把 ``governance.audit.AuditSystem`` 接成**产品入口**。
117
118 GAP-7 修复点:此前 ``AuditSystem`` 已构造并注入 ``GovernanceContext``,但
119 ``causal_chain`` / ``denied_report`` / ``reconcile`` / ``legacy_session_audit``
120 在 ``pyharness/`` 内**零调用点** ⇒ 治理数据在盘上、查询面不可达,桌面
121 "审计"页只能用旧遥测(类型计数,无决策因果)顶上。
122
123 读侧纪律(**replay-only**):每条结论都由**重放会话日志**现算得出,不读
124 任何内存缓存、不订阅事件 ⇒ 重启后同一日志得同一结果(可复现)。
125
126 返回:``decisions``(最近 N 条 decision.issued,倒序)·
127 ``denied``(被拒清单,含 ``executed`` 证据位:该 call_id 是否有配对
128 tool.result)· ``findings``(``reconcile=True`` 时的三类一致性对账)·
129 ``chain``(给定 ``decision_id`` 的因果链还原)。
130 """
131 log_, spine = await self.public_spine_for(sid)
132 audit = getattr(getattr(spine, "governance", None), "audit", None)
133 if audit is None:
134 raise_code("CYC-999", module="application.service",
135 field="governance.audit",
136 why="治理审计面未接线(AuditSystem 未注入 GovernanceContext)")
137 from pyharness.governance import verify_receipt
138 events = list(log_.events_after(0))
139 decisions = []
140 tenant = getattr(log_, "tenant_id", None)
141 for e in events:
142 if e.type != "decision.issued":
143 continue
144 p = getattr(e, "payload", None) or {}
145 decisions.append({
146 "decision_id": str(p.get("decision_id") or ""),
147 "verdict": str(p.get("verdict") or ""),
148 "tool": str(p.get("tool") or ""),
149 "guard_ids": list(p.get("guard_ids") or []),
150 "policy_refs": list(p.get("policy_refs") or []),
151 "principal_kind": p.get("principal_kind"),
152 "principal_id": p.get("principal_id"),
153 "principal_channel": p.get("principal_channel"),
154 "approval_ref": p.get("approval_ref"),
155 "supersedes": p.get("supersedes"),
156 "seq": getattr(e, "seq", None),
157 "call_id": call_id_of(e),
158 # 租户归属(GAP-10):取自**事件信封**(框架侧落盘的事实),
159 # 不是从服务实例的 self.tenant_id 反推 —— 后者是"当前请求"的租户,
160 # 前者是"该事件属于谁"的事实。
161 "tenant_id": getattr(e, "tenant_id", None),
162 })
163 n = max(1, int(limit or 100))
164 # 凭证核验面(离线可验):README 曾自陈 ``verify_receipt`` "未接 CLI/HTTP
165 # /工具外壳"—— 库级可达而产品不可达。此处接上:**从事件日志重建**全部凭证
166 # (``rebuild_from_log``,INV-G2/R2:不依赖任何内存缓存,重启后同样成立),
167 # 逐条重算哈希(篡改任一受保护字段即 False),再校验 prev_hash 链单调。
168 from pyharness.governance import verify_chain
169 from pyharness.governance.receipt import rebuild_from_log
170 all_rec = rebuild_from_log(log_)
171 receipts: dict = {
172 "count": len(all_rec),
173 "verified": sum(1 for r in all_rec if verify_receipt(r)),
174 "chain_valid": verify_chain(all_rec) if all_rec else True,
175 "items": [{"receipt_id": r.receipt_id,
176 "decision_id": r.decision_id, "kind": r.kind,
177 "verified": verify_receipt(r),
178 "approval_ref": r.approval_ref,
179 "content_hash": r.content_hash,
180 "prev_hash": r.prev_hash} for r in all_rec],
181 }
182 return {
183 "sid": str(sid), "ok": True,
184 "tenant_id": tenant,
185 "decisions": decisions[-n:][::-1],
186 "denied": audit.denied_report(since_seq=int(since_seq or 0)),
187 "findings": (await audit.reconcile()) if reconcile else [],
188 "chain": audit.causal_chain(str(decision_id)) if decision_id else None,
189 "receipts": receipts,
190 }

202 append-only 日志**现场重建**——故重启后同一日志得同一答案。``indexed`` 与

209 field="governance.evidence",
210 why="证据面未接线(EvidenceCollector 未注入 GovernanceContext)")

212 rebuilt = await EvidenceCollector.from_log(log_) # INV-E3:由日志重建
213 out: dict = {
214 "sid": str(sid), "ok": True, "task_id": str(task_id),
215 "indexed": coll.evidence_count(),
216 "rebuilt": rebuilt.evidence_count(),
217 "artifacts": [
218 {"evidence_id": str((getattr(e, "payload", None) or {})
219 .get("evidence_id") or ""),
220 "claim": str((getattr(e, "payload", None) or {}).get("claim") or ""),
221 "refs": list((getattr(e, "payload", None) or {}).get("refs") or []),

228 {"evidence_id": ev.evidence_id, "claim": ev.claim,
229 "refs": [{"kind": r.kind, "locator": r.locator}
230 for r in ev.refs]}
231 for ev in rebuilt.collect_for_task(str(task_id))]
