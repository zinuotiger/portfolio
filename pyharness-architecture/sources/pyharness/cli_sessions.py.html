86def _repairable_unhealthy(hits: list, sessions_dir: Path) -> Optional[str]:
87 """从 auto_scan 结果取首个**可修**的损坏会话;**跳过被其他进程持有的活会话**。
88
89 R14-2 / R14-7:索引是 200ms **攒批**落的派生视图 ⇒ 活会话在攒批窗口内必然
90 ``view_last != last`` 而被判 ``index_stale``。这类会话**既不是损坏**、本进程也
91 **修不动**(``repair_session`` 取锁即 PERS-202)⇒ 必须跳过;否则调用方要么
92 启动中止(启动自检,R14-2),要么整条命令以 PERS-202 失败、**永远够不到真正损坏
93 的会话**(``repair``,R14-7 —— 修复前的实测:只有活会话时不报"无损坏会话"而直接
94 抛 PERS-202)。
95
96 **唯一候选筛选点**:自检与 repair 子命令都必须走这里。
97 """
98 from pyharness.persistence import session_lock_held, session_lock_path
99 for h in hits:
100 if getattr(h, "healthy", False):
101 continue
102 sid = getattr(h, "sid", None)
103 if not sid:
104 continue
105 if session_lock_held(session_lock_path(sessions_dir / f"{sid}.jsonl")):
106 log.info("repair 候选:会话正被其他进程持有,跳过 sid=%s", sid)
107 continue
108 return sid
109 return None
