1"""pyharness/core/plugin_loader.py — 插件目录装载器 + 热重载(#50/#51/#52 Consumer 面)。
2
3契约(单文件插件):cfg.plugins.dir/<pid>/plugin.py,模块内声明:
4 MANIFEST = {"id","version","api_version":"1","requires":[]}
5 async def register_tools(registry) -> list[str] # 可选:桥接 core ToolRegistry
6PluginManager(bus Registry)管五态生命周期 + plugin.installed/uninstalled 留痕;
7register_tools 把可执行工具注册进 engine 的 core ToolRegistry(executor 消费面)
8——两套注册表语义桥(总线索引 vs 执行契约)。

11(importlib 重新 exec,sys.modules 驱逐)→ install/activate/register;loader 级
12热更(进程内模块级),非全框架热插拔(诚实口径)。

31def discover_plugins(plugins_dir: str | Path) -> list[Path]:
32 """扫描插件目录:含 plugin.py 的子目录即一个插件包。"""
33 root = Path(plugins_dir).expanduser()
34 if not root.exists():
35 return []
36 return [d for d in sorted(root.iterdir())
37 if d.is_dir() and (d / "plugin.py").exists()]

40def load_spec(pkg_dir: Path) -> SimpleNamespace:
41 """装载单插件模块(独立命名空间,防 sys.modules 污染主命名)。
42
43 mod_name 带 nonce:同秒重写文件时不命中 __pycache__/sys.modules 旧缓存
44 (HMR 确定性——代码热更必须真加载新字节码)。"""
45 pid = pkg_dir.name
46 py = pkg_dir / "plugin.py"
47 # pyc 按源路径缓存(mtime 同秒会命中旧字节码,nonce 名绕不过)→ 装载前清缓存
48 pycache = pkg_dir / "__pycache__"
49 if pycache.exists():
50 import shutil
51 shutil.rmtree(pycache, ignore_errors=True)
52 nonce = f"{id(pkg_dir):x}{__import__('time').time_ns() & 0xffff:x}"
53 mod_name = f"{_MODULE_NS}.{pid}.{nonce}"
54 loaded = False
55 try:
56 try:
57 spec = importlib.util.spec_from_file_location(mod_name, py)
58 if spec is None or spec.loader is None:
59 raise_code("BUS-003", pid=pid, detail="plugin.py 不可装载(spec 为空)")
60 mod = importlib.util.module_from_spec(spec)
61 sys.modules[mod_name] = mod
62 spec.loader.exec_module(mod)
63 except PyHError:
64 raise
65 except Exception as e: # noqa: BLE001 模块级错误定位
66 raise_code("BUS-003", pid=pid,
67 detail=f"plugin.py 导入失败:{type(e).__name__}:{e}")
68 manifest = dict(getattr(mod, "MANIFEST", {}) or {})
69 manifest_id = str(manifest.get("id") or pid)
70 if manifest_id != pid:
71 raise_code("BUS-003", pid=pid, manifest_id=manifest_id,
72 detail="MANIFEST.id 必须与插件目录名一致")
73 manifest["id"] = pid
74 manifest.setdefault("version", "0.1.0")
75 manifest.setdefault("api_version", "1")
76 loaded = True
77 return SimpleNamespace(id=pid, module=mod, manifest=manifest)
78 finally:
79 if not loaded:
80 sys.modules.pop(mod_name, None)

87async def load_plugin(manager: Any, spec: SimpleNamespace, ctx: Any,
88 tool_reg: Any, *, state: dict) -> None:
89 """安装 + 激活 + 桥接工具(幂等前提:同 pid 已存在则先卸载)。"""
90 pid = spec.id
91 prev = state.get(pid)
92 if prev is not None:
93 await unload_plugin(manager, spec, ctx, tool_reg, state=state)
94 names: list[str] = []
95 before_names = set(tool_reg.names()) if callable(getattr(tool_reg, "names", None)) else set()
96 installed = False
97 active = False
98 try:
99 await manager.install(spec.manifest, ctx)
100 installed = True
101 await manager.activate(pid, ctx)
102 active = True
103 fn = _tool_reg_fn(spec)
104 if callable(fn):
105 ret = fn(tool_reg)
106 if inspect.isawaitable(ret):
107 ret = await ret
108 names = list(ret or [])
109 state[pid] = {"spec": spec, "tool_names": names,
110 "path": str(spec.module.__file__ or "")}
111 except Exception:
112 after_names = set(tool_reg.names()) if callable(getattr(tool_reg, "names", None)) else set()
113 for name in set(names) | (after_names - before_names):
114 try:
115 tool_reg.unregister_definition(name)
116 except Exception: # noqa: BLE001 回滚尽力
117 log.debug("plugin rollback unregister miss %s", name)
118 try:
119 if active:
120 await manager.deactivate(pid, ctx)
121 if installed:
122 await manager.uninstall(pid, ctx)
123 except Exception: # noqa: BLE001 不掩盖原始错误
124 log.warning("plugin rollback cleanup failed pid=%s", pid, exc_info=True)
125 state.pop(pid, None)
126 raise

129async def unload_plugin(manager: Any, spec: SimpleNamespace, ctx: Any,
130 tool_reg: Any, *, state: dict) -> None:
131 """热卸载(工具注销 → deactivate → uninstall → 模块驱逐)。"""
132 pid = spec.id
133 rec = state.pop(pid, None)
134 if rec is not None:
135 for name in rec.get("tool_names", ()):
136 try:
137 tool_reg.unregister_definition(name)
138 except Exception: # noqa: BLE001 注销尽力
139 log.debug("plugin tool unregister miss %s", name)
140 try:
141 st = manager.state(pid)
142 if st in ("active", "inactive"):
143 if st == "active":
144 await manager.deactivate(pid, ctx)
145 await manager.uninstall(pid, ctx)
146 except Exception: # noqa: BLE001 热卸尽力
147 log.warning("plugin unload cleanup failed pid=%s", pid)
148 _evict(spec)

151async def reload_plugin(manager: Any, spec: SimpleNamespace, ctx: Any,
152 tool_reg: Any, *, state: dict,
153 pkg_dir: Path) -> None:
154 """HMR:卸旧载新(importlib 重新 exec,进程内模块级热更)。"""
155 await unload_plugin(manager, spec, ctx, tool_reg, state=state)
156 fresh = load_spec(pkg_dir) # 新代码重新装载
157 await load_plugin(manager, fresh, ctx, tool_reg, state=state)

160def _evict(spec: SimpleNamespace) -> None:
161 """sys.modules 驱逐(防 reload 拿到旧模块缓存;nonce 名一并摘除)。"""
162 for name in (f"{_MODULE_NS}.{spec.id}",
163 getattr(spec.module, "__name__", "")):
164 if name:
165 sys.modules.pop(name, None)
