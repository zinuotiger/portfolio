90 for key in ('node_id','attempt_id','execution_id'):

92 for node in self.nodes:
93 p=node.prerequisites

95 p.project_revision!=self.project_revision):raise ValueError('graph node authority scope')
96 if node.environment.boundary=='host_unrestricted' and not self.selection.confirm_host_boundary:raise ValueError('graph host confirmation missing')
97 if total_allocation(self.nodes)!=self.allocated_budget:raise ValueError('graph allocation sum')
