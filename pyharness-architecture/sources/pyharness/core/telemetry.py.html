30def session_audit(session: Any) -> dict:
31 """事件日志 → 审计报告(纯只读派生)。"""
32 events = list(session.events_after(0))
33 counts: dict[str, int] = {}
34 usage = {"requests": 0, "in_tokens": 0, "out_tokens": 0}
35 tools: dict[str, int] = {}
36 guards = {"rejected": 0, "evaluated": 0}
37 approvals: dict[str, int] = {}
38 errors: list[str] = []
39 first_ts = last_ts = None
40 for ev in events:
41 counts[ev.type] = counts.get(ev.type, 0) + 1
42 ts = getattr(ev, "ts", None)
43 if ts is not None:
44 first_ts = first_ts or ts
45 last_ts = ts
46 p = ev.payload or {}
47 if ev.type == "llm.usage":
48 usage["requests"] += 1
49 usage["in_tokens"] += int(p.get("in_tokens") or 0)
50 usage["out_tokens"] += int(p.get("out_tokens") or 0)
51 elif ev.type == "tool.result":
52 tools[p.get("name", "?")] = tools.get(p.get("name", "?"), 0) + 1
53 elif ev.type == "guard.rejected":
54 guards["rejected"] += 1
55 elif ev.type == "guard.evaluated":
56 guards["evaluated"] += 1
57 elif ev.type.startswith("approval."):
58 key = ev.type.split(".", 1)[1]
59 approvals[key] = approvals.get(key, 0) + 1
60 elif ev.type == "system.error":
61 errors.append(_brief(p.get("hint") or p.get("code") or ""))
62 return {
63 "session_id": getattr(session, "sid", ""),
64 "seq_max": events[-1].seq if events else 0,
65 "event_types": counts,
66 "llm_usage": usage,
67 "tools": tools,
68 "guards": guards,
69 "approvals": approvals,
70 "errors": errors[:10],
71 "span": {"first_ts": first_ts, "last_ts": last_ts},
72 }

81def session_summary_lines(session: Any) -> list[str]:
82 """人读摘要行(CLI stats/桌面审计卡)。"""
83 a = session_audit(session)
84 u = a["llm_usage"]
85 lines = [

88 ]
89 if a["tools"]:
90 lines.append("工具调用: " + ", ".join(
91 f"{k}×{v}" for k, v in sorted(a["tools"].items())))
92 if a["approvals"]:
93 lines.append("审批: " + ", ".join(
94 f"{k}×{v}" for k, v in sorted(a["approvals"].items())))
95 if a["errors"]:
96 lines.append("错误: " + "; ".join(a["errors"][:3]))
97 return lines
