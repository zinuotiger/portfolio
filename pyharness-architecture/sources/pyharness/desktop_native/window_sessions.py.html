143 def _render_chat(self) -> None:
144 parts = ["<html><body style='font-family:Segoe UI;font-size:13px'>"]
145 for msg in self._messages:
146 role = msg.get("role", "system")
147 content = html.escape(str(msg.get("content") or ""))
148 if role == "user":
149 parts.append(f"<p><b>你</b><br>{content}</p><hr>")
150 elif role == "assistant":
151 parts.append(f"<p><b>Agent</b><br>{content}</p><hr>")
152 else:
153 parts.append(f"<p style='color:#777'><b>{role}</b> {content}</p>")
154 if self._stream_text:
155 parts.append(f"<p><b>Agent(streaming)</b><br>{html.escape(self._stream_text)}</p>")
156 parts.append("</body></html>")
157 self.chat.setHtml("".join(parts))
158 self.chat.verticalScrollBar().setValue(self.chat.verticalScrollBar().maximum())
