107def _cfg_get(ctx: Any, dotted: str, default: Any) -> Any:
108 """读 ctx 配置点分键(属性链与 dict 访问双兼容;未接线/缺键回落默认)。
109
110 2026-09-21 R11-7 修:此前**只认 `ctx.cfg`**,而装配层设的是 **`ctx.config`**
111 (`create_agent`)⇒ 本模块**所有配置阈值恒回落模块默认**:`loop.content.
112 file_spill_bytes`(读阈值)与 `security.policy.read_extra_dirs`(**只读例外目录**)
113 都不生效 —— 后者是安全相关面:运维配了例外目录仍会被拒(方向 fail-closed,但
114 文档承诺的授权能力不可达)。同族三处(session_query/spill/tool_web)一直兼容
115 `("cfg","config")` 两名字,**仅本处漏了一处判据**(同类第五次)。
116 """
117 cfg = None
118 for name in ("cfg", "config", "settings"):
119 cfg = getattr(ctx, name, None)
120 if cfg is not None:
121 break
122 if cfg is None:
123 log.debug("ctx 配置面未接线,配置键 %s 回落默认 %r", dotted, default)
124 return default
125 cur: Any = cfg
126 for part in dotted.split("."):
127 if isinstance(cur, dict):
128 if part not in cur:
129 return default
130 cur = cur[part]
131 else:
132 nxt = getattr(cur, part, None)
133 if nxt is None:
134 return default
135 cur = nxt
136 return cur if cur is not None else default

165 """workspace 唯一根(偏离 3:ctx.scope.workspace → policy.workspace_root 回落)。

184def _under(p: Path, root: Path) -> bool:
185 """p 是否几何落在 root 内(含根本身;normcase 后带分隔符边界前缀比较)。
186
187 同 guard._inside_workspace 语义(大小写/分隔符归一,Windows 同源比较)。
188 """
189 rs = os.path.normcase(str(root))
190 ps = os.path.normcase(str(p))
191 if ps == rs:
192 return True
193 return ps.startswith(rs.rstrip("\\/") + os.sep)

208def resolve_in_workspace(raw: str, ctx: Any, *, writable: bool = False) -> Path:
209 """路径单点归一(第二道几何闸,F034/F055 语义;模块内私有不注册为工具)。
210
211 三道拒(与管道 g3 同码同判,防 TOCTOU 与内层直调,INV-04):
212 POL-FS-1 绝对路径词法越界(不在 workspace 内);
213 POL-FS-2 规范化后含 ".." 逃逸(相对路径归一后越界);
214 POL-FS-3 symlink/junction 最终解析越界(writable=True 无只读例外;
215 writable=False 允许 security.policy.read_extra_dirs 只读例外)。
216 失败一律 GRD-401(reason=POL-FS-x)结构化上抛,零副作用。
217 """
218 root = _workspace_root(ctx)
219 p = Path(str(raw)).expanduser()
220 if p.is_absolute():
221 # POL-FS-1:绝对路径越界 —— **只读例外目录**(security.policy.read_extra_dirs)
222 # 命中且本调用为只读时放行(A2/R24:此前该例外在 provider 层也不可达 ⇒ 配了仍拒)。
223 if not _under(p, root) and not (not writable
224 and _in_extra_read(p, ctx)):
225 raise_code("GRD-401", reason="POL-FS-1", path=str(raw),
226 advice="只允许 workspace 内路径(绝对路径越界;"
227 "只读例外目录未命中)")
228 cand = p
229 else:
230 cand = root / p # 相对路径锚定 workspace 根
231 norm = Path(os.path.normpath(cand))
232 if not _under(norm, root) and not (not writable
233 and _in_extra_read(norm, ctx)):
234 raise_code("GRD-401", reason="POL-FS-2", path=str(raw),
235 advice="路径含 .. 越界(只读例外目录未命中)")
236 try:
237 real = norm.resolve() # 解开 symlink/junction
238 except OSError:
239 raise_code("GRD-401", reason="POL-FS-3", path=str(raw),
240 advice="符号链接解析失败,按越界拒绝(无法安全归一)")
241 inside = _under(real, root)
242 if not inside and not (not writable and _in_extra_read(real, ctx)):
243 raise_code("GRD-401", reason="POL-FS-3", path=str(raw),
244 advice="符号链接指向 workspace 外(只读例外目录亦未命中)")
245 return real

307async def read_file(args: dict, ctx: Any) -> dict:
308 """fs.read_file(F034):读 workspace 内文本文件(UTF-8 自动 BOM)。
309
310 流程:第二道几何闸 → 凭据纵深 → stat 判存在/目录/大小(>10MB spill 上限
311 PERS-223 拒)→ 整读(≤10MB)→ 二进制探测拒读 TLB-806 → utf-8-sig 解码;
312 内容 ≤64KB(loop.content.file_spill_bytes)全量返回;>64KB 经 ctx.storage.
313 spill.put 落盘只返回摘要+引用,全文不进上下文/历史(输出有界 T-09)。
314 """
315 p = resolve_in_workspace(args["path"], ctx, writable=False)
316 _reject_credential(p, ctx) # g4 同码兜底(POL-CRED-1)
317 try:
318 st = p.stat()
319 except FileNotFoundError:
320 raise_code("TLB-802", path=str(p),
321 advice="文件不存在,先 list_dir 自查路径拼写")
322 except OSError as e:
323 raise_code("TLB-805", path=str(p), cause=e,
324 advice="文件不可访问(权限/IO),请重试")
325 if stat_mod.S_ISDIR(st.st_mode):
326 raise_code("TLB-805", path=str(p),
327 advice="目标是目录,本工具只读文件,先 list_dir 侦查")
328 if st.st_size > _spill_max_bytes(ctx): # 预检:超 spill 单文件上限
329 raise_code("PERS-223", path=str(p), size=st.st_size,
330 advice="文件超过 spill 上限,换 F052 分块或缩小目标")
331 try:
332 data = p.read_bytes() # ≤10MB 才整读
333 except OSError as e:
334 raise_code("TLB-805", path=str(p), cause=e,
335 advice="文件读取失败(权限/IO),请重试")
336 if _looks_binary(data):
337 raise_code("TLB-806", path=str(p),
338 advice="二进制文件,不能用文本读取(换专门工具)")
339 try:
340 text = data.decode("utf-8-sig") # 自动 BOM
341 except UnicodeDecodeError:
342 raise_code("TLB-806", path=str(p), # 偏离 4:非 UTF-8 拒读
343 advice="非 UTF-8 文本(解码失败),不能用文本读取")
344 if len(data) <= _read_spill_bytes(ctx): # 小文件全量返回
345 return {"content": _redact(ctx, text), "truncated": False,
346 "bytes": len(data), "path": str(p)}
347 redacted = _redact(ctx, text) # 出口脱敏后再落盘(INV-09)
348 meta = await _spill_put(ctx, redacted, kind="read")
349 return {"content": str(meta.get("preview", ""))[:500],
350 "truncated": True, "spill_ref": meta,
351 "bytes": len(data), "path": str(p)} # 大文件只给摘要+引用

354# =============================================================== fs.write_file
355async def write_file(args: dict, ctx: Any) -> dict:
356 """fs.write_file(F035):新建/追加文本到 workspace 内,自动建目录,原子写。
357
358 原子写:同目录临时文件写毕 → os.replace 原子替换(崩溃不留半文件);
359 失败清临时文件后原样上抛(executor 收敛 TLB-805)。覆写审批由管道 g7
360 在 Provider 前完成(工具内不二判、不降级);content >1MB 已在契约层
361 (schema maxLength)拒,零执行(TLB-803,INV-06)。
362 """
363 p = resolve_in_workspace(args["path"], ctx, writable=True) # 写:只认 workspace
364 content = args["content"] # 契约已验 ≤1MB
365 if args.get("mode", "write") == "append" and p.exists():
366 try:
367 if p.stat().st_size > _APPEND_MAX_BYTES:
368 raise_code("TLB-805", path=str(p),
369 advice=f"追加目标超过 {_APPEND_MAX_BYTES} 字节上限"
370 "(整读+重写代价过大);改用 write 或分段")
371 content = p.read_text(encoding="utf-8") + content # append=复制+追加
372 except (OSError, UnicodeDecodeError) as e:
373 raise_code("TLB-805", path=str(p), cause=e,
374 advice="追加读取原文件失败(权限/IO/编码非 UTF-8),请重试")
375 try:
376 p.parent.mkdir(parents=True, exist_ok=True) # 自动建目录
377 except OSError as e:
378 raise_code("TLB-805", path=str(p.parent), cause=e,
379 advice="父目录创建失败(权限/IO),请重试")
380 tmp = p.with_name(p.name + f".tmp-{uuid.uuid4().hex[:8]}") # 同目录临时文件
381 try:
382 tmp.write_text(content, encoding="utf-8", newline="") # 防换行改写
383 os.replace(tmp, p) # 原子替换
384 except BaseException:
385 tmp.unlink(missing_ok=True) # 失败清临时文件
386 raise
387 return {"bytes": len(content.encode("utf-8")), "path": str(p)}

390# =============================================================== fs.list_dir
391async def list_dir(args: dict, ctx: Any) -> dict:
392 """fs.list_dir(F036):列 workspace 内目录条目(名称/类型/大小/修改时间)。
393
394 隐藏文件默认隐藏(all=True 显示);glob 为名称前缀过滤(伪码 startswith,偏离 8);
395 recursive=1 层递归(偏离 7);条目 >500 截断留痕 truncated=True(偏离 6),
396 防巨目录打爆上下文(输出有界 T-09)。
397 """
398 p = resolve_in_workspace(args.get("path") or ".", ctx, writable=False)
399 if not p.is_dir():
400 raise_code("TLB-802", path=str(p),
401 advice="目录不存在或路径不是目录,先 list_dir 父目录侦查")
402 show_all = bool(args.get("all"))
403 recursive = bool(args.get("recursive"))
404 gpat = args.get("glob")
405
406 def _visible(name: str) -> bool:
407 """隐藏过滤(点开头即隐藏;all=True 显示)。"""
408 return show_all or not name.startswith(".")
409
410 cands: list[Path] = []
411 try:
412 top = [c for c in p.iterdir() if _visible(c.name)]
413 cands = list(top)
414 if recursive: # 1 层递归:仅直接子目录的条目
415 for d in top:
416 if d.is_dir():
417 cands.extend(c for c in d.iterdir() if _visible(c.name))
418 except OSError as e:
419 raise_code("TLB-805", path=str(p), cause=e,
420 advice="目录列举失败(权限/IO),请重试")
421 cands.sort(key=lambda c: c.name) # 稳定排序(确定性输出)
422
423 entries: list[dict] = []
424 truncated = False
425 for child in cands:
426 if gpat and not child.name.startswith(gpat):
427 continue # 通配过滤(前缀匹配,偏离 8)
428 if len(entries) >= LIST_MAX_ENTRIES: # >500 截断留痕(偏离 6)
429 truncated = True
430 break
431 try:
432 st = child.stat()
433 except FileNotFoundError:
434 continue # 竞态消失/断链:单条跳过(偏离 9)
435 except OSError as e:
436 raise_code("TLB-805", path=str(p), cause=e,
437 advice="目录条目读取失败(权限/IO),请重试")
438 entries.append({
439 "name": child.name,
440 "type": "dir" if child.is_dir() else "file",
441 "size": st.st_size if child.is_file() else None, # size 仅文件
442 "modified": datetime.fromtimestamp(st.st_mtime)
443 .isoformat(timespec="seconds"), # ISO 8601 mtime
444 })
445 return {"entries": entries, "truncated": truncated}

448# ============================================================= fs.delete_file
449async def delete_file(args: dict, ctx: Any) -> dict:
450 """fs.delete_file(critical 锚点,PRD §6.3/T-SEC)。
451
452 注册为 danger=critical——管道 g-danger 对 critical 直接拒绝、不可审批
453 (POL-DGR-1),正常路径 Provider 恒零执行(审计样本:"拦了且没执行")。
454 本函数体是真实删除实现,仅策略被显式下调或测试直调 Provider 时才可达;
455 即便下调也守第二道几何闸(writable=True 只认 workspace 根)。
456 """
457 p = resolve_in_workspace(args["path"], ctx, writable=True)
458 if not p.exists():
459 raise_code("TLB-802", path=str(p),
460 advice="文件不存在,先 list_dir 确认")
461 if p.is_dir():
462 raise_code("TLB-805", path=str(p),
463 advice="本工具只删文件;目录删除不在工具面")
464 try:
465 p.unlink() # 真实副作用仅策略下调后发生
466 except OSError as e:
467 raise_code("TLB-805", path=str(p), cause=e,
468 advice="删除失败(权限/IO),请重试")
469 return {"deleted": str(p)}
