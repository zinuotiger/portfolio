113def _os_lock_fd(fd: int) -> None:
114 """对 fd 取 OS 级**非阻塞**独占锁;已被占 → OSError。Windows msvcrt / POSIX fcntl。
115
116 锁独立于 JSONL 追加句柄(锁文件另开),故 store 内 close/reopen 句柄(轮转/截断
117 重写)不会丢锁。msvcrt 锁首字节,fcntl 锁整文件。
118 """
119 if os.name == "nt":
120 import msvcrt
121 os.lseek(fd, 0, os.SEEK_SET)
122 msvcrt.locking(fd, msvcrt.LK_NBLCK, 1) # 锁首字节;被占抛 OSError
123 else:
124 import fcntl
125 fcntl.flock(fd, fcntl.LOCK_EX | fcntl.LOCK_NB)

128def _os_unlock_fd(fd: int) -> None:
129 """释放 fd 上的 OS 级锁(尽力;失败不阻断 close)。"""
130 try:
131 if os.name == "nt":
132 import msvcrt
133 os.lseek(fd, 0, os.SEEK_SET)
134 msvcrt.locking(fd, msvcrt.LK_UNLCK, 1)
135 else:
136 import fcntl
137 fcntl.flock(fd, fcntl.LOCK_UN)
138 except OSError as e: # noqa: BLE001 释放失败不阻断
139 log.warning("PERS-202 域:文件锁释放失败 why=%s", e)

142class _FileLock:
143 """会话日志跨进程独占锁(INV-07 单写者)。
144
145 在 lock_path 上取 OS 级非阻塞独占锁,防两进程同写一 JSONL 造成 seq 重复 /
146 轮转 os.replace 互覆 / 重写时他进程活句柄写的字节静默消失。锁文件独立于 JSONL
147 且**不删除**(留 0 字节 .lock 属标准做法;删锁反有 release↔unlink 竞态)。
148 """
149
150 def __init__(self, lock_path: Path) -> None:
151 self._path = lock_path
152 self._fd: Optional[int] = None
153
154 def try_acquire(self) -> bool:
155 """非阻塞取锁;成功 True。冲突/IO 失败 → False(不抛,由调用方转 PERS-202)。"""
156 try:
157 self._path.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
158 fd = os.open(str(self._path), os.O_RDWR | os.O_CREAT, 0o600)
159 except OSError:
160 return False
161 try:
162 _os_lock_fd(fd)
163 except OSError:
164 os.close(fd)
165 return False
166 self._fd = fd
167 return True
168
169 def release(self) -> None:
170 """释放锁(幂等);不删锁文件。"""
171 if self._fd is None:
172 return
173 try:
174 _os_unlock_fd(self._fd)
175 finally:
176 fd, self._fd = self._fd, None
177 os.close(fd)

186def acquire_session_lock(lock_path: Path) -> None:
187 """取会话日志独占锁(进程内引用计数可重入);跨进程冲突 → PERS-202。
188
189 调用方须在 finally 里配对 release_session_lock(lock_path)。
190 """
191 with _LOCKS_MUTEX:
192 key = str(lock_path)
193 ent = _LOCKS.get(key)
194 if ent is not None:
195 ent[1] += 1
196 return
197 lock = _FileLock(lock_path)
198 if not lock.try_acquire():
199 raise_code("PERS-202", op="lock", path=str(lock_path),
200 advice="另一进程正在写/修复该会话;先结束该进程再打开")
201 _LOCKS[key] = [lock, 1]
202

204def release_session_lock(lock_path: Path) -> None:
205 """释放会话日志独占锁(引用计数归零才真正解锁;幂等)。"""
206 with _LOCKS_MUTEX:
207 key = str(lock_path)
208 ent = _LOCKS.get(key)
209 if ent is None:
210 return
211 ent[1] -= 1
212 if ent[1] <= 0:
213 del _LOCKS[key]
214 ent[0].release()

217def session_lock_held(lock_path: Path) -> bool:
218 """该会话是否**正被其他持有者占用**(只探测,不改语义;R14-2)。
219
220 启动自检不应把**活会话**当作可修对象:被别的进程持有的会话既不该被报为"待修",
221 本进程也修不动(``repair_session`` 取锁即 PERS-202)。此前自检会把"索引攒批窗口内
222 的正常会话"报为损坏 → ``_ensure_repair`` 撞 PERS-202 并**上抛** ⇒ 第二个 CLI 启动
223 直接失败(两进程实测复现)。
224
225 语义:同进程已持有(``_LOCKS`` 命中)→ True;锁文件不存在 → False(**不**新建);
226 跨进程 ``try_acquire`` 成功 → 立即归还并返回 False。纯探测,不留副作用。
227
228 **残余**(如实标注):探测在被占用时会**失败取锁**(不持有,无窗口);只有在会话
229 **空闲**时才会短暂持有再归还 —— 该微秒级窗口内若有他进程正好 ``open_store`` 同一
230 会话,会收到一次 PERS-202(可重试)。量级远小于它修掉的"第二个 CLI 启动即失败",
231 且与既有取锁语义同源(无新的锁类型)。
232 """
233 if str(lock_path) in _LOCKS:
234 return True
235 if not lock_path.exists():
236 return False # 从未取过锁:未被占用(不新建锁文件)
237 lock = _FileLock(lock_path)
238 if lock.try_acquire():
239 lock.release() # 探针成功:立即归还(锁文件不删,见 _FileLock)
240 return False
241 return True

368 修复策略并存**」。R24 依此把本方法由「第二套流水线」改为**薄适配器**（此前两套
369 并存：备份命名不同、隔离表示不同、派生视图重建缺失）。
370
371 安全委托序（**关键**）：本对象正持有自己的**追加句柄**，而 `repair_session` 会
372 整体重写主文件（它自己以 `ctx.session` 守卫**拒绝**活会话修复，正是防句柄失
373 同步）⇒ 委托前必须 flush + **关本句柄**，委托后**重开**：
374
375 ① `flush()`（攒批落盘）→ ② 关 `_fh` → ③ 委托 `repair_session`（同进程
376 会话锁可重入，不冲突）→ ④ 重开追加句柄 + 按事实重建 `_torn_tail`
377 → ⑤ 解除暂停（写通道已随修复恢复）。
378
379 返回**本模块的** `RepairReport`（`fixed`/`quarantined: list[int]`/`backup_path`），
380 由权威报告投影而来，保持既有的声明返回类型不变。
381 """
382 sid = session_id or self.session_id
383 path = self.path
384 if not path.exists():
385 return RepairReport() # 空会话:空报告(新会话)
386 await self.flush() # ① 攒批先落盘
387 close = getattr(self._fh, "close", None)
388 if callable(close):
389 close() # Do not rewrite while ownership of the old handle is uncertain.
390 from pyharness.repair import repair_session # 惰性:避免模块级回边
391 ctx = SimpleNamespace(
392 storage=SimpleNamespace(sessions_dir=path.parent))
393 primary = None
394 try:
395 rep = await repair_session(ctx, sid, interactive=False)
396 except BaseException as exc:
397 primary = exc
398 raise
399 finally:
400 self._invalidate_scan()
401 try:
402 self._fh = open(path, "a", encoding="utf-8", newline="\n")
403 self._torn_tail = detect_truncation(path) is not None
404 if primary is None: self._suspended = False
405 except BaseException as exc:
406 if primary is None: raise
407 primary.add_note(f'repair reopen failed: {type(exc).__name__}')
408 kept = [int(getattr(e, "line_no", 0)) for e in
409 (getattr(rep, "quarantined", None) or [])]
410 self._quarantine.clear()
411 self._quarantine.update(kept)
412 return RepairReport(fixed=list(getattr(rep, "fixed", None) or []),
413 quarantined=sorted(self._quarantine),
414 backup_path=getattr(rep, "backup_path", None))

416 def _rewrite_without_tail(self, cut_offset: int) -> None:
417 """把主文件截至 cut_offset 的内容原子重写(临时文件 + fsync + rename)。
418
419 同目录 tmp → fsync → os.replace 原子替换 → 重开追加句柄;rename 原子性
420 保证任何时刻磁盘上要么旧完整文件要么新完整文件,杜绝半写。Windows 先关
421 句柄再 rename(持开句柄 replace 会 PermissionError),失败不留临时残留。
422 """
423 tmp = self.path.with_name(self.path.name + f".tmp-{os.getpid()}")
424 primary = None
425 cleanup_errors = []
426 try:
427 try:
428 self._fh.close()
429 with open(self.path, "rb") as src, open(tmp, "wb") as dst:
430 dst.write(src.read(cut_offset))
431 dst.flush()
432 os.fsync(dst.fileno())
433 os.replace(tmp, self.path)
434 except OSError as exc:
435 raise_code("PERS-202", op="rewrite_without_tail", why=str(exc),
436 advice="原子截断失败；保留原文件及故障证据")
437 except BaseException as exc:
438 primary = exc
439 raise
440 finally:
441 try: tmp.unlink(missing_ok=True)
442 except BaseException as exc: cleanup_errors.append(exc)
443 if self._fh.closed:
444 try: self._fh = open(self.path, "a", encoding="utf-8", newline="\n")
445 except BaseException as exc: cleanup_errors.append(exc)
446 self._invalidate_scan()
447 if primary is not None:
448 for exc in cleanup_errors: primary.add_note(f'cleanup failed: {exc!r}')
449 elif cleanup_errors:
450 raise BaseExceptionGroup('rewrite cleanup failed', cleanup_errors)

452 def _rotate(self) -> None:
453 """主文件 >50MB → 原子 rename 为 {sid}.{n}.jsonl → 开新主句柄。"""
454 self._fh.flush()
455 n = max((int(p.stem.rsplit('.', 1)[1]) for p in self._rotated_paths()), default=0) + 1
456 target = self.path.with_name(f"{self.session_id}.{n}.jsonl")
457 self._fh.close() # Windows:先关句柄再 rename
458 try:
459 os.replace(self.path, target) # 原子 rename(同文件系统)
460 except OSError:
461 self._fh = open(self.path, "a", encoding="utf-8", newline="\n")
462 raise # 主文件未动,恢复句柄
463 self._fh = open(self.path, "a", encoding="utf-8", newline="\n")
464 self._torn_tail = False # 新文件无残片(R13-2)
465 self._invalidate_scan()
466 log.info("jsonl rotated: %s (size_mb=%d)",
467 target, self.rotate_bytes // 1_048_576)

490def session_recorder(store: Any, session_id: str) -> Callable:
491 """总线 → 存储订阅工厂(**唯一实现**;engine 与桌面壳共用,禁各自实现)。
492
493 **sid 过滤是必须的**:多会话**共用一条总线**时,订阅者会收到**所有**会话的信封;
494 不过滤则 B 的事件会串写进 A 的 JSONL(实测:两个 ``session.created`` 落同一文件,
495 回放错序并让 repair 误判空洞/误隔离)。``store.append`` 侧只写字节、**不做** sid
496 校验,故过滤责任**只在此单点**(桌面壳曾各自实现过一份,engine 侧长期缺失 ——
497 2026-09-21 R8 补)。
498
499 另:瞬时类型(``llm.chunk`` 等以 dict 上总线)不进 append-only JSONL。
500 强同步清单**唯一真源** = ``events.vocab.SYNC_TYPES``(ADR-019 P-3),不本地维护副本。
501 """
502 async def _record(type_: str, payload: Any) -> None:
503 if not hasattr(payload, "model_dump_json"):
504 return
505 if session_id and getattr(payload, "session_id", None) != session_id:
506 return # 跨会话事件:不落本会话文件
507 await store.append(payload, sync=type_ in SYNC_TYPES)
508 return _record

529def open_store(session_id: str, *, dir: Optional[Path] = None,
530 flush_interval_s: Optional[float] = None,
531 flush_batch: Optional[int] = None) -> SessionStore:
532 """创建/打开会话存储(工厂):目录 mkdir(700) → 主文件 append 句柄。
533
534 文件已存在(重启恢复):由 session.open_session 重放取 _seq 基线;检测到尾部
535 半行 → 告警提示先跑 repair(open 不自动截断——修复前强制备份,F060)。
536 异常:目录/文件不可写 → PERS-202。
537
538 ``flush_interval_s`` / ``flush_batch``(N1):**已解析的标量**,由**装配层**从
539 ``Settings`` 取值后传入(见 ``flush_kwargs_of``);缺省 ``None`` ⇒ 用模块常量
540 (DEFAULTS)。修复前本工厂**不接收也不传递**这两个值 ⇒ ``config.log.jsonl.*``
541 是**死配置**(实测:配置 3.0/7,store 实为 0.5/64)。
542 """
543 d = dir or default_sessions_dir()
544 try:
545 d.mkdir(parents=True, exist_ok=True, mode=0o700)
546 except OSError as e:
547 raise_code("PERS-202", op="mkdir", path=str(d), why=str(e),
548 advice="查磁盘/权限;repair 后恢复")
549 path = d / f"{session_id}.jsonl" # 会话日志锚点 ~/.pyharness/sessions
550 lock_path = session_lock_path(path) # 跨进程独占锁(INV-07 单写者)
551 try:
552 acquire_session_lock(lock_path) # 冲突 → PERS-202(不建第二写者)
553 except PyHError:
554 raise
555 except Exception as e: # noqa: BLE001 非预期锁故障:拒开
556 raise_code("PERS-202", op="lock", path=str(lock_path), why=str(e),
557 advice="会话锁获取异常;查磁盘/权限")
558 fh = None
559 try:
560 trunc = detect_truncation(path)
561 if trunc is not None:
562 log.warning("PERS-201:尾部半行待 repair file=%s offset=%s", path, trunc)
563 # newline="\n":JSONL 统一 \n 行尾,Windows 下禁 \r\n 翻译(半行检测按字节)
564 fh = open(path, "a", encoding="utf-8", newline="\n")
565 return SessionStore(session_id=session_id, path=path, fh=fh,
566 lock_path=lock_path, flush_interval_s=flush_interval_s,
567 flush_batch=flush_batch)
568 except BaseException as primary:
569 for cleanup in ([fh.close] if fh is not None else []) + [lambda: release_session_lock(lock_path)]:
570 try:
571 cleanup()
572 except BaseException as secondary:
573 primary.add_note(f'initialization cleanup failed: {type(secondary).__name__}')
574 if isinstance(primary, OSError):
575 raise_code('PERS-202', op='open', path=str(path), why=str(primary),
576 advice='查磁盘/权限;repair 后恢复')
577 raise
