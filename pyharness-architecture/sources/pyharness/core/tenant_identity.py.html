24def normalize_tenant_id(value: Any) -> str:
25 tenant = str(value or "default").strip()
26 if tenant in _RESERVED or not _TENANT_RE.fullmatch(tenant):
27 raise_code("EVT-100", field="tenant_id", tenant=tenant,
28 advice="tenant_id 仅允许 1-64 位字母/数字/._-")
29 return tenant

41def register_session_tenant(session_id: str, tenant_id: str) -> None:
42 """登记「会话 → 租户」进程内记忆(sid → **租户集合**;不覆盖、可多认领)。
43
44 2026-09-21 修(两轮):①此前**无条件覆写** ⇒ 跨租户 sid 撞名时后写者夺走归属;
45 ②改成"先写者胜"后仍不对 —— 进程内会话被删/服务重启时段落不清,旧认领会**挡住**
46 新租户的合法登记。会话目录本就**按租户分目录**(sid 只在租户内唯一),故正确模型是
47 **集合**:同一 sid 允许被多个租户认领,查询时"多认领 ⇒ 归属未知" → 由
48 ``tenant_for_session`` 返回空串,消费方按**未知**走 fail-closed(见
49 ``event_tenant_allowed``),绝不给一个**错误的确定答案**。
50 """
51 sid = str(session_id or "")
52 if not sid:
53 return
54 tid = normalize_tenant_id(tenant_id)
55 holders = _SESSION_TENANTS.setdefault(sid, set())
56 if holders and tid not in holders:
57 log.warning("会话 sid 被多租户认领 sid=%s 已有=%s 新增=%s"
58 "(按租户分目录 ⇒ sid 可跨租户撞名;归属转未知,fail-closed)",
59 sid, sorted(holders), tid)
60 holders.add(tid)

62def tenant_for_session(session_id: str) -> str:
63 """会话归属租户;**多认领(歧义)⇒ 返回 ""**(未知,由消费方 fail-closed)。"""
64 holders = _SESSION_TENANTS.get(str(session_id or ""))
65 if not holders or len(holders) != 1:
66 return ""
67 return next(iter(holders))

78def tenant_of_log(path: Any) -> str:
79 """从**落盘日志**取会话租户(GAP-10 的已持久化事实);取不到 → ``""``(未知)。
80
81 与 ``open_session`` 同口径:**最后一条有 ``tenant_id`` 的信封胜出**(日志即既成
82 事实,重启后仍可恢复)。只读**尾部窗口**(租户逐条盖在信封上,末条即可代表),
83 不解析 payload、**不抛**(不可读 → 未知,由调用方 fail-closed)。
84
85 L-1 闭合用:自此"未登记的磁盘会话"也能拿到**服务端权威**租户,不再只能回落
86 客户端自报头。
87 """
88 import json as _json
89 from pathlib import Path as _Path
90
91 p = _Path(str(path))
92 try:
93 size = p.stat().st_size
94 if size <= 0:
95 return ""
96 window = min(size, 65536)
97 with open(p, "rb") as fh:
98 if size > window:
99 fh.seek(size - window)
100 fh.readline() # 丢弃半行
101 raw = fh.read()
102 except OSError:
103 return ""
104 found = ""
105 for line in raw.split(b"\n"):
106 line = line.strip()
107 if not line:
108 continue
109 try:
110 env = _json.loads(line.decode("utf-8"))
111 except (ValueError, UnicodeDecodeError):
112 continue
113 t = str(env.get("tenant_id") or "")
114 if t:
115 found = t # 最后一条有值者胜出
116 if not found or found in _RESERVED:
117 return ""
118 try:
119 return normalize_tenant_id(found)
120 except Exception: # noqa: BLE001 非法租户名 → 视为未知
121 return ""

123def event_tenant_of(payload: Any, *, session_id: str = "") -> str:
124 """事件租户归属**唯一派生点**:优先**落盘信封** ``tenant_id``,回落进程内映射。
125
126 为什么优先信封:信封租户是 GAP-10 的**已持久化事实**(随 JSONL 落盘、可按日志
127 恢复),对"进程重启后映射为空""跨租户 sid 撞名"都免疫;进程内映射只是记忆。
128 """
129 env_tenant = str(getattr(payload, "tenant_id", "") or "")
130 if env_tenant:
131 return normalize_tenant_id(env_tenant)
132 sid = session_id or str(getattr(payload, "session_id", "") or "")
133 return tenant_for_session(sid) if sid else ""

135def event_tenant_allowed(client_tenant: str, event_tenant: str) -> bool:
136 """租户事件可见性**唯一判据**(两个外壳共用,禁止各自实现)。
137
138 规则(失败方向统一为**关闭**,与投影层既有语义一致):
139 1. 事件租户**已知**:相同 → 允许;不同 → 拒绝;
140 2. 事件租户**未知**(既无信封租户又无会话归属):仅 ``default`` 客户端可见。
141
142 2026-09-21 修:此前投影层(``EventStreamHub``)与原生壳(``_event_allowed``)
143 **各自实现**且失败方向**相反** —— 原生壳 ``not owner or owner == tenant`` 在
144 归属未知时**放行**,跨租户事件可达 UI;投影层 fail-closed。同 N5(通道解析
145 六层五种实现)的横切分裂,故收成单点判据。
146 """
147 client = normalize_tenant_id(client_tenant)
148 event = str(event_tenant or "")
149 if not event:
150 return client == "default"
151 return client == normalize_tenant_id(event)

153def unregister_session_tenant(session_id: str,
154 tenant_id: Optional[str] = None) -> None:
155 """撤销会话租户认领。
156
157 ``tenant_id`` 给定 ⇒ **只撤该租户的认领**(会话删除只需撤自己那份,别把别人的
158 认领一起抹掉 —— 多租户同名 sid 时那会让归属从"歧义"变成"错误确定");
159 缺省 ⇒ 撤全部认领(测试/整表清理用)。
160 """
161 sid = str(session_id or "")
162 if not sid:
163 return
164 if tenant_id is None:
165 _SESSION_TENANTS.pop(sid, None)
166 return
167 holders = _SESSION_TENANTS.get(sid)
168 if not holders:
169 return
170 holders.discard(normalize_tenant_id(tenant_id))
171 if not holders:
172 _SESSION_TENANTS.pop(sid, None)

174def resolve_tenant_secret(ref: str) -> str:
175 parts = str(ref or "").split(":", 2)
176 if len(parts) != 3 or parts[0] != "tenant":
177 raise_code("CRED-701", ref=ref, reason="tenant_ref")
178 tenant = normalize_tenant_id(parts[1])
179 store = _STORES.get(tenant)
180 if store is None:
181 raise_code("CRED-701", ref=ref, reason="tenant_store_unavailable",
