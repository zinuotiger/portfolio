98 async def stream(self, req: dict) -> AsyncIterator[Any]:
99 """SSE 流式(include_usage):逐行 data: 块 → raw chunk 命名空间;断流异常直传。"""
100 body = self._body(req)
101 self._check_send()
102 async with self._client.stream("POST", self._req_path, json=body) as resp:
103 self._response_metadata(resp)
104 if resp.status_code >= 400:
105 metadata = self._response_metadata(resp)
106 data = bytearray()
107 try:
108 async for chunk in resp.aiter_bytes():
109 if len(data) + len(chunk) > 16384:
110 data.clear()
111 break
112 data.extend(chunk)
113 body = json.loads(data)
114 from pyharness.core.llm_diagnostics import atom
115 if isinstance(body, dict) and isinstance(body.get('error'), dict):
116 metadata['provider_code'] = atom(body['error'].get('code'))
117 except (ValueError, httpx.TransportError):
118 pass
119 raise ProviderStatusError(resp.status_code, diagnostics=self._safe_diagnostics(metadata))
120 try:
121 done = False
122 async for line in resp.aiter_lines():
123 if not line.startswith("data:"):
124 continue
125 payload = line[5:].strip()
126 if not payload:
127 continue
128 if payload == "[DONE]":
129 done = True
130 break
131 try:
132 data = json.loads(payload)
133 except ValueError:
134 raise ProviderProtocolError("SSE 块非 JSON") from None
135 if not isinstance(data, dict):
136 raise ProviderProtocolError("Invalid stream response")
137 if 'error' in data:
138 from pyharness.core.llm_diagnostics import atom
139 exc = ProviderProtocolError("Provider stream error")
140 error = data['error']
141 exc.diagnostics = {'provider_code': atom(error.get('code')) if isinstance(error, dict) else 'unknown'}
142 raise exc
143 yield _to_ns(data)
144 if not done:
145 raise ProviderProtocolError("Incomplete stream")
146 except (httpx.TransportError, ProviderProtocolError) as exc:
147 metadata = self._response_metadata(resp)
148 metadata.update(getattr(exc, 'diagnostics', {}))
149 exc.diagnostics = self._safe_diagnostics(metadata)
