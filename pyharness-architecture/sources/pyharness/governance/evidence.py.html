1"""pyharness/governance/evidence.py — 证据工件与归档入口(S5-1)。
2
3一句话职责:把「支持某个 claim 的**引用集合** + 脱敏摘要」升格为一等对象,并以
4``evidence.archived`` 事件留痕。**核心纪律(INV-G4 / INV-E1):只存引用,绝不复制
5事件内容** —— 事件日志仍是唯一真源,Evidence 只是**索引**。
6
7依赖方向(ADR-018:308,强制):本模块只 import 标准库、``pyharness.errors`` 与本包;
8**禁止** ``pyharness.core.*`` / ``engine`` / ``bus`` / ``persistence``。会话经
9``ctx.session``/构造注入**鸭子类型**只读使用(与 ``receipt.py`` 同型)。
10
11**边界**:
12
13- **S5-1**:数据契约 + ``archive()``(唯一写点)。
14- **S5-2a**:``on_event()``(**只读**索引) + ``from_log()``(日志重建) +
15 ``collect_for_task()``(段锚聚合)——**索引只是派生缓存,可由事件日志完全重建**
16 (``from_log``),**不是事实源**(INV-E3);``on_event`` 体内**零 ``append``**。
17 订阅面 5 类:``evidence.archived`` / ``segment.start`` / ``segment.end`` /
18 ``decision.issued`` / ``receipt.emitted``;**不订阅** ``tool.result``(属工具执行
19 因果关系,由 S5-3 Audit 承担)。
20- **S5-2b**:``engine`` 装配 + 订阅 + 注入 ``GovernanceContext.evidence``。
21- **S7**:``TraceabilityMatrix``。
22
23INV-E2(引用可解析):``archive()`` 在发射前校验每条 ``EvidenceRef.locator`` 指向
24**真实存在**的真源锚(seq / decision_id / receipt_id / segment);不可解析则
25``CYC-999`` fail-closed。``kind="test"`` 是**外部**引用(测试路径),不做运行时校验。
26"""

89def _coerce_refs(refs: Iterable[Any]) -> tuple[EvidenceRef, ...]:
90 """规范化引用集合:接受 ``EvidenceRef`` 或 ``{"kind","locator"}`` 字典。"""
91 out: list[EvidenceRef] = []
92 for r in refs or ():
93 if isinstance(r, EvidenceRef):
94 out.append(r)
95 elif isinstance(r, dict) and set(r) == {"kind", "locator"}:
96 out.append(EvidenceRef(kind=str(r["kind"]),
97 locator=str(r["locator"])))
98 else:
99 raise_code("CYC-999", module="governance.evidence", field="refs",
100 why="引用必须是 EvidenceRef 或 {kind, locator} 字典",
101 got=type(r).__name__)
102 return tuple(out)

105class EvidenceCollector:
106 """证据归档入口(S5-1)。
107
108 当前只提供 **``archive()``(唯一写点)**;``on_event`` / ``collect_for_task``
109 属 S5-2,``TraceabilityMatrix`` 属 S7。
110 """
111
112 def __init__(self, *, session: Any = None,
113 id_factory: Optional[Callable[[], str]] = None,
114 clock: Optional[Callable[[], str]] = None) -> None:
115 self._session = session
116 self._id_factory = id_factory or (lambda: uuid4().hex)
117 self._clock = clock or _utc_now
118 # ---- 派生缓存(S5-2a):**可由事件日志完全重建**,非事实源(INV-E3) ----
119 # ① 证据索引:evidence_id → (事件 seq, Evidence)
120 self._evidence: dict[str, tuple[int, Evidence]] = {}
121 # ② 段索引:task_id → [start_seq, end_seq](end 未闭合时为 None)
122 self._segments: dict[str, list[Optional[int]]] = {}
123 # ③ 锚 seq:decision_id / receipt_id → 其事件 seq(用于把引用解析到段)
124 self._anchor_seq: dict[str, int] = {}
125
126 # ------------------------------------------------------------ 校验
127 @staticmethod
128 def _resolve(session: Any, ref: EvidenceRef) -> bool:
129 """INV-E2:该引用能否指向**真实存在**的真源锚(只读,不写)。"""
130 if ref.kind == "test":
131 return True # 外部引用:不做运行时校验
132 events = _read_events(session)
133 if ref.kind == "seq":
134 tail = ref.locator.rsplit(":", 1)[-1]
135 try:
136 want = int(tail)
137 except ValueError:
138 return False
139 return any(int(getattr(e, "seq", -1)) == want for e in events)
140 if ref.kind == "decision_id":
141 return any(getattr(e, "type", None) == "decision.issued"
142 and str((getattr(e, "payload", None) or {})
143 .get("decision_id") or "") == ref.locator
144 for e in events)
145 if ref.kind == "receipt_id":
146 return any(getattr(e, "type", None) == "receipt.emitted"
147 and str((getattr(e, "payload", None) or {})
148 .get("receipt_id") or "") == ref.locator
149 for e in events)
150 if ref.kind == "segment":
151 task = ref.locator.rsplit(":", 1)[0]
152 return any(getattr(e, "type", None) == "segment.start"
153 and str((getattr(e, "payload", None) or {})
154 .get("task_id") or "") == task
155 for e in events)
156 return False

158 # ------------------------------------------------------------ 归档
159 async def archive(self, claim: str, refs: Iterable[Any], *,
160 ctx: Any = None, summary: str = "",
161 artifact_path: Optional[str] = None) -> Optional[Evidence]:
162 """归档一条证据 → ``append("evidence.archived")``(**本模块唯一写点**)。
163
164 - ``ctx`` 省略时用构造期注入的 ``session``;
165 - 无会话(纯内存/单测)⇒ 返回 ``None``(降级,不发射);
166 - 引用不可解析(INV-E2)⇒ ``CYC-999`` **fail-closed**;
167 - 事件为**普通攒批**(非强同步)——丢失可由既有事件**再派生**(INV-E3)。
168 """
169 if not isinstance(claim, str) or not claim:
170 raise_code("CYC-999", module="governance.evidence", field="claim",
171 why="claim 必须为非空字符串")
172 norm = _coerce_refs(refs)
173 for r in norm:
174 if r.kind not in REF_KINDS:
175 raise_code("CYC-999", module="governance.evidence",
176 field="refs.kind", got=str(r.kind),
177 why=f"引用种类必须是 {REF_KINDS} 之一")
178 if not isinstance(r.locator, str) or not r.locator:
179 raise_code("CYC-999", module="governance.evidence",
180 field="refs.locator", why="locator 必须为非空字符串")
181 sess = getattr(ctx, "session", None) if ctx is not None else self._session
182 if sess is None:
183 return None # 未接线:降级不发射
184 for r in norm:
185 if r.kind in _RESOLVABLE_KINDS and not self._resolve(sess, r):
186 raise_code("CYC-999", module="governance.evidence",
187 field="refs.locator", kind=r.kind,
188 locator=r.locator,
189 why="引用不可解析(INV-E2):locator 未指向真实存在的真源锚")
190 evidence_id = self._id_factory()
191 ts = self._clock()
192 payload = {"evidence_id": evidence_id, "claim": claim,
193 "refs": [{"kind": r.kind, "locator": r.locator}
194 for r in norm],
195 "artifact_path": artifact_path}
196 r = sess.append(EVENT_EVIDENCE_ARCHIVED, payload, actor="system",
197 trace=None)
198 if hasattr(r, "__await__"):
199 r = await r
200 return Evidence(evidence_id=evidence_id, claim=claim, refs=norm,
201 summary=summary, artifact_path=artifact_path, ts=ts)

203 # ------------------------------------------------------------ 只读索引
204 async def on_event(self, env: Any) -> None:
205 """**只读**消费一条事件,更新派生索引(S5-2a)。
206
207 约束:① 只消费不产生——**本方法体内零 ``append``**;② 索引仅是缓存,
208 可由日志 replay 完全重建(``from_log``);③ 未知类型静默忽略;④ 畸形
209 事件不得抛穿(总线的 EVT-103 隔离之外,**此处亦不抛**)。
210
211 订阅面(5 类):``evidence.archived``(证据索引) · ``segment.start`` /
212 ``segment.end``(段范围) · ``decision.issued`` / ``receipt.emitted``
213 (锚 → seq,用于把引用解析到段)。
214 **不订阅** ``tool.result``(属工具执行因果关系,由 S5-3 Audit 承担)。
215 """
216 try:
217 t = getattr(env, "type", None)
218 p = getattr(env, "payload", None) or {}
219 seq = int(getattr(env, "seq", 0) or 0)
220 if t == EVENT_EVIDENCE_ARCHIVED:
221 ev = self._evidence_of(p, seq)
222 if ev is not None:
223 self._evidence[ev.evidence_id] = (seq, ev)
224 elif t == "segment.start":
225 task = str(p.get("task_id") or "")
226 if task:
227 self._segments.setdefault(task, [seq, None])[0] = seq
228 elif t == "segment.end":
229 task = str(p.get("task_id") or "")
230 if task:
231 st = int(p.get("start_seq") or 0) or seq
232 self._segments.setdefault(task, [st, None])
233 self._segments[task][0] = st
234 self._segments[task][1] = seq # 闭区间右端 = end 事件 seq
235 elif t == "decision.issued":
236 did = str(p.get("decision_id") or "")
237 if did:
238 self._anchor_seq[f"decision_id:{did}"] = seq
239 elif t == "receipt.emitted":
240 rid = str(p.get("receipt_id") or "")
241 if rid:
242 self._anchor_seq[f"receipt_id:{rid}"] = seq
243 except Exception: # noqa: BLE001 只读索引:绝不抛穿
244 return

246 @classmethod
247 async def from_log(cls, session: Any, **kw: Any) -> "EvidenceCollector":
248 """**由事件日志重建** collector(INV-E3:索引可完全再派生)。"""
249 col = cls(session=session, **kw)
250 for env in _read_events(session):
251 await col.on_event(env)
252 return col

270 def _task_at(self, seq: int) -> Optional[str]:
271 """该事件 seq 落在哪个段内(闭区间;未闭合段**不得**向右无限延伸)。
272
273 段在时间轴上互不重叠且按 seq 递增开合,故判据 = **起点 ≤ seq 的最近一段**
274 (未闭合段的右界由此天然止于下一段起点之前)。
275
276 2026-09-21 R14-10:此前把未闭合段(崩溃现场:进程被杀 → ``_run_task`` 的
277 ``finally`` 未执行 → 无 ``segment.end``)的右界当作**无穷**,又按插入序遍历
278 ``_segments`` ⇒ 该段**吞掉其后所有任务的证据**(实测:未闭合的 taskA 之后,
279 taskB 的证据被算到 taskA 上,``collect_for_task("taskB")`` 返回空)。属**恢复
280 一致性**缺陷:崩溃重启后证据按任务查错。
281 """
282 best_task: Optional[str] = None
283 best_start = -1
284 for task, (s, e) in self._segments.items():
285 if s is None:
286 continue
287 s = int(s)
288 if s > seq or (e is not None and seq > int(e)):
289 continue
290 if s > best_start: # 起点最近者胜(段不重叠)
291 best_task, best_start = task, s
292 return best_task

294 def _tasks_of(self, ev: Evidence) -> set[str]:
295 """该证据归属的 task 集合(按引用解析;无法解析的引用被忽略)。"""
296 out: set[str] = set()
297 for r in ev.refs:
298 if r.kind == "segment":
299 task = r.locator.rsplit(":", 1)[0]
300 if task:
301 out.add(task)
302 continue
303 if r.kind == "seq":
304 tail = r.locator.rsplit(":", 1)[-1]
305 try:
306 t = self._task_at(int(tail))
307 except ValueError:
308 t = None
309 else: # decision_id / receipt_id
310 anchor = self._anchor_seq.get(f"{r.kind}:{r.locator}")
311 t = self._task_at(anchor) if anchor else None
312 if t:
313 out.add(t)
314 return out

316 def collect_for_task(self, task_id: str) -> tuple[Evidence, ...]:
317 """按**段锚**聚合该任务的证据(S5-2a)。
318
319 来源**只有** ``evidence.archived`` 事件(D-1(a));返回按事件 seq 升序的
320 不可变元组;未知/空 ``task_id`` 或无线索 ⇒ ``()``(**不抛**)。
321 """
322 if not task_id:
323 return ()
324 hits = [(seq, ev) for seq, ev in self._evidence.values()
325 if task_id in self._tasks_of(ev)]
326 hits.sort(key=lambda it: it[0])
327 return tuple(ev for _seq, ev in hits)
