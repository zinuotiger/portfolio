18@dataclass(frozen=True)
19class ErrorSpec:
20 """错误码注册表条目:code/name/advice 必填 + 三轴分类 (ERR §1.3)。"""
21
22 code: str
23 name: str
24 advice: str
25 retryable: str = "N" # R=自动退避 / F=修复后重试 / N=不可重试
26 action: str = "拒写" # 拒写|回喂|降级|暂停|终态|隔离
27 fatal: bool = False

34def register(code: str, name: str, advice: str, *,
35 retryable: str = "N", action: str = "拒写", fatal: bool = False) -> None:
36 """登记单码;重复登记拒绝 (码一经发布不改含义,禁覆盖)。"""
37 if code in ERRORS:
38 raise ValueError(f"错误码重复登记:{code}")
39 ERRORS[code] = ErrorSpec(code, name, advice, retryable, action, fatal)

107def raise_code(code: str, **ctx: Any) -> NoReturn:
108 """全系统唯一抛出入口:查表→按码域抛对应域类;未登记码兜底 CYC-999。
109
110 双落纪律:每次 raise_code ≥1 行日志 (code=xxx key=value)。
111 ctx 只放脱敏结构化现场;secret 原文永不出凭据模块。
112 """
113 spec = ERRORS.get(code)
114 if spec is None:
115 spec = ERRORS.get("CYC-999", ErrorSpec("CYC-999", "未知内部错误", "查日志归因"))
116 ctx = {**ctx, "unknown_code": code}
117 code = "CYC-999"
118 log.error("code=%s name=%s %s", code, spec.name,
119 " ".join(f"{k}={v}" for k, v in ctx.items()))
120 raise _domain_class(code)(code, ctx=ctx, spec=spec)

123# ---------------------------------------------------------------- PyHError
124class PyHError(Exception):
125 """结构化错误基类:code/message/retryable/cause/context 五要素。
126
127 str() 只含 [code] name,不吐 ctx 敏感值。
128 """
129
130 def __init__(self, code: str, *, ctx: Optional[dict] = None,
131 spec: Optional[ErrorSpec] = None):
132 self.code = code
133 self.ctx = dict(ctx or {})
134 self.cause = self.ctx.pop("cause", None) # 底层异常仅本地,不外泄文案
135 self.spec = spec or ERRORS.get(code) or ERRORS.get(
136 "CYC-999", ErrorSpec("CYC-999", "未知内部错误", "查日志归因"))
137 self.retryable = self.spec.retryable
138 self.message = self.spec.name
139 super().__init__(f"[{self.code}] {self.message}")
140
141 def to_event(self) -> dict:
142 """事件化:system.error payload;堆栈永不进事件。"""
143 return {"type": "system.error", "actor": "system",
144 "payload": {"code": self.code,
145 "hint": f"{self.message}:{self.spec.advice}",
146 "ctx": self.ctx}}
147
148 def to_model_message(self) -> str:
149 """给 LLM 的可行动文本:≤2000 字符,含 [code]+name+建议。
150
151 建议的**取值优先级**(2026-09-21 修,O-2):
152 ``ctx["advice"]``(调用点现场最贴切的指引)→ ``ctx["hint"]``(调用点自查提示)
153 → ``spec.advice``(错误码表的"处置"列)。
154 此前恒用 ``spec.advice``,而该列是**运维处置话术**(如 TLB-802 的
155 "tool.error 回喂 LLM 自查,不静默"),对模型毫无可行动性,且把调用点已经
156 写好的正确指引(如"目录不存在…先 list_dir 父目录侦查")整条丢弃 ——
157 实测:``fs.list_dir`` 读不存在目录时 LLM 收到的是那句运维话术。
158 """
159 advice = str(self.ctx.get("advice") or self.ctx.get("hint")
160 or self.spec.advice)
161 msg = f"错误[{self.code}]:{self.message}。建议:{advice}"
162 tip = self.ctx.get("detail") or self.ctx.get("hint")
163 if tip and str(tip) != advice: # 已作建议的不重复进明细
164 msg += f"明细:{str(tip)[:200]}"
165 return msg[:2000]
166
167 def to_user_message(self) -> str:
168 """给用户/远端:只含 code+advice,绝不吐 ctx 明细。"""
169 return f"{self.code}:{self.spec.advice}"

204def wrap_unexpected(e: Exception, module: str) -> PyHError:
205 """非 PyHError 未预期异常兜底:归 CYC-999;堆栈仅本地 debug。"""
206 log.debug("unexpected error in %s:\n%s", module, traceback.format_exc())
207 return PyHError("CYC-999", ctx={"module": module,
208 "cause": e, "type": type(e).__name__})
