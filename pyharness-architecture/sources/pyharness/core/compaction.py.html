2
3接近窗口上限时把**早期完整任务段**折叠为摘要。物理日志仍只追加(INV-01 铁律):
4折叠区间原事件一行不删不改,只追加一条强同步 `context.compacted`
5(ranges/summary/tokens_before/tokens_after) 声明,折叠标记由 session 在吸收
6声明事件时同步入 `_folded` 索引——"遮蔽"与"声明"是同一动作,任何时刻可按
7ranges 展开日志核对原文(压缩但不撒谎)。

9模块职责(对应 spec §模块职责):
10 1. 触发判定 should_compact:派生历史 ≥75% 窗口 **且** 距上次压缩新增 ≥10 轮

13 segment.end 配对齐全、end_seq ≤ 保留锚),整段不切半(F044);含未完成
14 plan/goal/todo、待审批 approval、guard.rejected、已折叠区间的段整体剔除。
15 3. 压缩执行 compact/_apply:逐段独立 LLM 摘要(预算 400 token,失败重试 1 次
16 仍败 → _degrade_summary 规则降级,不阻塞对话)→ 强同步追加 context.compacted
17 (声明不落盘则回放误报空洞,EVENT-SCHEMA §8.1)。
18 4. run_if_needed:请求前门面(agent-loop 消费;非空闲边界不压不炸)。
19 5. kv_prefix_plan:两次压缩之间派生历史头部前缀稳定,供 llm 层标注 cache 命中区。

207 def should_compact(self, ctx: Any) -> bool:
208 """触发判定(F058 双条件;纯判定,不抛):
209 ① 派生历史 ≥75% 窗口;② 距上次压缩新增 ≥10 轮。缺一不触发。"""
210 window = self._window(ctx)
211 if window <= 0:
212 return False
213 hist_tokens = estimate_tokens(ctx.session.derive_history())
214 if hist_tokens < self._window_ratio * window: # ① 未到 75%
215 return False
216 if self._turns_since_last_compact(ctx) < self._min_new_rounds: # ② <10
217 return False
218 return True

229 def _segments_before(self, ctx: Any, anchor: int) -> list[dict]:
230 """重建完整任务段(F044):扫描 seq ≤ anchor 的事件,segment.start/end
231 按 task_id + start_seq 配对;仅收 start/end 均 ≤ anchor 的完整段。"""
232 if anchor <= 0:
233 return []
234 starts: dict[str, Envelope] = {}
235 segs: list[dict] = []
236 for ev in self._events(ctx):
237 if ev.seq > anchor:
238 break
239 if ev.type == "segment.start":
240 starts[ev.payload["task_id"]] = ev
241 elif ev.type == "segment.end":
242 start = starts.get(ev.payload["task_id"])
243 if start is not None and ev.seq <= anchor and \
244 start.seq == ev.payload.get("start_seq"):
245 segs.append({"task_id": start.payload["task_id"],
246 "start": start.seq, "end": ev.seq,
247 "start_ts": start.ts})
248 del starts[ev.payload["task_id"]]
249 return sorted(segs, key=lambda s: s["start"]) # seq 升序

251 def _protected(self, events: list[Envelope]) -> bool:
252 """不可折叠集命中判定(SECURITY.md §3.1 + F058):段内含以下任一内容
253 → 整段保护、不折叠(整段不切半):
254 ① 未完成 plan(proposed/exec.step 无终态 rejected/done/aborted)
255 / goal(未 done/abandoned)/ todo(末次更新仍有未 done 项)
256 ② 待审批 approval.requested 段内无裁决
257 ③ guard.rejected 拒绝记录(单调审计链,永不进摘要)
258 段边界假设:完整任务段内自启自结;跨段引用(他段 plan 在本段 exec)一律
259 保守保护(引用外部在跑计划 = 不可折叠)。"""
260 plan_closed: set[int] = set() # 已终结 plan_id(proposed 的 seq)
261 plans_open: set[int] = set() # 未终结 plan_id
262 goals_open: dict[str, str] = {} # goal_id → 状态(created/active/paused)
263 approvals_pending: set[int] = set()
264 todos_unfinished: bool = False
265 for ev in events:
266 t = ev.type
267 if t == "guard.rejected":
268 return True # ③ 单调审计链
269 if t == "approval.requested":
270 approvals_pending.add(ev.seq) # ② 待审批
271 elif t in ("approval.granted", "approval.denied",
272 "approval.timeout"):
273 approvals_pending.discard(ev.payload["approval_id"])
274 elif t == "plan.proposed":
275 plans_open.add(ev.seq) # ① plan 开启
276 elif t == "plan.exec.step":
277 pid = ev.payload["plan_id"] # 执行中(段内未见
278 if pid not in plan_closed: # 终态即未完成)
279 plans_open.add(pid)
280 elif t in ("plan.rejected", "plan.done", "plan.aborted"):
281 pid = ev.payload["plan_id"] # 终态:closed
282 plan_closed.add(pid)
283 plans_open.discard(pid)
284 elif t == "plan.approved":
285 plans_open.add(ev.payload["plan_id"]) # 批准 ≠ 完成
286 elif t == "goal.created":
287 goals_open[ev.payload["goal_id"]] = "created"
288 elif t == "goal.updated":
289 gid = ev.payload["goal_id"]
290 if ev.payload["status"] in ("done", "abandoned"):
291 goals_open.pop(gid, None) # 终态
292 else:
293 goals_open[gid] = ev.payload["status"] # 仍 active/paused
294 elif t == "goal.completed":
295 goals_open.pop(ev.payload["goal_id"], None)
296 elif t == "todo.updated":
297 todos = ev.payload.get("todos") or []
298 # payload 经 model_dump 归一:todo 项为 dict(兼容对象形态)
299 todos_unfinished = any(
300 not (it.get("done") if isinstance(it, dict)
301 else getattr(it, "done", False))
302 for it in todos) # 末次更新口径
303 if approvals_pending or plans_open or goals_open or todos_unfinished:
304 return True
305 return False

307 def candidates(self, ctx: Any) -> list[FoldCandidate]:
308 """折叠候选选择:最近 12 轮以前的完整任务段,命中不可折叠集/已折叠区间
309 的段整体剔除;按 seq 升序返回(纯读,不抛)。"""
310 anchor = self._recent_anchor_seq(ctx)
311 declared = self._declared_ranges(ctx)
312 out: list[FoldCandidate] = []
313 for seg in self._segments_before(ctx, anchor):
314 rng = (seg["start"], seg["end"])
315 events = list(ctx.session.events_between(*rng))
316 if self._protected(events): # 不可折叠集 → 整段排除
317 continue
318 if any(self._ranges_overlap(rng, d) for d in declared):
319 continue # 已折叠区间 → 防重复
320 out.append(FoldCandidate(task_id=seg["task_id"], range=rng,
321 start_ts=seg["start_ts"],
322 n_events=len(events)))
323 return out # 升序,不切半

358 prompt = ("请将以下任务段压缩为不超过 400 token 的中文摘要:保留用户意图、"
359 "最终结论、关键事实与目标/待办结果;省略工具原始输出与安全审计"
360 "细节(guard/审批记录一律不写)。\n" + text)
361 try:
362 summ = await self._call_summarize(ctx, prompt)
363 except PyHError:
364 summ = await self._call_summarize(ctx, prompt) # 重试 1 次
365 summary = (summ or "").strip()
366 if not summary:
367 raise_code("LLM-399", reason="empty-summary",
368 hint="摘要模型返回空文本;该段转规则降级")
369 return SegmentSummary(range=cand.range, summary=summary,
370 degraded=False, tokens=estimate_tokens(summary))
371
372 async def _apply(self, ctx: Any,
373 folds: list[SegmentSummary]) -> Optional[Envelope]:
374 """遮蔽 + 声明(幂等:空 folds → None):tokens 会计 → 强同步 append
375 context.compacted;遮蔽标记随声明事件被 session 吸收(入 _folded)。
376
377 异常:EVT-100(区间与既有折叠重叠 = 内部 bug,拒写防双摘要,调用前由
378 compact 预检);PERS-202(session.append 强同步落盘失败上抛,压缩视为
379 未发生,repair 后重试)。"""
380 if not folds:
381 return None
382 session = ctx.session
383 before = estimate_tokens(session.derive_history()) # 全量投影
384 raw_total = 0
385 for f in folds: # 区间原文投影 token(掩码口径)
386 evs = list(session.events_between(*f.range))
387 raw_total += estimate_tokens(_projection(evs))
388 ranges = [list(f.range) for f in folds]
389 summary = "\n".join(f.summary for f in folds)
390 # 摘要消息 = reducer 实际插桩形态(含 ranges 包装);after 为其净效果
391 added = estimate_tokens(f"[已压缩 {ranges}] {summary}") + _MSG_OVERHEAD
392 after = max(0, before - raw_total) + added
393 env = await session.append(
394 "context.compacted",
395 {"ranges": ranges, "summary": summary,
396 "tokens_before": before, "tokens_after": after},
397 actor="system", sync=True) # 强同步:声明不丢(§8.1)
398 log.info("session=%s compaction applied ranges=%s before=%d after=%d",
399 getattr(session, "sid", "?"), ranges, before, after)
400 return env

402 async def compact(self, ctx: Any) -> CompactReport:
403 """压缩主流程:候选逐段摘要(失败降级)→ 防重叠预检 → 强同步声明 →
404 报告;无可折叠候选 → noop 不写事件;全程不碰 JSONL 原文。
405
406 异常表:EVT-100(折叠区间与既有声明重叠 = candidates 应已过滤的内部
407 bug)、BUSY(会话 running 中直调)、PERS-202(强同步失败,压缩视为未发生)。"""
408 self._assert_boundary(ctx) # 仅请求边界可压(BUSY 防竞争)
409 cands = self.candidates(ctx)
410 if not cands:
411 return CompactReport(folds=[], tokens_before=0, tokens_after=0,
412 kept_recent_rounds=self._keep_recent_rounds,
413 reason="no-candidates", noop=True)
414 session = ctx.session
415 folds: list[SegmentSummary] = []
416 failures = 0
417 for cand in cands: # 逐段独立摘要(失败→降级不阻塞)
418 try:
419 folds.append(await self._summarize_segment(ctx, cand))
420 except PyHError:
421 failures += 1
422 evs = list(session.events_between(*cand.range))
423 txt, _ = self._degrade_summary(evs)
424 if not txt: # 空降级兜底:诚实占位,不伪造
425 txt = f"[任务段 {cand.range} 无可保留的决策类文本]"
426 folds.append(SegmentSummary(range=cand.range, summary=txt,
427 degraded=True,
428 tokens=estimate_tokens(txt)))
429 declared = self._declared_ranges(ctx) # 防双摘要预检(内部 bug 拒写)
430 for f in folds:
431 if any(self._ranges_overlap(f.range, d) for d in declared):
432 raise_code("EVT-100", type_="context.compacted",
433 reason="range-overlap",
434 detail=f"折叠区间 {f.range} 与既有声明重叠"
435 f"(candidates 应已过滤;内部 bug,拒写)")
436 env = await self._apply(ctx, folds) # 遮蔽+强同步声明
437 p = env.payload
438 return CompactReport(folds=folds,
439 tokens_before=p["tokens_before"],
440 tokens_after=p["tokens_after"],
441 kept_recent_rounds=self._keep_recent_rounds,
442 reason=f"window:{failures} degraded",
443 noop=False)

445 # ==================================================== ④ 前缀规划
446 def kv_prefix_plan(self, ctx: Any) -> PrefixPlan:
447 """KV 缓存前缀规划:稳定前缀终点 = 最新 compacted 之后(其后只有尾部
448 追加,前缀零变动 → LLM 服务端前缀缓存可整段复用)。只读派生计算。"""
449 events = self._events(ctx)
450 comps = [ev for ev in events if ev.type == "context.compacted"]
451 if not comps:
452 return PrefixPlan(stable_upto_seq=0, prefix_tokens=0,
453 cacheable=False, blocks=[])
454 comp = comps[-1] # 最新压缩声明
455 head = [ev for ev in events if ev.seq <= comp.seq]
456 blocks = [list(r) for r in comp.payload["ranges"]]
457 msgs = _projection(head) # 稳定前缀的派生消息形态
458 return PrefixPlan(stable_upto_seq=comp.seq,
459 prefix_tokens=estimate_tokens(msgs),
460 cacheable=True, blocks=blocks)

462 # ==================================================== 请求前门面
463 async def run_if_needed(self, ctx: Any) -> Optional[CompactReport]:
464 """请求前门面(DIS-CORE §5 消费点):agent-loop 每次 LLM 请求前调用——
465 非空闲边界/双条件未齐 → None(不压不炸);压缩成功且非 noop → 经 ctx.bus
466 广播 sysprompt.compacted(通知装配层重试,缓存已失效)。"""
467 if self._busy_state(ctx): # 仅请求边界可压(BUSY 防竞争)
468 return None
469 if not self.should_compact(ctx): # 双条件未齐 → 不压
470 return None
471 report = await self.compact(ctx) # 强同步声明后返回
472 if not report.noop:
473 self._signal(ctx, "sysprompt.compacted",
474 {"tokens_before": report.tokens_before,
475 "tokens_after": report.tokens_after})
476 return report

478 @staticmethod
479 def _signal(ctx: Any, type_: str, payload: dict) -> None:
480 """系统声明信号出口:ctx.bus(emit_sync 优先;异步则排程;失败日志降级)。"""
481 bus = getattr(ctx, "bus", None)
482 if bus is None:
483 log.debug("compaction 信号 %s 未投递:ctx.bus 未装配", type_)
484 return
485 sink = getattr(bus, "emit_sync", None) or getattr(bus, "emit", None)
486 if not callable(sink):
487 return
488 try:
489 result = sink(type_, payload)
490 if hasattr(result, "__await__"):
491 import asyncio
492 try:
493 asyncio.get_running_loop().create_task(result)
494 except RuntimeError: # 无运行中事件循环:日志降级
495 log.debug("compaction 信号 %s 未排程(无事件循环)", type_)
496 except Exception as exc: # noqa: BLE001 信号失败不阻断
497 log.warning("compaction 信号 %s 投递失败:%s", type_, exc)
