122 async def sync_knowledge(self, connection_id=None):
123 """Reconcile explicit sync-enabled connections. State survives restart.
124
125 No startup network dependency, hidden retries or manual-mode uploads.
126 Local mutation remains authoritative after a remote failure; the next
127 explicit sync/query retries the pending desired version idempotently.
128 """
129 data = self.store.read()
130 ident = connection_id or data['preferences'].get('knowledge_provider')
131 row = data['connections'].get(ident)
132 if not row or row['type'] != 'remote_rag' or row['config'].get('sync_mode','manual') != 'sync-enabled':
133 return {'status':'manual','operations':0}
134 if not row['config'].get('enabled'):
135 return {'status':'disabled','operations':0}
136 lock = self.locks.setdefault('knowledge-sync:'+ident, asyncio.Lock())
137 async with lock:
138 with self.store.transaction() as current:
139 sync = current['preferences'].setdefault('knowledge_sync', {}).setdefault(ident,
140 {'owner':uuid.uuid4().hex,'documents':{},'endpoint':row['config']['endpoint']})
141 if sync['endpoint'] != row['config']['endpoint']:
142 fail('knowledge_sync_endpoint_conflict')
143 desired = self.store.read()['knowledge']
144 provider = self.rag_provider(row)
145 completed = 0
146 try:
147 for source in sorted(set(desired) | set(sync['documents'])):
148 saved = sync['documents'].get(source)
149 document = desired.get(source)
150 if document and saved and saved['sha256'] == document['sha256'] and saved['version'] == document['version']:
151 continue
152 if completed >= 100:
153 fail('knowledge_sync_batch_pending')
154 path = '/platform-sources/'+sync['owner']+'/'+source
155 if document:
156 result = await provider.request('PUT',path,{k:document[k] for k in ('text','version','sha256')})
157 answer = result.get('data',{})
158 if (result.get('status') != 'ok' or answer.get('source_id') != source
159 or answer.get('version') != document['version'] or answer.get('sha256') != document['sha256']):
160 fail('knowledge_sync_invalid_response')
161 entry = {k:document[k] for k in ('version','sha256')}
162 else:
163 result = await provider.request('DELETE',path+'?version='+str(saved['version']))
164 if result.get('status') != 'ok' or result.get('data',{}).get('deleted') is not True:
165 fail('knowledge_sync_invalid_response')
166 entry = None
167 with self.store.transaction() as current:
168 records = current['preferences']['knowledge_sync'][ident]['documents']
169 if entry is None: records.pop(source, None)
170 else: records[source] = entry
171 current['preferences']['knowledge_sync'][ident]['last_error'] = None
172 completed += 1
173 except Exception as error:
174 from pyharness.core.knowledge import KnowledgeError
175 from pyharness.errors import PyHError
176 code = error.code if isinstance(error, KnowledgeError) else (
177 error.code if isinstance(error, PyHError) and error.code in (
178 'knowledge_sync_batch_pending','knowledge_sync_invalid_response') else 'knowledge_sync_failed')
179 with self.store.transaction() as current:
180 current['preferences']['knowledge_sync'][ident]['last_error'] = code
181 fail(code)
182 return {'status':'synced','operations':completed}
