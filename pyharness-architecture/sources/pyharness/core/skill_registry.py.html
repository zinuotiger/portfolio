29def _safe_name(name) -> str:
30 """技能名/版本段安全校验(CND-04:外部输入不得直接作资源路径依据)。
31
32 非空 + `_NAME_RE` + **拒 "."/".."**——`_NAME_RE` 字符集含 `.`,只靠它会放行
33 `name="."/".."` → `dir / ".."` 越界(remove 甚至 `rmtree` 父目录)。
34 """
35 nm = str(name or "")
36 if not _NAME_RE.fullmatch(nm) or nm in (".", ".."):
37 raise_code("EVT-100", field="name", name=nm[:64],
38 advice="Skill 名称/版本段非法(仅 [A-Za-z0-9._-] 且不得为 . 或 ..)")
39 return nm

42def _safe_under(root: Path, *segments: str) -> Path:
43 """`root/segments…` 归一后必须仍在 root 内(纵深防御,SECURITY §文件影响域单点归一)。
44
45 防 symlink/junction 越界(POL-FS-3)与拼接绕过;逃逸 → POL-FS-2 拒。
46 """
47 root_r = root.resolve()
48 target = root_r.joinpath(*segments).resolve()
49 if target != root_r and root_r not in target.parents:
50 # POL-FS-* 是 **guard 拒绝原因**(ERR.md §2.11),不是错误码;与 tool_fs
51 # ``resolve_in_workspace`` 同款:GRD-401 + reason=POL-FS-2。此前把原因当码用
52 # ⇒ ``raise_code`` 按未登记码**静默改写为 CYC-999**(R19):路径穿越的拒绝在
53 # 审计/HTTP 上退化成"未知内部错误",丢了安全语义。
54 raise_code("GRD-401", reason="POL-FS-2", path=str(target),
55 root=str(root_r),
56 advice="Skill 路径逃逸出根目录,拒绝(路径穿越)")
57 return target

78 async def _read_url(self, url: str, *, max_bytes: int) -> bytes:
79 parsed = urlparse(str(url or ""))
80 if parsed.scheme == "file":
81 path = Path(unquote(parsed.path.lstrip("/"))) if os.name == "nt" else Path(unquote(parsed.path))
82 try:
83 data = path.read_bytes()
84 except OSError as exc: # noqa: BLE001
85 raise_code("TLB-805", url=url, stage="file",
86 cause=exc, advice="本地 Skill 包无法读取")
87 if len(data) > max_bytes:
88 raise_code("TLB-805", url=url, stage="size",
89 advice=f"Skill 包超过上限 {max_bytes} bytes")
90 return data
91 if parsed.scheme not in ("http", "https"):
92 raise_code("CFG-601", field="url", url=url,
93 advice="Skill Registry 仅支持 http/https/file URL")
94 chunks: list[bytes] = []
95 size = 0
96 try:
97 async with httpx.AsyncClient(follow_redirects=True, timeout=30) as client:
98 async with client.stream("GET", url) as resp:
99 if resp.status_code >= 400:
100 raise_code("TLB-805", url=url, http_status=resp.status_code,
101 advice="Skill Registry 下载失败")
102 async for chunk in resp.aiter_bytes():
103 size += len(chunk)
104 if size > max_bytes:
105 raise_code("TLB-805", url=url, stage="size",
106 advice=f"下载超过上限 {max_bytes} bytes")
107 chunks.append(chunk)
108 except httpx.HTTPError as exc:
109 raise_code("TLB-805", url=url, stage="transport",
110 cause=exc, advice="Skill Registry 网络失败")
111 return b"".join(chunks)

113 async def fetch_registry(self, url: str) -> dict:
114 raw = await self._read_url(url, max_bytes=_MAX_REGISTRY_BYTES)
115 try:
116 data = json.loads(raw.decode("utf-8"))
117 except (UnicodeDecodeError, ValueError) as exc: # noqa: BLE001
118 raise_code("TLB-805", url=url, stage="registry",
119 cause=exc, advice="Skill Registry 不是合法 JSON")
120 if not isinstance(data, dict) or not isinstance(data.get("skills"), list):
121 raise_code("EVT-100", field="registry",
122 advice="Registry 必须包含 skills 数组")
123 return data

125 async def search(self, query: str, registry_url: str, *, limit: int = 20) -> list[dict]:
126 data = await self.fetch_registry(registry_url)
127 q = str(query or "").strip().lower()
128 rows: list[dict] = []
129 for skill in data.get("skills", []):
130 if not isinstance(skill, dict):
131 continue
132 name = str(skill.get("name") or "")
133 desc = str(skill.get("description") or "")
134 tags = " ".join(str(x) for x in (skill.get("tags") or []))
135 hay = f"{name} {desc} {tags}".lower()
136 if q and q not in hay:
137 continue
138 versions = skill.get("versions") or []
139 version_rows = [v for v in versions if isinstance(v, dict)
140 and v.get("version")]
141 version_ids = [str(v.get("version")) for v in version_rows]
142 latest = str(skill.get("latest") or
143 (version_ids[-1] if version_ids else ""))
144 latest_entry = next((v for v in reversed(version_rows)
145 if str(v.get("version")) == latest), {})
146 rows.append({"name": name, "description": desc,
147 "tags": list(skill.get("tags") or []),
148 "latest": latest,
149 "versions": version_ids,
150 "sha256": str(latest_entry.get("sha256") or ""),
151 "source": str(latest_entry.get("source") or
152 latest_entry.get("url") or registry_url)})
153 return rows[:max(1, int(limit))]

165 if not versions:

178 def _safe_extract(self, raw: bytes, target: Path) -> None:
179 try:
180 zf = zipfile.ZipFile(io.BytesIO(raw))
181 except zipfile.BadZipFile as exc: # noqa: BLE001
182 raise_code("TLB-805", stage="unzip", cause=exc,
183 advice="Skill 包不是合法 zip")
184 infos = zf.infolist()
185 if len(infos) > self.max_files:
186 raise_code("TLB-805", stage="unzip",
187 advice=f"Skill 包文件数超过上限 {self.max_files}")
188 total = 0
189 for info in infos:
190 name = info.filename.replace("\\", "/")
191 path = PurePosixPath(name)
192 if path.is_absolute() or ".." in path.parts:
193 raise_code("TLB-805", stage="path",
194 advice=f"Skill 包包含非法路径:{name}")
195 if info.flag_bits & 0x1:
196 raise_code("TLB-805", stage="encrypted",
197 advice="不支持加密 Skill 包")
198 total += int(info.file_size)
199 if total > self.max_package_bytes:
200 raise_code("TLB-805", stage="size",
201 advice=f"解压后大小超过上限 {self.max_package_bytes}")
202 target.mkdir(parents=True, exist_ok=True)
203 try:
204 zf.extractall(target)
205 except (OSError, zipfile.BadZipFile) as exc: # noqa: BLE001
206 raise_code("TLB-805", stage="unzip", cause=exc,
207 advice="Skill 包解压失败")
208 finally:
209 zf.close()

221 from pyharness.core.skill import _parse_frontmatter
222 text = (skill_root / "SKILL.md").read_text(encoding="utf-8")

229 def _replace_active(self, name: str, version_dir: Path, version: str) -> None:
230 active = self.skills_dir / name
231 backup = self.versions_dir / name / self._current_version(name)
232 if active.exists() and not backup.exists():
233 try:
234 shutil.copytree(active, backup)
235 except OSError: # noqa: BLE001
236 log.warning("active skill backup failed name=%s", name, exc_info=True)
237 tmp = self.skills_dir / f".{name}.installing-{os.getpid()}"
238 if tmp.exists():
239 shutil.rmtree(tmp)
240 shutil.copytree(version_dir, tmp)
241 (tmp / ".pyharness-version").write_text(version, encoding="utf-8")
242 if active.exists():
243 shutil.rmtree(active)
244 tmp.replace(active)

253 async def install(self, name: str, *, registry_url: str,
254 version: Optional[str] = None,
255 approved_by: Optional[str] = None) -> dict:
256 if not approved_by:
257 raise_code("GRD-401", reason="skill-install-approval",
258 advice="Skill 安装必须由人类显式批准")
259 name = _safe_name(name) # CND-04:外部 name 不得直接作路径依据
260 registry = await self.fetch_registry(registry_url)
261 entry = self._entry(registry, name)
262 selected = self._version_entry(entry, version)
263 package_url = str(selected.get("url") or "")
264 expected = str(selected.get("sha256") or "").lower()
265 if not package_url or not re.fullmatch(r"[0-9a-f]{64}", expected):
266 raise_code("EVT-100", skill=name, version=selected.get("version"),
267 advice="版本必须包含 url 与合法 sha256")
268 raw = await self._read_url(package_url, max_bytes=self.max_package_bytes)
269 actual = self._sha256(raw)
270 if actual != expected:
271 raise_code("GRD-401", reason="skill-hash-mismatch", skill=name,
272 expected=expected, actual=actual,
273 advice="SHA-256 不匹配，拒绝安装")
274 version_id = _safe_name(selected.get("version")) # CND-04:版本段亦外部输入
275 with tempfile.TemporaryDirectory(prefix="skill-", dir=self.quarantine_dir) as td:
276 extract_root = Path(td) / "extract"
277 self._safe_extract(raw, extract_root)
278 skill_root = self._find_skill_root(extract_root)
279 fm_name = self._frontmatter_name(skill_root)
280 if fm_name != name:
281 raise_code("EVT-100", skill=name, actual=fm_name,
282 advice="Registry name 与 SKILL.md frontmatter 不一致")
283 version_dir = _safe_under(self.versions_dir, name, version_id)
284 if version_dir.exists():
285 shutil.rmtree(version_dir)
286 version_dir.parent.mkdir(parents=True, exist_ok=True)
287 shutil.copytree(skill_root, version_dir)
288 self._replace_active(name, version_dir, version_id)
289 return {"ok": True, "name": name, "version": version_id,
290 "sha256": actual, "source": package_url,
291 "approved_by": approved_by, "path": str(self.skills_dir / name)}

293 def rollback(self, name: str, version: str, *, approved_by: Optional[str] = None) -> dict:
294 if not approved_by:
295 raise_code("GRD-401", reason="skill-rollback-approval",
296 advice="Skill 回滚必须由人类显式批准")
297 name = _safe_name(name) # CND-04:外部 name/version 不得直接作路径依据
298 version = _safe_name(version)
299 version_dir = _safe_under(self.versions_dir, name, version)
300 if not version_dir.is_dir():
301 raise_code("EVT-101", skill=name, version=version,
302 advice="目标版本不存在")
303 previous = self._current_version(name)
304 self._replace_active(name, version_dir, version)
305 return {"ok": True, "name": name, "from_version": previous,
306 "to_version": version, "approved_by": approved_by}

308 def remove(self, name: str, *, approved_by: Optional[str] = None) -> dict:
309 if not approved_by:
310 raise_code("GRD-401", reason="skill-remove-approval",
311 advice="Skill 卸载必须由人类显式批准")
