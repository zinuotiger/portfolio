1"""pyharness/governance/receipt.py — 决策凭证(S4/M4)。
2
3一句话职责:把一次**已经发生并持久化**的治理决策,升格为**可离线核验的凭证**
4——`DecisionReceipt` 是**证明对象**,不是新业务状态;持久化形态只有
5``receipt.emitted`` 事件(薄引用 + 完整性链),**事件日志仍是唯一真源**。
6
7依赖方向(ADR-018:308,强制):本模块只 import 标准库、``pyharness.errors`` 与本包;
8**禁止** ``pyharness.core.*`` / ``engine`` / ``bus`` / ``persistence``。需要会话时
9经 ``ctx`` 鸭子类型只读使用(与 ``context.authorize`` 同型)。
10
11关键裁定(2026-09-15 M4 Entry Analysis,R-1/R-2/R-3):
12
131. **R-1**:``receipt.emitted.payload.digest`` == ``Receipt.content_hash``(与相邻
14 ``prev_hash`` 同属完整性链)。``inputs_digest`` 语义**不变**(表达"本次授权针对
15 什么 inputs"),**不得**混用。不新增 payload 字段。
162. **R-2**:approval 凭证在 **D2 处**发射(``approval.granted`` 早于 D2;D1 无
17 ``approval_ref``)。链路:``approval.granted.approval_id → D2.approval_ref →
18 D2.decision_id → approval Receipt``。
193. **R-3**:``Receipt.ts`` = ``receipt.emitted`` 的 ``Envelope.ts``(凭证**生成**时刻);
20 与 ``Decision.ts``(决策**形成**时刻)语义不同,两者并存不混。
21
22**content_hash 确定性**(M4 硬约束):受保护集合**排除**随机 ``receipt_id``
23(否则同一事实每次 hash 不同)与 append 时才产生的 ``ts``/``event_seq``。
24规范化复用 ``policy.compute_fingerprint`` 同款 canonical JSON 规则
25(``sort_keys=True`` / ``ensure_ascii=False`` / ``separators=(",",":")`` /
26``default=str``)+ sha256——**不新建第三套 canonicalization/hash**。
27
28**prev_hash 链**:链头由 ``ReceiptStore`` 首次 emit 时**从会话日志 replay 推出**
29(只读使用唯一真源);内存 ``_last_hash`` 仅为缓存,重启可由日志重建 ⇒ 不构成第二
30persistence source of truth(INV-R5)。
31"""

52 """规范化 JSON(与 ``policy.compute_fingerprint`` 同款规则;不新建第二套)。"""
53 return json.dumps(payload, sort_keys=True, ensure_ascii=False,

61def receipt_kind_of(decision: Decision) -> Optional[str]:
62 """决策 → 凭证类别(M4 产生规则)。``None`` = **不产生凭证**。
63
64 - ``approval_ref is not None`` ⇒ ``approval``(D2:审批后重新授权,携带真实
65 approval identity ⇒ 闭合 INV-G2 链);
66 - ``verdict=APPROVAL`` 且 ``approval_ref is None`` ⇒ **D1(审批请求决策)**:
67 不产生凭证——它是"需要进一步审批"的治理中间态,不授权任何执行;
68 - 其余(ALLOW / REJECT)⇒ ``decision``。
69 """
70 if decision.approval_ref is not None:
71 return KIND_APPROVAL
72 if decision.verdict is Verdict.APPROVAL:
73 return None
74 return KIND_DECISION

107@dataclass(frozen=True)
108class DecisionReceipt:
109 """★ 可离线核验的裁决凭证(设计 §3.4)。
110
111 运行时对象**字段较丰富**;持久化的 ``receipt.emitted`` payload **只 5 字段**
112 (``receipt_id``/``decision_id``/``kind``/``digest``/``prev_hash``)——其余由
113 ``decision_id`` 指向的 ``decision.issued`` **派生**(INV-G4:只含引用与哈希)。
114 """
115
116 receipt_id: str
117 decision_id: str
118 kind: str
119 verdict: str
120 tool: str
121 inputs_digest: str
122 policy_fingerprint: str
123 principal: Optional[Principal]
124 approval_ref: Optional[int]
125 supersedes: Optional[str]
126 event_seq: Optional[int]
127 call_id: str
128 ts: str
129 content_hash: str
130 prev_hash: Optional[str] = None
131 signature: Optional[str] = None # v1.0 留空(设计 §3.4)
132
133 @property
134 def is_approval(self) -> bool:
135 return self.kind == KIND_APPROVAL
136
137 def protected(self) -> dict:
138 """本凭证的受保护集合(供 verify 重算哈希)。"""
139 p = self.principal
140 return {
141 "decision_id": str(self.decision_id),
142 "kind": str(self.kind),
143 "verdict": str(self.verdict),
144 "tool": str(self.tool),
145 "inputs_digest": str(self.inputs_digest),
146 "policy_fingerprint": str(self.policy_fingerprint),
147 "principal_kind": str(p.kind) if p is not None else "",
148 "principal_id": str(p.id) if p is not None else "",
149 "principal_channel": (p.channel if p is not None else None),
150 "approval_ref": self.approval_ref,
151 "supersedes": self.supersedes,
152 "prev_hash": self.prev_hash,
153 }
154
155 def to_payload(self) -> dict:
156 """持久化形态(冻结 5 字段;``digest`` == content_hash)。"""
157 return {"receipt_id": self.receipt_id, "decision_id": self.decision_id,
158 "kind": self.kind, "digest": self.content_hash,
159 "prev_hash": self.prev_hash}

162def _read_events(sess: Any) -> list:
163 """只读取出会话事件(鸭子类型)。
164
165 优先 ``events_after(0)``(``SessionLog``);回落 ``replay()``(``SessionStore``
166 等持久化面)。二者均为**唯一真源**的只读视图——本模块不 import 任何 core 模块
167 (ADR-018:308),也不缓存为第二真源。
168 """
169 for name in ("events_after", "replay"):
170 reader = getattr(sess, name, None)
171 if reader is None:
172 continue
173 try:
174 return list(reader(0) if name == "events_after" else reader())
175 except Exception: # noqa: BLE001 只读探测:尽力
176 return []
177 return []

184def verify_receipt(receipt: Any) -> bool:
185 """离线核验(**fail-closed**,ADR-018:318;INV-R4)。
186
187 返回 ``False`` ⇒ 调用方**必须**拒绝执行(沿用 INV-05 纪律)。检查:
188
189 1. 类型与 ``kind`` 合法;
190 2. ``content_hash`` 与受保护集合**重算值**一致(任一受保护字段被改 → False);
191 3. ``kind`` 与 ``approval_ref`` 语义自洽(approval↔非 None / decision↔None)。
192 """
193 if not isinstance(receipt, DecisionReceipt):
194 return False
195 if receipt.kind not in RECEIPT_KINDS:
196 return False
197 if not receipt.content_hash or not receipt.decision_id:
198 return False
199 expected = _sha256(_canonical(receipt.protected()))
200 if expected != receipt.content_hash:
201 return False
202 if receipt.kind == KIND_APPROVAL and receipt.approval_ref is None:
203 return False
204 if receipt.kind == KIND_DECISION and receipt.approval_ref is not None:
205 return False
206 return True

209class ReceiptStore:
210 """凭证的生成、持久化、查询、核验(M4)。
211
212 **不持有第二份持久化**:唯一写动作 = ``session.append("receipt.emitted", …)``;
213 索引与哈希链头均为**可由日志重建的缓存**(INV-R5)。
214 """
215
216 def __init__(self, *, id_factory: Optional[Callable[[], str]] = None,
217 clock: Optional[Callable[[], str]] = None) -> None:
218 self._id_factory = id_factory or (lambda: uuid4().hex)
219 self._clock = clock or _utc_now
220 self._index: dict[str, DecisionReceipt] = {}
221 self._last_hash: Optional[str] = None
222 self._chain_loaded = False
223
224 # ------------------------------------------------------------ 链头
225 def _load_chain(self, sess: Any) -> None:
226 """从会话日志 replay 出链头(只读;唯一真源=JSONL)。幂等,仅首次生效。"""
227 if self._chain_loaded:
228 return
229 self._chain_loaded = True
230 for e in _read_events(sess):
231 if getattr(e, "type", None) != EVENT_RECEIPT_EMITTED:
232 continue
233 d = (getattr(e, "payload", None) or {}).get("digest")
234 if d:
235 self._last_hash = str(d)
236
237 # ------------------------------------------------------------ 发射
238 async def emit(self, ctx: Any, decision: Decision, *,
239 call_id: str = "") -> Optional[DecisionReceipt]:
240 """按产生规则生成凭证并落盘(强同步)。``None`` = 该决策不产生凭证(D1)。
241
242 异常:落盘失败按 PERS-202 语义上抛(F-SYNC-1 修复后**不再 silent success**),
243 调用方 fail-closed。
244 """
245 kind = receipt_kind_of(decision)
246 if kind is None:
247 return None # D1:审批请求决策,无凭证
248 sess = getattr(ctx, "session", None)
249 if sess is None:
250 return None # 未接线(纯内存/单测):降级
251 self._load_chain(sess)
252 prev = self._last_hash
253 chash = content_hash_for(decision=decision, kind=kind, prev_hash=prev)
254 receipt_id = self._id_factory()
255 payload = {"receipt_id": receipt_id, "decision_id": decision.decision_id,
256 "kind": kind, "digest": chash, "prev_hash": prev}
257 trace = {"call_id": call_id} if call_id else None
258 r = sess.append(EVENT_RECEIPT_EMITTED, payload, actor="system",
259 sync=True, trace=trace)
260 if hasattr(r, "__await__"):
261 r = await r
262 receipt = DecisionReceipt(
263 receipt_id=receipt_id, decision_id=decision.decision_id, kind=kind,
264 verdict=str(decision.verdict), tool=decision.tool,
265 inputs_digest=decision.inputs_digest,
266 policy_fingerprint=decision.policy_fingerprint,
267 principal=decision.principal, approval_ref=decision.approval_ref,
268 supersedes=decision.supersedes,
269 event_seq=getattr(r, "seq", None), call_id=call_id,
270 ts=self._clock(), content_hash=chash, prev_hash=prev)
271 self._index[receipt_id] = receipt
272 self._last_hash = chash
273 return receipt
274
275 # ------------------------------------------------------------ 查询
276 def get(self, receipt_id: str) -> Optional[DecisionReceipt]:
277 return self._index.get(receipt_id)
278
279 def of_decision(self, decision_id: str) -> tuple[DecisionReceipt, ...]:
280 """该决策的全部凭证(**复数**——decision/approval 混合模型)。"""
281 return tuple(r for r in self._index.values()
282 if r.decision_id == decision_id)
283
284 def all(self) -> tuple[DecisionReceipt, ...]:
285 return tuple(self._index.values())

287 def verify(self, receipt: Any) -> bool:
288 """见模块级 ``verify_receipt``(fail-closed)。"""

292def rebuild_from_log(sess: Any) -> tuple[DecisionReceipt, ...]:
293 """从**事件真源**重建全部凭证(INV-G2/R2:可 replay、可核验)。
294
295 仅用 ``receipt.emitted``(定位 + 哈希链)与 ``decision.issued``(决策事实),
296 **不依赖任何内存缓存**;返回的凭证必须全部 ``verify_receipt`` 为真。
297 """
298 decisions: dict[str, Any] = {}
299 emitted: list[Any] = []
300 for e in _read_events(sess):
301 t = getattr(e, "type", None)
302 if t == "decision.issued":
303 d = getattr(e, "payload", None) or {}
304 did = d.get("decision_id")
305 if did:
306 decisions[str(did)] = (d, getattr(e, "trace", None) or {})
307 elif t == EVENT_RECEIPT_EMITTED:
308 emitted.append(e)
309 out: list[DecisionReceipt] = []
310 for e in emitted:
311 p = getattr(e, "payload", None) or {}
312 did = str(p.get("decision_id") or "")
313 d, trace = decisions.get(did, ({}, {}))
314 if not d:
315 continue # 无对应决策:跳过(不臆造)
316 kind = str(p.get("kind") or "")
317 pid = str(d.get("principal_id") or "")
318 pch = d.get("principal_channel")
319 principal = Principal.from_legacy_by(
320 f"{pch}:{pid}" if (pch and pch != pid) else (pid or "system"))
321 out.append(DecisionReceipt(
322 receipt_id=str(p.get("receipt_id") or ""), decision_id=did, kind=kind,
323 verdict=str(d.get("verdict") or ""), tool=str(d.get("tool") or ""),
324 inputs_digest=str(d.get("inputs_digest") or ""),
325 policy_fingerprint=str(d.get("policy_fingerprint") or ""),
326 principal=principal, approval_ref=d.get("approval_ref"),
327 supersedes=d.get("supersedes"),
328 event_seq=getattr(e, "seq", None),
329 call_id=str(trace.get("call_id") or ""),
330 ts=str(getattr(e, "ts", "") or ""),
331 content_hash=str(p.get("digest") or ""),
332 prev_hash=p.get("prev_hash")))
333 return tuple(out)

336def verify_chain(receipts: tuple[DecisionReceipt, ...]) -> bool:
337 """校验 ``prev_hash`` 链单调可验(INV-R8):首条创世为 ``None``,其余指向前一条。"""
338 prev: Optional[str] = None
339 for r in receipts:
340 if r.prev_hash != prev:
341 return False
342 if not verify_receipt(r):
343 return False
344 prev = r.content_hash
345 return True
