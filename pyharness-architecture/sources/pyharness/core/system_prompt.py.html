153def _escape_control(text: str) -> str:
154 """转义 C0 控制字符(防变量值伪造段分隔/角色标记的注入面)。
155
156 保留 \n \t;其余 C0 控制字符一律替换为空格——确定性替换(同输入同输出),

163def _count_text_tokens(text: str) -> int:
164 """纯文本 token 粗估(确定性):中文 ≈ 1.5 token/字,西文 ≈ 4 字符/token。
165
166 spec:token 估算器 est_tokens 内部实现,中文 ≈ 1.5 token/字粗估可配。
167 取整规则:1.5×中文 与 /4 西文均向上取整,保证 token 数 ≥ 1 且单调不减。
168 """
169 cjk = sum(1 for ch in text if "\u4e00" <= ch <= "\u9fff")
170 other = len(text) - cjk
171 cjk_tok = (cjk * 3 + 1) // 2 # ceil(cjk × 1.5)
172 other_tok = (other + 3) // 4 # ceil(other / 4)
173 return cjk_tok + other_tok

176def est_tokens(message: dict) -> int:
177 """单条消息 dict 的 token 粗估(窗口裁剪预算用)。
178
179 content 可为 str 或 list(tool_calls/多模态块):list 按紧凑 JSON 文本计;
180 每条消息 +_MSG_OVERHEAD 结构开销。纯确定性,无 IO。
181 """
182 content = message.get("content", "") if isinstance(message, dict) else message
183 if isinstance(content, (list, tuple)):
184 import json
185 text = json.dumps(list(content), ensure_ascii=False, sort_keys=True)
186 else:
187 text = str(content)
188 return _count_text_tokens(text) + _MSG_OVERHEAD

196def render_template(tpl: PromptTemplate, *, vars: dict) -> str:
197 """分段渲染(F010/DIS §5.3.2):按 parts 声明顺序拼装 text/var 段。
198
199 变量只收白名单表(scope/config 派生);缺变量/未知 kind 即渲染失败——确定性:
200 同输入同输出(可缓存/可测)。段间以 "\n\n" join。
201
202 异常:var 名不在变量表 → CFG-602(会话继续);part.kind 未知 → CFG-603(本次不发)。
203 """
204 if not isinstance(tpl, PromptTemplate):
205 raise_code("CFG-602", part="<template>",
206 advice="提示词装配缺少合法模板;查模板载入路径")
207 segs: list[str] = []
208 for part in tpl.parts:
209 if part.kind == "text": # 静态文本段
210 segs.append(_escape_control(part.body))
211 elif part.kind == "var": # 变量段(role/deny/domains…)
212 if part.name is None: # var 缺名 = 模板配置错误
213 raise_code("CFG-603", part="<unnamed-var>",
214 advice="模板变量段缺 name;修模板后重试")
215 val = vars.get(part.name) # 变量表之外的名字 = 未声明
216 if val is None: # 缺变量 → 渲染失败(不静默空串)
217 raise_code("CFG-602", part=part.name,
218 advice="提示词装配缺少变量;查 scope/config 派生变量表")
219 if str(val).strip() == "": # 空变量段不产空段(A4:turn_source
220 continue # 未设置时为 "",不该多出分隔)
221 segs.append(_escape_control(part.render(str(val)))) # 转义防段间注入
222 else: # 未知 kind = 模板配置错误
223 raise_code("CFG-603", part=part.name, kind=part.kind,
224 advice="模板段 kind 非法(仅 text/var);修模板后重试")
225 return "\n\n".join(segs) # 确定性:同输入同输出

228def build_guard_segment(scope: Any) -> Optional[str]:
229 """护栏段构建(F024/DIS §5.3.3):由 scope 数据生成,返回 None = 结构非法。
230
231 内容:禁止操作清单 / 外发域名 / 注入防御指令 / 危险操作提示 / 沙箱级别。
232 LLM 与工具结果不可改写本段(边界);scope 数据非法时返回 None,由 assemble
233 抛 CFG-603 拒绝本次 LLM 调用(缺护栏不发请求,安全优先)。
234 """
235 try:
236 policy = scope.policy # scope 策略唯一数据源
237 deny = sorted(policy.deny_tools) if policy.deny_tools else ["无"]
238 net = sorted(policy.allowed_domains) if policy.allowed_domains else []
239 level = policy.sandbox_level
240 lines = [
241 f"{GUARD_HEADER} 禁止操作:{', '.join(deny)}",
242 f"外发域名:{', '.join(net) if net else '无(禁止外发)'}",

285def truncate_history(hist: list[dict], budget: int) -> tuple[list, list]:
286 """窗口裁剪(DIS §5.3.3):从新到旧累计 token,旧消息先丢,返回 (kept, dropped)。
287
288 kept 还原时间序(保最近上下文完整);dropped = 被丢消息 seq 列表(从新到旧,
289 供上层落 sysprompt.truncated 留痕)。边界:单条消息 > budget 不切半条消息
290 (由 compaction hint 兜底 F058);budget 再小也至少保最新一条。
291 """
292 kept: list = []
293 dropped: list = []
294 acc = 0
295 for idx in range(len(hist) - 1, -1, -1): # 从最新往最旧扫(保近期)
296 m = hist[idx]
297 acc += est_tokens(m)
298 if acc > budget and kept: # 预算耗尽:更旧的全丢
299 dropped.append(m.get("seq", idx + 1)) # seq 优先,无 seq 用 1-based 位置
300 continue
301 kept.append(m)
302 kept.reverse() # 还原时间序
303 return kept, dropped

337class SystemPromptAssembler:
338 """系统提示词装配器(F010):模板 + ctx 派生变量 + 护栏 + 截窗历史 → messages。
339
340 用法(agent-loop,装配后挂 ctx.sysprompt):
341 prompt = ctx.sysprompt.assemble(hist, ctx=ctx) # ctx 门面鸭子注入
342
343 ctx 所需成员(均按需 getattr,装配缺位降级不炸):
344 scope: 策略/窗口数据源(必填;缺失 → CFG-603 拒请求)
345 session: title()/derived_tokens()/turns_since_last_compact()(可缺)
346 tools: schemas_for(scope)|schemas() 能力清单源(可缺 = 无可用工具)
347 bus: emit_sync/emit 事件出口(可缺 = 信号降级为日志)

349
350 def __init__(self, template: Optional[PromptTemplate] = None, *,
351 config: Any = None,
352 window_tokens: Optional[int] = None,
353 compact_ratio: float = COMPACT_RATIO,
354 min_new_rounds: int = COMPACT_MIN_NEW_ROUNDS) -> None:
355 """template:渲染模板(缺省内置 DEFAULT);config:Settings 快照(读
356 loop.max_context_tokens 回落窗宽);window_tokens:显式窗宽覆盖。
357 """
358 self.template: PromptTemplate = template or default_template()
359 self.config: Any = config

379 # ------------------------------------------------------- 变量表构建
380 def _tools_schemas(self, ctx: Any) -> list:
381 """从 ctx.tools 注册表取当前会话可见的工具 schema 列表(能力描述源)。
382
383 优先 schemas_for(scope)(scope 过滤后视图),回落 schemas()/裸列表;
384 任何形态缺位 → [] = 无可用工具。只读,不触碰工具执行面。
385 """
386 tools = getattr(ctx, "tools", None)
387 if tools is None:
388 return []
389 try:
390 for fn_name, args in (("schemas_for", (getattr(ctx, "scope", None),)),
391 ("schemas", ())):
392 fn = getattr(tools, fn_name, None)
393 if callable(fn):
394 out = fn(*args)
395 if isinstance(out, (list, tuple)):
396 return list(out)
397 if isinstance(tools, (list, tuple)): # 裸 schema 列表注入
398 return [t for t in tools if t is not None]
399 defs = getattr(tools, "definitions", None) # name → schema 字典形态
400 if isinstance(defs, dict) and defs:
401 return [{**schema} if isinstance(schema, dict)
402 else schema for schema in defs.values()]
403 return []
404 except Exception as exc: # noqa: BLE001 注册表视图异常
405 log.warning("tools schema 视图读取失败,按无工具处理:%s", exc)
406 return []

408 def _vars(self, ctx: Any) -> dict:
409 """变量表构建(白名单数据源):role/deny/domains/workspace/sandbox_level/
410 window_tokens/title/capabilities 全部来自 scope 策略与 config/tools schema。
411
412 数据源唯一——禁止任何 LLM 输入/工具结果入表(注入防御)。
413 scope 缺失时返回 {}:模板若引用变量将由 render_template 抛 CFG-602
414 (spec 异常表:补 scope/config 派生变量表,会话继续)。
415 """
416 scope = getattr(ctx, "scope", None)
417 if scope is None:
418 return {}
419 try:
420 policy = scope.policy
421 except (AttributeError, TypeError):
422 return {} # 策略缺失:交由渲染层 CFG-602 / 护栏层 CFG-603
423 role = getattr(policy, "role", None) or getattr(scope, "role", None) or ""
424 deny_raw = getattr(policy, "deny_tools", None) or []
425 domains_raw = getattr(policy, "allowed_domains", None) or []
426 ws = getattr(policy, "workspace_root", None) or getattr(scope, "workspace_root", "")
427 runtime = getattr(ctx, 'platform_runtime', None)
428 if runtime is not None and runtime.definition.sandbox_profile.mode == 'isolated':
429 # Model-facing paths describe the container, never its host backing directory.
430 ws = 'Controlled private workspace. File tools accept relative paths only: calculator.py; use . to list the root. Never supply absolute paths, .., drive letters, UNC or URI.'
431 level = getattr(policy, "sandbox_level", None) or ""
432 session = getattr(ctx, "session", None)
433 title = ""
434 title_fn = getattr(session, "title", None)
435 if callable(title_fn):
436 try:
437 title = str(title_fn() or "")
438 except Exception: # noqa: BLE001 会话标题异常不阻断
439 log.debug("session.title() 读取失败,按无主题处理")
440 title = ""
441 # 排序保确定性:deny/domains 为 set 真源(scope 规格),跨进程 join 顺序锁死
442 deny = ", ".join(sorted(str(x) for x in deny_raw)) or "无"
443 domains = ", ".join(sorted(str(x) for x in domains_raw)) or "无(禁止外发)"
444 schemas = self._tools_schemas(ctx)
445 return {
446 "role": f"[角色] {role}" if role else "[角色] 未指定",
447 "capabilities": f"[可用能力(来自工具 schema)]\n{build_capabilities(schemas)}",
448 "deny": f"[禁止操作] {deny}",
449 "domains": f"[外发域名] {domains}",
450 "workspace": f"[工作区] {ws}",
451 "sandbox_level": f"[沙箱级别] {level or '未指定'}",
452 "window_tokens": f"[上下文窗口] {self._window(ctx)} tokens",
453 "title": f"[会话主题] {title}" if title else "[会话主题] 无",
454 # KF-C(A4):本轮**框架侧**来源(受控:engine 从 task.meta.source 取,
455 # 非模型/工具文本)。到点触发轮 ⇒ 明确告知"这是定时触发",否则模型
456 # 只会复述任务状态而不产出提醒本身。空串 = 普通轮(渲染时跳过)。
457 "turn_source": self._turn_source_line(ctx),
458 }

470 # ------------------------------------------------------- 压缩触发判定
471 def _needs_compaction(self, ctx: Any, hist: list[dict]) -> bool:
472 """压缩触发判定(F058):派生历史 ≥75% 窗口 且 上次压缩后新增 ≥10 轮。
473
474 派生 token 优先 ctx.session.derived_tokens()(spec 数据源),缺位回落对
475 hist 逐条估算;距上次压缩轮数(turns_since_last_compact)未知时保守不触发
476 (防抖优先,不猜数)——宁可多留一次截断留痕,不误发压缩信号。
477 """
478 win = self._window(ctx)
479 if win <= 0:
480 return False
481 hist_tokens: Optional[int] = None
482 session = getattr(ctx, "session", None)
483 derived_fn = getattr(session, "derived_tokens", None)
484 if callable(derived_fn):
485 try:
486 hist_tokens = int(derived_fn())
487 except Exception: # noqa: BLE001 派生接口异常回落估算
488 hist_tokens = None
489 if hist_tokens is None:
490 hist_tokens = sum(est_tokens(m) for m in (hist or []))
491 if hist_tokens < self.compact_ratio * win: # <75%:窗口充裕,不触发
492 return False
493 since_fn = getattr(session, "turns_since_last_compact", None)
494 if not callable(since_fn): # 上次压缩轮数未知 → 不猜不触发
495 return False
496 try:
497 since = int(since_fn())
498 except Exception: # noqa: BLE001 同上,保守不触发
499 return False
500 return since >= self.min_new_rounds # 新增 ≥10 轮才压缩(防抖)

502 # ------------------------------------------------------- 信号出口
503 @staticmethod
504 def _signal(ctx: Any, type_: str, payload: dict) -> None:
505 """系统声明信号出口:经 ctx.bus 广播(emit_sync 优先;emit 为异步则降级)。
506
507 事件投递失败仅记日志不阻断组装(信号是建议性的,审计由调用方按
508 last_report 兜底)——与 EVT-103 订阅者隔离、config 热更事件降级同构。
509 """
510 bus = getattr(ctx, "bus", None)
511 if bus is None:
512 log.debug("sysprompt 信号 %s 未投递:ctx.bus 未装配", type_)
513 return
514 sink = getattr(bus, "emit_sync", None) or getattr(bus, "emit", None)
515 if not callable(sink):
516 log.warning("sysprompt 信号 %s 未投递:bus 无 emit/emit_sync", type_)
517 return
518 try:
519 result = sink(type_, payload)
520 if hasattr(result, "__await__"): # 异步 emit:同步上下文不 await
521 log.debug("sysprompt 信号 %s 交由异步分发", type_)
522 except Exception as exc: # noqa: BLE001 信号失败不阻断请求
523 log.warning("sysprompt 信号 %s 投递失败:%s", type_, exc)
524
525 # ------------------------------------------------------- 主装配
526 def assemble(self, hist: list[dict], user_msg: str = "", *, ctx: Any) -> list[dict]:
527 """提示词主装配(F010):核心模板段 + 护栏段 + 截窗历史 → messages。
528
529 hist:session.derive_history() 产物(含 user_msg 已入历史);
530 user_msg:保留参数(供断言/调试,不重复入列);
531 ctx:scope 策略/窗口数据源(护栏缺则拒请求 CFG-603)。
532
533 异常表:
534 CFG-602 模板渲染缺变量(会话继续)
535 CFG-603 护栏段构建失败 → 拒绝本次 LLM 调用(缺护栏不发请求)
536 """
537 hist = hist or []
538 core = render_template(self.template, vars=self._vars(ctx)) # 分段确定性渲染
539 guard = build_guard_segment(getattr(ctx, "scope", None)) # F024:恒末
540 if guard is None: # 护栏失败:缺护栏不发请求
541 raise_code("CFG-603", scope=str(getattr(ctx, "scope", None)),
542 advice="护栏段构建失败,本次请求未发送;修 scope 数据源")
543 window = self._window(ctx) # 默认 64k tokens(scope 数据源)
544 hist_budget = window - reserve(core) - reserve(guard) # 先保 system 段
545 kept, dropped = truncate_history(hist, hist_budget) # 头部截断(旧先丢)
546 kept_tokens = sum(est_tokens(m) for m in kept)
547 compact = self._needs_compaction(ctx, hist)
548 self.last_report = TruncReport(dropped_range=list(dropped),
549 kept_tokens=kept_tokens,
550 compact_hint=compact)
551 if dropped: # 裁剪留痕可审计(F010 边界)
552 self._signal(ctx, "sysprompt.truncated",
553 {"dropped_seq_range": list(dropped),
554 "kept_tokens": kept_tokens})
555 if compact: # F058:派生 ≥75% 窗口且新增 ≥10 轮
556 self._signal(ctx, "sysprompt.compact_hint", {"reason": "window"})
557 goal_seg = _active_goal_segment(ctx) # F047 活动目标板(空串零注入)
558 skills_seg = _active_skills_segment(ctx) # F073 技能目录(空库零注入)
559 parts = [core]
560 if goal_seg:
561 parts.append(goal_seg)
562 if skills_seg:
563 parts.append(skills_seg)
564 parts.append(guard) # F024 护栏段恒在最末
565 content = "\n\n".join(parts)
566 return [{"role": "system", "content": content}] + list(kept)
