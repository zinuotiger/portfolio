40# 估 token 启发式(无权威 est_tokens 实现;仅窗口截断用,不影响 reducer 映射):
41# CJK 逐字 1 token;ASCII 词按 4 字符 ≈ 1 token;每条消息 +4 role 开销常量。
42_ASCII_WORD = re.compile(r"[A-Za-z0-9_./\\@:+-]+")

45def _estimate_tokens(content: str) -> int:
46 """消息文本 token 粗估(中文逐字,西文按词;截断阈值仅需量级近似)。"""
47 cjk = sum(1 for ch in content if "\u4e00" <= ch <= "\u9fff")
48 words = sum(max(1, (len(w) + 3) // 4) for w in _ASCII_WORD.findall(content))
49 return cjk + words + 4 # +4: role/content 等结构开销(启发式常量)

121 def _absorb(self, env: Envelope) -> None:
122 """吸收一条事件入缓存(append/回放共用):seq 推进 + 声明索引同步。"""
123 self._cache.append(env)
124 self._seqs.append(env.seq)
125 self._seq = env.seq
126 if env.type == "context.compacted":
127 # 折叠声明入索引(compaction 的"已折叠"语义;区间合法性由 payload 模型保证)
128 for pair in env.payload["ranges"]:
129 self._folded.append(list(pair))
130 self._folded.sort(key=lambda r: r[0])
131 if env.type in DECLARE_TYPES:
132 # 空洞合法化声明(**唯一判据** `declared_ranges`):含 compacted.ranges 与
133 # recovered.lost / "seq-holes:[…]"。R14-9 前此处只认 compacted ⇒ 已声明的
134 # 修复空洞在下次回放被重报为"未声明空洞(疑丢事件)"。
135 for lo, hi in declared_ranges(env):
136 self._holes_declared.append([lo, hi])
137 self._holes_declared.sort(key=lambda r: r[0])

139 def _seq_index(self) -> list[int]:
140 """seq 升序索引(与缓存平行;append-only 保证与 _cache 严格对齐)。"""
141 return self._seqs

148 def _gap_declared(self, gap_lo: int, gap_hi: int) -> bool:
149 """空洞区间 [gap_lo, gap_hi] 是否被**声明**完全覆盖(合法空洞不回填)。
150
151 判据源 = ``_holes_declared``(compacted + recovered),**不是** ``_folded``
152 (后者只是"已折叠",不含修复声明;R14-9)。
153 """
154 return all(any(lo <= s <= hi for lo, hi in self._holes_declared)
155 for s in range(gap_lo, gap_hi + 1))

287 @staticmethod
288 def _drop_unpaired_tool_calls(msgs: list[dict],
289 paired: set) -> list[dict]:
290 """ADR-022 终局兜底:仍无配对 tool 消息的 `assistant.tool_calls` 予以剔除。
291
292 为什么必须有:拒绝类事件在**旧日志/异常时序**下可能缺失(如历史遗留会话、
293 反查失败),此时投影仍会悬空 —— 而 OpenAI/DeepSeek 端点对"`assistant.tool_calls`
294 无配对 `tool` 消息"直接 400(session.py 上文两处注释即作者真链实测)。
295 本兜底**只作用于投影**,不动日志(INV-01:真源仍是 append-only 事件流)。
296
297 无配对的 assistant 消息:仍有 content ⇒ 降级为纯文本 assistant;
298 连 content 也没有 ⇒ **整条丢弃**(等价于"该轮未发生"),避免空 assistant 消息。
299 """
300 out: list[dict] = []
301 for m in msgs:
302 calls = m.get("tool_calls")
303 if not calls:
304 out.append(m)
305 continue
306 kept = [c for c in calls if c.get("id") in paired]
307 if len(kept) == len(calls):
308 out.append(m)
309 elif kept:
310 out.append({**m, "tool_calls": kept})
311 elif m.get("content"):
312 out.append({"role": "assistant", "content": m["content"]})
313 return out

315 @staticmethod
316 def _truncate_head(msgs: list[dict], max_tokens: Optional[int]) -> list[dict]:
317 """超窗头部截断:自最新消息向前累计,超窗即丢更旧头部(整条粒度)。
318
319 最新一条恒保留(空上下文无意义);压缩摘要职责归 F058,本函数不承担语义裁
320 剪——仅窗口预算粗闸。None = 不截断(测试/无窗口场景)。
321 """
322 if max_tokens is None:
323 return msgs
324 kept: list[dict] = []
325 acc = 0
326 for m in reversed(msgs): # 新 → 旧累计,旧消息先丢
327 t = _estimate_tokens(m.get("content") or "")
328 if acc > 0 and acc + t > max_tokens:
329 break # 预算已满:更旧的头部整体丢弃
330 kept.append(m)
331 acc += t
332 if not kept and msgs: # 单条即超窗:保最新一条
333 kept = [msgs[-1]]
334 kept.reverse()
335 return kept

338 async def append(self, type_: str, payload: dict, *, actor: str,
339 sync: bool = False, trace: Optional[dict] = None,
340 origin: Optional[str] = None,
341 task_id: Optional[str] = None) -> Envelope:
342 """唯一写路径(F009):校验 → 分配 seq → 入内存 → 总线分发 →(强同步)落盘。
343
344 校验链 5 步(events.validate_envelope:信封/注册/payload/seq/状态)任一失败
345 拒写——事件不进内存/总线/日志,零副作用。seq/ts 只由框架在此分配,调用方
346 无权自报(防伪造乱序)。
347
348 异常:EVT-104(终态后/重复 finished)、EVT-106(先于 created/重复 created)、
349 EVT-100(信封非法/瞬时入日志/BadTarget)、EVT-102(类型未注册)、EVT-101(seq
350 失步)、PERS-202(强同步落盘失败,repair 后重试)。
351 """
352 # 1) 终态拒写:结构上不可能(EVT-104,INV-01 闸)
353 if self._closed:
354 raise_code("EVT-104", type_=type_, hint="会话已结束(finished 仅一次);"
355 "查绕过 agent.close 的写路径(INV-01)")
356 # 2) 瞬时事件禁 append 入日志(EVENT-SCHEMA §1.3:llm.chunk 仅总线)
357 if is_transient(type_):
358 raise_code("EVT-100", type_=type_, hint="瞬时事件禁止 append 入日志:"
359 "llm.chunk/registry.updated 仅经总线分发")
360 # 暂停的存储会拒收新事件，但总线会隔离订阅者异常。
361 # 在分配 seq/写内存前拒绝，避免后续 flush 只恢复旧行却向新消息报成功。
362 check_writable = getattr(self._persistence, "ensure_writable", None)
363 if callable(check_writable):
364 check_writable()
365 # 3) finished 单次:置位防并发双写(终态不可逆;EVT-104 由第 1 步接住重复)。
366 # 第 1 步至此全为同步段(无 await),同一次事件循环回合内不可能插入第二条
367 # finished,故置位仍具并发防护力。
368 # S1-04 修复:置位必须**在失败时回滚**——否则 EVT-100/102(校验)或
369 # PERS-202(落盘)之后 _closed 永久为真,agent.close 的重试(其自身
370 # state 已回滚,agent.py:307-314)会被第 1 步 EVT-104 短路,"close
371 # 幂等可重入"不成立。回滚语义:未成功落盘的 finished 不构成事实。
372 finished_inflight = type_ == "session.finished"
373 if finished_inflight:
374 self._closed = True
375 absorbed = False
376 try:
377 # 4) session.created 必须为会话首事件且仅一条(seq=1 引导,EVT-106)
378 if type_ == "session.created" and self._seq > 0:
379 raise_code("EVT-106", type_=type_,
380 hint="session.created 必须为会话首事件(seq=1),"
381 "同会话仅一条")
382 # 5) 引用锚点预检:edited→user.message / feedback→agent.message(BadTarget)
383 if type_ == "user.message_edited" and not self._ref_exists(
384 payload.get("target_seq", 0), "user.message"):
385 raise_code("EVT-100", type_=type_,
386 hint=f"BadTarget:target_seq 必须指向已存在的 "
387 f"user.message,got {payload.get('target_seq')}")
388 if type_ == "user.feedback" and not self._ref_exists(
389 payload.get("target_seq", 0), "agent.message"):
390 raise_code("EVT-100", type_=type_,
391 hint=f"BadTarget:target_seq 必须指向已存在的 "
392 f"agent.message,got {payload.get('target_seq')}")
393 # 6) 框架打点 + 五步校验链(失败拒写:EVT-100/102/101/106)
394 env = make_envelope(self.sid, type_, actor, payload,
395 origin=origin, task_id=task_id, trace=trace,
396 tenant_id=self.tenant_id,
397 seq_state=self._seq_state)
398 # 7) 记账 + 入内存:订阅者/派生视图可即时读(先于总线,spec 顺序)
399 self._seq = env.seq
400 self._seq_state.commit(env) # 分配器前进(防漂移;失败由 repair 收尾)
401 if env.type == "session.created":
402 self._seq_state.open.add(self.sid) # pending → active 开闸
403 self._absorb(env)
404 absorbed = True
405 # 8) 总线分发:日志订阅者(§8)负责物理落盘
406 await self._dispatch(env)
407 # 9) 强同步三类/显式 sync:落盘成功才返回(PERS-202 上抛,repair 后重试)
408 if sync or env.type in SYNC_TYPES:
409 await self._flush(env)
410 except BaseException:
411 if finished_inflight:
412 # 回滚语义:未成功的 finished 不构成事实。判据取**内存事实**——
413 # absorbed 且缓存尾已是 finished(事件确已入内存并已分发,总线
414 # 订阅者可能已落盘)→ 保持终态位,维持"缓存有 finished ⇔ _closed"
415 # 的一致;否则(校验 EVT-100/102 等在入缓存前失败)→ 回滚置位,
416 # 使 agent.close 的重试可重入(修复目标)。
417 # 残留边界:同进程内"已分发但 flush 失败"仍为终态(retry 会 EVT-104),
418 # 跨进程 repair 后按日志事实重算 → 重试可入;与 agent.close
419 # docstring 6 的"repair 后重试"口径一致(非本次修复范围)。
420 absorbed_finished = (
421 absorbed and bool(self._cache)
422 and self._cache[-1].type == "session.finished")
423 self._closed = absorbed_finished
424 raise
425 # 10) 派生缓存整体失效(INV-03:缓存纪律)
426 self.history_cache = None
427 self._needs_rebuild = False
428 return env

430 # ============================================================ 派生视图工厂
431 def derive_messages(self, max_tokens: Optional[int] = None) -> list[dict]:
432 """派生历史:日志 → LLM 消息列表(§3.5 reducer 唯一权威;别名 derive_history)。
433
434 纯函数:输入仅为本类日志与 max_tokens,无副作用、无第二份状态(INV-01/02)。
435 history_cache 仅尾部未变时有效,append 即失效——调用方不得绕过本函数自建
436 历史。max_tokens = 窗口余量(scope.window_tokens 联动;None = 不截断)。
437 """
438 self._ensure_rebuilt()
439 if max_tokens is None and self.history_cache is not None:
440 return [dict(m) for m in self.history_cache] # 浅拷贝防外部改写缓存
441 out = self._truncate_head(self._fold_history(), max_tokens)
442 if max_tokens is None:
443 self.history_cache = out # 尾部未变时有效;append 即失效
444 return out
445
446 # derive_history 别名(§3.5/§4.1:同函数对象,禁止第二份实现)
447 derive_history = derive_messages

449 def events_after(self, after_seq: int = 0) -> Iterator[Envelope]:
450 """seq 升序增量读:返回 seq > after_seq 的全部事件(排他下界,0 = 全量)。
451
452 UI 增量同步/子代理拉新事件/审计尾随共用;内存缺失时惰性从 persistence
453 replay 补齐(可弃优化)。只读遍历,绝不在迭代中写日志。
454 """
455 self._ensure_rebuilt()
456 start = bisect.bisect_right(self._seq_index(), after_seq)
457 for env in self._cache[start:]:
458 yield env
459
460 def events_between(self, lo: int, hi: int) -> Iterator[Envelope]:
461 """[lo, hi] 闭区间事件迭代(lo > hi = 空);基于增量读,不另存状态。"""
462 for env in self.events_after(lo - 1): # 下界排他转闭区间
463 if env.seq > hi:
464 break # 上界截断
465 yield env

467 def get(self, seq: int) -> Optional[Envelope]:
468 """按 seq 取单条事件;越界/空洞(不存在)→ None 不抛。
469
470 空洞 = 日志物理缺失(compacted/repair 声明区间或坏行隔离);内存阶段缓存
471 全量保留原文,故折叠区间内仍可读到原事件(get 以缓存实际为准)。
472 """
473 self._ensure_rebuilt()
474 if not (1 <= seq <= self._seq):
475 return None
476 i = bisect.bisect_left(self._seq_index(), seq)
477 if i >= len(self._cache) or self._cache[i].seq != seq:
478 return None # 空洞(声明区间/隔离行)
479 return self._cache[i]

482 def rebuild_from_log(self) -> None:
483 """缓存整体重建(INV-03):弃内存 _cache/history_cache,从 persistence 重放。
484
485 append 后调用方可比对重建结果与缓存逐事件一致(测试钉死);坏行由
486 persistence.replay 记跳隔离(PERS-201),本函数不中断。
487 """
488 self._cache = []
489 self._seqs = []
490 self._folded = []
491 self._holes_declared = []
492 self._holes_warned = []
493 self._seq = 0
494 self.history_cache = None
495 self._closed = False
496 self._seq_state = SeqState()
497 last: Optional[Envelope] = None
498 if self._persistence is not None:
499 for env in self._persistence.replay():
500 if env.session_id != self.sid:
501 continue # 多会话文件过滤(轮转合并兜底)
502 self._absorb(env)
503 last = env
504 if last:
505 # 从最后完整点续写(max+1,空洞不回填)并开闸;终态按日志事实重算
506 self._seq_state.rebuild(self.sid, self._seq)
507 self._recompute_terminal()
508 self._needs_rebuild = False

531# ------------------------------------------------------------------ 工厂/恢复
532async def open_session(sid: str,
533 persistence: SessionStoreProtocol,
534 *, tenant_id: Optional[str] = None) -> SessionLog:
535 """启动/恢复重建:重放日志重建会话状态(崩溃恢复与审计回放同一条代码路径)。
536
537 repair(F060)先于本函数执行;文件不存在 = 新会话(等 session.created,由校验链
538 EVT-106 守卫);坏行 PERS-201 记跳不中断;无 compacted/recovered 声明的空洞 →
539 warn_hole 告警(F031);_seq 从最后完整点续写(空洞不回填);终态会话恢复后仍拒写。
540
541 ``tenant_id``(GAP-10)**两源合一,日志优先**:调用方传入的是"本次装配认为的
542 租户";而日志里已落的 ``Envelope.tenant_id`` 是**既成事实**。恢复时以日志为准
543 ——否则重启后换一个调用方就能给历史会话贴上不同租户(那正是 L-1 登记的"租户
544 未随会话落盘"残余)。两者不一致时保留日志值并告警,不回写、不覆盖事实。
545 """
546 log_ = SessionLog(sid=sid, persistence=persistence, tenant_id=tenant_id)
547 last: Optional[Envelope] = None
548 first_seq: Optional[int] = None
549 logged_tenant: Optional[str] = None
550 gaps: list[tuple[int, int]] = [] # 待判定空洞(声明可能落在其后)
551 for env in persistence.replay(): # 坏行由 replay 记跳隔离,不中断
552 if env.session_id != sid:
553 continue # 多会话文件过滤
554 if getattr(env, "tenant_id", None):
555 logged_tenant = env.tenant_id # 日志是既成事实:最后一条胜出
556 if first_seq is None:
557 first_seq = env.seq
558 if env.seq > 1:
559 # 首段连续缺失(轮转文件丢失/坏文件):seq 1..(env.seq-1) 整体缺失,
560 # 旧实现只查相邻差、对前缀失明(P1-5,与 repair.check_seq_gap 对齐)
561 gaps.append((1, env.seq - 1))
562 elif last and env.seq != last.seq + 1:
563 gaps.append((last.seq + 1, env.seq - 1))
564 log_._absorb(env)
565 last = env
566 # 空洞判定**必须在声明全部吸收之后**(R14-9):`session.recovered` 声明通常由 repair
567 # **追加在流尾**,边读边判会把"刚被声明的修复空洞"重新报成未声明空洞(假 F031 线索)。
568 for gap_lo, gap_hi in gaps:
569 if not log_._gap_declared(gap_lo, gap_hi):
570 log_.warn_hole(gap_lo) # 报**首个缺失的 seq**(此前误报空洞后那条)
571 if last:
572 log_._seq_state.rebuild(sid, log_._seq) # 开闸:从最后完整点续写
573 if last.type == "session.finished":
574 log_._closed = True # 终态会话恢复后仍拒写(EVT-104)
575 log_.history_cache = None # 派生缓存一律重建(INV-03)
576 log_._needs_rebuild = False
577 # GAP-10:租户以**日志**为准(既成事实)。日志无归属(旧会话/单租户)时保留
578 # 调用方注入值;两者冲突时日志胜出并告警——绝不因重开而改写历史归属。
579 if logged_tenant is not None:
580 if tenant_id is not None and tenant_id != logged_tenant:
581 log.warning("会话 %s 租户不一致:装配=%s 日志=%s(以日志为准)",
582 sid, tenant_id, logged_tenant)
583 log_.tenant_id = logged_tenant
584 return log_
