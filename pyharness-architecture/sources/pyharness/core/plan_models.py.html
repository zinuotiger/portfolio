38def _utcnow() -> datetime:
39 """当前 UTC aware 时间(会话事件 ts 同口径;避免 naive/aware 比较 TypeError)。"""

42def _iso(dt: datetime) -> str:
43 """datetime → ISO8601 UTC 串(Z 结尾,与 Envelope.ts 同形)。"""
44 return dt.astimezone(timezone.utc).isoformat(timespec="microseconds").replace(
45 "+00:00", "Z")

47def _parse_dt(s: Any) -> Optional[datetime]:
48 """ISO8601 串(含 Z 结尾)→ aware datetime;空/非法返回 None。"""
49 if not s:
50 return None
51 try:
52 return datetime.fromisoformat(str(s).replace("Z", "+00:00"))
53 except ValueError:
54 return None

82 """会话级方案(id = plan.proposed 事件 seq;状态只由事件回放重建,INV-01)。

91 why: str = "" # selfcheck 未过原因(空 = 通过)

93 created_at: Optional[datetime] = None

95 who: Optional[str] = None

97 def summary(self) -> str:
98 """用户可读一行摘要(UI/命令层展示待批方案用)。"""
99 steps_txt = " → ".join(f"{i}.{s.action}({s.tool})" for i, s in
100 enumerate(self.steps))
101 check = "自检通过" if self.selfcheck_ok else f"自检未过:{self.why}"
102 return (f"[plan {self.id}] {self.goal} | {self.status} | {check}\n"
103 f" 步骤({len(self.steps)}):{steps_txt}")
