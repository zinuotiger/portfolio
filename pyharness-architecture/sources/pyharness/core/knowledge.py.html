37 async def query(self, query: str, documents: list[dict]) -> dict:
38 validate_query(query, self.capabilities)
39 words = set(re.findall(r'[\w-]+', query.casefold()))
40 hits = []
41 for document in documents:
42 for number, line in enumerate(document['text'].splitlines(), 1):
43 score = sum(word in line.casefold() for word in words)
44 if score:
45 hits.append({'source_id': document['source_id'], 'name': document['name'],
46 'line': number, 'quote': line[:1000], 'score': score,
47 'version': document.get('version', 1)})
48 hits.sort(key=lambda h: (-h['score'], h['name'], h['line']))
49 return {'provider': 'exact_keyword', 'citations': hits[:10],
50 'answer_allowed': bool(hits), 'reason': None if hits else 'no_reliable_evidence'}
