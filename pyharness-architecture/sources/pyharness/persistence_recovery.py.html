23 def _rotated_paths(self) -> list[Path]:
24 """主目录下 {sid}.{n}.jsonl(n 从 1)按序号升序(重放合并顺序)。"""
25 pat = re.compile(r"^" + re.escape(self.session_id) + r"\.(\d+)\.jsonl$")
26 found: list[tuple[int, Path]] = []
27 try:
28 children = list(self.path.parent.iterdir())
29 except OSError:
30 return []
31 for child in children:
32 m = pat.match(child.name)
33 if m:
34 found.append((int(m.group(1)), child))
35 return [p for _n, p in sorted(found)]

37 def _maybe_rotate(self) -> None:
38 """轮转检查(append 写入后调用):>rotate_bytes → _rotate。
39
40 大小取 OS 文件字节数(fstat),不用 `_fh.tell()`——文本模式 tell() 返回的是
41 不透明游标(cookie)而非字节偏移,与 rotate_bytes(字节)不可比(P3:_rotate 判定)。
42 轮转失败只记日志不抛(数据无损:事件已先落主文件;rename 原子性保证主文件
43 仍在),下次 append 再触发——避免在 append 成功路径上叠加二次异常。
44 """
45 if self._retry_q or self._pending:
46 return
47 try:
48 size = os.fstat(self._fh.fileno()).st_size
49 if size > self.rotate_bytes:
50 self._rotate()

55 def seq_holes(self, path: Path) -> list[int]:
56 """对照连续序列找缺失号(含轮转合并回放;path 参数保留 spec 形态)。
57
58 只找相邻差 >1 的缺口;seq 只前进,空洞只解释不回填(§3.4)。重复/倒退
59 (range 空)不入洞——append 保证单调,此处只定位缺失,声明判定在 repair 内。
60 """
61 seqs = [env.seq for env in self.replay()] # 复用坏行隔离读取
62 holes: list[int] = []
63 if seqs and seqs[0] > 1:
64 # 前缀段连续缺失(轮转文件丢失/坏文件):seq 1..(seqs[0]-1) 整体缺失,
65 # 旧实现只查相邻差、对首段失明(P1-5,与 repair.check_seq_gap 口径对齐)
66 holes.extend(range(1, seqs[0]))
67 for expect, got in zip(seqs, seqs[1:]):
68 if got != expect + 1:
69 holes.extend(range(expect + 1, got)) # 相邻差 >1 = 空洞
70 return holes
