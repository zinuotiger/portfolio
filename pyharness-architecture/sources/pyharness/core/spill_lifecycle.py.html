8async def detach(ctx: Any) -> None:
9 """幂等摘除(DIS-SEAM §6.1 ⑤):摘读工具/摘定位器/摘订阅,文件不删。
10
11 会话关闭(ctx.close)时调用;**文件保留**——留给 F060 归档/repair 在会话
12 结束后仍可读的窗口,删除统一由 purge_session 在 session.finished 后执行。
13 摘除失败仅本地日志告警,不阻断会话关闭(文件系统无副作用残留)。
14 """
15 storage = getattr(ctx, "storage", None)
16 # ---- 1) 幂等闸:locator 自报未挂载即已摘;无 locator 面按挂载痕迹判
17 if storage is not None:
18 gate = getattr(storage, "_unmounted", None)
19 if callable(gate):
20 try:
21 if await _maybe_await(gate("spill")):
22 return # 已摘除(幂等早退)
23 except Exception: # noqa: BLE001 locator 异常:
24 pass # 继续尽力摘除,不阻断
25 else:
26 services = getattr(storage, "services", None) or {}
27 if not (hasattr(storage, "spill") or "spill" in services):
28 return # 无 spill 挂载痕迹 → 已摘
29 # ---- 2) 摘 LLM 可见读工具(TLB-802 = 已摘,静默幂等)
30 tools = getattr(ctx, "tools", None)
31 if tools is not None:
32 unreg = getattr(tools, "unregister", None)
33 if not callable(unreg):
34 unreg = getattr(tools, "unregister_definition", None)
35 if callable(unreg):
36 try:
37 await _maybe_await(unreg(TOOL_NAME))
38 except PyHError as e:
39 if e.code != "TLB-802": # 已摘(重入/双摘)→ 静默
40 _log_warn(ctx, "spill detach unregister 失败 code=%s", e.code)
41 except Exception as e: # noqa: BLE001
42 _log_warn(ctx, "spill detach unregister 异常: %s", e)
43 # ---- 3) 摘定位器(ctx.storage.spill;真实 locator 走 unmount)
44 if storage is not None:
45 unmount = getattr(storage, "unmount", None)
46 if callable(unmount):
47 try:
48 await _maybe_await(unmount("spill"))
49 except Exception as e: # noqa: BLE001
50 _log_warn(ctx, "spill detach unmount 失败: %s", e)
51 elif hasattr(storage, "spill"): # 无 locator API:直接摘属性
52 try:
53 delattr(storage, "spill")
54 except Exception as e: # noqa: BLE001
55 _log_warn(ctx, "spill detach 摘除属性失败: %s", e)
56 # ---- 4) 摘 session.finished 清理订阅(按会话属主)
57 _unsubscribe_purge(ctx)
58 log.info("spill detached(文件保留,F060 读窗口)")

73async def purge_session(session_id: str, ctx: Any = None) -> None:
74 """清理策略主体(F039 随会话生命周期/I-3):删除该会话 spill 目录全部文件。
75
76 先 containment 校验(只删 spill 根下本会话子目录,逃逸即 PERS-222 拒);
77 目录不存在幂等;删除失败记日志+告警不阻断(F060 repair 兜底残留)。
78 订阅 session.finished 后由本函数收口"spill 随会话生命周期清理"。
79 """
80 root = _spill_root(ctx)
81 if root is None:
82 raise_code("PERS-222", session=session_id,
83 advice="spill 根未配置,无法校验清理目标,已拒绝")
84 root = root.resolve()
85 target = (root / session_id).resolve() # 归一后比对
86 try:
87 target.relative_to(root) # containment 单点(Windows
88 except ValueError: # 大小写不敏感,偏离 4)
89 raise_code("PERS-222", session=session_id,
90 advice="清理目标逃逸 spill 根,已拒绝")
91 if target == root: # 目标即根(空/`.` session)→ 拒
92 raise_code("PERS-222", session=session_id,
93 advice="清理目标逃逸 spill 根,已拒绝")
94 try:
95 if target.exists():
96 for f in target.glob(SPILL_PATTERN): # 逐文件删(可留审计计数)
97 f.unlink(missing_ok=True)
98 for f in target.glob(f"{SPILL_PATTERN}.tmp"): # 残留 tmp(偏离 5)
99 f.unlink(missing_ok=True)
100 target.rmdir() # 空目录收口
101 log.info("spill purge done session=%s dir=%s", session_id, target)
102 except OSError as e: # 残留不致命(F060 兜底)
103 _log_warn(ctx, "spill purge failed session=%s: %s", session_id, e)

105def subscribe_purge(ctx: Any, session_id: Optional[str] = None) -> Optional[str]:
106 """订阅本会话 session.finished → purge_session(返回 owner 标记,可摘除)。
107
108 装配于 Provider.enter(生命周期 enter→announce→detach);bus 未接线/无
109 subscribe 面 → 返回 None(清理留待装配层兜底,文件由 F060 repair 覆盖)。
110 """
111 sid = session_id or _session_id(ctx)
112 if not sid:
113 return None
114 bus = getattr(ctx, "bus", None)
115 subscribe = getattr(bus, "subscribe", None)
116 if not callable(subscribe):
117 return None
118 owner = f"spill:{sid}"
119
120 async def _on_finished(type_: str, payload: Any) -> None: # noqa: ARG001
121 """session.finished 回调:仅清本会话(总线跨会话共享,按 payload 过滤)。
122
123 声明 async:总线 _dispatch 对协程函数 handler 会 await(同步函数返回
124 协程则不会被等待,产生 never-awaited 告警)。
125 """
126 p_sid = getattr(payload, "session_id", None)
127 if p_sid is None and isinstance(payload, dict):
128 p_sid = payload.get("session_id")
129 if p_sid is not None and str(p_sid) != sid:
130 return # 其他会话的 finished:不动
131 await purge_session(sid, ctx) # 删除本会话 spill 目录
132
133 try:
134 bus.subscribe("session.finished", _on_finished, owner=owner)
135 except Exception as e: # noqa: BLE001 订阅失败不致命
136 _log_warn(ctx, "spill purge 订阅失败: %s", e)
137 return None
138 return owner
