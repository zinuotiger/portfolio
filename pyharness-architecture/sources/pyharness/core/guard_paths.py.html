12def _inside_workspace(target: str, workspace_root: str) -> bool:
13 """target(已 normpath)是否几何落在 workspace_root 内(含根本身)。
14
15 前缀比较带分隔符边界(防 ws=/a/b 误纳 /a/bc);Windows normcase 后比较。
16 """
17 ws = _normcase_path(os.path.abspath(os.path.expanduser(workspace_root)))
18 t = _normcase_path(os.path.abspath(target))
19 if t == ws:
20 return True

23def _norm_target(raw: Any, workspace_root: str) -> str:
24 """原始路径 → 参与几何判定的规范绝对串(相对路径以 workspace 为基拼接)。
25
26 只做词法规范化(normpath),不做 symlink 解析(那是 POL-FS-3 单独职责);
27 规范化后仍可能词法在界内而真实(realpath)在界外 → FS-3 接住。
28 """
29 s = str(raw).strip()
30 if not s:
31 return os.path.abspath(os.path.expanduser(workspace_root))
32 raw = os.path.expanduser(s)
33 ws = os.path.abspath(os.path.expanduser(workspace_root))
34 if os.path.isabs(raw):
35 return os.path.normpath(raw)
36 return os.path.normpath(os.path.join(ws, raw))

58def _is_junction(path) -> bool:
59 """Inspect one Windows path without following its junction target (3.11+)."""
60 if os.name != "nt":

68def _is_linkish(p: str) -> bool:
69 """p 或其现存祖先是否 symlink/junction(触发 FS-3 realpath 复核的门槛)。"""
70 def is_link(path: str) -> bool:
71 if os.path.islink(path):
72 return True
73 junction = getattr(os.path, "isjunction", None)
74 if junction is not None:
75 return junction(path)
76 try:
77 return _is_junction(path)
78 except (OSError, ValueError):
79 return False
80
81 cur = os.path.normpath(p)
82 if is_link(cur):
83 return True
84 parent = os.path.dirname(cur)
85 guard = 0
86 while parent and parent != cur and guard < 64: # 向上找现存链点
87 if is_link(parent):
88 return True
89 nxt = os.path.dirname(parent)
90 if nxt == parent:
91 break
92 cur, parent, guard = parent, nxt, guard + 1
93 return False

95def _in_read_extra(target: str, extra_read_dirs: Any) -> bool:
96 """target 是否落在**只读例外目录**内(A2;两层同判口径)。
97
98 例外目录条目按 `expanduser` + `abspath`(+ 尽力 `realpath`)归一后比较;
99 不可解析的条目跳过(不放大)。**只判包含**:是否"允许"由调用方按**读写**决定。
100 """
101 for entry in (extra_read_dirs or ()):
102 raw = str(entry or "").strip()
103 if not raw:
104 continue
105 try:
106 base = os.path.abspath(os.path.expanduser(raw))
107 real = os.path.realpath(base)
108 except (OSError, ValueError):
109 continue
110 if _inside_workspace(target, base) or _inside_workspace(target, real):
111 return True
112 return False

114def _resolve_geometry(raw: Any, workspace_root: str,
115 *, link_resolver: Optional[Callable[[str], str]] = None,
116 read_only: bool = False,
117 extra_read_dirs: Any = ()
118 ) -> tuple[str, Optional[str]]:
119 """路径几何单点判定(g3 权威语义;F055 单点 guard 侧实现)。
120
121 返回 (规范绝对路径, policy_ref);policy_ref 非空 = 越界(调用方须拒):
122 POL-FS-1 绝对路径越界(词法不在 workspace)
123 POL-FS-2 ".." 段逃逸(规范化后越界)
124 POL-FS-3 symlink/junction 终解析越界(词法在界内,真实在界外)
125 全过返回 (target, None)。纯只读:零 stat 之外副作用(os.path 探测)。
126
127 ``read_only`` + ``extra_read_dirs``(**A2/R24**):``security.policy.read_extra_dirs``
128 声明的例外目录**只对读类工具**放行(写类一律仍越界拒)。此前例外在两层都被忽略
129 ⇒ 配置了仍被拒(L-24)。此处与 provider ``resolve_in_workspace`` **同判**。
130 """
131 target = _norm_target(raw, workspace_root)
132 ws = os.path.abspath(os.path.expanduser(workspace_root))
133 exempt = bool(read_only) and _in_read_extra(target, extra_read_dirs)
134 # 1) POL-FS-1:绝对路径词法越界(不 startswith workspace)
135 if os.path.isabs(str(raw).strip()) and not _inside_workspace(target, ws) and not exempt:
136 return (target, "POL-FS-1")
137 # 2) POL-FS-2:".." 段逃逸(规范化后不在 workspace)
138 if _has_dotdot(str(raw)) and not _inside_workspace(target, ws) and not exempt:
139 return (target, "POL-FS-2")
140 # 3) POL-FS-3:symlink/junction 终解析越界(链点存在才查,防误伤普通路径)
141 if _is_linkish(target):
142 final = (link_resolver or _final_realpath)(target)
143 if not _inside_workspace(final, ws) and not exempt:
144 return (target, "POL-FS-3")
145 return (target, None)
