121async function performAndReadBack(action, run) {
122 // Observe lifecycle before the irreversible action, including same-URL
123 // reloads and redirect chains. Never repeat the action to obtain readback.
124 let pending = false, ready = Promise.resolve(), resolveReady, timer;
125 const deadline = Date.now() + options.timeout;
126 const requested = request => {
127 if (!request.isNavigationRequest() || request.frame() !== page.mainFrame()) return;
128 if (!pending) {
129 pending = true;
130 ready = new Promise(resolve => { resolveReady = resolve; });
131 }
132 };
133 const loaded = () => {
134 pending = false;
135 if (resolveReady) resolveReady();
136 resolveReady = undefined;
137 };
138 page.on('request', requested);
139 page.on('domcontentloaded', loaded);
140 try {
141 const result = await run();
142 if (result) return problem(result);
143 // Give immediate DOM handlers and navigation request events a turn.
144 await page.waitForTimeout(50);
145 while (pending) {
146 const remaining = deadline - Date.now();
147 if (remaining <= 0) throw Error('browser_navigation_unconfirmed');
148 try {
149 await Promise.race([ready, new Promise((_, reject) => {
150 timer = setTimeout(() => reject(Error('browser_navigation_unconfirmed')), remaining);
151 })]);
152 } finally { clearTimeout(timer); }
153 }
154 return {acted: action, readback: await observe(0)};
155 } finally {
156 clearTimeout(timer);
157 page.removeListener('request', requested);
158 page.removeListener('domcontentloaded', loaded);
159 }
