1"""pyharness/core/attachment.py — 图片附件校验与内容寻址落盘(F061 唯一实现)。
2
3**为什么单独成模块**:F061 的校验面(magic bytes / 大小 / 数量 / 内容寻址)此前
4**只到事件 schema 层**(mime 白名单由 payload 模型强制,但 mime 由客户端自报、
5缺省回落 `image/png`,且无魔数/大小/数量校验、`sha256` 可省略回落全零占位)。
6规格(PRD-Core §F061 验收伪代码)要求:魔数校验("不信扩展名")→ 单张 ≤10MB →
7单消息 ≤5 张 → `sha256(raw)[:16]` 内容寻址去重。本模块是该规格的**单点实现**,
8装配层(ApplicationService)与外壳一律经此,不得各自实现(同 core/channel.py 先例)。
9
10安全口径:入参 `path` 来自**客户端自报**,但读到的字节一律按**不可信内容**处理
11——先按魔数定类型(不看扩展名)、再按配置上限定大小,最后内容寻址落盘到会话工作区;
12任一环节不满足即 `ATT-001` 拒绝(ERR.md §2.11 功能特性码,零写盘)。
13"""

47def detect_image(raw: bytes) -> Optional[tuple[str, str]]:
48 """按**魔数**判类型(不看扩展名);未命中 → None。"""
49 for magic, mime, ext in _MAGIC:
50 if raw.startswith(magic):
51 return mime, ext
52 if len(raw) >= 12 and raw[:4] == b"RIFF" and raw[8:12] == b"WEBP":
53 return "image/webp", ".webp"
54 return None

57def _dims_png(raw: bytes) -> Optional[tuple[int, int]]:
58 if len(raw) < 24 or raw[12:16] != b"IHDR":
59 return None
60 w = int.from_bytes(raw[16:20], "big")
61 h = int.from_bytes(raw[20:24], "big")
62 return (w, h) if w > 0 and h > 0 else None

67 return None
68 w = int.from_bytes(raw[6:8], "little")
69 h = int.from_bytes(raw[8:10], "little")
70 return (w, h) if w > 0 and h > 0 else None

73def _dims_jpeg(raw: bytes) -> Optional[tuple[int, int]]:
74 """扫 SOFn 标记取尺寸(SOF0/1/2/3/5/6/7/9/10/11/13/14/15)。"""
75 i = 2
76 n = len(raw)
77 while i + 9 < n:
78 if raw[i] != 0xFF:
79 i += 1
80 continue
81 marker = raw[i + 1]
82 if marker in (0xD8, 0x01) or 0xD0 <= marker <= 0xD7:
83 i += 2
84 continue
85 seg_len = int.from_bytes(raw[i + 2:i + 4], "big")
86 if marker in (0xC0, 0xC1, 0xC2, 0xC3, 0xC5, 0xC6, 0xC7,
87 0xC9, 0xCA, 0xCB, 0xCD, 0xCE, 0xCF):
88 h = int.from_bytes(raw[i + 5:i + 7], "big")
89 w = int.from_bytes(raw[i + 7:i + 9], "big")
90 return (w, h) if w > 0 and h > 0 else None
91 if seg_len <= 0:
92 return None
93 i += 2 + seg_len
94 return None

97def _dims_webp(raw: bytes) -> Optional[tuple[int, int]]:
98 if len(raw) < 30:
99 return None
100 fourcc = raw[12:16]
101 if fourcc == b"VP8X":
102 w = int.from_bytes(raw[24:27], "little") + 1
103 h = int.from_bytes(raw[27:30], "little") + 1
104 return (w, h) if w > 0 and h > 0 else None
105 if fourcc == b"VP8L" and len(raw) >= 25 and raw[20] == 0x2F:
106 bits = int.from_bytes(raw[21:25], "little")
107 w = (bits & 0x3FFF) + 1
108 h = ((bits >> 14) & 0x3FFF) + 1
109 return (w, h) if w > 0 and h > 0 else None
110 if fourcc == b"VP8 " and len(raw) >= 30:
111 w = int.from_bytes(raw[26:28], "little") & 0x3FFF
112 h = int.from_bytes(raw[28:30], "little") & 0x3FFF
113 return (w, h) if w > 0 and h > 0 else None
114 return None

121def validate_image(raw: bytes, *,
122 mime_whitelist: Any = None,
123 max_bytes: Optional[int] = None) -> dict:
124 """**魔数校验 + 尺寸解析 + 大小/白名单**校验(不信扩展名/自报 mime)。
125
126 返回 ``{"mime","ext","w","h","size_bytes","sha256"}``;任一不满足 → ``ATT-001``。
127 """
128 limit = int(max_bytes or DEFAULT_MAX_BYTES)
129 if not raw:
130 _att_reject("empty", size_bytes=0)
131 if len(raw) > limit:
132 _att_reject("too-big", size_bytes=len(raw), limit=limit)
133 hit = detect_image(raw)
134 if hit is None:
135 _att_reject("magic-mismatch", size_bytes=len(raw),
136 advice="仅接受 jpeg/png/webp/gif(按魔数判定,不看扩展名)")
137 mime, ext = hit
138 allow = tuple(mime_whitelist or DEFAULT_MIME_WHITELIST)
139 if mime not in allow:
140 _att_reject("mime-not-allowed", mime=mime, allowlist=list(allow))
141 dims = _DIMS[mime](raw)
142 if dims is None:
143 _att_reject("dims-unreadable", mime=mime,
144 advice="无法解析图像尺寸:文件可能截断或伪装")
145 w, h = dims
146 return {"mime": mime, "ext": ext, "w": int(w), "h": int(h),
147 "size_bytes": len(raw),
148 "sha256": hashlib.sha256(raw).hexdigest()}

151def store_content_addressed(raw: bytes, info: dict, *,
152 workspace: Any) -> str:
153 """内容寻址落盘:`{workspace}/attachments/{sha256[:16]}{ext}`(已存在即去重)。
154
155 返回落盘绝对路径(即事件 ``file_path``)。工作区缺位 → ``ATT-001``(装配错误,
156 fail-closed:附件必须落在**会话可见边界内**,不接受任意宿主路径)。
157 """
158 if not workspace:
159 _att_reject("no-workspace",
160 advice="会话工作区未装配:附件必须落在工作区内(拒绝宿主任意路径)")
161 d = Path(str(workspace)).expanduser() / "attachments"
162 d.mkdir(parents=True, exist_ok=True)
163 target = d / f"{info['sha256'][:16]}{info['ext']}"
164 if not target.exists(): # 内容寻址 ⇒ 同内容天然去重
165 tmp = target.with_name(target.name + f".tmp-{os.getpid()}")
166 try:
167 tmp.write_bytes(raw)
168 os.replace(tmp, target)
169 except OSError as exc: # noqa: BLE001
170 try:
171 tmp.unlink(missing_ok=True)
172 except OSError:
173 pass
174 # 走**本模块统一拒绝出口**:ATT-001 是 ERR.md §2.11 功能特性码、`errors.py`
175 # **故意不登记**(同 QUE-001/BUSY 先例),须**直接构造**保住字面码 ——
176 # 用 `raise_code("ATT-001")` 会被静默改写为 CYC-999(R19)。
177 _att_reject("store-failed", path=str(target),
178 why=f"{type(exc).__name__}")
179 return str(target)

182def ingest_image_path(path: Any, *, workspace: Any,
183 mime_whitelist: Any = None,
184 max_bytes: Optional[int] = None) -> dict:
185 """从**客户端自报路径**读字节 → 校验 → 内容寻址落盘;返回事件载荷 dict。
186
187 路径不可读 → ``ATT-001``(不静默跳过:调用方需知道附件没进来)。
188 """
189 p = Path(str(path or "")).expanduser()
190 try:
191 raw = p.read_bytes()
192 except OSError as exc:
193 _att_reject("unreadable", path=str(p), why=f"{type(exc).__name__}")
194 info = validate_image(raw, mime_whitelist=mime_whitelist,
195 max_bytes=max_bytes)
196 stored = store_content_addressed(raw, info, workspace=workspace)
197 return {"file_path": stored, "mime": info["mime"], "sha256": info["sha256"],
198 "w": info["w"], "h": info["h"], "size_bytes": info["size_bytes"]}
