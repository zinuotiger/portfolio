1"""pyharness/core/workflow.py — 工作流编排器(#45 Consumer 面)。
2
3一句话职责:把意图序列(编排脚本/LLM 生成的步骤)依序交给任务队列逐条执行,
4每步落 workflow.step 事件(started/done/failed,词表字段级)供 UI 进度与断点续跑。
5
6submit_intent = 外壳注入适配器(desktop/CLI 各自:append user.message →
7TaskQueue.submit → wait_for),本模块不持有执行细节;失败默认继续(记录),
8stop_on_fail=True 则首败即停(编排语义二选一)。

20def queue_submit_adapter(session: Any, queue: Any) -> Callable[[str], Awaitable]:
21 """TaskQueue 适配器:user.message 强同步落盘 → submit → wait_for。
22
23 与 engine runner seam 同构(调用方负责首落盘,队列承载编排),返回
24 (ok: bool, summary: str)。"""

26 async def _run(intent: str, *, workflow_step=None) -> tuple:
27 await session.append("user.message", {"content": intent},
28 actor="user", sync=True)
29 tid = await queue.submit(intent)
30 if workflow_step is not None:
31 await record_workflow_task(session, workflow_step, tid)
32 res = await queue.wait_for(tid)
33 result = (bool(getattr(res, "ok", False)), str(getattr(res, "summary", "") or "完成"))
34 return (*result,tid) if workflow_step is not None else result
35 return _run

41 def __init__(self, session: Any, submit_intent: Callable[[str], Awaitable],
42 *, name: str = "main", bind_steps: bool = False) -> None:
43 self.session = session
44 self._submit = submit_intent
45 self.name = name
46 self.bind_steps = bind_steps

48 async def run(self, steps: list, *, stop_on_fail: bool = False) -> list[dict]:
49 """执行步骤序列;返回 [(index, intent, ok, summary)] 结果集。"""
50 if not steps:
51 raise_code("EVT-100", field="steps", hint="workflow 步骤为空")
52 results: list[dict] = []
53 for idx, intent in enumerate(steps):
54 intent = str(intent or "").strip()
55 if not intent:
56 continue
57 await self._step_event(idx, intent, "started")
58 ok, summary, task_id = False, "", None
59 try:
60 if self.bind_steps:
61 result = await self._submit(intent, workflow_step=(self.name, idx, intent))
62 ok, summary = result[:2]
63 task_id = result[2] if len(result)>2 else None
64 else:
65 ok, summary = await self._submit(intent)
66 except Exception: # noqa: BLE001 单步异常记录
67 # Transport exceptions may contain headers, URLs or credentials.
68 # Detailed diagnostics belong to their existing bounded owner.
69 summary = 'workflow_step_failed'
70 await self._step_event(idx, intent,
71 "done" if ok else "failed", summary[:2000], task_id=task_id)
72 results.append({"index": idx, "intent": intent, "ok": ok,
73 "summary": summary})
74 if stop_on_fail and not ok:
75 break
76 return results
