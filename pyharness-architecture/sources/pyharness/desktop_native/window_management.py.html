242 async def check_interactions(self) -> None:
243 approvals = await self.controller.pending_approvals()
244 for row in approvals.get("pending", []):
245 key = (str(row.get("session_id") or ""), int(row["approval_id"]))
246 if key in self._seen_approvals or str(row.get("session_id") or "") != self.sid:
247 continue
248 text = (f"工具:{row.get('tool')}\n风险:{row.get('risk')}\n参数:{row.get('args_summary')}")
249 ans = QMessageBox.question(self, "工具审批请求", text, QMessageBox.Yes | QMessageBox.No)
250 if ans == QMessageBox.Yes:
251 await self.controller.decide_approval(int(row["approval_id"]), "approve", sid=self.sid)
252 self._seen_approvals.add(key)
253 elif ans == QMessageBox.No:
254 await self.controller.decide_approval(int(row["approval_id"]), "deny", sid=self.sid)
255 self._seen_approvals.add(key)
256
257 asks = await self.controller.pending_asks()
258 for row in asks.get("pending", []):
259 key = (str(row.get("session_id") or ""), int(row["ask_id"]))
260 if key in self._seen_asks or str(row.get("session_id") or "") != self.sid:
261 continue
262 question = str(row.get("question") or "Agent 有问题")
263 options = list(row.get("options") or [])
264 if options:
265 value, ok = QInputDialog.getItem(self, "Agent 提问", question, options, 0, True)
266 choice, text = (value, None) if ok else (None, None)
267 else:
268 value, ok = QInputDialog.getText(self, "Agent 提问", question)
269 choice, text = (None, value) if ok else (None, None)
270 if ok:
271 await self.controller.answer_ask(int(row["ask_id"]), choice=choice, text=text, sid=self.sid)
272 self._seen_asks.add(key)
