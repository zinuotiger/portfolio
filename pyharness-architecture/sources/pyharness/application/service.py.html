356 def attachment_payloads(self, attachments: Any, *, sid: str = "") -> list[dict]:
357 """附件入站**单点校验**(F061):数量 → 魔数/大小 → 内容寻址落盘。
358
359 2026-09-21 修(L-22):此前只做 schema 形状组装(客户端自报 mime、缺省回落
360 `image/png`;无魔数/大小/数量校验;`sha256` 可省略回落全零占位)⇒ **伪装图片的
361 恶意文件可进事件流**,且同内容不去重。现按 PRD-Core F061 规格:魔数定类型
362 (不信扩展名/自报 mime)→ 单张 ≤`max_bytes` → 单消息 ≤`max_per_message`
363 → `sha256(raw)[:16]` 内容寻址去重落盘到**会话工作区内**;任一不满足 →
364 `ATT-001` 零写盘。schema 校验保留为第二道(形状)。
365 """
366 from pyharness.core import attachment as att_mod
367 refs = list(attachments or [])
368 if not refs:
369 return []
370 n_max, b_max, mimes = self._attachment_limits()
371 if len(refs) > n_max:
372 att_mod.reject_attachment("too-many", count=len(refs), limit=n_max)
373 workspace = self._session_workspace(sid)
374 payloads: list[dict] = []
375 for ref in refs:
376 if not isinstance(ref, dict):
377 raise_code("EVT-100", field="attachments", advice="附件须 dict 引用")
378 payload = att_mod.ingest_image_path(
379 ref.get("file_path") or ref.get("ref") or "", workspace=workspace,
380 mime_whitelist=mimes, max_bytes=b_max)
381 payloads.append(validate_payload("user.attachment.image", payload))
382 return payloads
