116class DocumentRevision(VersionOne):
117 document_id: DocumentId
118 revision: int = Field(ge=1, le=200)
119 project_id: ProjectId
120 project_revision: int = Field(ge=1)
121 created_by: str = Field(min_length=1, max_length=128)
122 title: str = Field(min_length=1, max_length=100)
123 source: Annotated[PrivateInputSource | PdfCapturedSource, Field(discriminator='kind')]
124 source_metadata: DocumentSourceMetadata | None = None
125 content_digest: Digest
126 size_bytes: int = Field(ge=0, le=10 * 1024 * 1024)
127 format: Literal['text', 'pdf']
128 parser_version: str = Field(min_length=1, max_length=100)
129 offset_unit: Literal['unicode-codepoint'] = 'unicode-codepoint'
130 page_count: int = Field(ge=0, le=200)
131 unlocatable_pages: list[int] = Field(max_length=200)
132 capability: Literal['locatable', 'unlocatable']
133 capability_reason: Literal['text', 'no_text_layer', 'empty_text', 'encrypted_pdf', 'invalid_pdf', 'unsupported_format']
134 blocks: list[DocumentBlock] = Field(max_length=400)
135 retired: bool = False
136
137 @model_validator(mode='after')
138 def coherent(self):
139 if len({b.block_id for b in self.blocks}) != len(self.blocks):
140 raise ValueError('duplicate block')
141 if sum(len(b.text.encode()) for b in self.blocks) > 98304:
142 raise ValueError('document text limit')
143 if self.unlocatable_pages != sorted(set(self.unlocatable_pages)) or any(not 1 <= p <= self.page_count for p in self.unlocatable_pages):
144 raise ValueError('invalid unlocatable pages')
145 if (self.capability == 'locatable') != bool(self.blocks):
146 raise ValueError('capability/block mismatch')
147 if self.format == 'text' and (self.page_count or self.unlocatable_pages or any(b.page is not None for b in self.blocks)):
148 raise ValueError('plain text has no physical pages')
149 if self.format == 'pdf' and any(b.page is None or b.page > self.page_count or b.page in self.unlocatable_pages for b in self.blocks):
150 raise ValueError('invalid physical page')
151 return self
