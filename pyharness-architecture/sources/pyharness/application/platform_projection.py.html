12def task_events(events, task_id):
13 """Resolve legacy untagged facts only within their explicit segment fence."""
14 active = None
15 for event in events:
16 payload = event.payload
17 if event.type == 'segment.start':
18 active = payload.get('task_id')
19 owner = getattr(event,'task_id',None) or payload.get('task_id')
20 if event.type == 'user.message':
21 owner = (payload.get('meta') or {}).get('platform',{}).get('task_id') or owner
22 if (owner or active) == task_id:
23 yield event
24 if event.type == 'segment.end':
25 active = None

27def error_kind(code):
28 if str(code).startswith("LLM-"): return "model"
29 if str(code).startswith("APR-"): return "approval"
30 if str(code).startswith("TLB-"): return "tool"
31 if "sandbox" in str(code): return "sandbox"
32 if code == "cancelled": return "cancellation"
33 return "execution"

35def finalize(run, status, ts, code=None, diagnostics=None):
36 if run.status in TERMINAL:
37 return
38 run.status, run.finished_at = status, ts
39 run.error_code = code
40 if code:
41 run.error_kind = error_kind(code)
42 run.error_diagnostics = diagnostics if run.error_kind == "model" else None
43 run.error_message = diagnostics["summary"] if diagnostics else "Execution failed (" + run.error_kind + ")"

50def summary(value, limit=800):
51 text = value if isinstance(value, str) else json.dumps(value, ensure_ascii=False, default=str)
52 text = re.sub(r'\bsk-[A-Za-z0-9_-]+', '[redacted]', text)
53 text = re.sub(r'(?i)(api[_-]?key|token|secret|password)\s*[:=]\s*[^\s,;]+', r'\1=[redacted]', text)
54 return redact_text(text)[:limit]

57def subtask_history(session_id, events, active_children=()):
58 """Read delegation facts without restarting children or trusting summaries as instructions."""
59 rows = {}
60 active = {row['sub_id'] for row in active_children}
61 anchors = {}
62 task = None
63 for event in events:
64 p = event.payload
65 if event.type == 'segment.start':
66 task = p.get('task_id')
67 anchors[event.seq] = getattr(event, 'task_id', None) or task
68 if event.type == 'subagent.spawned':
69 child = p['sub_id']
70 parent_task = anchors.get(p.get('parent_seq')) or task
71 rows[child] = {
72 'session_id': child, 'parent_session_id': session_id,
73 'parent_run_id': f'{session_id}~{parent_task}' if parent_task else None,
74 'task': summary(p.get('task', ''), 800), 'status': 'running',
75 'created_at': event.ts, 'finished_at': None, 'summary': '',
76 'error_code': None,
77 }
78 child = p.get('sub_id')
79 if event.type == 'system.cancelled' and str(p.get('what', '')).startswith('subagent:'):
80 child = p['what'].removeprefix('subagent:')
81 if child in rows and rows[child]['status'] == 'running':
82 status = {'subagent.joined':'completed', 'subagent.failed':'failed',
83 'system.cancelled':'cancelled'}.get(event.type)
84 if status:
85 rows[child].update(status=status, finished_at=event.ts,
86 summary=summary(p.get('summary', '')))
87 if event.type == 'segment.end':
88 task = None
89 for child, row in rows.items():
90 if row['status'] == 'running' and child not in active:
91 row.update(status='failed', error_code='runtime_interrupted')
92 return list(rows.values())

118 agent_version=meta.get('agent_version'), retry_count=meta.get('retry_count', 0))

139 if kind == 'executor.event' and p.get('action') == 'turn_completed' and p.get('record',{}).get('status') == 'completed':

143 run.token_usage += p['record'].get('input_tokens',0)+p['record'].get('output_tokens',0)
144 if kind == 'llm.request':

146 last_request[current] = {'request_seq': e.seq, 'model': p.get('model')}
147 if role in (None, 'main_agent'):
148 key = ('model', current, e.seq)
149 steps[key] = RunStep(step_id=f'{session_id}~{e.seq}', run_id=run.run_id,
150 name='model request', status='running', started_at=e.ts)
151 model_steps[current] = key
152 elif kind == 'llm.response' and role not in ('title_generation', 'probe'):

154 step = steps.get(key)
155 if step and step.status == 'running':
156 step.status, step.finished_at = 'completed', e.ts
157 if kind == 'llm.error' and role == 'main_agent':
158 step = steps.get(('model', current, diag.get('request_seq')))
159 if step and step.status == 'running':
160 step.status, step.finished_at, step.error_code = 'failed', e.ts, p.get('code')
161 if fatal and role not in ('title_generation', 'probe', 'other') and run.status not in TERMINAL:
162 d = public_diagnostics(diag, code=p.get('code'), role='main_agent', **last_request.get(current, {}))

165 if step is None:

168 run_id=run.run_id, name='main execution', status='running', started_at=e.ts))
169 if step.status not in ('failed', 'cancelled'):
170 step.status, step.finished_at, step.error_code = 'failed', e.ts, p.get('code')
171 if kind == 'task.started' and run.status not in TERMINAL:
172 run.status, run.started_at = 'running', e.ts
173 elif kind == 'task.completed':
174 if current in successful:
175 finalize(run, 'completed', e.ts)
176 elif kind == 'task.failed':
177 code = p.get('error') or 'unknown'
178 finalize(run, 'cancelled' if p.get('reason') == 'cancelled' else 'failed', e.ts, code,
179 public_diagnostics(code=code, role='main_agent'))
180 elif kind == 'approval.requested' and run.status not in {'completed','failed','cancelled'}:
181 run.status = 'waiting_approval'
182 elif kind in ('approval.granted', 'approval.denied', 'approval.timeout') and run.status not in {'completed','failed','cancelled'}:
183 run.status = 'running'
184 elif kind == 'queue.suspended' and run.status not in {'waiting_approval','completed','failed','cancelled'}:
185 run.status = 'paused'
186 elif kind == 'queue.resumed' and run.status not in {'completed','failed','cancelled'}:

188 elif kind == 'llm.usage':

194 run.estimated_cost = (run.estimated_cost or 0) + p['cost_est']

206 step.status='waiting_approval'
207 elif kind in ('approval.granted', 'approval.denied', 'approval.timeout'):

213 step.status='running' if kind=='approval.granted' else 'failed'
214 if kind == 'approval.reviewed' and p['approval_id'] in approvals:
215 approvals[p['approval_id']]['review'] = p
216 cid = call_id_of(e)
217 if kind == 'tool.call' and cid and cid not in steps:
218 steps[cid] = RunStep(step_id=f'{session_id}~{e.seq}',
219 run_id=run.run_id if run else '', name=p.get('name') or p.get('tool') or 'tool', status='running', started_at=e.ts)
220 elif kind in ('tool.result', 'tool.error') and cid in steps and steps[cid].status not in TERMINAL:
221 step = steps[cid]
222 step.status = 'completed' if kind == 'tool.result' and p.get('ok') else 'failed'
223 step.finished_at = e.ts
224 if (kind == 'system.error' or kind.startswith(('task.', 'executor.', 'tool.', 'llm.', 'approval.', 'decision.', 'platform.', 'agent.'))) and kind not in ('llm.chunk',):
225 # Never expose llm.request/response bodies or private reasoning fields.
226 public = {k: p[k] for k in ('name','code','reason','ok','model','in_tokens','out_tokens','cost_est') if k in p}
227 if kind == 'executor.event':
228 public = {'executor':p['executor'],'action':p['action'],
229 'record':redact(p['record'])}
230 if kind == 'platform.artifact':

233 if kind=='platform.sandbox':
234 record=p.get('record',{})

236 public.update({k:p[k] for k in ('action','command_summary','exit_code') if k in p})
237 if kind in ('llm.error', 'system.error'):
238 public = public_diagnostics(diag, code=p.get('code'), role=role or 'main_agent')
239 span=TraceSpan(span_id=f'{session_id}~{e.seq}', run_id=run.run_id if run else None,
240 type=kind, name=str(p.get('name') or kind), status=('warning' if role in ('title_generation','probe','other') else 'failed') if kind.endswith('error') else 'recorded',
241 parent_span_id=f"{session_id}~{e.trace['parent_seq']}" if getattr(e,'trace',None) and e.trace.get('parent_seq') else None,
242 started_at=e.ts, output_summary=summary(public), error_code=p.get('code'),
243 error_diagnostics=public if kind in ('llm.error','system.error') else None,

249 span.status='running'

261 for run in runs.values():

263 run.usage_unknown_requests = max(0, requests.get(run.task_id, 0) - usage_counts.get(run.task_id, 0))
264 run.usage_status = ('unknown' if run.usage_unknown_requests or not usage_counts.get(run.task_id)

270 if run.status in {'cancelled','failed'}:

272 if step.status in {'pending','running','waiting_approval'}:
273 step.status='cancelled' if run.status=='cancelled' else 'failed'
274 step.finished_at = run.finished_at
275 step.error_code = run.error_code
276 selected = [s for s in spans if trace_run_id is None or s.run_id == trace_run_id]

280 'trace': [s.model_dump() for s in selected],
