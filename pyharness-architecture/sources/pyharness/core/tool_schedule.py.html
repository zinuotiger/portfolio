91class _ScheduleHandle:
92 """Provider handle:op 分派到 ctx.schedule(现有 Scheduler)。"""

94 async def handle(self, args: dict, ctx: Any) -> str:
95 sched = getattr(ctx, "schedule", None)
96 if sched is None:
97 raise_code("CYC-999", module="tool_schedule",
98 hint="ctx.schedule 未装配(引擎未挂 Scheduler),schedule 不可用")
99 op = str(args.get("op") or "").strip()
100 if op == "create":
101 return await self._create(sched, args, ctx)
102 if op == "list":
103 return await self._list(sched, ctx)
104 # 显式分派表(不用 getattr(op) —— 外部输入不得直接选方法,关 4/CND-04)
105 fn = {"pause": sched.pause, "resume": sched.resume,
106 "remove": sched.remove}.get(op)
107 if fn is None:
108 raise_code("EVT-100", field="op",
109 advice=f"op ∈ {'/'.join(_OPS)}")
110 name = _require_name(args, op)
111 await fn(name, ctx=ctx)
112 return _dump({"ok": True, "op": op, "name": name})

114 async def _create(self, sched: Any, args: dict, ctx: Any) -> str:
115 """建任务:参数完整性检查后交 Scheduler.register(校验/持久化归其所有)。"""
116 name = _require_name(args, "create")
117 missing = [f for f in ("kind", "expr", "intent")
118 if not str(args.get(f) or "").strip()]
119 if missing:
120 raise_code("EVT-100", field=missing[0],
121 hint=f"create 缺必填参数:{', '.join(missing)}",
122 advice=_MISSING_ADVICE)
123 kind = str(args["kind"]).strip()
124 expr = str(args["expr"]).strip()
125 intent = str(args["intent"]).strip()
126 # is_risky **单调**:只能升级为危险,不能下调。本工具不传 meta(INV-09),
127 # 故非显式声明时 Scheduler 的推导恒为 False;若允许传 False,等于让调用方
128 # 单方面关掉"深夜禁触危险模板"这一安全默认 → 故 False/缺省一律交回推导。
129 override = True if args.get("is_risky") is True else None
130 await sched.register(name, kind, expr, {"intent": intent},
131 is_risky=override, ctx=ctx)
132 info = next((j for j in await sched.list_jobs(ctx=ctx)
133 if j.name == name), None)
134 return _dump({"ok": True, "op": "create", "name": name, "kind": kind,
135 "expr": expr, "paused": False,
136 "next_fire_at": _iso(info.next_fire_at) if info else None})

138 async def _list(self, sched: Any, ctx: Any) -> str:
