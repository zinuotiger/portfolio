1"""pyharness/core/proc.py — 子进程会话管理(#58 exec 沙箱族后端)。
2
3一句话职责:会话级子进程登记表——start 拉起(沙箱 cwd/裁剪 env/隐藏窗口)、
4status 轮询读尾部、kill 结束;进程树按 session_id 隔离(会话关闭即清)。
5
6应用层约束(诚实口径,见 tool_exec docstring):cwd/env/输出配额/超时 +
7CREATE_NO_WINDOW。**这不是安全沙箱**:命令可绕过 cwd 并访问网络/绝对路径;
8安全边界由 high 风险审批承担,恶意代码必须使用 OS 级隔离。
9PTY 语义:Windows 无内建 pty,行式会话(stdin 写/stdout 读)为等价降级。
10
11线程安全:每会话一个 asyncio.Lock;输出经后台线程读入 deque(reader daemon),
12不阻塞事件循环;进程对象只在 _PROCS 内可达(会话结束 close_session 清理)。
13"""

51 def __init__(self, token: str, proc: subprocess.Popen, *, tail: deque,
52 reader: threading.Thread,
53 timer: Optional[threading.Timer] = None) -> None:
54 self.token = token

61 @property

65 def status(self) -> dict:
66 owner = getattr(self, 'owner', None)
67 rc = owner.poll() if owner is not None else self.proc.poll()
68 return {
69 "pid": self.pid,
70 "token": self.token,
71 "running": rc is None,
72 "exit_code": rc,
73 "chars_total": self.chars_total,
74 "tail_lines": len(self.tail),
75 "tail": list(self.tail)[-20:],
76 }

101def start_session(sid: str, sandbox: Path, command: str, *,
102 timeout_total_s: int) -> ProcSession:
103 """沙箱内拉起子进程(shell 语义:cmd /c;POSIX sh -c)。"""
104 sandbox.mkdir(parents=True, exist_ok=True)
105 if os.name == "nt":
106 argv = ["cmd", "/d", "/s", "/c", command]
107 else:
108 argv = ["/bin/sh", "-c", command]
109 from pyharness.core.process_owner import ProcessOwner
110 owner = ProcessOwner(argv, cwd=sandbox, env=sandbox_env(sandbox),
111 text=True, merge_stderr=True, input_open=True)
112 proc = owner.proc
113 tail: deque = deque(maxlen=MAX_TAIL_LINES)
114 sess = ProcSession(token=f"{sid}-{proc.pid}", proc=proc, tail=tail,
115 reader=None)
116 reader = threading.Thread(target=_pump, args=(proc, tail, sess),
117 daemon=True, name=f"proc-{sid}-{proc.pid}")
118 sess.owner = owner
119 sess.reader = reader
120 timer = threading.Timer(max(1, int(timeout_total_s)), _kill_tree, args=(sess,))
121 timer.daemon = True
122 sess.timer = timer
123 def reap():
124 try:
125 owner.wait_exit()
126 owner.terminate()
127 except BaseException as exc:
128 sess.cleanup_error = exc
129 watcher = threading.Thread(target=reap, daemon=True, name=f"proc-owner-{sid}")
130 sess.watcher = watcher
131 started = []
132 try:
133 for worker in (reader, timer, watcher):
134 worker.start()
135 started.append(worker)
136 _table().setdefault(sid, {})[sess.token] = sess
137 except BaseException as primary:
138 timer.cancel()
139 try: owner.terminate()
140 except BaseException as exc: primary.add_note(f'process cleanup: {type(exc).__name__}')
141 for worker in started:
142 try: worker.join(timeout=5)
143 except BaseException as exc: primary.add_note(f'thread cleanup: {type(exc).__name__}')
144 for stream in (proc.stdin, proc.stdout, proc.stderr):
145 if stream is not None:
146 try: stream.close()
147 except BaseException as exc: primary.add_note(f'pipe cleanup: {type(exc).__name__}')
148 raise
149 return sess

152def _pump(proc: subprocess.Popen, tail: deque, sess: ProcSession) -> None:
153 """后台读线程:行追加(配额内),进程退出后补读残余并保底清 stdin。"""
154 try:
155 assert proc.stdout is not None
156 for line in proc.stdout:
157 tail.append(line.rstrip("\n")[-4000:])
158 sess.chars_total += len(line)
159 if sess.chars_total > MAX_TAIL_CHARS:
160 _kill_tree(sess)
161 tail.append("[输出超配额,进程已终止]")
162 break
163 except Exception: # noqa: BLE001 读线程兜底
164 tail.append("[输出读异常]")
165 finally:
166 if sess.timer is not None:
167 sess.timer.cancel()
168 for stream in (proc.stdout, proc.stderr):
169 if stream is not None:
170 try:
171 stream.close()
172 except OSError:
173 pass
174 try:
175 proc.stdin.close()
176 except OSError:
177 pass

180def find_session(sid: str, token_or_pid: Any) -> ProcSession:
181 """定位会话(token 或裸 pid);缺失 → TLB-802(给 LLM 明确指引)。"""
182 table = _table().get(sid) or {}
183 if isinstance(token_or_pid, str) and token_or_pid in table:
184 return table[token_or_pid]
185 for s in table.values():
186 if str(s.pid) == str(token_or_pid):
187 return s
188 raise_code("TLB-802", module="proc",
189 hint=f"会话 {token_or_pid} 不存在或不属于本会话(sid={sid})")

192def kill_session(sid: str, token_or_pid: Any) -> dict:
193 """结束进程(树级:Windows taskkill /T 兜底;POSIX 组)。"""
194 sess = find_session(sid, token_or_pid)

199def _kill_tree(sess: ProcSession) -> None:
200 """结束父进程及其子进程;失败再退化到父进程 kill。"""
201 if sess.timer is not None:
202 sess.timer.cancel()
203 owner = getattr(sess, "owner", None)
204 if owner is not None:
205 owner.terminate()
206 elif sess.proc.poll() is None:
207 # Legacy injected process without a tree capability: only its live handle.
208 sess.proc.kill()
209 sess.proc.wait(timeout=5)

217def close_session(sid: str) -> int:
218 """会话收尾:结束全部子进程,清登记(agent close 时调用)。返回清理数。"""
219 table = _table().get(sid, {})
220 n = 0
221 errors = []
222 for token, sess in list(table.items()):
223 n += 1
224 item_errors = []
225 if sess.timer is not None:
226 sess.timer.cancel()
227 try: _kill_tree(sess)
228 except BaseException as exc: item_errors.append(exc)
229 for worker in (sess.reader, sess.timer, getattr(sess, 'watcher', None)):
230 if worker is not None and worker is not threading.current_thread():
231 try:
232 worker.join(timeout=5)
233 if worker.is_alive(): raise RuntimeError('owned process worker did not stop')
234 except BaseException as exc: item_errors.append(exc)
235 if not item_errors: table.pop(token, None)
236 errors.extend(item_errors)
237 if not table: _table().pop(sid, None)
238 if errors: raise BaseExceptionGroup('process session cleanup failed', errors)
239 return n

258async def wait_result_async(sandbox: Path, command: str, *, timeout_s: int, python: bool = False) -> dict:
259 """Cancellation stops and reaps this owned tree before propagating cancellation."""
260 from pyharness.core.process_owner import ProcessOwner
261 sandbox.mkdir(parents=True, exist_ok=True)
262 owner = ProcessOwner(_argv(command, python), cwd=sandbox, env=sandbox_env(sandbox))
263 collector = asyncio.create_task(asyncio.to_thread(owner.collect, timeout_s))
264 try:
265 return await asyncio.shield(collector)
266 except BaseException as primary:
267 async def cleanup(error=primary):
268 try: await asyncio.to_thread(owner.terminate)
269 except BaseException as exc: error.add_note(f'process cleanup: {type(exc).__name__}')
270 try: await collector
271 except BaseException as exc: error.add_note(f'collector cleanup: {type(exc).__name__}')
272 cleanup_task = asyncio.create_task(cleanup())
273 while not cleanup_task.done():
274 try: await asyncio.shield(cleanup_task)
275 except asyncio.CancelledError: continue
