3Reference commit 874637a0dbc24050f72d1385add4fc6170a2d726, Tencent/openclaw-weixin.
4This is an independent Python implementation of the public protocol. It never
5reads another client's login files, obtains a password, or exports chat history.

25def official_url(url: str) -> str:
26 parsed = urlsplit(url)
27 host = parsed.hostname or ''
28 if (parsed.scheme != 'https' or parsed.username or parsed.password or parsed.port not in (None,443)
29 or parsed.query or parsed.fragment or parsed.path not in ('','/')
30 or not (host == 'ilinkai.weixin.qq.com' or host.endswith('.ilinkai.weixin.qq.com'))):
31 raise ILinkError('wechat_untrusted_endpoint')
32 return url.rstrip('/')+'/'

48class ILinkClient:
49 def __init__(self, *, base_url=BASE, token='', transport=None):
50 self.base_url = official_url(base_url)
51 self.token = token
52 self.http = httpx.AsyncClient(timeout=40, follow_redirects=False, transport=transport)
53
54 def headers(self):
55 return {'Content-Type':'application/json','AuthorizationType':'ilink_bot_token',
56 'Authorization':'Bearer '+self.token if self.token else '',
57 'X-WECHAT-UIN':base64.b64encode(str(secrets.randbits(32)).encode()).decode(),
58 'iLink-App-Id':'bot','iLink-App-ClientVersion':str((2<<16)|(4<<8)|10)}
59
60 async def post(self, path, body, *, timeout=15):
61 response = await self.http.post(self.base_url+path, headers=self.headers(),
62 json={**body,'base_info':{'channel_version':VERSION,'bot_agent':'PyHarness/0.1.0'}},timeout=timeout)
63 response.raise_for_status()
64 value = response.json()
65 code = value.get('errcode') or value.get('ret') or 0
66 if code:
67 raise ILinkError('wechat_session_expired' if code == -14 else 'wechat_api_'+str(code))
68 return value

70 async def login_qr(self):
71 return await self.post('ilink/bot/get_bot_qrcode?bot_type=3', {'local_token_list':[]})

73 async def login_status(self, qrcode, *, verification_code=None):
74 response = await self.http.get(self.base_url+'ilink/bot/get_qrcode_status',
75 headers={k:v for k,v in self.headers().items() if k.startswith('iLink-')},
76 params={'qrcode':qrcode,**({'verify_code':verification_code} if verification_code else {})},timeout=38)
77 response.raise_for_status()
78 return response.json()

83 async def send(self, user, context_token, text, ident):
