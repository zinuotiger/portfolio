117 def _register_internal_types(cls, bus: "EventBus") -> None:
118 """tool.registered 为词表外瞬时事件(见模块 docstring 偏离 1)。
119

126 return

135 cls._registered_buses[bus] = None

154 def _validate_contract(self, defn: Any) -> None:
155 """注册前置契约校验(T-05 五要素/T-07;失败 → TLB-801)。
156
157 覆盖伪码之外的字段级规则:description 非空 ≤200 且无注入特征(T-07)、
158 owner 必填、danger 合法;名称 regex 与 approval×danger 一致性也在
159 register_tool 内直查(伪码顺序),此处为集中收口(鸭子 defn 与
160 ToolDefinition 实例同谓词)。
161 """
162 name = getattr(defn, "name", None)
163 danger = getattr(defn, "danger", None)
164 desc = getattr(defn, "description", None)
165 owner = getattr(defn, "owner", None)
166 approval = getattr(defn, "approval", "auto")
167
168 if not isinstance(name, str) or not NAME_RE.fullmatch(name):
169 raise_code("TLB-801", tool=name, rule=NAME_RE.pattern,
170 why="名称非法(小写字母开头,仅 a-z0-9_. ,长 2~64)")
171 if danger not in DANGER_LEVELS:
172 raise_code("TLB-801", tool=name, danger=danger,
173 why="缺 danger 或非法危险级(T-05 五要素之一)")
174 if approval == "never" and danger not in ("none", "low"):
175 raise_code("TLB-801", tool=name,
176 why="approval=never 只允许 danger ∈ none/low")
177 if not isinstance(desc, str) or not desc.strip():
178 raise_code("TLB-801", tool=name, why="缺 description(T-05 五要素之一)")
179 if len(desc) > 200:
180 raise_code("TLB-801", tool=name, length=len(desc),
181 why="description 超 200 字上限")
182 low = desc.lower()
183 hit = next((m for m in INJECTION_MARKERS if m.lower() in low), None)
184 if hit is not None:
185 raise_code("TLB-801", tool=name, marker=hit,
186 why="description 含注入特征(T-07 拒)")
187 if not isinstance(owner, str) or not owner.strip():
188 raise_code("TLB-801", tool=name, why="缺 owner(能力/插件 id 必填)")
189

191 def register_tool(self, defn: Any, *, provider: Any = None) -> str:
192 """工具登记(F008):查重 → 契约校验 → schema 编译(TLB-803 早失败)→ 入表
193 → 绑 Provider → 事件留痕。返回工具名。
194
195 拒绝路径(重名/保留名/非法名/矛盾契约/坏 schema)注册表零变更、不发
196 tool.registered(GWT-T7-01:无脏数据)。Definition 入表即不可变(frozen)。
197 """
198 name = getattr(defn, "name", "")
199 if name in self._tools or name in self.RESERVED:
200 raise_code("TLB-801", tool=name,
201 advice="重名或脊柱保留名(BUS-002);注销旧定义后重注册")
202 self._validate_contract(defn) # T-05/T-07 集中收口
203
204 schema = getattr(defn, "schema", None)
205 try:
206 model = _compile_model(name, schema) # F026 编译期校验(早失败)
207 except Exception as e: # noqa: BLE001 编译器内聚,
208 log.debug("tools_registry schema 编译失败 name=%s: %s", name, e)
209 raise_code("TLB-803", tool=name, why="schema 不可编译")
210 self._tools[name] = defn # Definition frozen,入表不可变
211 self._models[name] = model
212 if provider is not None:
213 self._providers[name] = provider # 绑定实现(可后补/换绑)
214 self._emit(TOOL_REGISTERED, {
215 "name": name, "danger": getattr(defn, "danger", None),
216 "owner": getattr(defn, "owner", None),
217 "hooks": list(getattr(defn, "guard_hooks", ()))})
218 self._emit(REGISTRY_UPDATED, {"op": "add", "kind": "tool", "key": name})
219 return name

229 def require_approval(self, name: str) -> None:
230 """Tighten a registered contract without exposing or replacing its Provider.
231
232 The original frozen definition remains immutable. Only governance fields
233 of its replacement change; schema, compiled validator and binding stay
234 intact. Critical tools remain critical (and therefore denied).
235 """
236 old = self.lookup(name)
237 if name in self._running:
238 _raise_busy(tool=name, advice="有在途调用,稍后重试")
239 data = old.model_dump()
240 data['schema'] = data.pop('schema_')
241 data.update(danger='critical' if old.danger == 'critical' else 'high',
242 approval='always')
243 replacement = ToolDefinition.model_validate(data)
244 self._validate_contract(replacement)
245 self._tools[name] = replacement
246 self._emit(REGISTRY_UPDATED, {'op': 'tighten', 'kind': 'tool', 'key': name,
247 'danger': replacement.danger, 'approval': 'always'})

367 按 scope 可见性过滤(scope.can_use(name)==False 者不下发——LLM 拿到的
368 是当前 scope/变体下注定能过 scope 关的集合,减少注定被拒的无效调用)
369 + toolset 子集过滤(子 agent 变体 F049)+ 确定性排序(同输入同输出,
370 可缓存可测)。scope=None(装配期未接线)→ 跳过 scope 过滤(偏离 6)。
371 """
372 out: list[dict] = []
373 for name in sorted(self._tools): # 确定性排序
374 if scope is not None and not scope.can_use(name):
375 continue # scope 不可见 → 不下发
376 if toolset is not None and name not in toolset:
377 continue # agent 变体子集外
378 out.append(self.to_openai_schema(self._tools[name]))
379 return out
380
381 def to_openai_schema(self, defn: Any) -> dict:
382 """Definition → OpenAI function schema(描述/参数;不泄露 danger/
383 guard_hooks/approval 给 LLM——安全元数据只在管道内消费)。"""
384 return {"type": "function",
385 "function": {"name": defn.name,
386 "description": defn.description,
387 "parameters": defn.schema}}
