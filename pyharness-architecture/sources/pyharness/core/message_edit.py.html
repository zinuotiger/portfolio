19def _edit_window_s(session: Any) -> int:
20 """编辑窗秒数(功能键 `loop.message_edit_window_s`;取不到 → L1 默认 1800)。
21
22 R24:该键此前**零读取者**(L-20 死配置)⇒ 文档承诺的 `EDT-001`「编辑超 30 分钟
23 窗拒写」在实现里**根本不存在**。现接到唯一编辑入口。
24 """
25 from pyharness.config import DEFAULTS
26
27 default = int(DEFAULTS["loop"]["message_edit_window_s"])
28 for holder in (getattr(session, "_cfg", None),
29 getattr(session, "settings", None),
30 getattr(session, "ctx", None)):
31 loop = getattr(holder, "loop", None) if holder is not None else None
32 win = getattr(loop, "message_edit_window_s", None)
33 if win is not None:

38def _age_s(ts: Any) -> Optional[float]:
39 """事件 ts(UTC ISO 毫秒 Z)距今秒数;不可解析 → None(不拦,如实降级)。"""
40 raw = str(ts or "").strip()
41 if not raw:
42 return None
43 try:
44 when = datetime.fromisoformat(raw.replace("Z", "+00:00"))
45 except ValueError:
46 return None
47 if when.tzinfo is None:
48 when = when.replace(tzinfo=timezone.utc)
49 return (datetime.now(timezone.utc) - when).total_seconds()

52async def edit_user_message(session: Any, target_seq: int,
53 new_content: str) -> Any:
54 """修正一条 user.message(编辑):追加 edited 事件,强同步落盘。
55
56 target 必须是已落盘 user.message(EVT-101);空内容 EVT-100;**超出编辑窗
57 `loop.message_edit_window_s`(默认 1800s)→ `EDT-001` 拒写**(R24 接线;此前该
58 窗只有文档承诺、实现缺失)。原文仍在日志(append-only),审计可重建。
59 """
60 if not str(new_content or "").strip():
61 raise_code("EVT-100", field="new_content", hint="修正内容不能为空")
62 ev = session.get(int(target_seq))
63 if ev is None or ev.type != "user.message":
64 raise_code("EVT-101", seq=int(target_seq),
65 hint="目标不是可编辑的 user.message"
66 "(编辑面=用户消息;投影修正由 _fold_history 应用)")
67 window = _edit_window_s(session)
68 age = _age_s(getattr(ev, "ts", None))
69 if window > 0 and age is not None and age > window:
70 raise PyHError(EDIT_WINDOW_CODE, ctx={
71 "seq": int(target_seq), "age_s": round(age, 3), "window_s": window,
72 "advice": f"超过编辑窗 {window}s(loop.message_edit_window_s);"
73 "如需改历史请新发一条用户消息"})
74 return await session.append("user.message_edited",
75 {"target_seq": int(target_seq),
76 "new_content": str(new_content)},
77 actor="user", sync=True)

80def last_user_message(session: Any) -> Optional[dict]:
81 """最近一条 user.message 摘要(重发用;只读,无状态)。"""
82 for ev in reversed(list(session.events_after(0))):
