35async def configuration_body(request):
36 raw = bytearray()
37 async for chunk in request.stream():
38 if len(raw)+len(chunk) > 2048:
39 raise HTTPException(422, 'configuration_request_limit')
40 raw.extend(chunk)
41 try:
42 value = json.loads(raw)
43 except (ValueError, UnicodeError, RecursionError):
44 raise HTTPException(422, 'configuration_invalid_request') from None
45 if type(value) is not dict:
46 raise HTTPException(422, 'configuration_invalid_request')
