10def codex_completeness(events,message):
11 """Codex has public turn facts rather than an OpenAI llm.response trace.
12
13 Require the same explicit task/turn, a confirmed terminal and exact public
14 bytes. Neither a missing finish reason nor an unrelated turn is success.
15 """
16 task=getattr(message,'task_id',None)
17 if not task:return 'unknown'
18 facts=[e for e in events if e.seq<message.seq and e.type=='executor.event'
19 and getattr(e,'actor',None)=='system' and getattr(e,'task_id',None)==task
20 and e.payload.get('executor')=='codex']
21 starts=[e for e in facts if e.payload['action']=='turn_started']
22 if not starts:return 'unknown'
23 start=starts[-1];turn=start.payload['record'].get('turn_id')
24 terminals=[e for e in facts if e.seq>start.seq and e.payload['action']=='turn_completed'
25 and e.payload['record'].get('turn_id')==turn]
26 if not turn or not terminals or terminals[-1].payload['record'].get('status')!='completed':return 'unknown'
27 end=terminals[-1]
28 texts=[e.payload['record'].get('text') for e in facts if start.seq<e.seq<end.seq
29 and e.payload['action']=='item_completed' and e.payload['record'].get('type')=='agentMessage']
30 texts=[s for s in texts if isinstance(s,str) and s]
31 content=message.payload['content']
32 if texts and (content in texts or content=='\n\n'.join(texts)):return 'complete'
33 if any(len(s)>32000 and content==s[:32000] for s in texts):return 'incomplete'
34 return 'unknown'

37def quality_projection(events):
38 messages=[e for e in events if e.type=='agent.message']
39 errors=[e for e in events if e.type=='system.error']
40 from pyharness.core.budget_diagnostics import public_budget
41 budget=next((public_budget(e.payload.get('ctx',{}).get('diagnostics')) for e in reversed(errors)
42 if e.payload.get('code')=='LLM-305'),None)
43 base=dict(output_completeness='unknown',finish_reason='unknown',calculation_status='not_configured',
44 delivery_readiness='needs_review',budget_diagnostics=budget,effective_output_limit=None,
45 configured_output_limit=None,calculation=None)
46 invalid=[e for e in events if e.type=='platform.resource' and e.payload.get('action')=='order_review_invalid']
47 if invalid:base.update(calculation_status='invalid_input',calculation=invalid[-1].payload['record'])
48 if not messages:return base
49 message=messages[-1];parent=(message.trace or {}).get('parent_seq')
50 response=next((e for e in events if e.seq==parent and e.type=='llm.response'),None)
51 if response is not None:
52 reason=response.payload['finish_reason'];base.update(finish_reason=reason,output_completeness=response_completeness(reason))
53 request=next((e for e in events if e.seq==response.payload.get('request_seq') and e.type=='llm.request'),None)
54 if request:base.update(effective_output_limit=request.payload.get('effective_output_limit'),
55 configured_output_limit=request.payload.get('configured_output_limit'))
56 else:
57 completeness=codex_completeness(events,message)
58 if completeness!='unknown':base.update(output_completeness=completeness,
59 finish_reason='executor_completed' if completeness=='complete' else 'executor_public_text_clipped')
60 calculation=(message.payload.get('quality') or {}).get('calculation')
61 if calculation:base.update(calculation_status=calculation['status'],calculation=calculation)
62 if base['output_completeness']=='complete' and base['calculation_status'] in ('passed','not_configured'):
63 base['delivery_readiness']='ready_for_independent_review'
64 return base
