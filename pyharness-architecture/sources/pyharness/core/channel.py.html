1"""pyharness/core/channel.py — 通道解析唯一实现(Channel Contract;N5)。
2
3**为什么要这一层**:修复前"通道"在 6 层被**独立解释**了 5 次
4(Service 的 ``or`` 合并、Engine 的硬编码默认、Approval 的**精确名**匹配 + 回落、
5Governance 的 ``hasattr`` 哨兵 + ``or`` 合并),导致同一 ctx 状态在不同层得到
6**互相矛盾**的判定。实测反例:``"acp:client-7"`` 在 ``approval._ensure_channel``
7被判为 headless,而在同一文件的 ``_require_human`` 被接受;``""`` 在治理层静默
8降级为 ``system``。
9
10本模块给出**唯一**的"原始值 → 语义"映射。核心层(approval/service/engine)直接
11消费它;治理层因 ADR-018:308 依赖方向禁令**不得** import ``pyharness.core.*``,
12故经其**已有的** ``Principal.from_legacy_by`` 分类消费(该函数已按前缀正确识别
13``acp:<id>``),并由 ``tests/invariants`` 的真值表断言二者**逐状态一致**
14(沿用 ``decision.HUMAN_CHANNELS`` 的"受守卫的重复"先例)。
15
16**五种状态的冻结语义**(任何改动都是契约变更):
17
18=========== ========================================== ==========================
19状态 判定条件 语义
20=========== ========================================== ==========================
21MISSING ``ctx`` 无 ``channel`` 属性(从未声明) **fail-closed(APR-503)**
22HEADLESS ``ctx.channel is None``(显式声明无人类通道) 合法:SYSTEM 主体,无审批通道
23INTERACTIVE ``"<name>"`` 或 ``"<name>:<id>"``, 合法:人类通道,可发起审批
24 ``name ∈ HUMAN_CHANNELS``
25INVALID 非空字符串但 ``name ∉ HUMAN_CHANNELS`` **fail-closed(APR-503)**
26UNKNOWN 空串 ``""``(或非字符串) **fail-closed(APR-503)**
27=========== ========================================== ==========================
28
29**归一化规则(关键)**::``"acp:client-7"`` → ``name="acp"``, ``ident="client-7"``
30—— **保留 ACP 身份**,既**不降级为 desktop**,也**不降级为 headless**。
31"""

48class ChannelKind(StrEnum):
49 """通道原始值的五分类(见模块 docstring 的冻结语义表)。"""
50
51 MISSING = "missing" # 属性缺失 ⇒ fail-closed
52 HEADLESS = "headless" # 显式 None ⇒ 合法无通道
53 INTERACTIVE = "interactive" # 合法人类通道
54 INVALID = "invalid" # 非空串且前缀不在白名单 ⇒ fail-closed
55 UNKNOWN = "unknown" # 空串/非字符串 ⇒ fail-closed
56

58@dataclass(frozen=True)
59class ChannelState:
60 """解析结果(不可变)。``raw`` 保留原值以便报错时如实呈现。"""
61
62 kind: ChannelKind
63 name: Optional[str] = None # 归一化通道名(INTERACTIVE 时非 None)
64 ident: Optional[str] = None # 主体 id(``acp:c7`` → ``"c7"``;裸名 → 同 name)
65 raw: Any = None
66
67 @property
68 def is_interactive(self) -> bool:
69 return self.kind is ChannelKind.INTERACTIVE
70
71 @property
72 def is_headless(self) -> bool:
73 return self.kind is ChannelKind.HEADLESS
74
75 @property
76 def is_acceptable(self) -> bool:
77 """可接受 = 合法状态(HEADLESS 或 INTERACTIVE)。其余三种 fail-closed。"""
78 return self.kind in (ChannelKind.HEADLESS, ChannelKind.INTERACTIVE)

81def normalize_channel(raw: Any) -> ChannelState:
82 """原始值 → ``ChannelState``(**纯函数,无副作用,不抛**)。
83
84 调用方按 ``kind`` 自行决定 fail-closed,使本函数可在需要"分类而非拒绝"的
85 场景(如装配期留痕)复用。
86 """
87 if raw is None:
88 return ChannelState(ChannelKind.HEADLESS, raw=None)
89 if not isinstance(raw, str):
90 return ChannelState(ChannelKind.UNKNOWN, raw=raw)
91 if raw == "":
92 return ChannelState(ChannelKind.UNKNOWN, raw=raw)
93 name, sep, ident = raw.partition(":")
94 if name not in HUMAN_CHANNELS:
95 return ChannelState(ChannelKind.INVALID, raw=raw)
96 if sep and ident == "":
97 # ``"desktop:"`` —— 前缀后**缺 id**:与 ``governance.decision.Principal.
98 # from_legacy_by`` 的判据一致(它对此抛 APR-503)。不对称会让同一 ctx 状态
99 # 在核心层与治理层得到相反判定,正是本契约要消除的分歧。
100 return ChannelState(ChannelKind.INVALID, raw=raw)
101 # ``"acp:client-7"`` → name="acp", ident="client-7";裸 ``"cli"`` → ident=name
102 return ChannelState(ChannelKind.INTERACTIVE, name=name,
103 ident=(ident if sep else name), raw=raw)

106def resolve_channel(ctx: Any) -> ChannelState:
107 """由 ``ctx`` 解析通道状态(处理"属性缺失"这一维)。"""
108 if not hasattr(ctx, "channel"):
109 return ChannelState(ChannelKind.MISSING, raw=None)
110 return normalize_channel(getattr(ctx, "channel", None))

113def require_channel(ctx: Any, *, module: str, field: str = "channel") -> ChannelState:
114 """解析并**对不可接受状态 fail-closed**(MISSING / INVALID / UNKNOWN → APR-503)。
115
116 合法状态原样返回:``HEADLESS``(显式无通道)与 ``INTERACTIVE``(人类通道)。
117 """
118 st = resolve_channel(ctx)
119 if st.is_acceptable:
120 return st
121 why = {
122 ChannelKind.MISSING: "ctx.channel 未声明:通道身份缺失,禁止静默降级为 system"
123 "(fail-closed)。外壳须显式声明 "
124 "channel='cli'|'web'|'acp:<id>'|'desktop',或显式声明 "
125 "headless(channel=None)。",
126 ChannelKind.INVALID: f"通道名非法(非白名单前缀):{st.raw!r};"
127 f"合法前缀 {HUMAN_CHANNELS}(fail-closed,不静默降级)",
128 ChannelKind.UNKNOWN: "通道值为空串/非字符串:视为未知通道,"
129 "禁止 falsy 回退改写声明语义(fail-closed)",
130 }[st.kind]
131 raise_code("APR-503", module=module, field=field, got=repr(st.raw), why=why)
132 raise AssertionError("unreachable") # raise_code 恒抛;供类型检查收敛

135def default_channel_of(value: Any) -> Optional[str]:
136 """装配期"缺省通道"归一化(provider 级默认,**非** ctx 回退)。
137
138 ``"acp:<id>"`` → ``"acp"``;``None``/非法/未知 → ``None``。
139 用途:``ApprovalProvider`` 的 ``user_choice``/信任名单按"本 provider 服务哪个
140 交互外壳"判定。**不得**把它当作 ctx 未声明通道时的回退源(那正是修复前的缺陷)。
141 """
142 st = normalize_channel(value)
143 return st.name if st.is_interactive else None
