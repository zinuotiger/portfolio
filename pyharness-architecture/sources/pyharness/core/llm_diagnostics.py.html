22async def observed_wait(awaitable, timeout, *, observation=None):
23 """Share only this request's safe observations with wait_for's child task.
24
25wait_for replaces cancellation with TimeoutError; retain headers received before
26that cancellation without putting mutable state on a shared adapter/transport.
27"""
28 if observation is None:
29 observation = {}
30 token = response_observation.set(observation)
31 try:
32 return await asyncio.wait_for(awaitable, timeout=timeout)
33 except Exception as exc:
34 exc.diagnostics = {**observation, **getattr(exc, 'diagnostics', {})}
35 raise
36 finally:
37 response_observation.reset(token)

49def atom(value, limit=128):
50 """Reject free text, paths, credentials and oversized opaque metadata."""
51 if not isinstance(value, (str, int)) or isinstance(value, bool):
52 return 'unknown'
53 text = str(value)
54 if (len(text) > limit or not re.fullmatch(r'[A-Za-z0-9_.:+-]+', text)
55 or re.search(r'(?i)sk-|bearer|authorization|secret|password|api.?key', text)):
56 return 'unknown'
57 return text or 'unknown'

60def endpoint_origin(value):
61 try:
62 url = urlsplit(str(value))
63 if url.scheme in {'https', 'http'} and url.hostname:
64 host = atom(url.hostname, 253)
65 return f'{url.scheme}://{host}' if host != 'unknown' else 'unknown'
66 except ValueError:
67 pass

71def response_metadata(response):
72 """Only three response headers and a bounded JSON error code are inspected."""
73 headers = response.headers
74 code = 'unknown'
75 # Streaming responses may not have been read; do not buffer arbitrary bodies.
76 try:
77 content = response.content
78 except httpx.ResponseNotRead:
79 content = None
80 if content is not None and len(content) <= 16384:
81 try:
82 body = response.json()
83 error = body.get('error') if isinstance(body, dict) else None
84 if isinstance(error, dict):
85 code = atom(error.get('code'))
86 except (ValueError, httpx.ResponseNotRead):
87 pass
88 retry = headers.get('retry-after')
89 # Retry-After accepts delta seconds or an HTTP date, bounded and grammar checked.
90 if not isinstance(retry, str) or not re.fullmatch(r'(\d{1,9}|[A-Za-z]{3}, \d{2} [A-Za-z]{3} \d{4} \d{2}:\d{2}:\d{2} GMT)', retry):
91 retry = 'unknown'
92 return {'http_status': response.status_code, 'provider_code': code,
93 'request_id': atom(headers.get('x-request-id') or headers.get('request-id')),
94 'retry_after': retry}

123def diagnostics(exc, *, code, endpoint=None, model=None, request_seq=None):
124 meta = dict(getattr(exc, 'diagnostics', {}) or {})
125 response = getattr(exc, 'response', None)
126 if isinstance(response, httpx.Response):
127 meta.update(response_metadata(response))
128 status = meta.get('http_status', getattr(exc, 'status_code', None))
129 category = {401:'authentication', 403:'permission', 429:'rate_limit'}.get(status)

131 category = 'provider_server' if status >= 500 else 'protocol' if status >= 400 else None
132 if category is None:
133 if isinstance(exc, (TimeoutError, httpx.TimeoutException)):
134 category = 'timeout'
135 elif isinstance(exc, httpx.ProtocolError):
136 category = 'protocol'
