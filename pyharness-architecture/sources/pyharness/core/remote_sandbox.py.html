11def validate_address(address):
12 parsed = urlsplit(address)
13 loopback = parsed.hostname == 'localhost'
14 try: loopback |= ipaddress.ip_address(parsed.hostname).is_loopback
15 except ValueError: pass
16 if (parsed.scheme != 'https' and not (parsed.scheme == 'http' and loopback)) or not parsed.hostname or parsed.username or parsed.password or parsed.query or parsed.fragment or parsed.path not in ('', '/'):
17 fail('remote_address_requires_https')
18 return address.rstrip('/')

27 async def request(self, connection, method, path, body=None):
28 address, token = self.settings(connection)
29 if not token: fail('remote_credential_missing')
30 try:
31 async with httpx.AsyncClient(base_url=validate_address(address),
32 headers={'Authorization': 'Bearer ' + token}, timeout=15,
33 follow_redirects=False, trust_env=False, transport=self.transport) as client:
34 response = await client.request(method, path, json=body)
35 if response.status_code >= 400:
36 fail('remote_request_failed', 'HTTP ' + str(response.status_code))
37 return response.json()
38 except (httpx.HTTPError, ValueError):
39 fail('remote_disconnected', '远程服务连接中断；未改用本机执行')

51 async def execute(self, record, command, *, python=False):
52 connection, task, root, profile = self.handles[record.sandbox_id]
53 operation = uuid.uuid4().hex
54 try:
55 await self.request(connection, 'POST', f'/v1/tasks/{task}/execute',
56 {'command': command, 'python': python, 'operation': operation})
57 async with asyncio.timeout(profile.timeout + 20):

59 result = await self.request(connection, 'GET', '/v1/tasks/' + task)
