11async def _preload_plugins(spine: EngineSpine) -> None:
12 """插件预载(async 入口用;load_plugin 需 await,同步装配段不碰)。
13
14 扫 examples/plugins/* 逐个装载进该会话工具表;单个失败只告警不阻断
15 (会话照常可用;事件 plugin.installed 落该会话日志=真实审计)。
16 """
17 from pyharness.core import plugin_loader as _pl
18 plug_ctx = SimpleNamespace(
19 session=getattr(spine, "session", None),
20 bus=getattr(spine, "bus", None))
21 pdir = Path(__file__).resolve().parents[1] / "examples" / "plugins"
22 if not pdir.exists():
23 await _preload_mcp(spine)
24 return
25 # `plugins.priority`（R24 接线）:同层无依赖插件的装载次序,**小者先**;
26 # 未列入者按默认 0(平序回落到**名字序**,保持确定性)。该键此前零读取者。
27 prio = dict(getattr(getattr(spine, "settings", None), "plugins", None)
28 and (getattr(spine.settings.plugins, "priority", None) or {}) or {})
29 for pd in sorted(pdir.glob("*/"),
30 key=lambda d: (int(prio.get(d.name, 0) or 0), d.name)):
31 try:
32 spec = _pl.load_spec(pd)
33 await _pl.load_plugin(spine.plugins, spec, plug_ctx,
34 spine.tool_registry,
35 state=spine.plugin_state)
36 except Exception as e: # noqa: BLE001 预载失败不阻断
37 log.warning("plugin preload failed %s: %s", pd.name,
38 type(e).__name__)
39 await _preload_mcp(spine)

41async def _preload_mcp(spine: EngineSpine) -> None:
42 """装载配置中的 MCP stdio servers,并桥接工具到会话 registry。"""
43 plugins = getattr(getattr(spine, "settings", None), "plugins", None)
44 servers = list(getattr(plugins, "mcp_servers", None) or [])
45 if not servers:
46 return
47 from pyharness.core.mcp import McpClient, StdioTransport
48 for server in servers:
49 if not getattr(server, "enabled", True):
50 continue
51 client = McpClient(server.name, StdioTransport(
52 list(server.command), timeout_s=int(server.timeout_s)))
53 try:
54 await client.connect()
55 client.register_into(spine.tool_registry, allowed_tools=server.allowed_tools)
56 spine._mcp_clients.append(client)
57 except Exception as e: # noqa: BLE001 单个 server 隔离
58 await client.close()
59 log.warning("mcp preload failed name=%s: %s", server.name,
60 type(e).__name__)
