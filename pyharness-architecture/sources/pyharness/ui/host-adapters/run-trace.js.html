13 const spans = (data.trace || []).map(s => {
14 const step = steps.get(s.span_id), start = time(s.started_at), finish = time(s.finished_at) ?? time(step?.finished_at);
15 const explicit = typeof s.duration_ms === 'number' && Number.isFinite(s.duration_ms) && s.duration_ms >= 0;
16 const measured = explicit ? s.duration_ms : start != null && finish != null && finish >= start ? finish-start : null;
17 return {...s, status:step?.status || s.status, offset:start == null ? null : Math.max(0,start-origin), measured,
18 duration_source:explicit ? '轨迹记录' : measured != null ? (s.finished_at ? '轨迹起止时间' : '同标识步骤起止时间') : '未记录闭合时间',
19 end:measured != null && start != null ? Math.max(0,start-origin)+measured : null};
