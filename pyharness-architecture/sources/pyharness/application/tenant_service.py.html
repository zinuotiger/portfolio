67 def effective_settings(self) -> Any:
68 """Settings overridden by this tenant's active model profile **and storage scope**.
69
70 ``2026-09-21 R14-8``:非 ``default`` 租户的**会话目录**本就按租户分
71 （调用点一律用 ``tenant_root()/sessions``），但其**派生索引库**
72 （``storage.db_path``）仍是**全局**的 ⇒ 两个租户共用一个 FTS 库：
73 ① ``query`` 没有任何租户维度 ⇒ A 的 agent 能检索到 B 的会话内容
74 （实测：A 侧不带 ``session_id`` 查询直接命中 B 的条目，`session.fts_query`
75 的 ``session_id`` 又取自 LLM 自报参数）；
76 ② 同名 sid 跨租户在 ``fts_rows`` 主键 ``(session_id, seq)`` 上**碰撞**
77 ⇒ 后写者被 ``INSERT OR IGNORE`` 静默丢弃、其内容根本不入索引（实测）。
78 **派生视图必须与真源同域**：索引库随会话目录一起按租户分。
79 """
80 cfg = _session_module()._cfg_of(self.ctx)
81 profile = self._settings_store.active_profile(self.tenant_id)
82 scoped = self.tenant_id != "default"
83 if (profile is None and not scoped) or not hasattr(cfg, "model_dump"):
84 return copy.deepcopy(cfg) # Runtime registration must not mutate caller settings.
85 raw = cfg.model_dump()
86 if scoped: # 索引库随会话目录同域（R14-8）
87 storage = dict(raw.get("storage") or {})
88 storage["db_path"] = str(self.tenant_root() / "index.db")
89 raw["storage"] = storage
90 skills = dict(raw.get('skills') or {})
91 skills['dir'] = str(self.skill_catalog_root)
92 raw['skills'] = skills
93 if profile is not None:
94 llm = dict(raw.get("llm") or {})
95 from pyharness.application.model_profile_configuration import profile_values
96 llm.update(profile_values(profile, self.tenant_id, llm))
97 raw["llm"] = llm
98 from pyharness.config import Settings
99 return Settings.model_validate(raw)
