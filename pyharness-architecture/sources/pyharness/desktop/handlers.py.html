170 async def stream_sse(self, request: Request, sid: str = "",
171 after_seq: int = 0) -> StreamingResponse:
172 """SSE 事件流:注册 StreamClient(游标 after_seq)→ 总线事件按 seq 序 fan-out;
173 断开 → 客户端带 last_seq 重连续拉(F065 边界);15s 心跳注释行保活。"""
174 from pyharness.application.product_execution_sessions import reject_product_mutation
175 reject_product_mutation(sid)
176 me = StreamClient(sid=str(sid or ""), tenant=self.current_tenant(),
177 last_seq=max(0, int(after_seq)))
178
179 async def gen():
180 # 注册生命周期必须覆盖响应体实际迭代期:ASGI/uvicorn 在视图函数返回后才
181 # 流式读取 body,若像伪码那样在视图内 `async with register` 再 return,
182 # __aexit__ 会在流开始前就把游标注销 → SSE 只见心跳收不到事件。故注册
183 # 移入生成器内部:首个 anext 进入注册,生成器净退/断连/异常时 __aexit__
184 # 自动注销(行为契约修正,偏离 6 同条)。
185 async with self.hub.register(me):
186 while not self.stopping.is_set():
187 if await request.is_disconnected():
188 break # 前端断开 → 回收游标
189 try:
190 item = await asyncio.wait_for(me.queue.get(),
191 timeout=self._sse_heartbeat_seconds())
192 except asyncio.TimeoutError:
193 yield ": ping\n\n" # 心跳注释行(SSE 保活)
194 continue
195 if item is None: # hub.close_all 哨兵 → 净退
196 break
197 type_, text = item
198 yield f"event: {type_}\ndata: {text}\n\n"
199
200 return StreamingResponse(
201 gen(), media_type="text/event-stream",
202 headers={"Cache-Control": "no-cache",
203 "X-Accel-Buffering": "no"})

247 async def webhook_ingest(self, body: dict = Body(default={}),
248 x_webhook_token: Optional[str] = Header(default=None),
249 authorization: Optional[str] = Header(default=None)) -> dict:
250 body = body or {}
251 self._webhook_authorize(body, self._api_token,
252 x_webhook_token=x_webhook_token,
253 authorization=authorization)
254 text = str(body.get("content") or body.get("text") or "").strip()
255 if not text:
256 raise_code("EVT-100", advice="webhook content 不能为空")
257 sid = str(body.get("session_id") or "")
258 if not sid or sid not in self._logs:
259 sid = await self.service.create_session()
260 log_ = await self.service.require_session(sid)
261 resend = await self.service.resend_text(log_, sid, text)
262 return {"sid": sid, **resend}

267 @staticmethod
268 def _webhook_authorize(body: dict, expected: str, *,
269 x_webhook_token: Optional[str] = None,
270 authorization: Optional[str] = None) -> None:
271 if not expected:
272 return
273 given = str(body.get("token") or x_webhook_token or "")
274 if not given and authorization and authorization.lower().startswith("bearer "):
275 given = authorization[7:].strip()
276 if not given or not secrets.compare_digest(given, expected):
277 raise_code("CRED-701", reason="webhook-token",
278 hint="webhook token 缺失或不匹配(拒绝注入)")
