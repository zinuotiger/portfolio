108 """子 Agent 进程内派发管理器(F049;编排层,绑定父会话)。

110 生命周期 = DIS-SEAM §2.4 能力三动作(enter→announce→detach,均可 None ctx/幂等);

173 def _child_workspace_root(self, parent_root: Any, sub_id: str) -> str:
174 """F055 子会话独立根 = {storage.workspaces_dir}/{sub_id}(唯一派生点)。
175
176 权威:`specs/subagent.py.md` §约束常量"child workspace 根=
177 `~/.pyharness/workspaces/{sub_id}/`"、`CFG.md §3.6`。根目录取 config 权威键
178 `storage.workspaces_dir`;cfg 未注入(单测注入替身)时按**父根同族**推导
179 ——父根形态即 `{workspaces_dir}/{parent_sid}`,取 `.parent` 回到树根。
180 两条路径都经 `scope.session_workspace` 拼接,禁止就地字符串拼接。
181
182 2026-09-21 修(N10):此前恒用 `base.parent/{sub_id}`,而装配层父根曾为**裸**
183 workspaces_dir → 子根落到 `~/.pyharness/{sub_id}`(越出 workspaces 树)且
184 无人 mkdir,子代理文件类工具全废(治理判 allow、provider 报 TLB-802)。
185 """
186 ws_dir = getattr(getattr(self._cfg, "storage", None),
187 "workspaces_dir", None)
188 if not ws_dir:
189 from pathlib import Path
190 base = Path(str(parent_root or
191 "~/.pyharness/workspaces")).expanduser()
192 ws_dir = str(base.parent)
193 from pyharness.core.scope import session_workspace
194 return session_workspace(str(ws_dir), sub_id)
