122# ------------------------------------------------------------ AdapterHealth
123@dataclass
124class AdapterHealth:
125 """单适配器健康状态机(F033):healthy → 1 败 degraded → 连 3 败 down → 连 2 好 healthy。
126
127 mark(ok) 由探针 ping 结果驱动,纯状态机、无副作用(不计数、不发事件、不落日志)
128 ——llm.recovered 等通知由 probe_loop 依据状态翻转发出。防抖动:down 需连 3 败,
129 回切 healthy 需连 2 好(瞬时 1 败仅 degraded;degraded 单次成功即解除,见偏离 6)。
130 """
131
132 state: str = "healthy" # healthy / degraded / down
133 fail_streak: int = 0
134 ok_streak: int = 0
135
136 def mark(self, ok: bool) -> None:
137 """按一次探针结果推进状态机。ok=True=ping 成功,ok=False=ping 失败。"""
138 if ok:
139 self.fail_streak = 0
140 self.ok_streak += 1
141 if self.state == "down":
142 if self.ok_streak >= 2: # 连 2 好 → 回切(防抖动)
143 self.state = "healthy"
144 elif self.state == "degraded":
145 self.state = "healthy" # 瞬时单败解除(偏离 6)
146 else:
147 self.ok_streak = 0
148 self.fail_streak += 1
149 if self.state == "healthy" and self.fail_streak == 1:
150 self.state = "degraded" # 1 败仅 degraded,不 down
151 if self.fail_streak >= 3:
152 self.state = "down" # 连 3 败 down
153
154 def is_down(self) -> bool:
155 """down 态判定(down 者被 pick()/降级循环跳过)。"""
156 return self.state == "down"

206 # ====================================================== 链式降级编排(F013)
207 async def chat_with_fallback(self, messages, tools=None, *, ctx,
208 method: str = "chat") -> Any:
209 """从 chain[idx:] 起逐适配器经 _retry_adapter 尝试,返回 LLMResponse。
210
211 每适配器进入前过 BudgetGuard(预算前置闸);302/303 且满足降级条件 → 留痕 +
212 idx 前移,本轮请求继续下一适配器;304 等业务错不降级直接上抛;链尾 302/303
213 (或 degrade.enabled=false 主模型败)→ LLM-310 终态(err_hist 携带排查)。
214 """
215 err_hist: list[tuple[str, str]] = []
216 last_diagnostics = None
217 enabled = bool(ctx.config.llm.degrade.enabled)
218 # 降级总开关(CFG llm.degrade.enabled):关 = 只走当前 idx 适配器,败即 LLM-310
219 end = len(self.chain) if enabled else min(self.idx + 1, len(self.chain))
220 start = self.idx
221 sleep_left = [_MAX_CHAIN_BACKOFF_S] # 全链退避总预算(跨适配器共享,P3)
222 for i in range(start, end):
223 name = self.chain[i]
224 # 探针 down → 跳过(仅跳过循环起始位之后的;起始位不跳:仍尝试以测恢复,
225 # 与 pick() 全 down 保底一致——主模型 down 不跳,见 spec 注释)
226 if self.health[name].is_down() and i > start:
227 continue
228 await BudgetGuard.check(ctx) # 每请求前置预算闸(F032)
229 try:
230 return await self._retry_adapter(name, messages, tools, ctx,
231 method=method,
232 sleep_left=sleep_left)
233 except PyHError as e:
234 err_hist.append((name, e.code))
235 last_diagnostics = e.ctx.get("diagnostics")
236 if (e.code in _DEGRADE_CODES and enabled
237 and i + 1 < len(self.chain)
238 and self._may_degrade(e, ctx)):
239 # 降级:留痕 + idx 前移(单调;对后续请求生效),本轮继续下一适配器
240 self._record_degradation(name, self.chain[i + 1], ctx)
241 self.idx = i + 1
242 continue
243 if e.code in _DEGRADE_CODES:
244 break # 链尾/降级关:全链耗尽 → LLM-310(偏离 5)
245 raise # LLM-304 业务错等:不降级直接上抛
246 raise_code("LLM-310", err_hist=err_hist, diagnostics=last_diagnostics,
247 advice="全链失败,任务终止(不无限降级)")

249 # ====================================================== 指数退避(F028)
250 async def _retry_adapter(self, name: str, messages, tools, ctx, *,
251 method: str = "chat", sleep_left=None) -> Any:
252 """单适配器内重试循环:只重试 LLM-301/303;基数 1s×2 递增、上限 attempts
253 (默认 4)、±jitter(默认 30%)抖动;每次等待落 llm.retry 事件(sleep 可被取消
254 F025);退避耗尽 → LLM-303(exhausted)交 chat_with_fallback 降级决策。
255 sleep_left = 链级退避总预算 [剩余秒] 可变单元(chat_with_fallback 跨适配器共享),
256 累计 sleep 不得超过;耗尽即视为该层退避失败,交降级(P3:退避 sleep 在总闸外)。
257 """
258 cfg = ctx.config.llm.retry
259 n_attempts = max(1, int(cfg.attempts)) # 0 视同 1(偏离 9)
260 delay = float(cfg.base_delay_s) # 基数 1s ×2(CFG llm.retry)
261 jitter = float(getattr(cfg, "jitter", 0.3))
262 for attempt in range(n_attempts):
263 await BudgetGuard.check(ctx) # 每次重试前再查预算(防烧钱)
264 try:
265 fn = getattr(self.adapters[name], method, None)
266 if not callable(fn):
267 raise_code("CFG-601", reason="structural", adapter=name,
268 missing=method,
269 detail="适配器缺降级链请求方法")
270 from pyharness.core.llm_diagnostics import call_attempt
271 token = call_attempt.set(attempt + 1)
272 try:
273 from pyharness.core.llm_requests import ModelRequests
274 if not isinstance(self.adapters[name], ModelRequests):
275 from pyharness.core.context_observation import observe_adapter_request
276 await observe_adapter_request(ctx, name, method, messages, tools)
277 from pyharness.core.product_model_gate import check_model_attempt
278 await check_model_attempt(ctx, name, method)
279 return await fn(messages, tools, ctx=ctx)
280 finally:
281 call_attempt.reset(token)
282 except PyHError as e:
283 if e.code not in _RETRYABLE_CODES:
284 raise # LLM-302/304:不重试
285 if attempt == n_attempts - 1:
286 # 退避预算耗尽:交 chat_with_fallback 降级决策
287 raise_code("LLM-303", model=name, exhausted=True, diagnostics=e.ctx.get("diagnostics"))
288 await ctx.session.append(
289 "llm.retry",
290 {"model": name, "attempt": attempt,
291 "delay_ms": int(delay * 1000)},
292 actor="llm")
293 # sleep 可被取消(F025);总重试时长计入请求预算
294 wait = delay * random.uniform(1 - jitter, 1 + jitter)
295 if sleep_left is not None: # 链级退避总预算闸
296 if sleep_left[0] <= 0:
297 raise_code("LLM-303", model=name, exhausted=True, diagnostics=e.ctx.get("diagnostics"))
298 wait = min(wait, sleep_left[0])
299 sleep_left[0] -= wait
300 await asyncio.sleep(wait)
301 delay *= 2

303 # ====================================================== 降级条件判定(ADI §3.3)
304 def _may_degrade(self, e: PyHError, ctx) -> bool:
305 """按码判定:LLM-302 无条件降;LLM-303 exhausted 或连续限流达阈值(默认 2)

307 if e.code == "LLM-302": # 认证错:不重试直接降级

311 return True # 退避耗尽(attempts 次)后降
312 consec = ctx.counters.rate_limit_streak() # 连续 429 计数(事件可重建)
313 return consec >= int(
314 ctx.config.llm.degrade.rate_limit_consecutive) # 默认 2
315 return False # LLM-301/304:301 已在退避内消化

341
342 探针成功不计 F029 用量、失败不落 llm.retry——只判可达性,真实请求仍按错误码
343 走降级(ADI §3.5);主模型 down→healthy(2 好回切)时 idx 归零自动回切主模型
344 (偏离 4)。取消即退出(不吞 CancelledError,F025)。
345
346 偏离(相对 spec 伪码"对全部适配器 ping"):探针**只走 ``self.health`` 的键**
347 (= 本链成员),注册表里存在但不在链内的模型不探。理由:①``health`` 在
348 ``__post_init__`` 里按 ``chain`` 初始化,按注册表遍历会在"注册了链外模型"时
349 ``self.health[name]`` 直接 KeyError —— 该异常在 try 之外,会**打死整个探针
350 任务**(等于静默关掉回切);②回切判据 ``pick()`` 只读链内成员的健康位。
351 链成员尚未注册时跳过(不判失败),等其注册后自然纳入。
352 """
353 reg = self.adapters if adapters is None else adapters
354 if not reg:
355 log.warning("probe_loop: 无适配器注册表,探针空转退出")
356 return
357 while True:
358 for name in list(self.health): # 只探**链内**适配器(见下注)
359 adp = reg.get(name)
360 if adp is None: # 链成员尚未注册:跳过,不算失败
361 continue
362 prev = self.health[name].state
363 try:
364 # Keep the deadline in the owner task: Python 3.11 wait_for
365 # can lose external cancellation when its child finishes
366 # concurrently, leaving shutdown waiting for this loop.
367 async with asyncio.timeout(5.0):
368 await adp.ping()
369 self.health[name].mark(True)
370 except (PyHError, asyncio.TimeoutError):
371 self.health[name].mark(False)
372 except Exception: # noqa: BLE001 未预期同判失败
373 log.debug("probe %s 未预期异常", name, exc_info=True)
374 self.health[name].mark(False)
375 # 2 好回切(down→healthy):通知 + 主模型 idx 归零自动回切
376 if (prev == "down"
377 and self.health[name].state == "healthy"):
378 _fire_emit(self.bus, "llm.recovered", {"model": name})
379 if name == self.chain[0]:
380 self.idx = 0
381 await asyncio.sleep(interval_s)
382
383 # ====================================================== 适配器择优(F033)
384 def pick(self) -> str:
385 """返回 chain 中从 idx 起第一个非 down 适配器;全部 down → 返回 idx 当前
386 元素(交由降级流程暴露 LLM-310)。主模型 down 不跳过(保底仍尝试,以测恢复)。"""
387 for m in self.chain[self.idx:]:
388 if not self.health[m].is_down():
389 return m
390 return self.chain[self.idx]

393# -------------------------------------------------------------- BudgetGuard
394class BudgetGuard:
395 """单任务预算闸(F032 前置;CONSTRAINTS-07 §10):发请求前与每次重试前读
396 UsageCounters(与 scope.budget_state 同源,report_usage 唯一写入)与 BudgetLimits
397 (ctx.config.budget.task),返回 ok/warn;超任一硬闸 → budget.paused(exhausted)
398 事件 + 抛 BUDGET-EXHAUSTED(agent-loop 据此强制终态 reason=budget)。
399
400 偏离说明:check 为 async(spec 伪码同步签名与内部 await session.append 矛盾,
401 见偏离 2);paused/exhausted 态由 agent-loop 的 scope.budget_state 三闸强制,
402 本闸只做前置拦截,返回面为 ok/warn + raise。
403 """
404
405 # 硬闸:输出/输入 token 或估算成本任一超限 → 终态(判定以 token 为准,F032)
406 @staticmethod
407 async def check(ctx) -> BudgetState:
408 """前置检查:超限 raise BUDGET-EXHAUSTED;达 warn_ratio(默认 80%)返回
409 "warn";否则 "ok"。"""
410 used = ctx.counters.task_total() # 与 scope.budget_state 同一数据源
411 lim = ctx.config.budget.task # in≤200万 / out≤5万 / cost≤1 元(F032)
412 if (used.out_tokens >= lim.max_out_tokens
413 or used.in_tokens >= lim.max_in_tokens
414 or used.cost_est >= lim.max_cost_yuan):
415 await ctx.session.append(
416 "budget.paused", {"state": "exhausted",
417 "used": used.model_dump()}, actor="system")
418 raise PyHError("BUDGET-EXHAUSTED",
419 ctx={"used": used.model_dump()}) # 偏离 3:未登记码直构
420 warn_ratio = float(getattr(ctx.config.budget, "warn_ratio", 0.8))
421 if used.out_tokens >= warn_ratio * lim.max_out_tokens: # warn=80% 提醒
422 _fire_emit(getattr(ctx, "bus", None), "budget.warn",
423 {"used_out": used.out_tokens})
424 return "warn"
425 return "ok"
426
427 @staticmethod
428 def report(ctx) -> dict:
429 """预算读数汇总(/cost 命令与审计数据源,CONSTRAINTS-07 §9;纯只读)。"""
430 used = ctx.counters.task_total()
431 lim = ctx.config.budget.task
432 return {"used": used.model_dump(), "limit": lim.model_dump(),
433 "pct_out": used.out_tokens / lim.max_out_tokens * 100}
