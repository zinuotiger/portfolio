27def _resolve_by_seq(rows: list[tuple[int, str]]) -> list[tuple[int, str]]:
28 """写前按 seq 归并(保序)并做**一致性校验**(fail-closed)。
29
30 ``seq`` 是事件身份的一部分——框架经 ``SeqState`` 单调分配、``SessionLog``
31 另有 EVT-101 失步闸,故同一 seq 只可能代表同一事件。据此:
32
33 - 同 seq + **相同行** ⇒ 去重(同 seq 重放同一事件,幂等);
34 - 同 seq + **不同行** ⇒ **数据一致性冲突** → ``PERS-202`` 上抛;
35 **不 first-wins、不 last-wins、不静默覆盖**(丢哪一份都是丢事件)。
36
37 返回按 seq 升序的写序列(物理写序 = seq 序)。**本函数无副作用**:在任何
38 队列状态变更之前调用,冲突时原队列保持不动。
39 """
40 seen: dict[int, str] = {}
41 for seq, line in rows:
42 prev = seen.get(seq)
43 if prev is None:
44 seen[seq] = line
45 elif prev != line:
46 raise_code("PERS-202", op="seq_conflict", seq=seq,
47 why="同一 seq 出现不同内容:seq 是事件身份的一部分,"
48 "同一 seq 只能是同一事件",
49 advice="拒绝静默覆盖(不 first/last-wins);查是否绕过"
50 "框架自报 seq 或存在重复写入")
51 return [(s, seen[s]) for s in sorted(seen)]
