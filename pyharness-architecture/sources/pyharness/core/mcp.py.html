55 def __init__(self, argv: list, *, timeout_s: int = DEFAULT_TIMEOUT_S) -> None:
56 self._argv = list(argv)
57 self._timeout = timeout_s
58 self._proc: Optional[asyncio.subprocess.Process] = None
59 self._seq = 0

65 async def _ensure(self) -> None:
66 if self._closed or (self._proc is not None and self._proc.returncode is not None):
67 raise_code("TLB-805", module="mcp", hint="MCP 连接已关闭;需显式重新初始化。"
68 "此前未确认的远端操作结果不确定,副作用未撤销;先查询核对,不要盲目重试写入")
69 if self._proc is not None and self._proc.returncode is None:
70 return
71 flags = 0x08000000 if os.name == "nt" else 0 # CREATE_NO_WINDOW
72 self._proc = await asyncio.create_subprocess_exec(
73 *self._argv, stdin=asyncio.subprocess.PIPE,
74 stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.PIPE,
75 env=_mcp_env(), creationflags=flags)
76 self._stderr_task = asyncio.create_task(self._drain_stderr())
77
78 async def _drain_stderr(self) -> None:
79 proc = self._proc
80 if proc is None or proc.stderr is None:
81 return
82 while True:
83 line = await proc.stderr.readline()
84 if not line:
85 return
86 log.debug("mcp stderr: %s", line.decode("utf-8", errors="replace").rstrip())

88 async def request(self, method: str, params: dict) -> Any:
89 async with self._request_lock:
90 await self._ensure()
91 proc = self._proc
92 assert proc is not None and proc.stdin is not None and proc.stdout is not None
93 self._seq += 1
94 req_id = self._seq
95 message = {"jsonrpc": "2.0", "method": method, "params": params}
96 if not method.startswith("notifications/"):
97 message["id"] = req_id
98 envelope = json.dumps(message, ensure_ascii=False)
99 try:
100 async with asyncio.timeout(self._timeout):
101 proc.stdin.write((envelope + "\n").encode("utf-8"))
102 await proc.stdin.drain()
103 log.debug("mcp sent pid=%s id=%s method=%s tool=%s", proc.pid,
104 message.get("id"), method, params.get("name"))
105 if method.startswith("notifications/"):
106 return None # JSON-RPC 通知无 id、无响应
107 while True:
108 resp = await proc.stdout.readline()
109 if not resp:
110 raise ConnectionError("server closed stdout")
111 data = json.loads(resp.decode("utf-8"))
112 if not isinstance(data, dict) or data.get("jsonrpc") != "2.0":
113 raise ValueError("invalid JSON-RPC envelope")
114 if "id" not in data and str(data.get("method", "")).startswith("notifications/"):
115 continue
116 if data.get("id") != req_id:
117 raise ValueError("response id mismatch")
118 break
119 except asyncio.CancelledError:
120 # Discard late responses; a new stream must initialize explicitly.
121 log.warning("MCP cancelled id=%s method=%s; remote outcome unknown, no rollback", req_id, method)
122 await self.close()
123 raise
124 except (OSError, asyncio.TimeoutError, ValueError) as exc:
125 await self.close()
126 raise_code("TLB-805", module="mcp", method=method, request_id=req_id,
127 hint=f"MCP 传输失败({type(exc).__name__}),连接已关闭;"
128 "远端操作结果不确定,可能仍在运行,副作用未撤销;先查询核对,不要盲目重试写入")
129 if "error" in data:
130 raise_code("TLB-805", module="mcp",
131 hint=f"{method} 远端错误:{data['error']}")
132 return data.get("result")

134 async def close(self) -> None:
135 self._closed = True
136 task = self._close_task
137 if task is None or (task.done() and (task.cancelled() or task.exception())):
138 task = self._close_task = asyncio.create_task(self._close_owned())
139 cancelled = None
140 while True:
141 try:
142 await asyncio.shield(task)
143 break
144 except asyncio.CancelledError as exc:
145 if task.cancelled():
146 raise
147 cancelled = exc
148 except BaseException as exc:
149 if cancelled is not None:
150 cancelled.add_note(f'MCP cleanup failed: {type(exc).__name__}')
151 raise cancelled from exc
152 raise
153 if cancelled is not None:
154 raise cancelled
155
156 async def _close_owned(self) -> None:
157 # The cleanup task, rather than any individual waiter, owns these handles.
158 # Keep them available for a subsequent close if cleanup itself fails.
159 if self._proc is not None:
160 proc = self._proc
161 if proc.stdin is not None and not proc.stdin.is_closing():
162 proc.stdin.close()
163 try:
164 await proc.stdin.wait_closed()
165 except (BrokenPipeError, ConnectionResetError):
166 pass
167 if proc.returncode is None:
168 try:
169 # EOF is the normal stdio shutdown; allow the server to flush.
170 await asyncio.wait_for(proc.wait(), timeout=3)
171 except asyncio.TimeoutError:
172 try:
173 proc.terminate()
174 await asyncio.wait_for(proc.wait(), timeout=3)
175 except (ProcessLookupError, asyncio.TimeoutError):
176 try:
177 proc.kill()
178 except ProcessLookupError:
179 pass
180 await proc.wait()
181 if self._stderr_task is not None:
182 self._stderr_task.cancel()
183 try:
184 await self._stderr_task
185 except asyncio.CancelledError:
186 # 自己 cancel 的 stderr 排空任务:await 被掐断抛 CancelledError,
187 # 属清理段,吞掉即可——旧实现 re-raise 会让 request 超时路径的
188 # await self.close() 被打断,掩盖本应抛的 TLB-805(P2 修复)。
189 pass
190 except Exception: # noqa: BLE001 stderr 排空失败不阻断
191 pass
192 self._stderr_task = None
193 self._proc = None

230 async def connect(self) -> dict:
231 """MCP 握手:initialize → notifications/initialized → tools/list。"""
232 init = await self.transport.request("initialize", {
233 "protocolVersion": "2024-11-05",
234 "capabilities": {"tools": {}},
235 "clientInfo": {"name": "pyharness", "version": self.version}})
236 self.initialization = dict(init or {})
237 negotiated = self.initialization.get("protocolVersion")
238 if negotiated is not None and negotiated != "2024-11-05":
239 await self.close()
240 raise_code("TLB-805", module="mcp", hint=f"不支持协商版本 {negotiated}")
241 await self.transport.request("notifications/initialized", {})
242 listed = await self.transport.request("tools/list", {})
243 self.tools_listing = dict(listed or {})
244 self._tools = list((listed or {}).get("tools") or []) # result:null 判空(P2)
245 self.connected = True
246 return {"server": (init or {}).get("serverInfo"), "protocolVersion": negotiated,
247 "tools": len(self._tools)}

249 def register_into(self, tool_reg: Any, *, allowed_tools: Optional[list[str]] = None) -> list[str]:
250 """list 出的工具桥接进 core ToolRegistry(mcp.<name>.<tool> 前缀)。
251
252 danger=high:远端工具=代码执行面,一律走审批;返回注册名清单。"""
253 from pyharness.core.tools_registry import ToolDefinition
254 registered: list[str] = []
255 selected = self._tools
256 if allowed_tools is not None:
257 allowed = set(allowed_tools)
258 missing = allowed - {t.get("name") for t in self._tools}
259 if missing:
260 raise_code("TLB-802", module="mcp", tool=sorted(missing),
261 advice="本地允许的 MCP 工具未被服务发现")
262 selected = [t for t in self._tools if t.get("name") in allowed]
263 try:
264 for t in selected:
265 tool_name = f"mcp.{self.name}.{t.get('name')}"
266 schema = t.get("inputSchema") or {"type": "object",
267 "properties": {}}
268 tool_reg.register_tool(
269 ToolDefinition(name=tool_name, danger="high",
270 description=(t.get("description")
271 or f"MCP 远端工具 {tool_name}")[:190],
272 schema=schema, timeout_s=DEFAULT_TIMEOUT_S,
273 owner=f"mcp:{self.name}", ctx_path="mcp",
274 version="1.0.0"),
275 provider=_McpProvider(self, t.get("name")))
276 registered.append(tool_name)
277 except Exception:
278 for name in reversed(registered):
279 try:
280 tool_reg.unregister_definition(name)
281 except Exception: # noqa: BLE001 回滚尽力
282 log.debug("mcp rollback unregister miss %s", name)
283 raise
284 self.tool_reg = tool_reg
285 return registered

305 async def handle(self, args: dict, ctx: Any) -> str:
306 result = await self._client.call_tool(self._tool, args)
307 result = result or {} # result:null 判空(P2)
