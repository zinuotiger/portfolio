14 def shutdown_gracefully(self) -> None:
15 """窗口关闭 = 优雅停服(F065 边界;幂等:双击关闭只走一次)。
16
17 序:stopping 置位 → 停事件 fan-out → hub 关闭(SSE 客户端收 close)→ 兜底 flush
18 (强同步事件已随 append 落盘)→ server.should_exit=True 停 uvicorn → webview.destroy。
19 幂等判据用 _graceful_done(伪码 `if stopping.is_set(): return` 会在 closing 事件
20 先置位时让首次 graceful 空转——伪码缺陷,以行为契约修正,见模块 docstring)。
21 """
22 if self._graceful_done:
23 return
24 self._graceful_done = True
25 self.stopping.set()
26 loop = self.loop
27 if loop is not None and loop.is_running():
28 try:
29 fut = asyncio.run_coroutine_threadsafe(self._shutdown_async(), loop)
30 fut.result(timeout=15)
31 except Exception: # noqa: BLE001
32 log.warning("desktop async shutdown failed", exc_info=True)
33 else:
34 try:
35 asyncio.get_running_loop()
36 except RuntimeError:
37 asyncio.run(self._shutdown_async())
38 else:
39 # 已处于另一个事件循环(通常仅启动失败路径):用短线程隔离执行。
40 errors: list[BaseException] = []
