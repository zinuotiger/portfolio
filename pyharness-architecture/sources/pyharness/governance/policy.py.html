2

4算出内容哈希指纹、管理策略版本、发 policy.updated 事件——**只描述策略,不做决策、
5不执行**(G-4/INV-G5;关 2 出口升格属 S3)。
6

11``ToolCall``/``Scope`` 一律用 ``Any``——不得用 TYPE_CHECKING 变通(保留 import

56def _canonical_params(params: Optional[Mapping[str, Any]]) -> dict:
57 """声明式参数规范化 + 边界校验:含 callable 一律拒(防指纹失稳,R-A)。
58
59 list/tuple/set 归一为**有序**字符串列表——集合序不稳定会让同一配置算出不同
60 指纹,直接破坏 M2 判据 ①。
61 """
62 out: dict = {}
63 for k, v in (params or {}).items():
64 if _is_callable(v):
65 raise_code("CYC-999", module="governance.policy", field=str(k),
66 hint="policy params 只允许声明式值(函数对象不入指纹)")
67 if isinstance(v, (list, tuple, set, frozenset)):
68 out[str(k)] = sorted(str(x) for x in v)
69 else:
70 out[str(k)] = v
71 return out

74def compute_fingerprint(*, policy_id: str, version: str,
75 rules: Iterable["PolicyRule"],
76 disabled: Iterable[str],
77 params: Optional[Mapping[str, Any]]) -> str:
78 """策略内容哈希 = sha256(规范化 JSON)。
79
80 ★ **只含声明式内容**:policy_id / version / **元组序** rule_id / policy_refs /
81 forced / disabled / params。
82 ★ **排除 match/check**(函数对象):否则每次装配指纹都变,M2 判据 ①
83 ("指纹随规则内容变化")会退化为"每次都变"(R-A)。
84 """
85 payload = {
86 "policy_id": str(policy_id),
87 "version": str(version),
88 "rules": [{"rule_id": r.rule_id,
89 "policy_refs": list(r.policy_refs),
90 "forced": bool(r.forced)} for r in rules],
91 "disabled": sorted(str(x) for x in disabled),
92 "params": _canonical_params(params),
93 }
94 canon = json.dumps(payload, sort_keys=True, ensure_ascii=False,
95 separators=(",", ":"), default=str)
96 return hashlib.sha256(canon.encode("utf-8")).hexdigest()

110@dataclass(frozen=True)
111class PolicyRule:
112 """单条策略规则描述(``match``/``check`` 由装配层注入,本层只持有)。
113
114 ``rule_id`` 全链唯一(重名注册被拒 TLB-801);``policy_refs`` 是该规则可能产出的
115 策略引用(如 ("POL-FS-1","POL-FS-2","POL-FS-3")),供审计与凭证引用;
116 ``forced`` 对应 ``tools_guard.FORCED_GUARDS``(恒在不可禁)。
117 """
118
119 rule_id: str
120 policy_refs: tuple[str, ...] = ()
121 match: Any = None
122 check: Any = None
123 forced: bool = False
124
125 @classmethod
126 def from_descriptor(cls, desc: Any) -> "PolicyRule":
127 """鸭子描述符(如 ``tools_guard.RuleDescriptor``)→ PolicyRule。
128
129 只按属性读取,**不 import 描述符来源模块**(Δ-4:治理层不得反向依赖 core);
130 缺 ``rule_id`` 视为非法描述 → TLB-801。
131 """
132 rid = getattr(desc, "rule_id", None)
133 if not isinstance(rid, str) or not rid:
134 raise_code("TLB-801", rule=type(desc).__name__,
135 advice="规则描述符缺 rule_id(非法描述);须含 "
136 "rule_id/policy_refs/forced/match/check")
137 refs = tuple(str(r) for r in (getattr(desc, "policy_refs", ()) or ()))
138 return cls(rule_id=rid, policy_refs=refs,
139 match=getattr(desc, "match", None),
140 check=getattr(desc, "check", None),
141 forced=bool(getattr(desc, "forced", False)))

144@dataclass(frozen=True)
145class Policy:
146 """策略集**一等对象**(不可变):规则 + 求值序 + 禁用面 + 声明式参数 + 指纹。
147
148 ``rules`` 的**元组序即求值序**(对齐 ``tools_guard._BUILTIN_IDS``);
149 ``fingerprint`` 在构造期算定(见 ``compute_fingerprint``),内容变则指纹变。
150 派生(禁用/启用)只经 ``with_rule_*`` 返回**新对象**,不改本对象。
151 """
152
153 policy_id: str
154 version: str
155 rules: tuple[PolicyRule, ...] = ()
156 disabled: frozenset = frozenset()
157 params: Mapping[str, Any] = field(default_factory=dict)
158 fingerprint: str = ""
159
160 def __post_init__(self) -> None:
161 if not self.fingerprint: # 构造即算(不可变对象)
162 object.__setattr__(self, "fingerprint", compute_fingerprint(
163 policy_id=self.policy_id, version=self.version,
164 rules=self.rules, disabled=self.disabled, params=self.params))
165
166 def enabled_rules(self) -> tuple[PolicyRule, ...]:
167 """当前生效规则(排除禁用项);求值序保持。"""
168 return tuple(r for r in self.rules if r.rule_id not in self.disabled)
169
170 def rule_of(self, rule_id: str) -> Optional[PolicyRule]:
171 return next((r for r in self.rules if r.rule_id == rule_id), None)
172
173 def _derive(self, *, disabled: frozenset) -> "Policy":
174 """派生新 Policy(重算指纹;原对象不变)。"""
175 return Policy(policy_id=self.policy_id, version=self.version,
176 rules=self.rules, disabled=disabled, params=self.params)
177
178 def with_rule_disabled(self, rule_id: str, *, config_ref: str) -> "Policy":
179 """禁用一条规则(治理动作,Δ-2)。
180
181 必填 ``config_ref``(谁在何处声明了这次放宽)——缺则 CFG-601 拒;
182 规则不在策略内 → CFG-601;``forced`` 规则恒在不可禁 → CFG-601
183 (与既有 ``GuardChain.disable`` 的 FORCED_GUARDS 复核同源)。
184 幂等:已禁用 → 原样返回(不产生新对象,不重复留痕)。
185 """
186 if not str(config_ref or "").strip():
187 raise_code("CFG-601", reason="越权", fields=["config_ref"],
188 detail="禁用规则必须携带 config_ref(治理动作须可审计,"
189 "INV-G6)")
190 rule = self.rule_of(rule_id)
191 if rule is None:
192 raise_code("CFG-601", guard=rule_id,
193 why="规则不在当前策略内(禁用无从谈起)")
194 if rule.forced:
195 raise_code("CFG-601", guard=rule_id,
196 why="forced 规则恒在不可禁(对应 FORCED_GUARDS)")
197 if rule_id in self.disabled:
198 return self # 幂等:不重复留痕
199 return self._derive(disabled=self.disabled | {rule_id})
200
201 def with_rule_enabled(self, rule_id: str) -> "Policy":
202 """重新启用一条规则(治理动作,Δ-2)。幂等:未禁用 → 原样返回。"""
203 if rule_id not in self.disabled:
204 return self
205 return self._derive(disabled=self.disabled - {rule_id})

208class PolicyRegistry:
209 """规则注册表(只增链尾,延续 ``register_plugin_guard`` 单调性)。
210
211 重名注册被拒(TLB-801);无移除/重排 API——规则面的变更只经 Policy 的
212 治理动作表达,并落 ``policy.updated``。
213 """
214
215 def __init__(self) -> None:
216 self._rules: list[PolicyRule] = []
217
218 def register_rule(self, rule: Any) -> None:
219 """登记一条规则(``PolicyRule`` 或同形描述符);重名 → TLB-801。"""
220 if not isinstance(rule, PolicyRule):
221 rule = PolicyRule.from_descriptor(rule)
222 if any(r.rule_id == rule.rule_id for r in self._rules):
223 raise_code("TLB-801", rule=rule.rule_id,
224 advice="rule_id 全链唯一(重复注册被拒)")
225 self._rules.append(rule)
226
227 def rules(self) -> tuple[PolicyRule, ...]:
228 """已登记规则(登记序 = 求值序)。"""
229 return tuple(self._rules)

231 async def emit_updated(self, policy: Policy, op: str, *, reason: str,
232 config_ref: Optional[str] = None,
233 session: Any = None) -> bool:
234 """发 ``policy.updated``(强同步,Q3)。返回 True=已落盘,False=降级未发。
235
236 - **op 白名单**:非 ``POLICY_OPS`` → CYC-999 拒(Δ-1);
237 - **op=disable 必带 config_ref** → 缺则 CFG-601 拒(Δ-3:禁止静默放宽);
238 - **词表未注册时降级**(S2-1 阶段常态):记日志、不抛——与
239 ``approval._emit_trust``/``tools_guard._record`` 同款惯例;S2-2 注册
240 ``policy.updated`` 后**同路径自动生效,不需改调用方**;
241 - ``config_ref`` 经 ``Envelope.trace`` 携带(Δ-6)。
242 """
243 if op not in POLICY_OPS:
244 raise_code("CYC-999", module="governance.policy", op=str(op),
245 hint=f"policy.updated 的 op 必须 ∈ {sorted(POLICY_OPS)}"
246 "(tighten 归 scope.updated)")
247 if op == "disable" and not str(config_ref or "").strip():
248 raise_code("CFG-601", reason="越权", fields=["config_ref"],
249 detail="op=disable 必须携带 config_ref(禁止静默放宽)")
250 from pyharness.events.vocab import is_registered # 允许:events(ADR-018)
251 if not is_registered(EVENT_POLICY_UPDATED):
252 log.debug("policy.updated 未入词表(S2-2 注册),降级未发 op=%s", op)
253 return False
254 if session is None:
255 log.warning("policy.updated 未接线 session,降级未发 op=%s", op)
256 return False
257 payload = {
258 "policy_id": policy.policy_id,
259 "version": policy.version,
260 "fingerprint": policy.fingerprint,
261 "op": op,
262 "added": sorted(policy.disabled) if op == "disable" else [],
263 "reason": str(reason),
264 }
265 trace = {"config_ref": str(config_ref)} if config_ref else None
266 await session.append(EVENT_POLICY_UPDATED, payload, actor="system",
267 sync=True, trace=trace)
268 return True

271class PolicyEngine:
272 """策略的装配、解析与版本治理。**只提供策略,不做决策、不执行**。
273
274 构造经 ``from_config``(Δ-4 注入式);执行对象(``GuardChain``)由装配层经
275 ``chain_factory`` 注入——治理层只持有引用,``chain`` 为 None 时表示"尚未接线
276 (S2-1 常态)"。
277 """
278
279 def __init__(self, *, policy: Policy, registry: PolicyRegistry,
280 session: Any = None, bus: Any = None, chain: Any = None,
281 chain_factory: Any = None) -> None:
282 self._policy: Policy = policy
283 self._registry: PolicyRegistry = registry
284 self._session: Any = session

286 self._chain: Any = chain

289 # ------------------------------------------------------------ 装配
290 @classmethod
291 def from_config(cls, cfg: Any, *, session: Any = None, bus: Any = None,
292 rules: Iterable[Any] = (), params: Optional[Mapping] = None,
293 chain_factory: Any = None,
294 validator: Any = None, credential_paths: Any = None,
295 path_exists: Any = None, link_resolver: Any = None,
296 approval_channel: Optional[bool] = None,
297 policy_id: str = "builtin:v1",
298 version: str = "1.0.0") -> "PolicyEngine":
299 """装配策略引擎(**注入式**,Δ-4)。
300
301 - ``rules``:规则描述(``PolicyRule`` 或鸭子描述符,如
302 ``tools_guard.describe_rules()`` 的产物)→ 登记进取注册表,登记序 = 求值序;
303 - ``chain_factory``:执行对象的构造入口,由装配层传入
304 (生产值 = ``tools_guard.from_config``);**治理层不 import 它**;
305 - ``validator``/``credential_paths``/``path_exists``/``link_resolver``/
306 ``approval_channel``:透传给 ``chain_factory``,并把**其声明式形态**
307 (布尔/清单/注入存在性)纳入指纹 params——函数对象本身不入指纹;
308 - 初始禁用面读 ``cfg.security.guards.disabled``(只读)。
309 """
310 registry = PolicyRegistry()
311 for r in rules:
312 registry.register_rule(r)
313 p = _canonical_params(params)
314 if approval_channel is not None:
315 p["approval_channel"] = bool(approval_channel)
316 if credential_paths is not None:
317 p["credential_paths"] = sorted(str(x) for x in credential_paths)
318 p["injected"] = sorted(n for n, v in (("validator", validator),
319 ("path_exists", path_exists),
320 ("link_resolver", link_resolver))
321 if v is not None)
322 policy = Policy(policy_id=policy_id, version=version,
323 rules=registry.rules(),
324 disabled=frozenset(_disabled_from_cfg(cfg)),
325 params=p)
326 chain = None
327 if callable(chain_factory):
328 chain = chain_factory(cfg, session=session, bus=bus,
329 validator=validator,
330 credential_paths=credential_paths,
331 path_exists=path_exists,
332 link_resolver=link_resolver,
333 approval_channel=approval_channel)
334 return cls(policy=policy, registry=registry, session=session, bus=bus,
335 chain=chain, chain_factory=chain_factory)
