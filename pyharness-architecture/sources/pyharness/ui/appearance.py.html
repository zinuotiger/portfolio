13def normalize(value, *, allow_scenes=False):

19 'particle-ocean-blue-light', 'particle-starry-sky-light',
20 'particle-aurora-light', 'particle-ocean-light')

32 result['count'] = int(math.floor(result['count']+.5))

40 def load(self):
41 try:
42 if self.path.stat().st_size > 4096: return normalize({})
43 return normalize(json.loads(self.path.read_text(encoding='utf-8')), allow_scenes=True)

46 def save(self, value):
47 name = None
48 try:
