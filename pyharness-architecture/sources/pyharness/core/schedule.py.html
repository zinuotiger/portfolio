192 async def register(self, name: str, kind: str, expr: str, template: dict, *,
193 is_risky: Optional[bool] = None, ctx: Any = None) -> None:
194 """注册/更新调度 job(F048)。
195
196 非法表达式(CFG-601)/重名(BUSY)/模板缺 intent(EVT-100)/名字非法(CFG-601)
197 一律拒;合法则写 schedule.registered 持久化定义(事件=唯一真源 INV-01)。
198 """
199 if not isinstance(name, str) or not NAME_RE.fullmatch(name):
200 raise_code("CFG-601", hint=f"job 名非法: {name}",
201 advice="小写字母开头,字母/数字/下划线/点,总长 2~64")
202 if name in self._jobs: # 重名:显式拒,不隐式覆盖
203 raise PyHError("BUSY", ctx={
204 "hint": f"job {name} 已存在",
205 "advice": "先 remove 再注册(防隐式覆盖)"})
206 spec = self._validate_spec(kind, expr) # 非法 → CFG-601(EVT-100)
207 if not isinstance(template, dict) or not isinstance(
208 template.get("intent"), str) or not template["intent"].strip():
209 raise_code("EVT-100", hint="template 缺 intent 字段",
210 advice="任务模板必须含非空意图文本")
211 risky = self._derive_risky(template) if is_risky is None else bool(is_risky)
212 sess = self._session_ctx(ctx)
213 first = self._first_fire(kind, expr, now())
214 # 事件先落,内存后对齐(偏离 1:append 拒写时零副作用,INV-01 更严)
215 env = await sess.append("schedule.registered", {
216 "name": name, "kind": kind, "expr": expr,
217 "template": {"intent": template["intent"]}, # meta 不入事件(INV-09)
218 "is_risky": risky, "paused": False,
219 "next_fire_at": iso(first)}, actor="system")
220 job = ScheduleJob(name=name, kind=kind, expr=expr,
221 template=dict(template), is_risky=risky, paused=False,
222 next_fire_at=first, created_seq=env.seq, spec=spec)
223 self._jobs[name] = job # 事件落定 → 注册表对齐

226 def _count_from_events(self) -> dict[str, dict[str, int]]:
227 """triggered/missed 计数聚合(schedule.trigger 计数;missed 事件 payload
228 missed 累计)——内存不存第二份(INV-01),list_jobs 每次回放聚合。"""

230 for e in self._iter_session_events():
231 t = getattr(e, "type", "")
232 p = getattr(e, "payload", {}) or {}
233 if t == "schedule.trigger":
234 c = counts.setdefault(p.get("job", ""),
235 {"triggered": 0, "missed": 0})
236 c["triggered"] += 1
237 elif t == "schedule.missed":
238 c = counts.setdefault(p.get("job", ""),
239 {"triggered": 0, "missed": 0})
240 c["missed"] += int(p.get("missed", 0))
241 return counts

310 """恢复:继续按重臂后的 next_fire_at 触发;重复恢复幂等。"""

313 return # 幂等(重复恢复无事件)

318 template=j.template, spec=j.spec)
319 nxt = self.next_fire(probe, now())
320 await sess.append("schedule.updated", {
321 "name": name, "paused": False, "next_fire_at": iso(nxt)},
322 actor="system")
323 j.paused = False # 事件落定 → 内存对齐

327 async def _fire(self, ctx: Any, job: ScheduleJob, fired_at: datetime) -> None:
328 """触发执行(F048 I/O):先写 schedule.trigger 留痕再 submit 任务模板(F043)。
329
330 模板意图经队列执行,复用 guard/预算/日志纪律;入队失败(如 QUE-001)不静默,
331 写 system.error 显式事件化,下一触发窗重试。
332 """
333 sess = self._session_ctx(ctx)
334 await sess.append("schedule.trigger", {
335 "job": job.name, "cron": job.expr,
336 "fired_at": fired_at.isoformat()}, actor="system")
337 q = self._queue_ctx(ctx)
338 if q is None: # 未装配队列:宁失败不假装入队
339 raise_code("CYC-999", hint=f"schedule:{job.name} 未注入 task_queue,"
340 "到点任务无法入队")
341 intent = job.template["intent"]
342 await sess.append("user.message", {"content": intent}, actor="user",
343 origin=f"schedule:{job.name}", sync=True)
344 try:
345 await q.submit(intent, meta={"source": f"schedule:{job.name}"})
346 except PyHError as e: # QUE-001 队满等:显式事件化不吞
347 try:
348 await sess.append("system.error", {
349 "code": e.code, "hint": f"schedule:{job.name} 入队失败"},
350 actor="system")
351 except Exception: # noqa: BLE001 留痕失败只记日志
352 log.exception("schedule error event failed job=%s", job.name)
353
354 async def _tick(self, ctx: Any = None, *, now_dt: Optional[datetime] = None
355 ) -> list[str]:
356 """每分钟检查(F048 核心):暂停跳过 → 未到期跳过 → 深夜+危险 blocked 留痕
357 → at 一次性触发后移除 → cron/interval 触发并推进窗口;返回触发名单。
358
359 单 job 异常隔离:任一 job 失败记 system.error 不中断其余(tick 快照遍历)。
360 """
361 now_dt = now_dt or now()
362 sess = self._session_ctx(ctx)
363 fired: list[str] = []
364 for job in list(self._jobs.values()): # 快照遍历(tick 中可 remove/注册)
365 if job.paused:
366 continue
367 if not self._due(job, now_dt):
368 continue
369 try:
370 if self._is_night(now_dt) and job.is_risky:
371 # 深夜禁触危险类模板:blocked 留痕 + 顺延下次,零入队(GWT-06)
372 await sess.append("schedule.blocked", {
373 "job": job.name, "reason": "night-window"}, actor="system")
374 job.next_fire_at = self.next_fire(job, now_dt)
375 continue
376 if job.kind == "at": # 一次性:触发后移除(不重复)
377 await self._fire(ctx, job, now_dt)
378 fired.append(job.name)
379 await self.remove(job.name, ctx)
380 continue
381 # 周期型:先 fire 成功再推进窗口(P2:此前内存窗口先推、事件后落,
382 # _fire 失败时崩溃恢复按事件重推 → 重复/错位触发)
383 await self._fire(ctx, job, now_dt)
384 job.last_fired_at = now_dt
385 job.next_fire_at = self.next_fire(job, now_dt)
386 fired.append(job.name)
387 except PyHError as e: # 单 job 异常隔离(原码事件化)
388 log.error("schedule tick job=%s code=%s", job.name, e.code)
389 try:
390 await sess.append("system.error", {
391 "code": e.code, "hint": f"schedule:{job.name} tick 失败"},
392 actor="system")
393 except Exception: # noqa: BLE001
394 log.exception("schedule error event failed job=%s", job.name)
395 except Exception: # 未预期:记日志,不中断其余 job
396 log.exception("schedule tick unexpected job=%s", job.name)
397 return fired

405 def stop(self) -> None:
406 """停止泵(会话关闭/装配拆卸用):置停闸并取消 ticker 任务。"""
407 self._stopped = True
408 t = self._ticker_task
409 if t is not None and not t.done():
410 t.cancel()

412 async def aclose(self):
413 task = self._ticker_task
414 self.stop()
415 if task is not None:
416 await asyncio.gather(task, return_exceptions=True)
417 self._ticker_task = None
