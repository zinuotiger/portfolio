47def _tokenize(text: str) -> list[str]:
48 """逐段扫描文本:中文连续段逐字出 token,ASCII 连续段整词出 token,其余丢弃。"""
49 out: list[str] = []
50 i, n = 0, len(text)
51 while i < n:
52 ch = text[i]
53 if _CJK_CHAR.fullmatch(ch):
54 j = i + 1
55 while j < n and _CJK_CHAR.fullmatch(text[j]):
56 j += 1
57 out.extend(text[i:j]) # 逐字成 token(每字一词)
58 i = j
59 elif ch.isascii() and (ch.isalnum() or ch == "_"):
60 j = i + 1
61 while j < n and text[j].isascii() and (text[j].isalnum() or text[j] == "_"):
62 j += 1
63 out.append(text[i:j]) # ASCII 整词成 token
64 i = j
65 else:
66 i += 1 # 标点/空白/其他:丢弃(分隔)
67 return out

119def _index_item(type_: str, env: Any) -> Optional[tuple]:
120 """事件 → 索引项(模块级单源;``SessionQueryIndex._extract`` 与 ``is_indexable`` 共用)。
121
122 返回 None = 不索引;("insert", row) = 新增内容行;("edit", (sid, tseq, newc))
123 = 级联修正(F063,不改行键、不新增行)。
124 """
125 if type_ == "user.message_edited": # F063:级联更新 target 行
126 sid = _field(env, "session_id")
127 tseq = _payload_field(env, "target_seq")
128 newc = _payload_field(env, "new_content")
129 if not sid or not tseq or not newc:
130 return None
131 return ("edit", (sid, int(tseq), str(newc)))
132 fn = _WHITELIST.get(type_)
133 if fn is None:
134 return None # 白名单外(含 guard.*/瞬时)
135 text = fn(env)
136 text = str(text if text is not None else "").strip()
137 if not text:
138 return None # 空文本不索引(llm.response 空轮)
139 sid = _field(env, "session_id")
140 seq = _field(env, "seq")
141 ts = _field(env, "ts", "")
142 actor = _field(env, "actor", "")
143 if not sid or not seq:
144 return None # 缺行键:拒(不产生孤儿行)
145 row = (str(sid), int(seq), str(type_), str(ts), str(actor),
146 _index_text(str(_payload_field(env, "new_title", "") or "")),
147 _index_text(text))
148 return ("insert", row)

160def read_max_seq(db_path: Any, session_id: str) -> int:
161 """**装配外**只读索引末 seq(F060 落后对账;repair 无 live 索引句柄时用)。
162
163 2026-09-21 R11-2 修:`repair._fts_last_seq_provider` 只认 ``ctx.session_query``
164 / ``ctx.fts_last_seq``,而**没有任何生产调用方注入它们**(CLI 的 ``auto_scan``
165 连 ctx 都不收)⇒ 索引落后对账**从未真正启用**(``max_seq`` 的旧 docstring 亦自认
166 "装配后仍无法启用")。本函数让"只知道 db 路径"的调用方也能读数。
167
168 语义:库/表不存在 → 0(未索引,调用方按 0 判"不算落后");读取失败 → PERS-202
169 (不静默吞:对账判据不可得时必须显式)。连接短开短合,只读打开。
170 """
171 import sqlite3 as _sql
172 db_file = Path(str(db_path)).expanduser()
173 if not db_file.exists():
174 return 0
175 try:
176 conn = _sql.connect(f"file:{db_file}?mode=ro", uri=True)
177 except _sql.Error as e:
178 raise_code("PERS-202", op="fts_max_seq_ro", path=str(db_file),
179 hint=f"只读打开索引库失败: {e}")
180 try:
181 row = conn.execute(_MAX_SEQ_SQL, (str(session_id),)).fetchone()
182 except _sql.Error as e:
183 if "no such table" in str(e).lower(): # 索引未建表:未索引
184 return 0
185 raise_code("PERS-202", op="fts_max_seq_ro", path=str(db_file),
186 hint=f"索引末 seq 读数失败: {e}")
187 finally:
188 conn.close()
189 return int(row[0] or 0) if row else 0

213class FtsQueryHandle:
214 """session.fts_query 工具的 Provider 实现(executor 契约 handle(args, ctx))。
215
216 只读面:把 LLM 参数转发到 ctx.session.fts 上挂载的 SessionQueryIndex.query;
217 未挂载(能力未 announce)→ TLB-802(不静默,防幻觉工具名)。

220 name: str = TOOL_NAME

222 async def handle(self, args: dict, ctx: Any) -> dict:
223 """executor 关3 Provider 调用面:经 ctx.session.fts 执行查询。"""
224 sess = getattr(ctx, "session", None)
225 index = getattr(sess, "fts", None) if sess is not None else None
226 if index is None or not callable(getattr(index, "query", None)):
227 raise_code("TLB-802", tool=TOOL_NAME,
228 advice="会话 FTS 索引未挂载(ctx.session.fts);先 announce 能力")
229 result = await index.query(args.get("q", ""),
230 limit=args.get("limit", _RESULT_LIMIT),
231 session_id=args.get("session_id"),
232 actor=args.get("actor"),
233 after_ts=args.get("after_ts"))
234 return {"hits": [h.__dict__ for h in result.hits],
235 "timed_out": result.timed_out}
