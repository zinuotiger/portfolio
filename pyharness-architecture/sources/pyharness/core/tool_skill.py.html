18class _SkillHandle:

21 async def handle(self, args: dict, ctx: Any) -> str:
22 mgr = getattr(ctx, "skills", None)
23 if mgr is None:
24 raise_code("CYC-999", module="skill",
25 hint="ctx.skills 未装配(引擎未挂 SkillManager)")
26 op = str(args.get("_op") or "")
27 if op == "list":
28 rows = mgr.list()
29 if not rows:
30 return "技能库为空"
31 return "可用技能:\n" + "\n".join(
32 f"- {r['name']}: {r['description']}" for r in rows)
33 name = str(args.get("name") or "").strip()
34 if not name:
35 raise_code("EVT-100", field="name", hint="skill.load 缺技能名")
36 data = mgr.load(name) # 未知名 → SKL-901
37 try: # 审计:装载留痕(body 不入日志)
38 await ctx.session.append("skill.used", {"name": data["name"]},
39 actor="tool")
40 except Exception as e: # noqa: BLE001 审计失败不阻断
41 log.debug("skill.used append failed: %s", type(e).__name__)
42 return (f"技能已装载:{data['name']}\n"
43 f"简介:{data['description']}\n"
44 f"正文:\n{data['body']}")
