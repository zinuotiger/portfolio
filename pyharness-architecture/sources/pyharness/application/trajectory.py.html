212def trajectory_page(sid, events, projection, *, after_seq=0, limit=200, run_id=None, workflow=None):
213 events=list(events)
214 later=[e for e in events if e.seq>after_seq]
215 batch=later[:limit]
216 end=batch[-1].seq if batch else after_seq
217 nodes=projection['nodes']
218 if workflow is not None:
219 workflow_nodes=[n for n in nodes if n.get('workflow_name')==workflow]
220 linked={n.get('linked_run_id') for n in workflow_nodes if n.get('linked_run_id')}
221 nodes=[n for n in nodes if n in workflow_nodes or n.get('run_id') in linked]
222 elif run_id is not None:nodes=[n for n in nodes if n.get('run_id')==run_id]
223 relevant=({n.get('run_id') for n in nodes}|{n.get('linked_run_id') for n in nodes})-{None}
224 runs=[r for r in projection['runs'] if run_id==r['run_id'] or (run_id is None and (workflow is None or r['run_id'] in relevant))]
225 active=next((r for r in reversed(runs) if r['status'] in {'running','waiting_approval','queued','paused','retrying'}),None)
226 status=active['status'] if active else runs[-1]['status'] if runs else 'recorded'
227 if workflow is None:
228 if any(n['status']=='waiting_approval' for n in nodes):status='waiting_approval'
229 elif any(n['status']=='running' for n in nodes):status='running'
230 elif not runs and any(n['status']=='failed' for n in nodes):status='failed'
231 if workflow is not None:
232 states=[n['status'] for n in nodes if n.get('workflow_name')==workflow]
233 status='failed' if 'failed' in states else 'unknown' if 'unknown' in states else 'running' if 'running' in states else 'completed' if states and all(s=='completed' for s in states) else 'recorded'
234 selected=[n for n in nodes if any(after_seq<s<=end for s in n['event_seqs'])]
235 return {'session_id':sid,'run_id':run_id or sid,'workflow':workflow,'name':workflow or (runs[0]['name'] if run_id and runs else '会话执行轨迹'),
236 'status':status,'started_at':runs[0]['started_at'] if runs else events[0].ts if events else None,
237 'finished_at':runs[-1]['finished_at'] if runs and not active and status in {'completed','failed','cancelled'} else None,
238 'trace':selected,'trace_total':len(nodes),'trace_truncated':len(later)>len(batch),
239 'trace_cursor':end,'to_seq':end,'latest_seq':events[-1].seq if events else 0,
240 'steps':[],'runs':runs,'artifacts':[a for a in projection['artifacts'] if workflow is None and run_id is None or a.get('run_id') in relevant]}
