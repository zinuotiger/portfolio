97def _last_seq(s: Any) -> int:
98 """事件最新 seq(SessionLog.stats()[\"seq\"];stats 缺面时回放尾事件)。"""
99 st = getattr(s, "stats", None)
100 if callable(st):
101 try:
102 return int(st().get("seq", 0) or 0)
103 except Exception: # noqa: BLE001 门面异常 → 回放兜底
104 pass
105 last = 0
106 for env in _iter_events(s):
107 last = env.seq

111def _session_title(s: Any) -> str:
112 """会话标题派生(F042):session.created 初值 + 最新 session.renamed 覆盖。"""
113 title = ""
114 for env in _iter_events(s):
115 p = env.payload
116 if env.type == "session.created":
117 title = p.get("title") or ""
118 elif env.type == "session.renamed":
119 title = p.get("new_title") or title
120 return title or "(未命名)"

123def _aggregate_usage(s: Any) -> dict:
124 """llm.usage 事件聚合(F029/F032):in/out tokens 与 cost_est 逐条累加。"""
125 in_tokens = out_tokens = 0
126 cost_est = 0.0
127 for env in _iter_events(s):
128 if env.type != "llm.usage":
129 continue
130 p = env.payload
131 in_tokens += int(p.get("in_tokens") or 0)
132 out_tokens += int(p.get("out_tokens") or 0)
133 cost_est += float(p.get("cost_est") or 0.0)
134 return {"in_tokens": in_tokens, "out_tokens": out_tokens, "cost_est": cost_est}

137def _latest_todos(s: Any) -> list[dict]:
138 """todo 派生(F040):todo.updated 为全量快照语义 → 每 task_id 取最新事件,扁平化。"""
139 latest: dict[str, list[dict]] = {}
140 for env in _iter_events(s):
141 if env.type != "todo.updated":
142 continue
143 latest[env.payload.get("task_id", "")] = [
144 {"id": it.get("id"), "text": it.get("text"),
145 "done": bool(it.get("done"))}
146 for it in (env.payload.get("todos") or [])]
147 out: list[dict] = []
148 for task_id in sorted(latest): # 多任务按 task_id 稳定排序(确定性输出)
149 for it in latest[task_id]:
150 out.append({**it, "task_id": task_id})
151 return out

169async def cmd_status(args: str, ctx: Any) -> str:
170 """/status:会话/任务/todo 进度快照(F040/F042)——事件派生,零 LLM 零第二份状态。"""
171 s = ctx.session
172 budget = getattr(ctx, "budget", None)
173 state = getattr(budget, "state", None) if budget is not None else None
174 lines = [

180 lines.append(f"任务进度(F040): {len(pending)} 项未完成")
181 lines += [f" [{t['id']}] {t['text']}" for t in pending]
182 else:
183 lines.append("任务进度: 当前无未完成 todo 项")
184 return "\n".join(lines)

187async def cmd_cost(args: str, ctx: Any) -> str:
188 """/cost:当前会话预算/成本报表(F032)——只读离线,预算闸未装配降级只输出用量段。"""
189 usage = _aggregate_usage(ctx.session)

199 line += f" | 硬闸 {float(lim):.2f} 元({pct:.0f}%){warn}"
200 return line

236 return "再见"

240def register_command(name: str, handler: Callable[..., Any], *,
241 danger: str = "none", help_text: str = "") -> None:
242 """扩展注册点(阶段 4 plan_mode/undo 等运行时挂载,INV-08:各自 import)。
243
244 重名/非法名(非 [a-z][a-z0-9]*)/非法 danger(∉{none, high},命令面无 critical)
245 → ValueError(装配期编程错误,早失败,不进事件系统,禁现场造码);注册即冻结。
246 """
247 if name in COMMANDS or name in ALIASES: # 重名/占用别名 → 拒
248 raise ValueError(f"command already registered: /{name}")
249 if not re.fullmatch(r"[a-z][a-z0-9]*", name): # 非法名 → 拒
250 raise ValueError(f"illegal command name: /{name}")
251 if danger not in ("none", "high"): # 命令面无 critical → 拒
252 raise ValueError(f"command danger must be none|high: /{name}")
253 COMMANDS[name] = CommandEntry(name=name, handler=handler,
254 danger=danger, help=help_text)

308async def handle_slash(raw: str, ctx: Any) -> bool:
309 """斜杠命令分发总入口(F041):解析 → 查别名/主表 → 分发执行。恒返回 True。
310
311 流程(调用方契约:仅 '/'-开头输入进入本函数):
312 1. body = raw[1:],首词 = 命令名(小写归一);
313 2. 未知 → system.error(EVT-105) 回显,会话继续,零 LLM(不写 user.command);
314 3. headless × danger=high → system.error(APR-501) 留痕,零执行零副作用,
315 不写 user.command(见偏离 1);命令面无审批请求;
316 4. 命中 → **先** user.command 落盘(EVT-104 安全序:保证 /new /exit 终态前
317 命令已留痕)再调 handler;handler 异常兜底回显,会话不崩;
318 5. ctx.render 回显 handler 文本;恒返回 True(命令输入永不落 user.message)。
319
320 零 LLM 硬保证:本路径不 import llm、不触达 agent-loop 的模型调用——命令后
321 llm.request 事件数不增为验收断言(DEP G5)。
322 """
323 body = raw[1:].strip() # 去掉首 "/"
324 name, _, arg = body.partition(" ") # 首词 = 命令名
325 name = name.lower().lstrip("/")
326 entry = COMMANDS.get(ALIASES.get(name, name)) # 别名 → 主名 → 查表
327 if entry is None: # 未知命令:EVT-105,会话继续
328 await _record_error(ctx, "EVT-105",
329 f"未知命令 /{name},可用 /help 查看")
330 return True # 已消费;零 LLM
331 if _gated(entry, ctx): # headless × high:直接拒(APR-501)
332 await _record_error(ctx, "APR-501",
333 f"无交互通道,/{entry.name} 已拒绝(headless)")
334 return True # 拒绝不执行(INV-05 精神)
335 # 命中即写(先落盘!):/new /exit 终态命令执行前命令已留痕(EVT-104 安全序)
336 await ctx.session.append("user.command",
337 {"name": entry.name, "args": arg.strip()},
338 actor="user")
339 try:
340 text = await entry.handler(arg.strip(), ctx) # 分发执行
341 except PyHError as e:
342 # 结构化失败:保留原码回显 + 本地堆栈,会话不崩(异常表 CYC-999 兜底精神)
343 log.error("slash handler failed command=/%s code=%s", entry.name,
344 e.code, exc_info=True)
345 text = f"命令 /{entry.name} 执行失败[{e.code}];详见本地日志,会话继续"
346 except Exception as e: # noqa: BLE001 未预期异常兜底:归 CYC-999,本地堆栈
347 wrapped = wrap_unexpected(e, f"commands.cmd_{entry.name}")
348 log.error("slash handler unexpected command=/%s code=%s", entry.name,
349 wrapped.code, exc_info=True)
350 text = f"命令 /{entry.name} 执行失败[CYC-999];详见本地日志,会话继续"
351 await _render(ctx, text) # 外壳回显(注入,非日志)
352 return True # 恒 True:命令输入已消费
