12 async def query(self, q: str, *, limit: int = 20,
13 session_id: Optional[str] = None,
14 session_ids: Optional[set[str]] = None,
15 actor: Optional[str] = None,
16 after_ts: Optional[str] = None) -> QueryResult:
17 """全文查询(spec 伪码,F057 验收直译)。
18
19 解析(中文 2-gram + 英文 ≥3 字符词)→ 拼 MATCH(词间 OR + 整串短语补偿)→
20 SQLite 查询带 snippet() → 按 rank 取 ≤min(limit, result_limit) 条 Hit;
21 词长低于下限 → EVT-100;5s 超时**软降级**返回 {hits: [], timed_out: True}
22 不抛不挂(WARNING 留痕);库通道故障(非超时)→ PERS-202。
23 """
24 if self._db is None:
25 raise_code("PERS-202", op="fts_query",
26 hint="索引未装载(enter 未执行或已 detach)",
27 advice="先 enter/重新装载索引")
28 if not isinstance(q, str) or not q.strip():
29 raise_code("EVT-100", hint=f"查询词非法: {q!r}",
30 advice="中文≥2 字、英文≥3 字符;或直接给整句短语")
31 terms = self._cjk_terms(q) + self._latin_terms(q)
32 if not terms: # 中文<2 且英文<3 → 拒
33 raise_code("EVT-100", hint=f"查询词过短: {q!r}",
34 advice="中文≥2 字、英文≥3 字符;或直接给整句短语")
35 expr = self._match_expr(q)
36 sql = ("SELECT session_id, seq, type, ts, "
37 f"snippet(events_fts, {_CONTENT_COL}, '[', ']', '…', 24), rank "
38 "FROM events_fts WHERE events_fts MATCH ? AND seq>0")
39 args: list = [expr]
40 if session_id:
41 sql += " AND session_id=?"
42 args.append(session_id)
43 if session_ids is not None:
44 # An explicit empty scope must never fall back to the whole index.
45 if not session_ids:
46 return QueryResult(hits=[], timed_out=False)
47 sql += " AND session_id IN (" + ",".join("?" for _ in session_ids) + ")"
48 args.extend(sorted(session_ids))
49 if actor:
50 sql += " AND actor=?"
51 args.append(actor)
52 if after_ts:
53 sql += " AND ts>=?" # ISO8601 UTC 字典序可比
54 args.append(after_ts)
55 cap = self._clamp_limit(limit)
56 sql += f" ORDER BY rank LIMIT {cap}"
57 try:
58 hits = await asyncio.wait_for(
59 asyncio.to_thread(self._execute_query, sql, args),
60 timeout=self.query_timeout_s)
61 return QueryResult(hits=hits, timed_out=False)
62 except asyncio.TimeoutError: # 超时软降级:不抛不挂
63 _log_warn("fts query timeout q=%r(>%ss,软降级返回空)", q,
64 self.query_timeout_s)
65 return QueryResult(hits=[], timed_out=True)
66 except sqlite3.Error as e: # 库通道故障(非超时)
67 raise_code("PERS-202", op="fts_query", hint=f"FTS 查询失败: {e}",
68 advice="运行 repair(F060) 重建索引")

70 def _clamp_limit(self, limit: Any) -> int:
71 """limit 钳制:1 ≤ limit ≤ result_limit(超限钳到 config result_limit)。"""
72 try:
73 lim = int(limit)
74 except (TypeError, ValueError):
75 lim = self.result_limit
76 return max(1, min(lim, self.result_limit))

78 def _execute_query(self, sql: str, args: list) -> list[Hit]:
79 """同步查询执行(线程内;锁内取尽结果防跨线程游标)。
80
81 MATCH 语法错(极端畸形串)→ 空集(检索场景降级优于报错);结果 snippet 做
82 中文相邻字空格回缩美化(偏离说明:索引侧每字成词的展示补偿)。
83 """
84 try:
85 with self._lock:
86 rows = self._db.execute(sql, args).fetchall()
87 return [Hit(session_id=r[0], seq=r[1], type=r[2], ts=r[3],
88 snippet=_pretty_snippet(r[4]), rank=float(r[5]))
89 for r in rows]
90 except sqlite3.OperationalError:
91 return [] # MATCH 语法错 → 空集
92 except sqlite3.Error:
93 raise

95 def _cjk_terms(self, q: str) -> list[str]:
96 """中文段 2-gram 滑窗(每段 ≥2 字才产词;单字段自然为空,上层 EVT-100)。
97
98 例:"整理文件夹" → [整理, 理文, 文件, 件夹](spec 伪码原样;含"夹?"整串
99 短语由 _match_expr 连续子串补偿兜住)。
100 """
101 out: list[str] = []
102 for seg in _CJK_RUN.findall(q):
103 for i in range(len(seg) - 1):
104 out.append(seg[i:i + 2])
105 return out
106
107 def _latin_terms(self, q: str) -> list[str]:
108 """拉丁/数字词抽取(≥3 字符才入词;英文下限 F057 边界)。"""
109 return _WORD.findall(q)

111 def _match_expr(self, q: str) -> str:
112 """MATCH 表达式:2-gram 相邻单字短语 OR 拉丁词短语 + 整串连续子串补偿。
113
114 每个 2-gram 转 quoted phrase("备份" → `"备 份"`,相邻 token 序列)——unicode61
115 把中文连续串当单 token,索引侧已逐字空格化,故短语即"原文相邻字"语义;
116 len(q) ≥ 4 时把整串按同一 tokenizer 切成短语 OR 上(补偿 2-gram 割裂,
117 spec "连续子串补偿":补召回保精度)。token 只来自中文单字/ASCII 词,天然
118 无引号注入;去重保序。
119 """
120 parts: list[str] = []
121 seen: set[str] = set()
122 for g in self._cjk_terms(q): # "备份" → "备 份"
123 ph = f'"{g[0]} {g[1]}"'
124 if ph not in seen:
125 seen.add(ph)
126 parts.append(ph)
127 for w in self._latin_terms(q):
128 ph = f'"{w}"'
129 if ph not in seen:
130 seen.add(ph)
131 parts.append(ph)
132 if len(q) >= 4: # 整串短语补偿(≥4 字符)
133 toks = _tokenize(q)
134 if toks:
135 ph = '"' + " ".join(toks) + '"'
136 if ph not in seen:
137 parts.append(ph)
138 return " OR ".join(parts)
