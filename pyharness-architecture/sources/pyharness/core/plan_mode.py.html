92class PlanManager:
93 """会话级 plan 注册表与状态机(F045/F046 编排内核)。
94
95 消费 ctx.session(append 唯一写口)/ctx.llm.json_chat(F045 唯一 LLM 出口)/
96 ctx.task_queue(F043 submit/wait_for)/ctx.approval.user_choice(F046 三选一)/
97 ctx.scope.policy.danger_marks(F023 危险分级可选覆盖);被命令层(/plan、approve/
98 reject/edit)、UI 方案弹窗、compaction(F058 不可折叠集)、agent-loop(expiry
99 sweep 周期驱动)消费。内存态只由事件派生:直接构造(空表)或 rebuild_from_events。
100
101 私有字段:_plans(注册表,plan_id=proposed seq)、_danger_marks(内置 F023 表,
102 可被 ctx.scope 运行时覆盖)、_exec_tasks(approve 自动建的执行协程句柄集)。
103 """

105 def __init__(self, *, danger_marks: Optional[list[dict]] = None) -> None:
106 self._plans: dict[int, Plan] = {}
107 self._danger_marks: list[dict] = list(
108 danger_marks if danger_marks is not None else DANGER_TABLE)
109 self._exec_tasks: set[asyncio.Task] = set()

112 def _require_session(self, ctx: Any) -> Any:
113 """写事件前置:ctx.session 未接线 → EVT-100(编排内核不背第二落点)。"""
114 sess = getattr(ctx, "session", None)
115 if sess is None or not callable(getattr(sess, "append", None)):
116 raise_code("EVT-100", hint="ctx.session 未接线:plan.* 事件无落点",
117 advice="装配层注入 SessionLog 或同型 append 协议对象")
118 return sess

120 def _get(self, plan_id: Optional[int]) -> Plan:
121 """定位方案;不存在 → EVT-101(必先 plan.proposed,事件引用校验)。"""
122 p = self._plans.get(plan_id) if plan_id is not None else None
123 if p is None:
124 raise_code("EVT-101", plan_id=plan_id,
125 hint=f"方案 {plan_id} 不存在:先 plan_propose 再裁决",
126 advice="先 /plan 提案;核对 plan_id")

132 policy = getattr(getattr(ctx, "scope", None), "policy", None)
133 marks = getattr(policy, "danger_marks", None)

138 def _spawn_exec(self, plan_id: int, ctx: Any) -> None:
139 """approve 自动建任务(F046 I/O):登记句柄 + done 回调清理/记异常。"""
140 task = asyncio.create_task(self.plan_execute(plan_id, ctx))
141 self._exec_tasks.add(task)
142
143 def _done(t: asyncio.Task) -> None:
144 self._exec_tasks.discard(t)
145 if not t.cancelled() and t.exception() is not None:
146 # 后台执行协程异常不静默:堆栈仅本地(事件已由 _abort 落 plan.aborted)
147 log.error("plan %s 执行协程异常: %r", plan_id, t.exception())
148
149 task.add_done_callback(_done)

153 sess = self._require_session(ctx)

157 p.who = "system"

179 # ============================================================ 方案生成(F045)
180 async def plan_propose(self, goal: str, ctx: Any, *, steps: Optional[list] = None,
181 max_steps: int = MAX_STEPS) -> Plan:
182 """方案提案主函数:默认只调一次 LLM 产出 ≤max_steps 步 JSON 方案。
183
184 steps 显式传入时跳过 LLM(plan_revise 单步编辑复用,零 LLM 重调);逐步
185 危险打标(F023 同源表)+ 字段预检 + 自检;写 plan.proposed(先落后展示,
186 EVENT-SCHEMA §3.5.2)并登记 Plan(id=事件 seq,status=pending,24h TTL)。
187 """
188 sess = self._require_session(ctx)
189 if not goal or not str(goal).strip():
190 raise_code("EVT-100", hint="plan 目标为空",
191 advice="提供 <目标> 后重新提案")
192 if steps is None: # 常规路径:只调一次 LLM(F045 边界)
193 raw = await self._json_chat(ctx, goal, max_steps)
194 if isinstance(raw, dict):
195 steps = raw.get("steps") # 兼容 {"steps": [...]} 包装
196 else:
197 steps = raw
198 if not isinstance(steps, list): # LLM 结构不达标(非步骤数组)
199 raise_code("LLM-304",
200 hint="json_chat 未返回 steps 数组,方案不可执行",
201 advice="让模型重出(提示词强制 JSON),或人工给出步骤")
202 if not steps: # 空方案(显式 steps 空同样拒)
203 raise PyHError("EVT-100", ctx={
204 "hint": f"方案步骤须 1..{max_steps} 步",
205 "advice": "精简方案后重新提案"})
206 if len(steps) > max_steps: # ≤8 步强校验(超限拒写不落盘)
207 raise PyHError("EVT-100", ctx={
208 "hint": f"方案步骤须 1..{max_steps} 步(实际 {len(steps)})",
209 "advice": "精简方案后重新提案"})
210 marks = self._scope_marks(ctx) # F023 表(F045 提案时刻定格)
211 # 规范化 + 危险打标 + 执行意图拼装(每步只认五键,杜绝脏键入事件)
212 ann: list[dict[str, str]] = []
213 for i, s in enumerate(steps):
214 if not isinstance(s, dict):
215 raise PyHError("EVT-100", ctx={
216 "hint": f"步骤 {i + 1} 不是对象", "advice": "检查方案结构"})
217 action = str(s.get("action") or "").strip()
218 tool = str(s.get("tool") or "").strip()
219 expected = str(s.get("expected") or "").strip()
220 # 字段预检:缺 action/tool/expected 的步骤 events.payload.PlanStep 必拒
221 # (min_length=1),此处前置给明确错误——零半成品不落盘(见偏离 6)
222 if not action or not tool or not expected:
223 miss = [k for k, v in (("action", action), ("tool", tool),
224 ("expected", expected)) if not v]
225 raise PyHError("EVT-100", ctx={
226 "hint": f"步骤 {i + 1} 缺字段: {','.join(miss)}(不可执行)",
227 "advice": "补齐 action/tool/expected 后重新提案"})
228 risk = self._classify(tool, marks) # F023 同源表打标
229 intent = str(s.get("intent") or "").strip() or \
230 f"用 {tool} {action}: {expected}" # 缺省由 action/tool 拼装
231 ann.append({"action": action, "tool": tool, "expected": expected,
232 "risk": risk, "intent": intent})
233 ok, why = selfcheck_plan(ann) # 可执行性/依赖自检(纯函数,展示用)
234 created_at = _utcnow()
235 expires_at = created_at + timedelta(hours=PLAN_TTL_HOURS) # 24h 未确认作废
236 # 事件载荷:steps 只落 events.payload.PlanStep 五字段子集(action/tool/
237 # expected/risk),intent 不随事件(见偏离 2);先落盘后展示(§3.5.2)
238 step_payloads = [{k: d[k] for k in ("action", "tool", "expected", "risk")}
239 for d in ann]
240 ev = await sess.append(
241 "plan.proposed",
242 {"goal": str(goal).strip(), "steps": step_payloads,
243 "selfcheck_ok": ok, "expires_at": _iso(expires_at)},
244 actor="agent")
245 p = Plan(id=ev.seq, goal=str(goal).strip(),
246 steps=[Step(**d) for d in ann],
247 selfcheck_ok=ok, why=why, status="pending", # 待批:等用户裁决
248 created_at=created_at, expires_at=expires_at)
249 self._plans[p.id] = p # 注册(plan_id = proposed 事件 seq)
250 return p

284 # ============================================================ 审批裁决(F046)
285 async def plan_approve(self, plan_id: int, *, who: str = "user",
286 ctx: Any) -> None:
287 """审批:批准(F046)。仅 pending 可批——写 plan.approved(who) 并置 approved
288 (方案冻结,steps 不可变),随即启动执行协程(approved 自动建任务,逐步入队);
289 已拒绝/已终态/不存在显式拒;批准前先查 24h 过期(过期自动转 rejected)。
290 """
291 p = self._get(plan_id) # 不存在 → EVT-101(必先 proposed)
292 if p.status == "rejected": # 拒绝后不再推进(F046 边界)
293 raise PyHError("BUSY", ctx={
294 "hint": f"方案 {plan_id} 已拒绝,不可批准",
295 "advice": "修订后重新提案(reject 重提)"})
296 if p.status in ("completed", "aborted"): # 终态不可再裁决
297 raise PyHError("BUSY", ctx={
298 "hint": f"方案 {plan_id} 已{p.status}",
299 "advice": "终态不可逆;新需求请重新提案"})
300 if p.status != "pending": # 只待批可批(防重复 approve)
301 raise PyHError("BUSY", ctx={
302 "hint": f"方案 {plan_id} 非待批态({p.status})",
303 "advice": "已裁决方案不可重复批准"})
304 if await self._expire_if_stale(p, ctx): # 24h 未确认 → 自动 rejected
305 raise PyHError("BUSY", ctx={
306 "hint": "方案已过期(24h 未确认),请重新提案",
307 "advice": "重新 /plan 提案后再批准"})
308 sess = self._require_session(ctx)
309 await sess.append("plan.approved",
310 {"plan_id": plan_id, "who": who}, actor="user")
311 p.status = "approved" # 事件落定 → 内存对齐(INV-01)
312 p.who = who # 批准后 steps 冻结不可变(改=reject 重提)
313 self._spawn_exec(plan_id, ctx) # approved 自动建任务(F046 I/O)

315 async def _expire_if_stale(self, p: Plan, ctx: Any) -> bool:
316 """24h 过期即时检查(approve 前置 + sweep 双路复用):过期 → 自动 rejected。
317
318 写 plan.rejected(who=system,reason=expired) 后置状态;返回是否已过期。
319 """
320 if p.status != "pending" or p.expires_at is None:
321 return False
322 if p.expires_at > _utcnow():
323 return False
324 sess = self._require_session(ctx)
325 await sess.append("plan.rejected",
326 {"plan_id": p.id, "who": "system", "reason": "expired"},
327 actor="system") # who=system 自动态(24h 未确认)
328 p.status = "rejected" # 自动作废,不执行
329 p.who = "system"
330 return True

332 async def plan_reject(self, plan_id: int, *, who: str = "user",
333 reason: str = "user-rejected", ctx: Any) -> None:
334 """审批:拒绝(F046)。拒绝方案回修订对话——写 plan.rejected(who,reason);
335 拒绝后不可再 approve/revise/execute;终态(completed/aborted/rejected)
336 不可再拒绝。执行中被拒 → 执行循环顶守卫停止后续步骤(见偏离 7)。
337 """
338 p = self._get(plan_id) # 不存在 → EVT-101
339 if p.status in ("completed", "aborted", "rejected"):
340 raise PyHError("BUSY", ctx={
341 "hint": f"方案 {plan_id} 已终态({p.status})",
342 "advice": "终态不可逆;新需求请重新提案"})
343 sess = self._require_session(ctx)
344 await sess.append("plan.rejected",
345 {"plan_id": plan_id, "who": who, "reason": reason},
346 actor="user")
347 p.status = "rejected" # 事件落定 → 内存对齐(INV-01)
348 p.who = who

351 async def plan_revise(self, plan_id: int, step_idx: int, patch: dict,
352 ctx: Any) -> Plan:
353 """单步编辑(F046):仅 pending 可改——旧方案 rejected(reason=revised)+
354 同目标重提新方案,零 LLM 重调(编辑是本地操作);批准后不可改(改=reject
355 重提,F046 边界);改动工具时按新工具重打危险标(F023 联动)。
356 """
357 p = self._get(plan_id)
358 if p.status != "pending": # 只待批可改(批准后冻结)
359 raise PyHError("BUSY", ctx={
360 "hint": "仅待批方案可单步修改",
361 "advice": "已批准方案不可变;请 reject 后重新提案"})
362 if not (0 <= step_idx < len(p.steps)): # 步下标越界
363 raise_code("EVT-100", plan_id=plan_id, step_idx=step_idx,
364 hint=f"step_idx {step_idx} 越界(共 {len(p.steps)} 步)",
365 advice="核对方案步数(0 起)")
366 unknown = set(patch or {}) - {"action", "tool", "expected",
367 "risk", "intent"}
368 if unknown: # 脏键拒改(防打字错误静默丢字段)
369 raise_code("EVT-100", plan_id=plan_id, step_idx=step_idx,
370 hint=f"patch 含未知字段: {','.join(sorted(unknown))}",
371 advice="patch 字段 ∈ action/tool/expected/intent")
372 new_steps = [s.to_dict() for s in p.steps] # 值拷贝后单步字段替换
373 merged = {**new_steps[step_idx], **(patch or {})}
374 # intent 是 action/tool/expected 的派生态:改了源字段而未显式给 intent →
375 # 丢弃旧 intent 让重提时按新字段拼装(防修订后执行意图残留旧文案)
376 if "intent" not in (patch or {}) and any(k in (patch or {})
377 for k in ("action", "tool", "expected")):
378 merged.pop("intent", None)
379 new_steps[step_idx] = merged
380 sess = self._require_session(ctx)
381 await sess.append("plan.rejected", {
382 "plan_id": plan_id, "who": "user",
383 "reason": f"revised:step:{step_idx}"}, actor="user") # 旧版作废
384 self._plans.pop(plan_id, None) # 旧 plan 出注册表(事件留痕,偏离 11)
385 return await self.plan_propose(p.goal, ctx, # 同目标重提新版,显式 steps
386 steps=new_steps) # 零 LLM 调用

388 # ============================================================ 执行(F043/F046)
389 async def plan_execute(self, plan_id: int, ctx: Any) -> None:
390 """执行切换:逐步入队。逐 step:写 plan.exec.step → 步骤意图 submit 入队
391 (F043)并 wait_for;失败用户三选一(跳过/重试/中止),重试 ≤2 次仍败默认
392 中止;全部成功写 plan.done;中止写 plan.aborted(step=i);headless 无通道
393 自动中止(APR-501 安全默认)。批准只定方向,每步仍走正常 guard/审批链。
394 """
395 p = self._get(plan_id)
396 if p.status not in ("approved", "executing"): # 硬条件:批准后才执行
397 if p.status == "rejected":
398 # 拒绝后不再推进(EVENT-SCHEMA §3.5.2):已裁决方案不可执行——
399 # approve 后执行协程开跑前被 plan_reject 的竞态同样落此静默退场
400 return
401 raise PyHError("BUSY", ctx={
402 "hint": f"方案 {plan_id} 未批准,禁止执行",
403 "advice": "先 plan_approve 批准方案后再执行"})
404 sess = self._require_session(ctx)
405 tq = getattr(ctx, "task_queue", None)
406 if (tq is None or not callable(getattr(tq, "submit", None))
407 or not callable(getattr(tq, "wait_for", None))):
408 raise_code("CYC-999",
409 hint="ctx.task_queue 未接线:执行=逐步入队(F043)需要队列",
410 advice="装配层注入 TaskQueue(runner=agent_loop)后再批准")
411 p.status = "executing" # 批准 → 执行中(事件态由 exec.step 派生)
412 for i, step in enumerate(p.steps):
413 if p.status == "rejected": # 执行中被拒:停止后续步(偏离 7)
414 return
415 await sess.append("plan.exec.step", {
416 "plan_id": plan_id, "step_idx": i, "action": step.action},
417 actor="agent")
418 tries = 0
419 while True: # 逐步入队:队列单飞,天然串行(F043)
420 meta = {"plan_id": plan_id, "step_idx": i} # Task.meta 溯源(偏离 12)
421 tid = await tq.submit(step.intent, meta=meta)
422 r = await tq.wait_for(tid)
423 if getattr(r, "ok", False):
424 break # 成功 → 下一步
425 if tries >= MAX_RETRY: # 重试耗尽 → 默认中止(汇报,F046 边界)
426 await self._abort(p, ctx, i)
427 return
428 try:
429 choice = await self._user_choice(["跳过", "重试", "中止"], ctx)
430 except PyHError as e:
431 if e.code == "APR-501": # headless 无通道:直接中止(安全默认)
432 await self._abort(p, ctx, i)
433 raise
434 raise
435 if choice == "跳过":
436 break # 跳过:继续下一步
437 if choice == "中止":
438 await self._abort(p, ctx, i)
439 return
440 tries += 1 # 重试:同一步重新入队(重走 guard/审批)
441 if p.status == "executing": # 全部成功(未被中止/拒绝打断)
442 await sess.append("plan.done", {"plan_id": plan_id}, actor="system")
443 p.status = "completed" # 事件落定 → 内存对齐(INV-01)

445 # ============================================================ 过期清理(F045)
446 async def plan_sweep_expired(self, ctx: Any, *,
447 now: Optional[datetime] = None) -> int:
448 """过期清理:扫全部 pending 方案,超 24h 未确认自动 rejected(who=system,
449 reason=expired)并返回清理数;由 agent-loop 周期驱动与 approve 前即时检查
450 双路调用。now = 测试注入时钟(aware UTC,偏离 4)。
451 """
452 sess = self._require_session(ctx)
453 now = now or _utcnow()
454 n = 0
455 for p in list(self._plans.values()): # 快照遍历(清理中可安全变更)
456 if (p.status == "pending" and p.expires_at is not None
457 and p.expires_at <= now): # 24h 未确认(F045 边界)
458 await sess.append("plan.rejected", {
459 "plan_id": p.id, "who": "system", "reason": "expired"},
460 actor="system") # who=system 自动态
461 p.status = "rejected" # 自动作废,不执行
462 p.who = "system"
463 n += 1
464 return n

480 @classmethod
481 def rebuild_from_events(cls, events: list[Any]) -> "PlanManager":
482 """事件溯源重建(INV-01):顺序重放 plan.proposed/approved/rejected/
483 exec.step/done/aborted 重建注册表与状态——崩溃恢复/会话恢复后内存态 ≡
484 日志;孤儿引用(approved 前无 proposed)由写入层 EVT-101 拦截,此处防御
485 跳过。完成态映射为 completed(偏离 5);steps 从事件载荷重建,缺省 intent
486 由 Step.__post_init__ 拼装(偏离 2)。
487 """
488 m = cls()
489 for e in events: # seq 升序重放(只读不写回)
490 t = getattr(e, "type", "")
491 p = getattr(e, "payload", {}) or {}
492 if t == "plan.proposed":
493 pid = getattr(e, "seq", 0) # plan_id = proposed 事件 seq(§3.5.2)
494 created_at = _parse_dt(getattr(e, "ts", None)) or _utcnow()
495 expires = _parse_dt(p.get("expires_at")) or \
496 (created_at + timedelta(hours=PLAN_TTL_HOURS)
497 if p.get("expires_at") is None else None)
498 if expires is None: # 旧日志无 ts/expires:按创建时刻兜底
499 expires = created_at + timedelta(hours=PLAN_TTL_HOURS)
500 steps = [Step(**dict(s)) for s in p.get("steps", []) or []]
501 m._plans[pid] = Plan(
502 id=pid, goal=p.get("goal", ""), steps=steps,
503 selfcheck_ok=bool(p.get("selfcheck_ok", True)), why="",
504 status="pending", created_at=created_at, expires_at=expires)
505 else:
506 g = m._plans.get(p.get("plan_id"))
507 if g is None:
508 log.warning("orphan plan event seq=%s type=%s(防御跳过)",
509 getattr(e, "seq", "?"), t)
510 continue # 孤儿引用:重放不中断
511 if t == "plan.approved":
512 g.status = "approved"
513 g.who = p.get("who")
514 elif t == "plan.rejected":
515 g.status = "rejected" # reason=expired/revised 等不细分
516 g.who = p.get("who")
517 elif t == "plan.exec.step":
518 g.status = "executing" # 任一步开跑即执行中
519 elif t == "plan.done":
520 g.status = "completed"
521 g.who = "system"
522 elif t == "plan.aborted":
523 g.status = "aborted"
524 g.who = "system"
525 return m # 内存态 ≡ 日志派生态(INV-01)

528# ------------------------------------------------------------------ 纯函数
529def selfcheck_plan(steps: list[dict]) -> tuple[bool, str]:
530 """方案自检(纯函数,F045):步骤 1..8、每步 action/tool/expected 齐全、
531 risk=critical 的工具本层不可用(F014)则标 why;结果只展示不阻断(软指导,
532 批准权在用户)。零写副作用,propose 与测试直接调用。"""
533 if not steps or len(steps) > MAX_STEPS: # 空方案/超上限
534 return False, "步骤数须为 1..8"
535 for i, s in enumerate(steps): # 逐步字段完整性
536 if not s.get("action") or not s.get("tool"):
537 return False, f"步骤 {i + 1} 缺 action 或 tool(不可执行)"
538 if not s.get("expected"):
539 return False, f"步骤 {i + 1} 缺预期结果 expected(不可核对)"
540 if s.get("risk") == "critical": # critical 工具 guard 层直接不可用
541 return False, f"步骤 {i + 1} 工具 {s['tool']} 为 critical 不可执行,需换招"
542 return True, "" # 通过(展示 selfcheck_ok=True)
