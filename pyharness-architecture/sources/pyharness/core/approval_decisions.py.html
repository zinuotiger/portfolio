77 def _prepare_outcome(self, verdict: str, approval_id: int, by: str, *, reviewed=False) -> tuple:
78 """同步完成身份与 pending 校验,保证无效裁决立即抛错而不是后台吞掉。"""
79 if not reviewed:
80 self._require_human(by)
81 elif self._reviewer is None or by != self._reviewer.identity:
82 raise_code('APR-503', why='无独立审核器授权')
83 if self._detached:
84 raise_code("APR-503", why="审批服务已关闭")
85 req = self._pending.get(int(approval_id))
86 if req is None or req.state != "pending":
87 raise_code("APR-503", approval_id=approval_id,
88 why="未知或已裁决的 approval_id;同一审批至多一个结果(防重放)")
89 if req.batch_mono and (time.monotonic()-req.batch_mono)*1000 >= req.ttl_ms:
90 raise_code('APR-503', approval_id=approval_id, why='审批请求已过期')
91 if req.decision_inflight is not None:
92 raise_code("APR-503", approval_id=approval_id,
93 why=f"审批正在处理({req.decision_inflight}),拒绝重复裁决")
94 sess = req.log or self._session
95 if sess is None:
96 raise_code("CYC-999", module="approval",
97 hint="请求无会话日志落点,裁决无法强同步落盘")
98 type_ = _OUTCOME_EVENTS[verdict]
99 req.decision_inflight = verdict
100 return req, sess, type_

168 async def on_verdict(self, type_: str, payload: Any) -> None:

211 def _settle(self, req: ApprovalRequest, verdict: str, by: str, *,
212 remove: bool = True) -> None:
213 """终态迁移单点:批内全部 pending 成员迁移 + 取消 TTL + 解决各自 waiter。
214
215 幂等:成员已终态跳过(防重复解决);lead 从 _pending 摘除后,同 id 再收
216 裁决即 APR-503(重放闸)。合并等待者各自 waiter 在此级联解决(每个等待者
217 各自配对裁决结果,granted 后各自独立重入 guard 链)。
218 """
219 lead_ref = req.approval_id # approval identity(仅 lead 有)
220 for w in list(req.batch):
221 if w.state != "pending":
222 continue
223 w.state = verdict # 一次性迁移,终态不再接受裁决
224 w.by = by
225 w.decision_inflight = None
226 self._cancel_timer(w)
227 if verdict == "granted" and w.call_id:
228 # 授权↔执行绑定:executor 重入 approval 决策时按 call_id 取此校验
229 self._grant_slots[w.call_id] = w.binding
230 if w.call_id and lead_ref is not None:
231 # S4-P1-2:绑定 call_id → approval identity(= approval.requested 的
232 # seq)。merged waiter 自身无 approval_id,按批共享 lead 的 identity;
233 # 不新造 uuid,identity 恒来自真实请求事件。
234 self._ref_slots[w.call_id] = lead_ref
235 if not w.waiter.done():
236 w.waiter.set_result(verdict) # 唤醒 request() 等待者
237 if remove and req.approval_id is not None and req.approval_id in self._pending:
238 self._pending.pop(req.approval_id, None)
