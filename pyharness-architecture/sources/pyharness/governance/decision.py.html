263class DecisionEngine:
264 """决策装配 + 治理决策语义(纯;无 I/O、无事件、无执行)。
265
266 **不重实现 g1–g7**(Δ-1):规则求值的唯一真源仍是
267 ``tools_guard.PolicyRule.check`` / ``GuardChain.evaluate``;本类**消费**
268 一条已算好的 ``EvaluationResult``,只施加**治理决策语义**:
269
270 1. verdict 归一(Verdict 值域外 → CYC-999);
271 2. **单调性**:前序决策若为终局 REJECT,批准不可将其放宽(G-5 / GRD-403);
272 3. **无通道即拒**:verdict=APPROVAL 但无可用审批通道 → REJECT(APR-501 语义);
273 4. **批准结果**:``approval_verdict`` 为 ``denied`` / ``timeout`` 时
274 APPROVAL → REJECT(``granted`` 保持 APPROVAL,由执行侧重入校验);
275 5. 装配 Decision(id/ts/supersedes 经注入的工厂与时钟产生)。
276
277 **决策语义所有权(S3-2-1 R1 裁定,强制)**:下列原始条件由**既有所有者**
278 负责,**DecisionEngine 不得重复计算**:
279
280 | 条件 | 唯一所有者 |
281 |---|---|
282 | critical → reject | `GuardChain`(g-danger 规则 + `_evaluate_full` 兜底闸) |
283 | high + 无通道 → reject | `approval.py` APR-501 → `ToolExecutor._approval_round` |
284 | denied / timeout → 不执行 | `ToolExecutor._approval_round` |
285
286 因此 **`authorize()` 的生产路径不得传 `approval_available` /
287 `approval_verdict`**——二者保留在本签名中只为稳定 S3-1 Decision Contract 与
288 兼容面,其分支在生产路径中**不触发**。若未来要让本层重新拥有这些语义,须经
289 新的架构裁定。
290 """

292 def __init__(self, *, id_factory: Optional[Callable[[], str]] = None,
293 clock: Optional[Callable[[], str]] = None) -> None:
294 self._id_factory = id_factory or (lambda: uuid4().hex)
295 self._clock = clock or _utc_now

297 async def decide(self, evaluation: EvaluationResult, *,
298 principal: Principal,
299 call: Any = None,
300 policy: Any = None,
301 tool: Optional[str] = None,
302 inputs_digest: str = "",
303 approval_available: bool = True,
304 approval_verdict: Optional[str] = None,
305 prior: Optional[Decision] = None,
306 approval_ref: Optional[int] = None) -> Decision:
307 """把一条求值结果升格为治理决策(纯函数;确定性由注入设施保证)。"""
308 try:
309 verdict = Verdict(evaluation.verdict)
310 except ValueError:
311 raise_code("CYC-999", module="governance.decision", field="verdict",
312 why="求值结果 verdict 不在 {allow,reject,approval} 内",
313 got=str(evaluation.verdict))
314
315 # 2) 单调性:终局 reject 不可被批准放宽。
316 if prior is not None and prior.verdict is Verdict.REJECT:
317 verdict = Verdict.REJECT
318 # 3) 无审批通道 → APPROVAL 降为 REJECT。
319 elif verdict is Verdict.APPROVAL and not approval_available:
320 verdict = Verdict.REJECT
321 # 4) 审批结果:denied / timeout 使 APPROVAL 作废。
322 elif verdict is Verdict.APPROVAL and approval_verdict in ("denied", "timeout"):
323 verdict = Verdict.REJECT
324
325 tool_name = tool if tool is not None else str(getattr(call, "name", "") or "")
326 fingerprint = str(getattr(policy, "fingerprint", "") or "")
327 return Decision(
328 decision_id=self._id_factory(),
329 verdict=verdict,
330 tool=tool_name,
331 policy_refs=tuple(evaluation.policy_refs),
332 guard_ids=tuple(evaluation.guard_ids),
333 policy_fingerprint=fingerprint,
334 inputs_digest=inputs_digest,
335 principal=principal,
336 ts=self._clock(),
337 approval_ref=approval_ref,
338 supersedes=(prior.decision_id if prior is not None else None),
339 )
