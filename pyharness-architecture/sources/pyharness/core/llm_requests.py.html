34 def _output_limit(self, ctx: Any, configured: int) -> int:
35 """Clamp Platform requests to remaining declared output allowance.
36
37 This bounds the submitted generation parameter, not provider billing.
38 The existing sequential run owns its counters; other callers retain
39 their existing limit and budget gate.
40 """
41 runtime = getattr(ctx, 'platform_runtime', None)
42 if runtime is None:
43 return configured
44 counters = getattr(ctx, 'counters', None)
45 if counters is None:
46 from pyharness.errors import raise_code
47 raise_code('CFG-601', reason='platform_usage_counter_missing')
48 remaining = runtime.definition.budget.output_tokens - counters.task_total().out_tokens
49 if remaining <= 0:
50 from pyharness.core.scope import BudgetExhausted
51 from pyharness.core.budget_diagnostics import budget_detail
52 raise BudgetExhausted('Platform output token budget exhausted',diagnostics=budget_detail('output',
53 limit=runtime.definition.budget.output_tokens,used=counters.task_total().out_tokens,requested=1))
54 return min(configured, remaining)

69 async def _budgeted_request(self, messages, tools, *, ctx, streaming):
70 limit = self._output_limit(ctx, self._llm_cfg(ctx).max_tokens)
71 counters = getattr(ctx, 'counters', None)
72 release = lambda: None
73 if isinstance(counters, UsageCounters):
74 config=self._llm_cfg(ctx)
75 prices=_price_table(getattr(getattr(config,'usage',None),'unit_price',{}))
76 price=prices.get(self.model,{})
77 # Conservative byte reservation, including tool schemas/framing.
78 # Actual provider usage still settles the ledger; unknown prices
79 # cannot provide a monetary guarantee and remain explicitly unpriced.
80 input_bound=len(json.dumps([messages,tools],ensure_ascii=False).encode('utf-8'))
81 limit, release = counters.reserve_output(limit,input_tokens=input_bound,
82 input_rate=price.get('in_per_million',0)/1e6,
83 output_rate=price.get('out_per_million',0)/1e6)
84 try:
85 method = self._chat_stream if streaming else self._chat
86 return await method(messages, tools, ctx=ctx, output_limit=limit)
87 finally:
88 # Includes transport failure, malformed response and cancellation.
89 release()

94
95 全系统除本模块外不存在直连模型端点路径(INV-02);工具轮 content=null 归一 "";
96 双空响应(content null 且无 tool_calls)= 协议异常,不回空文本冒充成功(ADI §1.3)。

101 req = {"model": self.model, "messages": list(messages),
102 "tools": tools_clean or None, # 无工具整字段省略在传输层(wire 规则)
103 "temperature": cfg_llm.temperature, # 0-1.5(越窗已在配置层 CFG-601 拒)
104 "max_tokens": output_limit}
105 request_event = await ctx.session.append("llm.request", dict(_request_payload(self.model, self._deg,
106 tools),effective_output_limit=output_limit,configured_output_limit=cfg_llm.max_tokens), actor="llm", task_id=getattr(ctx, "task_id", None))
107 from pyharness.core.llm_diagnostics import observed_wait
108 observation = {}
109 async def invoke_transport():
110 from pyharness.core.context_observation import observe_transport_request
111 await observe_transport_request(ctx,req,transport,request_event.seq)
112 from pyharness.core.product_model_gate import check_model_attempt
113 await check_model_attempt(ctx, self.model, 'chat')
114 return await transport.complete(req)
115 try:
116 raw = await observed_wait( # 总时长闸(F017);只取消本任务
117 invoke_transport(), timeout=self.timeout.total_s, observation=observation)
118 except Exception as e: # noqa: BLE001 传输细分统一归一(ADI §7.2)

132 usage = getattr(raw, "usage", None)
133 if usage is not None:
134 await self.report_usage(usage, self.model, ctx=ctx) # F029:落 llm.usage+计数器
135 env = await ctx.session.append( # 偏离 1:单条 llm.response 落盘
136 "llm.response", _response_payload(self.model, finish, content, calls, request_event.seq),
137 actor="llm")
138 return LLMResponse(content=content, tool_calls=calls or None, usage=usage,
139 model=self.model, finish_reason=finish, raw=raw, seq=env.seq)

153 "temperature": cfg_llm.temperature, "max_tokens": output_limit,
154 "stream": True, "stream_options": {"include_usage": True}}

156 tools),effective_output_limit=output_limit,configured_output_limit=cfg_llm.max_tokens), actor="llm", task_id=getattr(ctx, "task_id", None))
157 from pyharness.core.llm_diagnostics import observed_wait
158 observation = {}
159 progress = {}
160 async def invoke_stream():
161 from pyharness.core.context_observation import observe_transport_request
162 await observe_transport_request(ctx,req,transport,request_event.seq)
163 from pyharness.core.product_model_gate import check_model_attempt
164 await check_model_attempt(ctx, self.model, 'chat_stream')
165 stream = transport.stream(req)

167 try:
168 text, calls, finish, usage = await observed_wait(
169 invoke_stream(), timeout=self.timeout.total_s, observation=observation)
170 except BaseException as e: # Include cancellation after emitted tokens.
171 partial = ''.join(progress.get('parts', [])) + ''.join(
172 slot['arguments'] for slot in progress.get('slots', {}).values())

174 if observed_usage is not None or partial:
175 # Missing provider totals remain unknown. Charge only an output
176 # estimate from received bytes, through the same task/ancestor
177 # ledger; never invent unobserved input or claim exact billing.
178 observed_usage = observed_usage or SimpleNamespace(
179 prompt_tokens=0, completion_tokens=_est_tokens(partial),
180 prompt_cache_hit_tokens=0, prompt_cache_miss_tokens=0,
181 estimated=True, input_unknown=True)
182 try:
183 await self.report_usage(observed_usage, self.model, ctx=ctx)
184 except Exception:

186 if isinstance(e, asyncio.CancelledError):
187 raise
188 code = normalize_exc(e)
189 if code == "LLM-303":
190 self._note_rate_limit(ctx) # 归一 303 计入连续限流(降级判定用)
191 await self._raise_diagnostic(e, code, ctx, request_event.seq)
192 _restore_call_names(calls, name_map) # sanitized → fs.read_file 还原
193 if not text and not calls:
194 await self._raise_diagnostic(ProviderProtocolError(), "LLM-304", ctx, request_event.seq, observation=observation)
195 if usage is not None: # 流式计量在 include_usage 末块(F029)
196 await self.report_usage(usage, self.model, ctx=ctx)
197 elif text or calls:
198 # P1-3 盲区修复:端点忽略 include_usage 或断流缺末块 → 整轮流式零入账,
199 # 预算闸(F032)对流式全程无感知。按已拼内容兜底估算 out_tokens(仅 output;
200 # input 不猜 0,账目保守),杜绝"流式长对话超支无感"。
201 fallback_usage = SimpleNamespace(
202 prompt_tokens=0, completion_tokens=_est_tokens(text),
203 prompt_cache_hit_tokens=0, prompt_cache_miss_tokens=0,
204 estimated=True, input_unknown=True)
205 await self.report_usage(fallback_usage, self.model, ctx=ctx)
206 env = await ctx.session.append( # 聚合完成落单条 llm.response(F027)
207 "llm.response", _response_payload(self.model, finish or "unknown",
208 text, calls, request_event.seq), actor="llm")
209 return LLMResponse(content=text, tool_calls=calls or None, usage=usage,
210 model=self.model, finish_reason=finish or "unknown",
211 raw=None, seq=env.seq)
