199 def _spawn(self, coro: Any) -> None:
200 try:
201 loop = asyncio.get_running_loop()
202 except RuntimeError:
203 loop = None
204 if self._disposed or loop is None:
205 close = getattr(coro, 'close', None)
206 if callable(close):
207 close()
208 return
209 task = loop.create_task(coro)
210 task.add_done_callback(self._task_done)
211
212 def _task_done(self, task: asyncio.Task) -> None:
213 try:
214 task.result()
215 except Exception as exc: # noqa: BLE001
216 if not self._disposed:
217 self.statusBar().showMessage(f"错误:{type(exc).__name__}: {exc}", 8000)
