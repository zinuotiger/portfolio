86def _checked_file(root, name, *, before_read=None, max_bytes=LIMIT):
87 """Bound the descriptor read and compare it with safe path metadata on both sides."""
88 gate_failed=False
89 try:
90 path = safe_path(root, name)
91 before = path.lstat()
92 if not stat.S_ISREG(before.st_mode) or not before.st_ino or before.st_nlink != 1:
93 raise_code('GRD-403', reason='snapshot_file_rejected')
94 if before.st_size > max_bytes:
95 raise_code('EVT-100', reason='snapshot_size_limit')
96 fd = os.open(path, os.O_RDONLY | getattr(os, 'O_BINARY', 0) | getattr(os, 'O_NONBLOCK', 0) | getattr(os, 'O_NOFOLLOW', 0))
97 with os.fdopen(fd, 'rb') as stream:
98 opened = os.fstat(stream.fileno())
99 if not stat.S_ISREG(opened.st_mode) or opened.st_nlink != 1 or identity(before) != identity(opened):
100 raise_code('EVT-101', reason='snapshot_source_changed')
101 if before_read is not None:
102 try:before_read()
103 except BaseException:
104 gate_failed=True
105 raise
106 # Authority may perform blocking root/account I/O. Bind the FD and
107 # current path again after that I/O, immediately before first bytes.
108 if identity(before)!=identity(os.fstat(stream.fileno())) or identity(before)!=identity(safe_path(root,name).lstat()):
109 raise_code('EVT-101',reason='snapshot_source_changed')
110 raw = stream.read(max_bytes + 1)
111 after_fd = os.fstat(stream.fileno())
112 after = safe_path(root, name).lstat()
113 if len(raw) != before.st_size or identity(before) != identity(after_fd) or identity(before) != identity(after):
114 raise_code('EVT-101', reason='snapshot_source_changed')
115 return SnapshotFile(path=name, size_bytes=len(raw), sha256=hashlib.sha256(raw).hexdigest(),
116 mtime_ns=str(before.st_mtime_ns), mode=stat.S_IMODE(before.st_mode),
117 device=before.st_dev, inode=before.st_ino), raw
118 except OSError:
119 if gate_failed:raise
120 raise_code('GRD-403', reason='snapshot_file_unavailable')
121 except PyHError as exc:
122 if gate_failed:raise
123 if exc.code not in {'GRD-403', 'EVT-100', 'EVT-101'}:
124 raise_code('GRD-403', reason='snapshot_file_rejected')
125 raise

153def collect_selection(row, command, *, before_read=None):
154 observed = observe_selection(row, command)
155 if observed['summary']['excluded']:
156 raise_code('GRD-403', reason='snapshot_selection_rejected')
157 root = Path(row['grant']['canonical_root'])
158 files, contents = [], {}
159 for name in command.paths:
160 check_content(name.encode('utf-8'))
161 file, raw = read_selected(root, name, before_read=before_read) if before_read is not None else read_selected(root, name)
162 check_content(raw)
163 files.append(file)
164 if sum(f.size_bytes for f in files) > LIMIT:
165 raise_code('EVT-100', reason='snapshot_size_limit')
166 contents[file.sha256] = raw
167 verify_selection(row, files, before_read=before_read)
168 grant = row['grant']
169 digest = selection_digest(grant['project_id'], grant['grant_id'], grant['revision'], grant['root_identity'], files)
170 return files, contents, digest
