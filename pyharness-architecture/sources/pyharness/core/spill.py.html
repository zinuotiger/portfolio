110async def enter(ctx: Any) -> Path:
111 """会话首访建私有目录(F039,幂等):spill_dir/<session_id>,权限 700。
112
113 已存在则幂等返回;创建/权限失败 → PERS-221 结构化上抛(Consumer 转
114 tool.error 回喂"输出归档写入失败,请重试")。返回归一后的会话 spill
115 目录 Path(后续 containment 基准)。
116 """
117 root = _spill_root(ctx)
118 if root is None:
119 raise_code("PERS-221", hint="storage.spill_dir 未配置(ctx.cfg/config 缺失)",
120 advice=_WRITE_FAIL_ADVICE)
121 sid = _session_id(ctx)
122 if not sid:
123 raise_code("PERS-221", hint="ctx.session.session_id 缺失(spill 装配错误)",
124 advice=_WRITE_FAIL_ADVICE)
125 d = root / sid
126 try:
127 d.mkdir(parents=True, exist_ok=True, mode=_MODE_PRIVATE_DIR)
128 try:
129 os.chmod(d, _MODE_PRIVATE_DIR) # POSIX owner 可遍历；Windows ACL 尽力而为
130 except OSError as e: # 权限设置失败同样上抛(spec 伪码)
131 raise_code("PERS-221", dir=str(d), cause=e, advice=_WRITE_FAIL_ADVICE)
132 except OSError as e:
133 raise_code("PERS-221", dir=str(d), cause=e, advice=_WRITE_FAIL_ADVICE)
134 return d.resolve() # 归一根,后续 containment 基准

138async def put(text: str, kind: str = "output", ctx: Any = None) -> dict:
139 """超限输出落盘(F039,Consumer 唯一入口):原文 → 会话私有 spill 文件。
140
141 顺序(spec 伪码权威):单文件超 10MB 先拒(PERS-223)→ 惰性建目录
142 (enter,PERS-221)→ ctx.redact 出口脱敏(INV-09)→ 临时文件+rename 原子
143 落盘(坏行不落地)→ 返回 EVENT-SCHEMA 同形 SpillMeta
144 {spilled, ref, chars, lines, preview(头500)}。
145
146 kind = 来源标注(read/fetch/exec/pty…,入错误现场与审计扩展点,偏离 7)。
147 """
148 if ctx is None:
149 raise_code("PERS-221", hint="ctx 缺失:put 必须带会话门面", advice=_WRITE_FAIL_ADVICE)
150 if not isinstance(text, str): # 契约:Consumer 已 render 成 str
151 raise_code("PERS-221", hint="put 仅接受 str(Consumer 须先 render_result_text)",
152 advice=_WRITE_FAIL_ADVICE)
153 limit = _max_per_file(ctx)
154 if len(text.encode("utf-8")) > limit: # 单文件 ≤10MB(**字节**,与键名同口径)
155 raise_code("PERS-223", bytes=len(text.encode("utf-8")), max=limit,
156 advice=_TOO_BIG_ADVICE)
157 d = await enter(ctx) # 惰性建目录(幂等;失败 PERS-221)
158 # 会话总量闸(R24 接线 `storage.spill.max_per_session_mb`):单文件闸挡不住
159 # "同一会话反复 spill" ⇒ 此处按**目录现有字节 + 本次**判上限,超则拒写。
160 cap = _max_per_session_bytes(ctx)
161 if cap > 0:
162 incoming = len(text.encode("utf-8"))
163 if _session_dir_bytes(d) + incoming > cap:
164 raise_code("PERS-223", op="per_session",
165 session_bytes=_session_dir_bytes(d), incoming=incoming,
166 max=cap,
167 advice="会话 spill 总量超上限"
168 "(storage.spill.max_per_session_mb);清理会话区内"
169 " spill-*.txt 或调大该键")
170 text = _redact(ctx, text) # 出口脱敏(INV-09,spill=明文出口)
171 name = f"{FILE_PREFIX}{uuid.uuid4().hex[:8]}.txt"
172 tmp = d / (name + ".tmp") # 临时文件:坏行不落地(rename 前不可见)
173 created = False
174 try:
175 with open(tmp, "x", encoding="utf-8", newline="",
176 opener=lambda path, flags: os.open(path, flags, MODE_PRIVATE)) as fh:
177 created = True
178 fh.write(text)
179 os.replace(tmp, d / name) # 原子替换(Windows os.replace 同 rename)
180 except OSError as e:
181 try: # 清理残留临时文件(尽力而为)
182 if created: # 不删独占创建失败时他人已有的文件
183 tmp.unlink(missing_ok=True)
184 except OSError:
185 pass
186 raise_code("PERS-221", dir=str(d), kind=kind, cause=e,
187 advice=_WRITE_FAIL_ADVICE)
188 return {"spilled": True,
189 "ref": f"{d.name}/{name}", # 会话内相对引用:ref=<sid>/spill-*.txt
190 "chars": len(text),
191 "lines": text.count("\n") + 1,
192 "preview": text[:SUMMARY_PREVIEW]} # 头 500 字符固定

196async def read(ref: str, start: int = 0, limit: int = DEFAULT_LIMIT,
197 ctx: Any = None) -> dict:
198 """按行取回(F039/LLM 可见工具 storage.spill):行区间 [start, start+limit)。
199
200 ref 必须先过 _resolve_ref containment(PERS-222 零读取);文件不存在(已
201 清理/会话结束)→ TLB-802;start<0/limit 超界 → TLB-803 零执行(schema 已
202 拦,Provider 直调路径兜底)。返回 {content, more, start, returned} 并把
203 已读字符数累入 counters.spill_read_chars(防循环烧预算)。
204 """
205 if ctx is None:
206 raise_code("PERS-222", ref=ref, advice=_REFUSED_ADVICE)
207 try:
208 start = int(start)
209 limit = int(limit)
210 except (TypeError, ValueError):
211 raise_code("TLB-803", tool=TOOL_NAME, field="start/limit",
212 advice="start/limit 须为整数;未执行任何操作")
213 if start < 0 or limit < 1 or limit > MAX_READ_LIMIT:
214 raise_code("TLB-803", tool=TOOL_NAME, start=start, limit=limit,
215 advice=f"start≥0 且 1≤limit≤{MAX_READ_LIMIT};未执行任何操作")
216 f = _resolve_ref(ref, ctx) # containment 单点判定
217 if f is None:
218 raise_code("PERS-222", ref=ref, advice=_REFUSED_ADVICE) # 零读取
219 if not f.is_file(): # ref 失效(已清理/会话结束/非文件)
220 raise_code("TLB-802", tool=TOOL_NAME, ref=ref,
221 advice="spill 引用已失效;重跑源头工具重新生成输出")
222 try:
223 with f.open("r", encoding="utf-8", newline="") as fh:
224 raw = fh.read()
225 except OSError as e: # 读 IO 失败 → 同样按失效回喂
226 raise_code("TLB-802", tool=TOOL_NAME, ref=ref, cause=e,
227 advice="spill 文件读取失败;重跑源头工具重新生成输出")
228 lines = raw.split("\n") # 与 put 行数口径严格同源(偏离 3)
229 chunk = lines[start:start + limit]
230 more = (start + limit) < len(lines) # 还有后续行
231 body = "\n".join(chunk)
232 _bump_counter(ctx, COUNTER_KEY, len(body)) # 已读量防循环
233 return {"content": body, "more": more, "start": start,
234 "returned": len(chunk)}

238def _resolve_ref(ref: Any, ctx: Any) -> Optional[Path]:
239 """ref → 私有区内绝对 Path;任一逃逸形态返回 None(PERS-222 唯一判定)。
240
241 拒绝:空/非 str、绝对路径(含盘符)、`..` 段、符号链接解析后越出 spill
242 根、跨会话引用(当前会话 id 已知时,偏离 2);全部零读取语义由调用方兑现。
243 """
244 if not isinstance(ref, str) or not ref:
245 return None
246 if os.path.isabs(ref) or ".." in Path(ref).parts:
247 return None
248 root = _spill_root(ctx)
249 if root is None:
250 return None
251 root = root.resolve() # 根归一(符号链接在根级先解)
252 cand = (root / ref).resolve() # 解符号链接后归一(逃逸即越界)
253 try:
254 cand.relative_to(root) # 最终落点必须在根内(Windows
255 except ValueError: # 路径比较大小写不敏感,偏离 4)
256 return None
257 parts = Path(ref).parts
258 if not parts:
259 return None
260 if cand.parent.name != parts[0]: # 首段必须是实际所在目录
261 return None
262 sid = _session_id(ctx) # 会话级隔离(偏离 2):本会话
263 if sid is not None and parts[0] != sid: # ref 只指本会话 spill 文件
264 return None
265 return cand
