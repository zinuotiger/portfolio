14def _with_message_origin(log_: Any, msgs: list[dict]) -> list[dict]:
15 """给派生消息附**来源标记**(只供 UI 判断消息来自谁,不参与 LLM 请求)。
16
17 定时任务到点触发时,Scheduler 会以 ``actor="user"`` 写一条 ``user.message``,
18 其 ``origin="schedule:<name>"``(schedule._fire)。``derive_messages`` 只投影
19 role/content,且**同时喂给 agent_loop 的 LLM 请求与 compaction 的 token 估算**,
20 故**不改其输出契约**;改在本层按 1:1 对应关系**复制**并附加键:
21
22 - ``session._fold_history`` 对每条 ``user.message`` 恰产出一条 ``role=user``
23 消息(``user.message_edited`` 是替换内容而非增删),故第 k 条 user 消息 ↔
24 第 k 条 user.message 事件;
25 - 复制字典,避免污染 ``derive_messages`` 的 history 缓存对象。
26
27 普通用户输入同样写 ``origin=None``(键恒在,便于前端统一判断)。
28 """
29 events = [e for e in log_.events_after(0) if e.type == 'user.message']
30 origins = [getattr(e, 'origin', None) for e in events]
31 answers = [e for e in log_.events_after(0) if e.type == 'agent.message']
32 j = 0
33 out: list[dict] = []
34 k = 0
35 for m in msgs:
36 row = dict(m)
37 if row.get("role") == "user":
38 row["origin"] = origins[k] if k < len(origins) else None
39 if k < len(events):
40 row['seq'] = events[k].seq
41 row['artifact_ids'] = (events[k].payload.get('meta') or {}).get('platform', {}).get('artifact_ids', [])
42 k += 1
43 elif row.get('role') == 'assistant' and row.get('content') and not row.get('tool_calls'):
44 if j < len(answers): row['seq'] = answers[j].seq
45 j += 1
46 out.append(row)
47 if k != len(origins): # 不变式破损:宁可少标记也不错标
48 log.warning("user.message 与派生 user 消息数不一致(events=%d msgs=%d)",
49 len(origins), k)
50 for row in out:
51 row.pop("origin", None)
52 row.pop('seq', None)
53 row.pop('artifact_ids', None)
54 if j != len(answers):
55 for row in out:
56 if row.get('role') == 'assistant': row.pop('seq', None)
57 return out
