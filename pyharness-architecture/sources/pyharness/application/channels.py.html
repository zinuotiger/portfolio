106 async def receive(self, name, raw, headers):
107 self.service._ensure_open()
108 config = FeishuConfig.model_validate(self.configs().get(name, {}))
109 if not config.enabled:
110 fail('channel_disabled')
111 event = verify_event(raw, headers, config, self.secrets(config))
112 if 'challenge' in event or event.get('ignored'):
113 return event
114 owner = hashlib.sha256((name+'\0'+event['user']+'\0'+event['chat']).encode()).hexdigest()
115 ident = hashlib.sha256((name+'\0'+event['message_id']).encode()).hexdigest()
116 # Idempotency is message-based: different event IDs for the same message
117 # cannot create another task. Limits bound disk growth until operator cleanup.
118 self.enqueue(ident,name,owner,event['message_id'],event['text'])
119 self.start()
120 return {'ok':True}
