103def register_default_llm(cfg: Any, *, force: bool = False,
104 counters: Any = None, registry: Any = None) -> str:
105 """注册 cfg.llm.model 的真实适配器(OpenAI 兼容端点)。
106
107 已注册同名模型 → 跳过(幂等;force=True 强制覆盖)。key 缺失 → 抛 LLM-304
108 明确提示,不静默降级。counters:会话级 F029 计数器(engine 装配传入共享实例,
109 使 adapter 计量入账与 scope 预算闸同源);缺省 = 适配器私有计数(旧行为)。
110 返回注册的模型名。
111
112 秘密纪律:api_key 是 "env:NAME"/"file:PATH" 引用字符串,值由适配器在
113 secret_ref_ok 解析(CRED-701/702/703)——本函数不接触也不落盘明文。
114 """
115 llm_cfg = cfg.llm
116 registry = llm_mod.adapters if registry is None else registry
117 api_ref = str(llm_cfg.api_key or "")
118 if not api_ref.startswith(("env:", "file:", "tenant:")):
119 llm_mod.raise_code(
120 "CRED-701", model=llm_cfg.model,
121 advice="llm.api_key 须 env:NAME / file:PATH / tenant:租户:档案 秘密引用")
122
123 credential_store, binding = None, None
124 if api_ref.startswith('tenant:'):
125 from pyharness.core.tenant_settings import TenantSettingsStore, tenants_dir, _STORES
126 tenant = api_ref.split(':', 2)[1]
127 root = tenants_dir(cfg.storage.root)
128 credential_store = _STORES.get(tenant)
129 if credential_store is None or credential_store.root.resolve() != root.resolve():
130 credential_store = TenantSettingsStore(root)
131 from types import MappingProxyType
132 binding = MappingProxyType(credential_store.binding(api_ref, str(llm_cfg.base_url)))
133
134 def actual_model(name):
135 existing = registry.get(str(name)) or llm_mod.adapters.get(str(name))
136 return str(existing.model) if isinstance(existing, llm_mod.OpenAICompatAdapter) else str(name)
137
138 def key_for(model_name: str) -> str:
139 if model_name in registry and not isinstance(registry[model_name], llm_mod.OpenAICompatAdapter):
140 return model_name # explicitly supplied deterministic/plugin adapter
141 limits = llm_mod.TimeoutLimits.from_cfg(llm_cfg.timeout)
142 from pyharness.core.model_connection_identity import connection_registry_key
143 return connection_registry_key(api_ref,llm_cfg.base_url,model_name,limits,binding,
144 str(credential_store.root.resolve()) if credential_store else '')
145
146 main_model = actual_model(llm_cfg.model)
147 main_key = key_for(main_model)
148 fallback_models = [actual_model(m) for m in (getattr(llm_cfg, 'fallback_models', None) or [])]
149 fallback_keys = [key_for(m) for m in fallback_models]
150 for actual, registry_key in zip([main_model, *fallback_models], [main_key, *fallback_keys]):
151 if registry_key not in registry or force:
152 previous = registry.get(registry_key)
153 if isinstance(previous, llm_mod.OpenAICompatAdapter) and previous._transport is not None:
154 raise_code('CRED-703', reason='live_model_replacement_requires_runtime_close')
155 registry[registry_key] = llm_mod.OpenAICompatAdapter(
156 model=actual, triple=llm_mod.AdapterTriple(
157 base_url=str(llm_cfg.base_url), api_key_ref=api_ref, model=actual,
158 credential_store=credential_store, credential_binding=binding),
159 limits=llm_mod.TimeoutLimits.from_cfg(llm_cfg.timeout), cfg=cfg)
160 llm_cfg.model = main_key
161 llm_cfg.fallback_models = fallback_keys
162 return main_key

165async def build_spine(cfg: Any, *, sid: str, sessions_dir: Path,
166 bus: Optional[EventBus] = None,
167 channel: Any = CHANNEL_UNDECLARED,
168 tenant_id: Optional[str] = None) -> EngineSpine:
169 """装配脊柱 8 模块束(create_agent 输入)+ 持久化三层。
170
171 流程:store(JSONL 真源)→ SessionLog(事件源,总线分发)→ AgentLoop(真
172 LLM)→ Scope(strict)→ Registry → spine。不 activate(agent 由调用方
173 enter 后 submit)。llm/scope/loop 全部真实,零 mock。
174
175 ``channel``(GAP-11):外壳身份声明,透传 ``build_runner_components``。
176 ``tenant_id``(GAP-10):租户归属,交给 ``open_session`` 盖到每条事件信封。
177 """
178 bus = bus or EventBus(**bus_kwargs_of(cfg)) # F005 背压阈值由装配层注入(N1)
179 # N1:攒批/定时落盘标量由**装配层**从 Settings 解析后注入(工厂不读 config)。
180 store = open_store(sid, dir=sessions_dir, **flush_kwargs_of(cfg))
181 try:
182 log_ = await open_session(sid, store, tenant_id=tenant_id) # async 工厂
183 spine = build_runner_components(cfg, log_=log_, bus=bus,
184 sessions_dir=sessions_dir,
185 store=store, channel=channel)
186 spine.persistence = store
187 return spine
188 except BaseException as primary:
189 # Ownership transfers only when the complete spine is returned.
190 try:
191 store.close()
192 except BaseException as cleanup_error:
193 primary.add_note(f'store cleanup failed: {type(cleanup_error).__name__}')

197async def _rollback_spine(spine: EngineSpine, primary: BaseException) -> None:
198 """Finish unreturned assembly cleanup without replacing its original failure."""
199 cleanup = asyncio.create_task(spine.close())
200 while True:
201 try:
202 await asyncio.shield(cleanup)
203 break
204 except asyncio.CancelledError:
205 if cleanup.done():
206 if cleanup.cancelled():
207 primary.add_note('engine cleanup task was cancelled')
208 break
209 # Retrieve any failure from the completed task on the next pass.
210 primary.add_note('cancellation received while awaiting engine cleanup')
211 except BaseException as cleanup_error:
212 primary.add_note(f'engine cleanup failed: {type(cleanup_error).__name__}')
213 break
214 try:
215 spine.scope.release()
216 except BaseException as cleanup_error:
217 primary.add_note(f'scope cleanup failed: {type(cleanup_error).__name__}')
