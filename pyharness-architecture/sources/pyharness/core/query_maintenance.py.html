26 async def rebuild(self, *, session_id: Optional[str] = None) -> int:
27 """索引整体重建(spec 伪码;F060/repair 对账权威入口):清派生行 → 重放重插。
28
29 作用域:session_id 指定 = 单会话(先 DELETE 该会话索引行);None = 全量
30 (DELETE 全表——派生可弃,JSONL 一行不改)。重放逐事件走与 on_event 同一
31 抽取逻辑(含 user.message_edited 修正,偏离 3),整批单事务提交(读侧见
32 旧态或新态,无中间态);返回重建插入行数。回放遇源异常/落库失败 →
33 PERS-202(坏行隔离纪律由注入 replay 源承担,PERS-201 域)。
34 """
35 if self._db is None:
36 raise_code("PERS-202", op="fts_rebuild",
37 hint="索引未装载(enter 未执行或已 detach)",
38 advice="先 enter/重新装载索引")
39 await self.flush() # 先清缓冲防新旧混写
40 rows: list[tuple] = []
41 edits: list[tuple] = []
42 for env in self._iter_replay(session_id):
43 item = self._extract(_field(env, "type"), env)
44 if item is None:
45 continue
46 if item[0] == "edit":
47 edits.append(item[1])
48 else:
49 rows.append(item[1])
50 try:
51 with self._lock:
52 with self._db: # DELETE+重插同事务(原子)
53 if session_id:
54 self._db.execute(
55 "DELETE FROM events_fts WHERE session_id=?", (session_id,))
56 self._db.execute(
57 "DELETE FROM fts_rows WHERE session_id=?", (session_id,))
58 else:
59 self._db.execute("DELETE FROM events_fts")
60 self._db.execute("DELETE FROM fts_rows")
61 # 幂等闸同事务重填(偏离 6 修复:rebuild 只清不填会让后续增量
62 # flush 绕过闸产生重复行,行键唯一语义被破坏)
63 for r in rows:
64 self._db.execute(
65 "INSERT INTO fts_rows(session_id, seq) VALUES(?,?)",
66 (r[0], r[1]))
67 self._db.executemany(
68 "INSERT INTO events_fts VALUES(?,?,?,?,?,?,?)", rows)
69 for sid, tseq, newc in edits:
70 self._db.execute(
71 "UPDATE events_fts SET content=? "
72 "WHERE session_id=? AND seq=?",
73 (_index_text(str(newc)), sid, tseq))
74 except sqlite3.Error as e:
75 raise_code("PERS-202", op="fts_rebuild", session=session_id,
76 hint=f"FTS 重建落库失败: {e}",
77 advice="修存储后重试 rebuild;索引为派生副本可整体重建")
78 log.info("session fts rebuilt session=%s rows=%d edits=%d",
79 session_id or "*", len(rows), len(edits))

82 async def reconcile_sources(self, sources, *, authorized_ids=None):
83 """Incrementally project an explicitly authorized set; prune removed access.
84
85 Watermarks are derived metadata. A crash before watermark commit replays
86 idempotently, and recreation of the database rebuilds from sequence zero.
87 Caller serializes reconcile/query within its tenant's application lock.
88 """
89 db = self._db
90 if db is None:
91 raise_code('PERS-202', op='fts_reconcile')
92 db.execute('CREATE TABLE IF NOT EXISTS source_progress (sid TEXT PRIMARY KEY, seq INTEGER NOT NULL)')
93 previous = dict(db.execute('SELECT sid,seq FROM source_progress'))
94 known = set(previous) | {r[0] for r in db.execute('SELECT DISTINCT session_id FROM fts_rows')}
95 authorized = set(sources) if authorized_ids is None else set(authorized_ids)
96 if not set(sources) <= authorized:
97 raise_code('EVT-100', hint='Index sources exceed authorized scope')
98 for sid in known - authorized:
99 await self.delete_session(sid)
100 db.execute('DELETE FROM source_progress WHERE sid=?',(sid,))
101 scanned = 0
102 self._sources = dict(sources)
103 for sid, source in sources.items():
104 last = previous.get(sid, 0)
105 current = source.stats()['seq']
106 if current < last:
107 await self.delete_session(sid)
108 self._sources[sid] = source
109 last = 0
110 for event in source.events_after(last):
111 await self.on_event(event.type, event)
112 scanned += 1
113 await self.flush()
114 db.execute('INSERT OR REPLACE INTO source_progress VALUES (?,?)',(sid,current))
115 db.commit()
116 return scanned
