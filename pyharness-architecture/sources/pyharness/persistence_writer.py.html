16 async def append(self, env: Envelope, sync: bool = False) -> None:
17 """Envelope → 单行落盘(F011 双速写)。
18
19 sync=True(强同步三类:user.message/guard.rejected/approval.* 等):立即
20 write+flush,成功才返回(崩溃一致性锚点);否则入 _pending 攒批(满 batch
21 即刷,间隔定时器兜底 —— **定时器不在本类**:由 ``EngineSpine`` 持有
22 (``start_flush_ticker``,间隔读本实例 ``flush_interval_s``),见 GAP-13)。
23 异常:强同步 OSError → PERS-202 直接抛;异步 3 次重试仍败 → PERS-202 +
24 会话暂停;暂停中(拒新不丢旧)→ PERS-202。
25 """
26 self.ensure_writable() # 拒新不丢旧:暂停期拒绝新事件
27 line = env.model_dump_json() + "\n" # 信封+payload 拍平单行(§3.6)
28 if sync: # 强同步点:成功才返回
29 # 入 pending + 立即 flush:物理序 = 入队序 = seq 序。若直接 write,
30 # 会与非 sync 攒批行交错 → 文件物理序 ≠ seq 序(5 在 3/4 前),
31 # 回放遇 seq 倒退即弃 → 重启后 0 事件(实测 bug)。
32 try:
33 self._pending.append((env.seq, line))
34 await self._flush_pending_all()
35 except OSError as e:
36 self._fail_streak += 1
37 if self._failure_state_reached(): # GAP-2:强同步同样受阈值约束
38 await self._enter_failure_state(why=str(e))
39 raise_code("PERS-202", seq=env.seq,
40 advice="会话暂停(fail-streak/重试队列达阈值),"
41 "repair 后恢复;本行仍在重试队列")
42 raise_code("PERS-202", seq=env.seq, why=str(e),
43 advice="落盘通道故障;跑 repair(F060)")
44 else: # 普通事件:攒批(内存即对订阅者可见)
45 self._pending.append((env.seq, line))
46 if len(self._pending) >= self.flush_batch: # ≥64 条 → 立即批量 flush
47 await self._flush_batch()
48 # 间隔定时器兜底:由 EngineSpine.start_flush_ticker 周期性调 flush()
49 # (append 内不 sleep;GAP-13 前该定时器不存在,注释与实现不符)
50 self._maybe_rotate() # >50MB → 轮转(§8.3.4)

52 async def _flush_pending_all(self) -> None:
53 """内部:全量 pending + 重试队列写盘(强同步路径用;失败 → OSError 上抛)。
54
55 失败语义(**F-SYNC-1 修复**):**本批一行不丢**——写失败时把
56 ``retry_old + pending_now`` **完整回填** ``_retry_q``(保序)后上抛
57 ``OSError``;不丢任何一条、不重复任何一条。
58
59 为什么这样能恢复 caller-visible failure:本函数由装配层适配器(总线订阅者)
60 调用,其异常被总线按 EVT-103 隔离——若此处**丢弃**本批,``session.append``
61 步骤 9 的 ``flush(seq)`` 会面对空队列而**静默返回成功**(调用方以为已落盘,
62 实际丢行)。回填后,步骤 9 成为对同一批的**第二次真实尝试**:
63
64 - 成功 ⇒ 事件真正持久,``append`` 成功返回(且 replay 可读出);
65 - 仍失败 ⇒ ``PERS-202`` 上抛至调用方(**不再 silent success**),行仍留在
66 ``_retry_q`` 待 repair 后恢复。
67
68 排序:重试行恒先写(旧行 seq 更小),与 ``flush`` / ``_flush_batch`` 三处
69 写入路径语义一致——**seq 不会倒退**,replay 顺序正常。
70 """
71 if not self._pending and not self._retry_q:
72 return
73 retry_old = list(self._retry_q)
74 pending_now = list(self._pending)
75 # 一致性校验先行(无副作用):同 seq 异内容 → PERS-202 fail-closed,
76 # 此时尚未改动任何队列 → 两份数据都保留,绝不静默覆盖。
77 to_write = _resolve_by_seq(retry_old + pending_now)
78 self._pending = deque()
79 try:
80 self._write_batch(to_write)
81 self._fh.flush()
82 self._fail_streak = 0
83 self._retry_q.clear()
84 except BaseException:
85 self._retry_q = deque(retry_old)
86 self._retry_q.extend(pending_now) # F-SYNC-1:本批完整回填(保序)
87 raise

89 async def flush(self, up_to_seq: Optional[int] = None) -> None:
90 """公开 flush:把 _pending 与 _retry_q 中 seq ≤ up_to_seq 的行写盘+flush。
91
92 up_to_seq=None = 全量(定时器/关闭前);成功才返回(强同步契约);OSError →
93 行回重试队列(PERS-202,拒新不丢旧)。调用方:session 强同步点、
94 ``EngineSpine.start_flush_ticker`` 的间隔定时器(GAP-13)、进程收尾。
95 两队列同受 up_to_seq 过滤(P3:此前 retry_q 不过滤,>up_to_seq 的旧重试行会被
96 提前写出、破坏 seq 升序)。
97 """
98 if not self._pending and not self._retry_q:
99 return # 无积压:空操作
100 def _within(seq: int) -> bool:
101 return up_to_seq is None or seq <= up_to_seq
102 take = [it for it in self._pending if _within(it[0])]
103 keep = [it for it in self._pending if not _within(it[0])]
104 self._pending = deque(keep)
105 retry_original = list(self._retry_q) # 保序(失败回填需与写序一致)
106 retry_take = [it for it in retry_original if _within(it[0])]
107 retry_keep = [it for it in retry_original if not _within(it[0])]
108 to_write = retry_take + take # 先重试旧行再新取(seq 升序)
109 if not to_write:
110 self._retry_q = deque(retry_keep) # 全在 up_to_seq 之后:原位保留
111 return
112 try:
113 self._write_batch(to_write)
114 self._fh.flush()
115 self._fail_streak = 0 # 成功:复位连续失败计数
116 self._retry_q = deque(retry_keep) # 仅剩 >up_to_seq 的旧重试行待刷
117 # R14-15:写通道**确已恢复** ⇒ 解除暂停(recovering→normal)。此前解除点只有
118 # ``SessionStore.repair()``(旧面,生产零调用者)⇒ 一旦因磁盘故障达阈值暂停,
119 # 本进程内**永不恢复**:``append`` 恒被 "拒新不丢旧" 挡回 PERS-202,而重试队列
120 # 只靠定时器重写、永不解除暂停(实测)。现由**既有** flush 定时器
121 # (``EngineSpine.start_flush_ticker``)在通道恢复后自动解除,不需新调用方。
122 if self._suspended:
123 self._suspended = False
124 log.warning("会话写通道恢复:解除暂停(重试队列已落盘)path=%s",
125 self.path)
126 except BaseException as e:
127 self._retry_q = deque(retry_original) # 旧行原样保序保留
128 self._retry_q.extend(take) # 新取入队:不重不漏
129 if not isinstance(e, OSError):
130 raise
131 self._fail_streak += 1
132 if self._failure_state_reached(): # GAP-2:公开 flush 同样受阈值约束
133 await self._enter_failure_state(why=str(e))
134 raise_code("PERS-202", n=len(take),
135 advice="会话暂停(fail-streak/重试队列达阈值),"
136 "repair 后恢复;行不丢")
137 raise_code("PERS-202", n=len(take), why=str(e),
138 advice="行已入重试队列不丢;repair 后恢复")

140 def _failure_state_reached(self) -> bool:
141 """失败状态判据(**单一真源**;GAP-2):连续失败 ≥ N 或重试队列越界。
142
143 全部写路径(强同步 ``append`` / 公开 ``flush`` / 异步 ``_flush_batch``)
144 共用本判据——修复前只有异步路径检查,强同步路径的 ``_fail_streak``
145 只增不判 ⇒ 磁盘持续故障时永不暂停、队列无界增长。
146 """
147 return (self._fail_streak >= _FAIL_STREAK_LIMIT
148 or len(self._retry_q) > _RETRY_Q_LIMIT)

150 async def _enter_failure_state(self, *, why: str = "") -> None:
151 """进入失败状态:发**可观察证据**(system.error)并暂停会话(拒新不丢旧)。
152
153 ``_suspended=True`` 后的新 ``append`` 一律 ``PERS-202``(拒新),已入
154 ``_retry_q`` 的行**一行不丢**(repair → recovering → normal 后恢复)。
155 """
156 await self._on_system_event("system.error", {
157 "code": "PERS-202",
158 "advice": "落盘通道故障,会话暂停(拒新不丢旧);已入重试队列的行不会丢,"
159 "通道恢复后由定时 flush 自动解除暂停(R14-15);"
160 "若进程先退出,队列内事件需跑 repair(F060) 复核",
161 "fail_streak": self._fail_streak,
162 "retry_q": len(self._retry_q),
163 **({"why": why} if why else {})})
164 self._suspended = True

166 def _heal_torn_tail(self) -> None:
167 """首写前给崩溃残片补一个行尾分隔(2026-09-21 R13-2)。
168
169 崩溃可能留下**无换行的尾部半行**;此后任何 append 都会与该残片拼在**同一物理
170 行** ⇒ 新事件不可解析(内存以为已落盘、磁盘读不回 = **静默丢事件**,实测:
171 `…,"type":"agent.mess{"seq":3,…`)。此处**只补一个 `\\n`**:残片自成一(坏)行、
172 新事件保持完整可解析;**不删任何字节**(残片留待 repair 按"中部坏行"口径隔离),
173 保持"open/回放不修"的既有口径(§repair:先备份再截断)。
174 """
175 if not self._torn_tail:
176 return
177 self._fh.write("\n")
178 self._fh.flush()
179 self._torn_tail = False

181 def _write_batch(self, rows: list) -> None:
182 """Commit actual missing rows, reconciling uncertain writes before retry.
183
184 The cache is derived from complete physical rows, never from write()'s
185 return value. Drain uncertain TextIO buffers before reading the file;
186 flush/fsync failure cannot cause a visible prefix to be appended twice.
187 """
188 rows = _resolve_by_seq(rows)
189 self._fh.flush()
190 self._reconcile_rows()
191 if self._torn_tail:
192 self._heal_torn_tail()
193 # A short write may contain the entire JSON object except its newline.
194 # Reconcile the now complete row before deciding which suffix to retry.
195 self._reconcile_rows()
196 missing = []
197 for seq, line in rows:
198 expected = json.loads(line)
199 actual = self._committed_rows.get(seq)
200 if actual is not None:
201 if actual != expected:
202 raise_code('PERS-202', op='seq_conflict', seq=seq)
203 else:
204 if self._committed_rows and seq <= max(self._committed_rows):
205 raise_code('PERS-202', op='out_of_order', seq=seq)
206 missing.append((seq, line))
207 # An fsync failure is an unknown durability result, never business success.
208 os.fsync(self._fh.fileno())
209 for _seq, line in missing:
210 if self._fh.write(line) != len(line):
211 raise OSError('short JSONL write')
212 self._fh.flush()
213 os.fsync(self._fh.fileno())
214 self._reconcile_rows()

216 def _reconcile_rows(self) -> None:
217 if not self._segments_scanned:
218 for path in self._rotated_paths():
219 for env in _iter_replay([path], self._quarantine):
220 self._remember_committed(env.model_dump(mode='json'))
221 self._segments_scanned = True
222 with self.path.open('rb') as source:
223 if source.seek(0, os.SEEK_END) < self._scan_offset:
224 raise_code('PERS-202', op='unexpected_truncation')
225 source.seek(self._scan_offset)
226 while True:
227 start = source.tell()
228 line = source.readline()
229 if not line:
230 self._torn_tail = False
231 break
232 if not line.endswith(b'\n'):
233 self._torn_tail = True
234 self._scan_offset = start
235 break
236 try:
237 env = Envelope.model_validate_json(line)
238 validate_payload(env.type, env.payload)
239 value = env.model_dump(mode='json')
240 except Exception:
241 self._scan_offset = source.tell()
242 continue # damaged bytes remain for explicit repair
243 self._remember_committed(value)
244 self._scan_offset = source.tell()

246 def _remember_committed(self, value):
247 seq = value['seq']
248 if seq in self._committed_rows:
249 raise_code('PERS-202', op='duplicate_physical_seq', seq=seq)
250 if self._committed_rows and seq <= next(reversed(self._committed_rows)):
251 raise_code('PERS-202', op='out_of_order', seq=seq)
252 self._committed_rows[seq] = value
253
254 def _invalidate_scan(self):
255 self._scan_offset = 0
256 self._committed_rows.clear()
257 self._segments_scanned = False

259 def _tail_looks_torn(self) -> bool:
260 """写失败后判定"文件是否停在行中"(R13-3)。
261
262 句柄是**缓冲**的 ⇒ 半行可能还在缓冲里:先尽力 ``flush`` 把它推到盘上,再按
263 事实读尾部(``detect_truncation``)。推不出去(仍 OSError)⇒ 无法判定,保守
264 按"可能半行"处理(下轮写前补分隔,最坏多一个空行 —— 空行被 replay 跳过,
265 代价远小于丢事件)。
266 """
267 try:
268 self._fh.flush()
269 except OSError:
270 return True
271 return detect_truncation(self.path) is not None

273 async def _flush_batch(self) -> None:
274 """内部:攒批写 + 失败重试(摘批先行:并发 append 不阻塞)。
275
276 失败语义:行全部回填 ``_retry_q``(拒新不丢旧),``_fail_streak += 1``;
277 达 ``_failure_state_reached`` ⇒ 暂停会话 + ``PERS-202``;未达阈值 ⇒
278 **静默保留**(等下一次 flush 定时器重试,不在批量路径上抛——批量由
279 ``append`` 的非 sync 分支调用,抛会打断事件发射)。
280 """
281 batch, self._pending = self._pending, deque()
282 retry = list(self._retry_q)
283 to_write = retry + list(batch)
284 if not to_write:
285 return
286 try:
287 self._write_batch(to_write)
288 self._fh.flush()
289 self._fail_streak = 0 # 成功:复位失败计数
290 self._retry_q.clear()
291 except BaseException as e:
292 self._retry_q = deque(retry)
293 self._retry_q.extend(batch) # 拒新不丢旧:全部回重试队列
294 if not isinstance(e, OSError):
295 raise
296 self._fail_streak += 1
297 if self._failure_state_reached():
298 await self._enter_failure_state(why=str(e))
299 raise_code("PERS-202", advice="会话暂停,repair 后恢复")
