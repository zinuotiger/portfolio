341 def _queue(self) -> Any:
342 """惰性取本会话队列;取值异常上抛，禁止伪装成未装配。"""
343 getter = self._queue_getter
344 if getter is None:
345 return None

348 async def _pause_queue_or_record(self, sess: Any) -> None:
349 q = self._queue()
350 if q is not None:
351 # Record ownership before awaiting: failed/cancelled pause may have
352 # already changed the real queue, so cleanup must reach that queue.
353 self._paused_queue = q
354 self._queue_driven = True
355 await q.pause("approval", by="system")
356 else:
357 await sess.append("queue.suspended", {"reason": "approval"}, actor="system")
358
359 async def _resume_queue_or_record_async(self, sess: Any) -> None:
360 if self._queue_driven:
361 await self._paused_queue.resume("approval", by="system")
362 self._queue_driven = False
363 self._paused_queue = None
364 else:
365 await (self._suspend_log or sess).append(
366 "queue.resumed", {"reason": "approval"}, actor="system")
