31 def __post_init__(self) -> None:

117 def reserve_output(self, requested, *, input_tokens=0, input_rate=0.0, output_rate=0.0):
118 """Atomic on the runtime loop: reserve before awaiting any transport."""
119 from pyharness.core.scope import BudgetExhausted
120 from pyharness.core.budget_diagnostics import budget_detail
121 accounts = list(self.ancestors())
122 available = int(requested)
123 for account in accounts:
124 limits = account._limits
125 if limits is None:
126 continue
127 used = account.task_total()
128 owner='task' if account is self else 'parent'
129 if used.in_tokens >= limits.max_in_tokens or used.in_tokens + account._reserved_in + input_tokens > limits.max_in_tokens:
130 raise BudgetExhausted('Input allowance exhausted', diagnostics=budget_detail('input',
131 limit=limits.max_in_tokens,used=used.in_tokens,reserved=account._reserved_in,
132 requested=input_tokens,account=owner,basis='conservative_utf8_bytes'))
133 if used.cost_est >= limits.max_cost_yuan:
134 raise BudgetExhausted('Cost allowance exhausted',diagnostics=budget_detail('cost',
135 limit=limits.max_cost_yuan,used=used.cost_est,reserved=account._reserved_cost,
136 requested=input_tokens*input_rate+output_rate,account=owner,basis='estimated_yuan'))
137 available = min(available, limits.max_out_tokens-used.out_tokens-account._reserved_out)
138 cost_left = limits.max_cost_yuan-used.cost_est-account._reserved_cost-input_tokens*input_rate
139 if cost_left < 0:
140 raise BudgetExhausted('No unreserved task or parent cost allowance',diagnostics=budget_detail('cost',
141 limit=limits.max_cost_yuan,used=used.cost_est,reserved=account._reserved_cost,
142 requested=input_tokens*input_rate+output_rate,account=owner,basis='estimated_yuan'))
143 if output_rate > 0:
144 available = min(available, int(max(0,cost_left)/output_rate))
145 if available <= 0:
146 dimension='cost' if output_rate>0 and cost_left<output_rate else 'output'
147 raise BudgetExhausted('No unreserved task or parent output allowance',diagnostics=budget_detail(dimension,
148 limit=limits.max_cost_yuan if dimension=='cost' else limits.max_out_tokens,
149 used=used.cost_est if dimension=='cost' else used.out_tokens,
150 reserved=account._reserved_cost if dimension=='cost' else account._reserved_out,
151 requested=input_tokens*input_rate+output_rate if dimension=='cost' else 1,
152 account=owner,basis='estimated_yuan' if dimension=='cost' else 'tokens'))
153 for account in accounts:
154 account._reserved_out += available
155 account._reserved_in += input_tokens
156 account._reserved_cost += input_tokens*input_rate+available*output_rate
157 released = False
158 def release():
159 nonlocal released
160 if not released:
161 released = True
162 for account in accounts:
163 account._reserved_out -= available
164 account._reserved_in -= input_tokens
165 account._reserved_cost = max(0, account._reserved_cost-input_tokens*input_rate-available*output_rate)
166 return available, release

168 def task_add(self, payload: dict) -> None:
169 """按 llm.usage 事件 payload 入账(事件即事实;计数器可整体重建)。"""
170 model = payload.get("model") or "?"
171 m = self.by_model.setdefault(model, UsageCounters())
172 self.requests += 1
173 m.requests += 1
174 self.in_tokens += int(payload.get("in_tokens") or 0)
175 self.out_tokens += int(payload.get("out_tokens") or 0)
176 self.cache_hit += int(payload.get("cache_hit") or 0)
177 self.cost_est += float(payload.get("cost_est") or 0.0)
178 # 模型细分按各自增量累计(与总量并行,互不覆盖)
179 m.in_tokens += int(payload.get("in_tokens") or 0)
180 m.out_tokens += int(payload.get("out_tokens") or 0)
181 m.cache_hit += int(payload.get("cache_hit") or 0)
182 m.cost_est += float(payload.get("cost_est") or 0.0)
183 self._rate_limit_streak = 0
184 if self._session_total is not None:
185 self._session_total.task_add(payload)
186 parent = self._parent
187 while parent is not None:
188 # A legacy session counter already received the charge above.
189 if parent is not self._session_total:
190 if parent._descendants is None:
191 parent._descendants = UsageCounters()
192 parent._descendants.task_add(payload)
193 parent = parent._parent
