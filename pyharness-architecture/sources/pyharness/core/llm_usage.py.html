9def estimate_cost(model: str, usage: Any, prices: dict, hit_discount: float = 0.1) -> float:
10 """成本估算(ADI §8.2,元/百万):未命中输入 + 命中输入×折扣 + 输出;只进事件/报表(N14)。
11
12 usage 无缓存字段厂商按全价估算(保守);prices 缺模型直取 KeyError 属调用方先查表(偏离 4)。
13 hit_discount 默认 0.1(缓存命中官方约 1/10 价)——原默认 1.0 使命中按全价计,成本恒高估
14 (P3:estimate_cost 折扣)。调用方未传折扣时取真实折扣,而非全价。
15 """
16 p = prices[model] # {in_per_million, out_per_million}
17 hit = getattr(usage, "prompt_cache_hit_tokens", 0) or 0
18 miss = getattr(usage, "prompt_cache_miss_tokens", 0) or 0
19 if not (hit or miss): # 无缓存字段厂商:prompt 全按未命中
20 miss = getattr(usage, "prompt_tokens", 0) or 0
21 billed_in = miss + hit * hit_discount # 命中按折扣价(官方约 1/10)
22 out = getattr(usage, "completion_tokens", 0) or 0
23 return billed_in / 1e6 * p["in_per_million"] + out / 1e6 * p["out_per_million"]

25def _price_table(unit_price: Any) -> dict:
26 """config 单价表(pydantic UnitPriceCfg 或 dict)→ 纯 dict {模型: {in,out}}。"""
27 out: dict = {}
28 for name, p in (unit_price or {}).items():
29 if isinstance(p, dict):
30 out[name] = {"in_per_million": float(p["in_per_million"]),
31 "out_per_million": float(p["out_per_million"])}
32 else: # pydantic 模型:attr 取数
33 out[name] = {"in_per_million": float(p.in_per_million),
34 "out_per_million": float(p.out_per_million)}
35 return out

67async def report_usage(usage: Any, model: str, *, ctx: Any,
68 counters: Optional[UsageCounters] = None) -> Any:
69 """F029 token 计量:每成功请求落 llm.usage 事件 + 更新 UsageCounters(F029 唯一写入)。
70
71 usage 事件即事实,计数器可整体重建(INV-01);预算硬闸判据 = **token 与 cost 并列**
72 (out/in/cost_est 任一 ≥ 上限即 exhausted;cost 由**软约束**单价表估算,上限见 CFG §3.1
73 `budget.task.max_cost_yuan` —— 实测「纯成本超限」即拦,R21 更正了旧措辞「只读 token 数」);
74 价格表缺模型 → cost_est 记 0 照常记账(偏离 4);缺失缓存字段按 0,不做厂商猜。
75 """
76 llm_cfg = _llm_cfg_from_ctx(ctx)
77 unit_price = getattr(getattr(llm_cfg, "usage", None), "unit_price", None) or {}
78 prices = _price_table(unit_price)
79 cost: Optional[float] = None
80 if model in prices: # 缺单价表不炸记账(仅报表,偏离 4)
81 cost = estimate_cost(model, usage, prices)
82 payload = {
83 "model": model, # 实际服务模型(降级后记备用,不记主名)
84 "in_tokens": int(getattr(usage, "prompt_tokens", 0) or 0),
85 "out_tokens": int(getattr(usage, "completion_tokens", 0) or 0),
86 "cache_hit": int(getattr(usage, "prompt_cache_hit_tokens", 0) or 0),
87 "cost_est": cost,
88 "estimated": bool(getattr(usage, 'estimated', False)),
89 "input_unknown": bool(getattr(usage, 'input_unknown', False)),
90 }
91 ev = await ctx.session.append("llm.usage", payload, actor="llm")
92 cnt = counters if counters is not None else getattr(ctx, "counters", None)
93 if cnt is not None:
94 cnt.task_add(ev.payload) # 预算硬闸读 token(F032)
95 return ev
