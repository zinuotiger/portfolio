48 def _read(self, stream, *, responses):

69 def _start(self):
70 if self.owner:
71 return
72 argv = [self.options['node'], str(Path(__file__).with_name('browser_runtime.cjs')),
73 json.dumps(self.options, separators=(',', ':'))]
74 self.private_root = tempfile.TemporaryDirectory(prefix='pyharness-browser-')
75 env = _mcp_env()
76 root = Path(self.private_root.name)
77 # Keep the OS user identity unchanged: native IME hooks may outlive the
78 # browser and keep their user-directory log open. Chrome's own profile
79 # and TEMP remain private. No persistent browser context is used.
80 home = Path(env.get('USERPROFILE') or env.get('HOME') or root / 'home')
81 for key, relative in [('APPDATA','AppData/Roaming'),('LOCALAPPDATA','AppData/Local')]:
82 path = home / relative

85 for key, relative in [('TMP','tmp'),('TEMP','tmp')]:
86 path = root / relative

91 except BaseException:
92 self.close()
93 raise

100 def call(self, name, arguments):
101 with self.lock:
102 if self.closed:
103 raise ValueError('browser_carrier_closed')
104 arguments = validate_call(name, arguments)
105 if name == 'close':
106 self.close()
107 return {'closed': True}
108 self.seq += 1
109 request = {'id': self.seq, 'name': name, 'arguments': arguments}
110 try:
111 self._start()
112 self.owner.proc.stdin.write((json.dumps(request, ensure_ascii=False) + '\n').encode('utf-8'))
113 self.owner.proc.stdin.flush()
114 result = self.output.get(timeout=self.timeout + 2)
115 if type(result) is not dict or result.get('id') != self.seq or set(result) != {'id', 'result'}:
116 raise ValueError('browser_protocol_failed')
117 value = result['result']
118 if type(value) is not dict:
119 raise ValueError('browser_protocol_failed')
120 if value.get('fatal'):
121 self.close()
122 return value
123 except BaseException:
124 self.close()
125 raise ValueError('browser_carrier_unconfirmed') from None

128 with self.lock:
129 self.closed = True

132 if stream and not stream.closed:
133 stream.close()

136 except subprocess.TimeoutExpired:

159def serve(carrier, source, output):
160 try:
161 while True:
162 line = source.readline(16385)
163 if not line:
164 return
165 if len(line) > 16384 or not line.endswith(b'\n'):
166 return
167 message = None
168 try:
169 message = json.loads(line)
170 if type(message) is not dict or message.get('jsonrpc') != '2.0':
171 return
172 if 'id' not in message:
173 if message.get('method') == 'notifications/initialized':
174 continue
175 return
176 ident = message['id']
177 if type(ident) not in (int, str) or len(str(ident)) > 80:
178 return
179 method = message.get('method')
180 if method == 'initialize':
181 result = {'protocolVersion': '2024-11-05', 'capabilities': {'tools': {}},
182 'serverInfo': {'name': 'pyharness-controlled-browser', 'version': '1'}}
183 elif method == 'tools/list':
184 result = {'tools': TOOLS}
185 elif method == 'tools/call':
186 params = message.get('params', {})
187 if type(params) is not dict or set(params) != {'name', 'arguments'}:
188 raise ValueError('invalid_browser_request')
189 value = carrier.call(params['name'], params['arguments'])
190 text = json.dumps(value, ensure_ascii=False, separators=(',', ':'))
191 if len(text) > 1900:
192 raise ValueError('browser_output_limit')
193 result = {'content': [{'type': 'text', 'text': text}], 'isError': bool(value.get('error'))}
194 else:
195 raise ValueError('unknown_browser_method')
196 envelope = {'jsonrpc': '2.0', 'id': ident, 'result': result}
197 except Exception:
198 # Protocol/schema errors contain no page text, local path or args.
199 envelope = {'jsonrpc': '2.0', 'id': message.get('id') if type(message) is dict else None,
200 'error': {'code': -32000, 'message': 'browser_request_failed'}}
201 output.write((json.dumps(envelope, ensure_ascii=True) + '\n').encode('utf-8'))
202 output.flush()
203 finally:
204 carrier.close()
