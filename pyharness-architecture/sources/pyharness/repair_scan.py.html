28

39def _segment_damage(path: Path) -> list[tuple[Path, Optional[int], list[int]]]:
40 """各轮转段的损坏面:**只读**扫描 → [(段路径, 尾部半行偏移|None, 中部坏行行号[])]。
41
42 尾部半行与中部坏行分开交付:前者按"未完成即未发生"截断,后者按"隔离不删"处理
43 (与主文件同口径)。尾部有半行时不再把它算作坏行(避免同一条被处理两次)。
44 """
45 out: list[tuple[Path, Optional[int], list[int]]] = []
46 for seg in _rotated_segments(path):
47 torn = detect_truncation(seg)
48 bad: list[int] = []
49 if torn is None:
50 bad = [no for no, text in _iter_text(seg)
51 if _parse_or_none(text) is None]
52 if torn is not None or bad:
53 out.append((seg, torn, bad))
54 return out

56def _segment_scan(path: Path) -> tuple[list[int], int, list[tuple[int, int]]]:
57 """轮转段的 (全部 seq, 可索引内容末 seq, 声明区间) —— 供**会话级**seq 连续性判定。
58
59 只读;段文件缺失/不可读时**跳过**(不影响主文件判定)。坏行记跳(与主文件同口径)。
60 """
61 seqs: list[int] = []
62 last = 0
63 declared: list[tuple[int, int]] = []
64 for seg in _rotated_segments(path):
65 try:
66 for _no, text in _iter_text(seg):
67 if text is None:
68 continue
69 env = _parse_or_none(text)
70 if env is None:
71 continue
72 seqs.append(env.seq)
73 if env.type == "session.recovered":
74 declared.extend(_declared_ranges(env))
75 continue
76 if _indexable(env):
77 last = max(last, env.seq)
78 if env.type in _DECLARE_TYPES:
79 declared.extend(_declared_ranges(env))
80 except OSError as e: # noqa: BLE001 段不可读不放大
81 log.warning("repair: 轮转段不可读,跳过其 seq 连续性判定 file=%s why=%s",
82 seg, e)
83 return seqs, last, declared

90def _session_content_last(path: Path) -> int:
91 """**会话级**可索引内容末 seq = max(主文件, 轮转段)。
92
93 与 ``scan_session`` 的 ``last`` **同口径**(两者都按 ``_indexable`` 判据 + 轮转段
94 合并读)。2026-09-21 R14-1 二阶修:重建源 ``open_store(...).replay`` **合并读轮转段**
95 (persistence),故对账水位也必须跨段取 max —— 此前只算主文件 ⇒ 主文件被截空(整文件
96 半行)或其尾部全为非内容事件时,段内容使 ``view_last>0`` 而单文件 ``log_last=0``
97 ⇒ **假 PERS-201**(repair 已成功却被判失败:CND-06/08 判据多源必然漂移)。
98 """
99 best = 0
100 for _no, text in _iter_text(path):
101 if text is None:
102 continue
103 env = _parse_or_none(text)
104 if env is not None and _indexable(env) and env.seq > best:
105 best = env.seq
106 _seqs, seg_last, _declared = _segment_scan(path)
107 return max(best, seg_last)

167def _parse_or_none(text: str) -> Optional[Envelope]:
168 """单行信封解析(容错):任何解析/校验失败 → None(坏行语义,PERS-201)。
169
170 含 payload 按 type 强校验(P1-4,读侧与写侧同口径)——缺字段/错类型的行
171 视为坏行隔离,不再漏进 reducer(此前只验信封、payload 坏数据延迟崩溃)。
172 """
173 try:
174 from pyharness.events.vocab import validate_payload
175 env = Envelope.model_validate_json(text)
176 canonical = validate_payload(env.type, env.payload)
177 if canonical is not env.payload:
178 env = env.model_copy(update={"payload": canonical})
179 return env
180 except Exception: # noqa: BLE001 — 坏行记跳不中断
181 return None

206def _iter_text(path: Path) -> Iterator[tuple[int, Optional[str]]]:
207 """逐物理行读(行号从 1 起;PERS-201 语义):非 UTF-8 段 → (no, None) 记跳。
208
209 \r\n 遗留容忍(rstrip 行尾);空行以 "" 交付(JSON 解析必败 = 坏行,与坏块同路)。
210 """
211 with open(path, "rb") as fh:
212 for no, raw in enumerate(fh, 1):
213 line = raw.rstrip(b"\r\n")
214 if not line:
215 yield no, ""
216 continue
217 try:
218 yield no, line.decode("utf-8")
219 except UnicodeDecodeError:
220 yield no, None

282async def scan_session(path: Path, *,
283 fts_last_seq: Optional[Callable[[str], int]] = None
284 ) -> SessionHealth:
285 """单会话健康扫描(截断/坏行/空洞/索引对账),只读零修改。
286
287 - 尾部 4KB 判半行(detect_truncation 偏移,复用 persistence 原语);
288 - 严格行迭代:解码失败/JSON 信封解析失败记 bad_lines(不中断,PERS-201 语义);
289 - 收集 seq 序列与声明区间(compacted/recovered),check_seq_gap 求未声明空洞;
290 - 索引对账:注入 fts_last_seq(sid) 提供者时,视图末 seq ≠ 日志末 seq = stale
291 (落后=索引漏事件;超前=幽灵行,见 index_stale 注释);无提供者不算 stale
292 (偏离 4,避免未装配 FTS 时全会话误报)。
293 损坏全部落报告字段,不抛(打开失败 OSError 上抛,由 auto_scan/repair_session
294 各自按 PERS-202 语义处置)。
295 """
296 sid = path.stem
297 if not path.exists():
298 return SessionHealth(sid=sid, path=path, healthy=True)
299 tail = detect_truncation(path) # 非空且末字节 ≠ \\n → 半行起始偏移;None=完整
300 bad: list[int] = []
301 seqs: list[int] = []
302 declared: list[tuple[int, int]] = []
303 last = 0
304 for no, text in _iter_text(path):
305 if text is None: # 编码级坏块:按行记跳(PERS-201 语义)
306 bad.append(no)
307 continue
308 env = _parse_or_none(text)
309 if env is None: # JSON/信封非法:坏行
310 bad.append(no)
311 continue
312 seqs.append(env.seq)
313 if env.type == "session.recovered": # 声明不产生 FTS 内容:不计入对账末 seq
314 declared.extend(_declared_ranges(env)) # 但其 lost 声明仍合法化空洞
315 continue
316 if _indexable(env): # 水位与索引 max_seq 同口径(CND-06/08)
317 last = max(last, env.seq)
318 if env.type in _DECLARE_TYPES:
319 declared.extend(_declared_ranges(env))
320 # 空洞是**会话级**属性(R13-6):轮转段与主文件同属一个会话(replay 合并读),
321 # 故 seq 连续性必须**跨段**判定 —— 否则主文件从 seq N+1 起会被误判为"空洞 1..N"
322 # (实测:凡轮转过一次的会话恒判不健康;每次 repair 都报假空洞并追加一条
323 # session.recovered ⇒ 稳定态噪声 + 重复写入)。段级坏行/半行仍按**各文件**口径
324 # 分别处置(见 `_segment_damage`)。
325 seg_seqs, seg_last, seg_declared = _segment_scan(path)
326 if seg_seqs:
327 seqs = sorted(set(seqs) | set(seg_seqs))
328 declared = declared + seg_declared
329 last = max(last, seg_last)
330 holes = check_seq_gap(seqs, declared) if seqs else []
331 index_stale = False
332 if fts_last_seq is not None:
333 try:
334 view_last = fts_last_seq(sid)
335 # 对账口径 = 视图末 seq 与日志末 seq **必须一致**(!=,非仅落后 <):
336 # 落后(视图<日志)= 索引漏事件;超前(视图>日志)= 幽灵行——dispatch 先于
337 # flush,session.append 落盘失败时事件已入总线被 FTS 订阅者索引,日志却
338 # 没有该 seq。两者都判 stale → 触发 rebuild 整体对账(幽灵行随之清除)。
339 index_stale = view_last is not None and view_last != last
340 except Exception as e: # noqa: BLE001 — 对账提供者故障不误报
341 log.warning("repair: fts_last_seq(%s) 提供者异常,索引对账不判定: %s",
342 sid, e)
343 return SessionHealth(
344 sid=sid, path=path,
345 tail_truncated=tail is not None, tail_offset=tail,
346 bad_lines=bad, holes=holes, holes_declared=declared,
347 index_stale=index_stale,
348 healthy=not (tail is not None or bad or holes or index_stale))

350async def auto_scan(sessions_dir: Path, *,
351 fts_last_seq: Optional[Callable[[str], int]] = None,
352 db_path: Optional[str] = None
353 ) -> list[SessionHealth]:
354 """全会话自检(启动 bootstrap 与 repair 子命令无 sid 共用):只返回需修会话。
355
356 目录不存在(首次运行)→ 零报告;轮转/备份/隔离辅助文件跳过;单文件打开失败
357 (权限/竞态)→ 本地异常日志跳过,不崩启动。只读零修改。
358
359 ``db_path``(2026-09-21 R11-2):ctx-less 调用方(如 CLI 启动自检)只给库路径,
360 即可启用**索引落后对账**;``fts_last_seq`` 优先(显式注入时胜出)。
361 """
362 if fts_last_seq is None and db_path:
363 from pyharness.core.session_query import read_max_seq
364 fts_last_seq = lambda sid: read_max_seq(db_path, sid) # noqa: E731
365 d = Path(sessions_dir) if sessions_dir is not None else default_sessions_dir()
366 if not d.exists():
367 return []
368 out: list[SessionHealth] = []
369 for path in sorted(d.glob("*.jsonl")):
370 if _is_aux_file(path):
371 continue
372 try:
373 h = await scan_session(path, fts_last_seq=fts_last_seq)
374 except PermissionError as e:
375 log.warning("repair: 会话不可读跳过(目录不可读/被占用) file=%s why=%s",
376 path, e)
377 continue
378 except OSError as e: # glob/读竞态:跳过单文件
379 log.warning("repair: 会话扫描失败跳过 file=%s why=%s", path, e)
380 continue
381 if not h.healthy:
382 out.append(h)
383 return out
