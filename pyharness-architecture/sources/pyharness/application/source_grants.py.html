75def parse_expiry(value: str) -> datetime:
76 try:
77 parsed = datetime.fromisoformat(value.replace('Z', '+00:00'))
78 if not value.endswith(('Z', '+00:00')) or parsed.utcoffset() != timezone.utc.utcoffset(parsed):
79 raise ValueError
80 return parsed
81 except (ValueError, TypeError):
82 raise ValueError('expiry must be UTC ISO8601') from None

90def authorize_root(value: str) -> dict:
91 """One geometry boundary via safe_path, rooted at a local volume anchor."""
92 try:
93 text = value.replace('\\', '/')
94 path = Path(text)
95 if (text.startswith('//') or '\x00' in text or not path.is_absolute()
96 or not path.anchor or any(p in {'.', '..'} for p in text.split('/'))):
97 raise ValueError
98 relative = path.relative_to(path.anchor).as_posix()
99 checked = safe_path(Path(path.anchor), relative)
100 resolved = checked.resolve(strict=True)
101 info = resolved.stat()
102 if not resolved.is_dir() or not info.st_ino:
103 raise ValueError
104 canonical_root = str(resolved)
105 return {'canonical_root': canonical_root, 'device': info.st_dev, 'inode': info.st_ino,
106 'root_identity': root_identity(canonical_root, info.st_dev, info.st_ino)}
107 except (OSError, ValueError, PyHError):
108 raise_code('EVT-100', reason='source_root_invalid')
