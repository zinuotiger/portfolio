155 # ======================================================== 主循环入口
156 async def run(self, ctx: Any) -> Optional[RunResult]:
157 """F007 主循环入口:wake 拉起;断言 idle + 拿锁防重入;队内排水监督器。
158
159 每件 pending 输入 = 独立 RunContext(轮数/取消按单输入计),逐件交
160 _run_engine 驱动;输入终态后按 FIFO 取下一件(状态表 running→running),
161 直至队空或遇 cancelled/budget 边界终态;返回最后一件的 RunResult。
162 异常分级收尾在 _run_engine;本层 finally 无条件回 idle、清 current/ctx
163 (协作式取消路径的 CancelledError 不在此吞,沿任务传播由外层 close 收尾)。
164 """
165 async with self.lock:
166 if self.state != "idle": # 锁前置:非 idle 拒入
167 raise PyHError("BUSY", ctx={"advice": "loop 忙,输入已入队或被拒"})
168 self._ctx = ctx # 1:1 绑定:本 run 期间 cancel
169 self.state = "running" # /队列满留痕有 ctx 可依
170 try:
171 last: Optional[RunResult] = None
172 while True:
173 env = self._pop_pending() # None → 队空
174 if env is None:
175 return last # 无输入可处理(空排水)
176 run = RunContext(run_id=uuid.uuid4().hex, input_seq=env.seq)
177 self.current = run
178 # One input is one task budget, while the session ledger is
179 # cumulative. A child arrives with its own preallocated account.
180 from pyharness.core.llm import UsageCounters
181 previous = getattr(ctx, 'counters', None)
182 account = previous
183 if isinstance(previous, UsageCounters):
184 if previous._parent is None:
185 account = previous.new_task(ctx.scope.limits)
186 ctx.counters = account
187 ctx.scope.bind_task_usage(account)
188 try:
189 last = await self._run_engine(ctx, run) # 伪码 run() 主体
190 finally:
191 if isinstance(previous, UsageCounters):
192 ctx.counters = previous
193 self.current = None # 单输入收尾(与伪码逐件一致)
194 # 边界终态(取消/预算)后不再自动取队内下一件:用户/资源边界,
195 # 剩余输入保留在队(不静默丢),下次 wake 按 FIFO 继续消费。
196 if last is not None and last.reason in ("cancelled", "budget", "error", "max_turns", "stall"):
197 return last
198 finally:
199 self.current = None
200 self.state = "idle" # 回 idle 等下一输入/队内下一件
201 # 注:_ctx 不清零——loop 与会话 1:1,绑定终身有效(偏离说明 3)
202
203 async def _run_engine(self, ctx: Any, run: RunContext) -> RunResult:
204 """单输入 run 引擎 = specs/agent_loop.py.md run() 伪码主体。
205
206 逐轮三闸前置判定 → scope.check_budget(F032)→ run_turn(单轮);轮内异常
207 分级收尾:BudgetExhausted → budget、MaxTurnsExceeded → max_turns、
208 PyHError(LLM-310 等)→ system.error 留痕 → error、未预期 → CYC-999 →
209 error;CancelledError 不吞(置 cancelled 后 re-raise,外层收尾)。
210 """
211 try:
212 while True: # 轮 = 一次 LLM 调用及其全部工具
213 if (r := self._must_stop(ctx, run)) is not None: # 三闸前置
214 return await self._finish(ctx, run, r)
215 ctx.scope.check_budget() # F032:超限抛 BudgetExhausted
216 outcome = await self.run_turn(ctx, run) # 单轮(含工具步)
217 if outcome is not None:
218 return await self._finish(ctx, run, outcome)
219 except asyncio.CancelledError: # F025:不吞;外层(close/cancel)收尾
220 run.cancelled = True
221 raise
222 except BudgetExhausted as exc: # 预算硬闸:无"再给一次机会"
223 run.error_code='LLM-305'
224 run.error_diagnostics=exc.diagnostics
225 await ctx.session.append('system.error',dict(code='LLM-305',hint='预算不足；本次模型请求尚未发送',
226 ctx=dict(role='main_agent',fatal=True,diagnostics=exc.diagnostics)),actor='system',task_id=getattr(ctx,'task_id',None),sync=True)
227 return await self._finish(ctx, run, "budget")
228 except MaxTurnsExceeded: # 轮数闸(防御路径)
229 return await self._finish(ctx, run, "max_turns")
230 except PyHError as e: # LLM-310 等结构化错误
231 from pyharness.core.llm_diagnostics import public_diagnostics
232 run.error_code = e.code
233 run.error_diagnostics = public_diagnostics(e.ctx.get("diagnostics"), code=e.code, role="main_agent")
234 await ctx.session.append(
235 "system.error",
236 {"code": e.code, "hint": e.code + ": " + run.error_diagnostics["summary"],
237 "ctx": {"role": "main_agent", "fatal": True, "diagnostics": run.error_diagnostics}},
238 actor="system", task_id=getattr(ctx, "task_id", None), sync=True)
239 return await self._finish(ctx, run, "error")
240 except Exception as e: # 未预期兜底:CYC-999,堆栈仅本地
241 run.error_code = "CYC-999"
242 log.error("loop fatal run_id=%s type=%s", run.run_id, type(e).__name__)
243 await ctx.session.append(
244 "system.error",
245 {"code": "CYC-999", "hint": "Main execution failed",
246 "ctx": {"role": "main_agent", "fatal": True}},
247 actor="system", task_id=getattr(ctx, "task_id", None), sync=True)
248 return await self._finish(ctx, run, "error")

267 # 护栏段恒末、超窗头部截断(装配器内部,见 system_prompt.py);缺位 = 裸历史

273 runtime: Optional[MessageMaterializer] = getattr(ctx, 'platform_runtime', None)
274 if runtime is not None:

276 vision_input = any(isinstance(m.get('content'), list) and any(
277 part.get('type') == 'image_url' for part in m['content'] if isinstance(part, dict)) for m in hist)
278 # F027:ctx.streaming(桌面 SSE)走 chat_stream——chunk 仅上总线不入日志,
279 # 聚合后与 chat 同型返回(append-only 不变式不受影响);CLI/单测默认非流式
280 chat_fn = ("chat_stream" if getattr(ctx, "streaming", False) else "chat")
281 from pyharness.core.llm_diagnostics import model_call
282 with model_call("main_agent"),request_context_scope(ctx,hist,history_window,system_window,enabled=observing):
283 resp = await getattr(ctx.llm, chat_fn)(
284 hist, tools=[] if vision_input else ctx.tools.schemas_for(ctx.scope), ctx=ctx)
285 if vision_input and resp.tool_calls:
286 from pyharness.errors import PyHError
287 raise PyHError('vision_tools_not_allowed')
288 # llm 层落 request/usage 事件;chat_fn 选择不改变事件路径
289 if not resp.tool_calls: # 纯文本 → 自然终态
290 content = resp.content or ""
291 quality=None
292 if runtime is not None:
293 review=getattr(runtime,'review_answer',None)
294 if callable(review):content,quality=review(content,ctx,resp)
295 else:content = runtime.validate_answer(content, ctx)
296 if not content: # llm 双空响应:模型域未预期
297 raise_code("LLM-399", hint="llm.chat 返回空 content 且无 tool_calls")
298 await ctx.session.append(
299 "agent.message", {"content": content, **({'quality':quality} if quality is not None else {})}, actor="agent",
300 trace={"parent_seq": resp.seq} if getattr(resp, "seq", None) else None)
301 if quality is not None and (quality.get('output_completeness')=='incomplete' or
302 quality.get('calculation',{}).get('status')=='failed'):
303 run.error_code='LLM-304'
304 return 'error'
305 return "complete" # finished 由外壳 close(reason) 落
306 # 工具轮:校验/guard/审批/Provider 全在 tools.execute 内;单步失败→事件化
307 # 回喂(tool.error/guard.rejected),不中断本轮(见 run_step)

310 for call in resp.tool_calls:
311 res = await self.run_step(ctx, run, call) # 逐工具步执行(顺序,阶段1)
312 if self._turn_failure_capped(ctx, call, res):
313 break # F026:同工具连败≥2 → 终止本轮
314 run.turn += 1
315 fp = self._outcome_fingerprint(ctx, start_seq) # 指纹=(name,summary,ok)序列
316 run.stall_streak = run.stall_streak + 1 if fp == run.last_fp else 0
317 run.last_fp = fp
318 if run.stall_streak >= self.stall_limit:
319 return "stall" # 连续 N 轮无新信息 → 空转封顶
320 return None # 下一轮

324 """单工具步:登记子任务执行 ctx.tools.execute(call, ctx),返回 ExecResult。
325
326 call = llm 响应解析出的 ToolCall(name/raw_args/call_id);TLB-802/803/805
327 等失败已由 tools.execute 捕获写 tool.error 回喂 LLM,不向主循环抛——本轮
328 继续。协作式取消:子任务被 cancel() 取消时置 run.cancelled 并 re-raise,
329 已开始工具如实留 partial(副作用不假装回滚)。
330
331 返回值:ExecResult(ok/summary)供 run_turn 做 F026 轮内连败判定(S1-03);
332 取消路径不返回(await 抛 CancelledError 后 re-raise)。
333 """
334 t = asyncio.create_task(ctx.tools.execute(call, ctx))
335 self._child_tasks.add(t) # 登记:取消可传播(F025)
336 try:
337 return await t # 60s 工具超时在 tools 内部(F017)
338 except asyncio.CancelledError:
339 run.cancelled = True # 协作式取消:不吞,继续传播
340 raise
341 finally:
342 self._child_tasks.discard(t)

352 """

358 def _turn_failure_capped(ctx: Any, call: Any, res: Any) -> bool:

374 def _must_stop(self, ctx: Any, run: RunContext) -> Optional[str]:

384 if ctx.scope.budget_state() == "exhausted":
385 return "budget"
386 return None

403 async def _finish(self, ctx: Any, run: RunContext, reason: str) -> RunResult:
404 """汇总 RunResult(run_id/reason/turns/last_seq);不写 session.finished

421 # ======================================================== 取消传播(F025)
422 async def cancel(self, reason: str = "cancelled") -> None:
423 """声明式取消:置位 → 强同步留痕(system.cancelled)→ 取消子任务 →
424 gather 等待收尾 → 清登记。不吞 CancelledError;append 失败不被日志故障
425 阻塞(记日志继续)。reason ∈ {cancelled, timeout, close}。
426 """
427 run = self.current
428 if run is None: # 无在途 run:无事可取消
429 return
430 ctx = self._ctx
431 run.cancelled = True # 置位:闸2 下一轮即触发
432 if ctx is not None:
433 try:
434 await ctx.session.append(
435 "system.cancelled",
436 {"what": f"run:{run.run_id}", "reason": reason},
437 actor="system", sync=True) # 强同步:取消审计可重建
438 except PyHError as e: # 强同步失败:PERS-202 等
439 log.error("cancel log failed code=%s", e.code) # 取消不被日志故障阻塞
440 for t in self._child_tasks: # 传播:取消派生子任务
441 if not t.done():
442 t.cancel()
443 await asyncio.gather(*self._child_tasks, return_exceptions=True)
444 self._child_tasks.clear() # 已开始工具如实留 partial

446 # ======================================================== 输入唤醒/入队
447 async def wake(self, env: Envelope, ctx: Any = None) -> Optional[RunResult]:
448 """外壳(agent.submit)在 user.message 强同步落盘后调用。
449
450 idle → 入队并拉起排水 run(await 至队空/边界终态,返回 RunResult);
451 running → 入队(FIFO),返回 None,当前 run 结束后自动取下一件;
452 队满(>queue_limit)→ system.error[BUSY] 强同步 + 拒新(BUSY)。
453 ctx:本 loop 尚未经 run() 绑定时显式注入(1:1 绑定后可不传,偏离说明 3)。
454 会话关闭态拒入归 agent.submit(agent.state ∈ stopping/closed 时 BUSY),
455 本函数不再判 loop 自身的 stopping/terminated(S1-02 删除该死态分支)。
456 """
457 if ctx is None:
458 ctx = self._ctx # 已绑定(1:1)则复用
459 if self.state == "running" and len(self.pending) >= self.queue_limit:
460 # 队满:显式拒新,不静默丢(先事件留痕再抛)
461 if ctx is not None:
462 try:
463 await ctx.session.append(
464 "system.error",
465 {"code": "BUSY",
466 "hint": f"输入队列已满(>{self.queue_limit}),请稍后重试"},
467 actor="system", sync=True)
468 except PyHError as e: # 留痕失败不阻断拒新
469 log.error("busy log failed code=%s", e.code)
470 raise PyHError("BUSY", ctx={"advice": f"队列已满(>{self.queue_limit})"})
471 self.pending.append(env) # 空闲或 running:统一入队保 FIFO
472 if self.state == "idle": # 空闲:拉起排水 run
473 if ctx is None:
474 raise PyHError("CYC-999",
475 ctx={"hint": "wake 拉起 run 前 loop 未绑定 ctx"
476 "(请先经 run(ctx) 或 wake(env, ctx))"})
477 # F058 压缩在请求边界执行(loop idle;compactor 内部再判双条件/忙态):
478 # 折叠旧段为摘要 → context.compacted 强同步声明 → 本轮 derive 即压缩视图
479 comp = getattr(ctx, "compactor", None)
480 if comp is not None and callable(getattr(comp, "run_if_needed", None)):
481 try:
482 await comp.run_if_needed(ctx)
483 except PyHError as e: # 压缩失败不阻断对话(截窗兜底)
484 log.warning("pre-run compaction skipped code=%s", e.code)
485 return await self.run(ctx)
486 return None # running:当前 run 结束自动取下一件

488 def _pop_pending(self) -> Optional[Envelope]:
489 """取队首输入(FIFO);队空返回 None。"""
490 return self.pending.popleft() if self.pending else None
