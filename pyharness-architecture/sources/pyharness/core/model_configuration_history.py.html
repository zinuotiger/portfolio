38def _increases(current, target, active, resource):
39 result = []
40 if active != resource:
41 result.append('active_model_connection_changed')
42 for name in ('model', 'base_url'):
43 if current.get(name) != target.get(name):
44 result.append(name+'_changed')
45 if 'max_tokens' not in current or 'max_tokens' not in target:
46 result.append('legacy_output_limit_requires_confirmation')
47 elif target['max_tokens'] > current['max_tokens']:
48 result.append('max_tokens_increased')
49 if set(target.get('fallback_models', [])) - set(current.get('fallback_models', [])):
50 result.append('fallback_models_added')
51 if target.get('supports_vision') and not current.get('supports_vision'):
52 result.append('vision_capability_added')

231 def capture_model_configuration(self, tenant_id, profile_id, body, *, authorize):
232 c.request(body, {'command_key', 'source_sha256'})
233 with self._configuration_access(tenant_id, profile_id, authorize) as (tenant, profile, doc):
234 request_digest = c.digest({'tenant': tenant, 'resource': profile, 'body': body})
235 previous = c.command(c.history(doc), body['command_key'], 'capture_model', request_digest)
236 if previous is not None:
237 return previous
238 state = self._model_configuration_state(tenant, profile, doc)
239 if not state['snapshot']:
240 c.reject('configuration_resource_not_found')
241 if state['source_sha256'] != body['source_sha256']:
242 c.reject('configuration_source_stale')
243 c.public_tree(body, tuple(self._load_secrets(tenant).values()))
244 history_ = c.history(doc, create=True)
245 if len(history_['revisions']) >= c.MAX_REVISIONS:
246 c.reject('configuration_revision_history_limit')
247 prior = [v['number'] for v in history_['revisions'].values() if v['resource_id'] == profile]
248 value = c.revision(tenant, 'model', profile, max(prior, default=0)+1,
249 state['snapshot'], state['source_sha256'], state['_binding'])
250 history_['revisions'][value['revision_id']] = value
251 result = {'status': 'captured', 'revision': c.public_revision(value),
252 'historical_result': True, 'execution_authorized': False}
253 c.save_command(history_, body['command_key'], 'capture_model', request_digest, result)
254 authorize()
255 self._save_doc(tenant, doc)
256 return result

258 def _model_configuration_preview(self, tenant, profile, ident, doc):

260 state = self._model_configuration_state(tenant, profile, doc)
261 issues = []
262 if not state['snapshot']:
263 issues.append('resource_missing')

306 state = self._model_configuration_state(tenant, profile, doc)

310 'profile_id': profile, 'active': profile,
311 'new_source_sha256': state['source_sha256'], 'historical_result': True,
312 'old_attempts_changed': False, 'external_effects_rolled_back': False,
313 'execution_authorized': False}
314 c.save_command(c.history(doc, create=True), body['command_key'], 'apply_model', request_digest, result)
315 authorize()
316 self._save_doc(tenant, doc) # Sole commit: model metadata + receipt; secret file untouched.
317 return result
