30 def __init__(self, ctx: Any = None, *, service: Any = None,
31 parent: Any = None, tenant_id: str = "default") -> None:
32 super().__init__(parent)
33 self.ctx = ctx
34 self.tenant_id = normalize_tenant_id(tenant_id)
35 self._provided_service = service
36 self._registry = (None if service is not None else
37 ApplicationServiceRegistry(
38 ctx, channel="desktop"))
39 self.service = service or self._registry.get(self.tenant_id)
40 self._subs: list[Any] = []
41 self.selected_sid: str = ""
42 bus = getattr(ctx, "bus", None) if ctx is not None else None
43 if bus is not None:
44 owner = native_event_owner(self.tenant_id) # 按租户唯一(见 constants)
45 for type_ in EVENT_TYPES:
46 try:
47 self._subs.append(bus.subscribe(
48 type_, self._receive_bus_event, owner=owner))
49 except Exception: # noqa: BLE001
50 log.debug("native bus subscribe skipped %s", type_)

56 return str(getattr(getattr(cfg, "skills", None),
57 "registry_url", "") or "")
58 except Exception: # noqa: BLE001
59 return ""

61 def _event_allowed(self, payload: Any) -> bool:
62 """事件是否投给本壳(租户可见性**单点判据**,与 Web 壳同源)。
63
64 2026-09-21 修:此前写 ``not owner or owner == self.tenant_id`` —— 归属
65 **未知即放行**(fail-open),而 Web 壳投影层对同一问题是 fail-closed ⇒ 两壳
66 失败方向相反;且归属取进程内映射,重启后为空 ⇒ **跨租户事件可达原生 UI**。
67 现统一走 ``event_tenant_of``(优先落盘信封)+ ``event_tenant_allowed``。
68 """
69 sid = (getattr(payload, "session_id", "") if not isinstance(payload, dict)
70 else (payload or {}).get("session_id"))
71 if not sid:
72 return self.tenant_id == "default"
73 return event_tenant_allowed(self.tenant_id,
74 event_tenant_of(payload, session_id=str(sid)))

76 async def _receive_bus_event(self, type_: str, payload: Any) -> None:
77 # EventBus runs synchronous subscribers in a worker thread. Qt delivery
78 # belongs on the qasync owner loop, without yielding before persistence.
79 self._on_bus_event(type_, payload)
80
81 def _on_bus_event(self, type_: str, payload: Any) -> None:
82 if not self._event_allowed(payload):
83 return
84 if type_ == "llm.chunk":
85 self.chunk_received.emit(dict(payload or {}))
86 elif type_.startswith("job."):
87 self.jobs_changed.emit()
88 elif type_.startswith("schedule."):
89 self.schedules_changed.emit()
90 elif type_.startswith("subagent."):
91 self.subagents_changed.emit()
92 elif type_ in {"session.created", "session.renamed", "session.finished"}:
93 self.sessions_changed.emit()
94 self.event_received.emit(type_, payload)

96 async def close(self) -> None:
97 try:
98 workbench = getattr(self, "_native_workbench", None)
99 if workbench is not None:
100 await workbench.close()
101 finally:
102 await self._close_services()
103
104 async def _close_services(self) -> None:
105 bus = getattr(self.ctx, "bus", None) if self.ctx is not None else None
106 if bus is not None:
107 try:
108 bus.unsubscribe_all(native_event_owner(self.tenant_id))
109 except Exception: # noqa: BLE001
110 pass
111 self._subs.clear()
112 if self._registry is not None:
113 await self._registry.close_all()
114 else:
115 shutdown = getattr(self.service, "shutdown", None)
116 if callable(shutdown):
117 await shutdown()
