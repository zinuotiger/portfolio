22 def _candidate_session_dirs(self) -> list[Path]:
23 """会话可能落盘的**全部权威目录** = 默认 sessions 目录 + `<root>/tenants/*/sessions`。
24

27 这里只做**枚举**,不做归属判定 —— "同源规则多份实现必然漂移",故派生点唯一。
28 """

46 def _disk_owners(self, sid: str) -> tuple[set, bool]:
47 """磁盘上的**权威归属** → ``(租户集合, 是否存在"归属不可得"的会话文件)``。
48
49 第二项是 fail-closed 的枢轴:文件**在盘**但读不出租户(GAP-10 缺失/坏行)
50 ⇒ 归属**未知**,调用方必须拒绝;绝不能把"读不出归属"混同于"会话不存在"
51 (后者才允许回落客户端自报头)。
52 """
53 owners: set = set()
54 unknown = False
55 for d in self._candidate_session_dirs():
56 f = d / f"{sid}.jsonl"
57 try:
58 if not f.is_file():
59 continue
60 except OSError:
61 unknown = True # 目录不可读 ⇒ 无法断言归属
62 continue
63 t = tenant_of_log(f)
64 if t:
65 owners.add(t)
66 else:
67 unknown = True # 会话在盘但未载租户 ⇒ 归属未知
68 return owners, unknown

70 @staticmethod
71 def _request_sid(request: Request) -> str:
72 """本请求指向的**会话 id**:路径段优先,其次 ``?sid=``(SSE 流式面)。
73
74 SSE 把 sid 放在查询串(`/api/stream?sid=…`)⇒ 只认路径会**整条绕过**归属
75 判定,流式面就成了"声明即授权"(实测:声明任意租户即可订阅该租户会话事件)。
76 查询参数按 ``validate_session_id`` 同款正则校验,避免把普通查询参数误判为 sid。
77 """
78 m = _SESSION_ID_IN_PATH.search(request.url.path)
79 if m:
80 return m.group(1)
81 q = str(request.query_params.get("sid") or "")
82 return q if _SID_RE.fullmatch(q) else ""

84 def resolve_request_tenant(self, request: Request) -> Optional[str]:
85 """解析本请求的**服务端权威**租户;不可得 → ``None``(调用方须 fail-closed)。
86
87 **不变式(2026-09-21 R31-1)**:客户端自报头只声明"我要访问哪个租户的**分区**",
88 它**绝不决定一个已存在会话的归属**,服务端也**绝不据此把请求改判/重定向到
89 另一个租户**。解析序:
90
91 ① 无会话语境(路径与查询都没有 sid) ⇒ 头即声明(既有语义);
92 ② 会话的权威归属 = 进程内认领集合 ∪ 全部权威目录的落盘事实;
93 ③ **归属不可得**(文件在盘但未载租户)或**归属歧义**(多租户同名 sid)
94 ⇒ ``None``(**拒绝**);
95 ④ 归属确定:声明与归属**相同** ⇒ 放行;**不同** ⇒ ``None``(拒绝);
96 ⑤ 任何权威来源都查无此会话 ⇒ 回落声明(该路由只会 404/EVT-106,无数据可泄)。
97
98 为什么"不一致即拒"而不是"改判为真实归属":改判等于把**声明了 A 的请求交给
99 B 的租户服务作答** ⇒ 未声明 B 的客户端读到 B 的会话正文。两版实现均可复现:
100 HEAD 版在 default 分区(即全局 sessions 目录)按目录归属放行,不认信封租户;
101 本轮前版(按权威目录改判)更彻底 —— **任意头甚至不带头**都返回真实归属的数据
102 (实测 `header=alpha/default/无` 于 beta 会话一律 200 + `BETA-ONLY`)。
103 **鉴权不由本函数负责**:R31-2 在中间件里先行把关(``_tenant_authorized``),
104 本函数只回答"这个会话的**权威归属**是谁",不回答"该主体**能否**用这个租户"。
105 """
106 declared = self._declared_tenant(request)
107 sid = self._request_sid(request)
108 if not sid:
109 return declared # ① 无会话语境:头即声明
110 owners, unknown = self._disk_owners(sid)
111 owners |= session_tenants(sid) # ② 进程内认领(仅作**归属来源**)
112 if unknown or len(owners) > 1:
113 return None # ③ 归属不可得 / 歧义 ⇒ 拒
114 if not owners:
115 return declared # ⑤ 查无此会话(路由只会 404)
116 return declared if declared in owners else None # ④ 声明≠权威归属 ⇒ 拒

118 def bootstrap_url(self, host: str, port: int) -> str:
119 """壳加载 UI 的 URL(**带操作者令牌**,R31-2)。
120
121 ``/`` 不再无条件发放全局令牌,故壳必须自证持有它。令牌走 URL 查询串而非页面

194 def _tenant_authorized(self, request: Request, tenant: str) -> bool:
195 """**鉴权(R31-2)**:声明 ``default`` 沿用全局令牌(路由依赖再强制一次);
196 声明非 default ⇒ 必须出示**该租户**的令牌。
197
198 为什么 default 放行:它是**兼容口** —— 桌面壳与既有客户端只持全局令牌,
199 若 default 也要租户令牌,引导页与全部既有流程即刻断。收益全在非 default:
200 此前任何人只要**报对租户名**即可进入(只有归属校验 R31-1,没有资格校验)。
201 """
202 from .accounts_api import principal
203 account = principal(request)
204 if account:
205 return account['tenant'] == tenant
206 if tenant == "default":
207 return True
208 return self._verify_tenant_token(
209 tenant, self._presented_tenant_token(request))

211 async def _tenant_middleware(self, request: Request, call_next: Any) -> Any:
212 # The only public mutation is an independently signed Feishu callback.
213 # It cannot select a tenant or reach any management/API endpoint.
214 if getattr(request,'method','GET') == 'POST' and re.fullmatch(r'/hooks/feishu/[a-z][a-z0-9_]{1,30}', request.url.path):
215 return await call_next(request)
216 from .accounts_api import inspect_request, principal
217 rejected = inspect_request(self, request)
218 if rejected is not None:
219 return rejected
220 if request.url.path.startswith('/api/account/') or request.url.path == '/login':
221 return await call_next(request)
222 # ① 鉴权先行:未出示 / 出示错误的本租户令牌 ⇒ 401。**先于**归属判定,故不会向
223 # 未授权方泄露"该租户下是否存在某会话"(403 与 404 之差即是信息)。
224 declared = self._declared_tenant(request)
225 # 引导页 ``/`` **自带**更细的凭证规则(持操作者令牌者可铸所声明租户的令牌),
226 # 故此处放行 —— 否则中间件先 401,铸令牌那条路径根本到不了(实测)。
227 if request.url.path != "/" and not self._tenant_authorized(request, declared):
228 return JSONResponse(status_code=401, content={
229 "code": "CRED-701",
230 "advice": f"租户 {declared} 需要该租户的 API 令牌"
231 "(X-PyHarness-Tenant-Token,或 /?tenant= 下发的 cookie);"
232 "default 租户沿用全局令牌"})
233 # ② 归属判定(R31-1):不可得 / 歧义 / 与声明不符 ⇒ 403
234 tenant = self.resolve_request_tenant(request)
235 if tenant is None:
236 # 归属不可得 / 歧义 / 声明与权威归属不符 ⇒ **拒绝**;一律不按客户端自报头
237 # 放行,也不改判到别的租户(防伪造头跨租户读取,L-1/R31-1)
238 return JSONResponse(status_code=403, content={
239 "code": "GRD-401",
240 "advice": "会话租户归属不可得、存在歧义,或与请求声明的租户不符;"
241 "拒绝服务(既不按客户端自报头判定,也不跨租户改判)"})
242 token = self._tenant_var.set(tenant)
243 from pyharness.application.accounts import current_actor, public_actor
244 actor_token = current_actor.set(public_actor(principal(request)))
245 try:
246 from .accounts_api import serve_with_quota
247 return await serve_with_quota(self, request, call_next)
248 finally:
249 current_actor.reset(actor_token)
250 self._tenant_var.reset(token)

260 def _resolve_api_token(self) -> str:
261 """API token:配置 shell.web.token(env:/file: 引用)优先,缺省进程级随机令牌。
262
263 同时记录 ``_api_token_generated``:随机态下该令牌**无处可查**,须由
264 :meth:`publish_operator_token` 落盘,否则浏览器/外部客户端无从取得。
265 """
266 raw = ""
267 cfg = _cfg_of(self.ctx)
268 try:
269 web = getattr(getattr(cfg, "shell", None), "web", None)
270 raw = str(getattr(web, "token", None) or "")
271 except Exception: # noqa: BLE001 非 Settings 形状
272 raw = ""
273 val = _resolve_secret_value(raw)
274 self._api_token_generated = not bool(val)
275 return val if val else secrets.token_hex(16)

277 def _require_api_auth(self, request: Request,
278 x_pyharness_token: Optional[str] = Header(default=None),
279 authorization: Optional[str] = Header(default=None)) -> None:
280 """读/写端点统一鉴权依赖(P1-2 收紧):令牌缺失/不匹配 → CRED-701(401)。
281
282 令牌来源:① X-PyHarness-Token / Authorization: Bearer 头(pywebview 桥/
283 测试/外部客户端);② pyharness_token HttpOnly cookie(_index_page 下发,
284 JS 不可读、同源请求自动携带)。仅索引页 / 与 webhook 入站端点免鉴权
285 (引导/外联);全部数据端点(含读端与 SSE)强制鉴权——此前 39 个读端点
286 匿名、SSE 无鉴权,任何本地进程可枚举全部会话。
287 """
288 from .accounts_api import principal, operator_principal
289 if principal(request) or operator_principal(request):
290 return
291 expected = self._api_token
292 if not expected:
293 return
294 given = x_pyharness_token or ""
295 if not given and authorization and authorization.lower().startswith("bearer "):
296 given = authorization[7:].strip()
297 if not given:
298 given = read_cookie(request, "pyharness_token")
299 ok = bool(given) and secrets.compare_digest(given, expected)
300 if not ok:
301 # ``R31-2``:非 default 租户可用**该租户令牌**通过本依赖。租户**授权**已在
302 # 中间件强制(它先跑);此处只是第二层,不代替它 —— 只有租户令牌、没有全局
303 # 令牌的浏览器因此可用,而它拿不到全局令牌,故换不到别的租户的 cookie。
304 tenant = self.current_tenant()
305 ok = (tenant != "default"
306 and self._verify_tenant_token(
307 tenant, self._presented_tenant_token(request)))
308 if not ok:
309 raise_code("CRED-701", reason="api-token",
310 hint="缺少或非法 API token"
311 "(X-PyHarness-Token / Bearer / pyharness_token cookie;"
312 "非 default 租户亦可出示该租户令牌)")
313
