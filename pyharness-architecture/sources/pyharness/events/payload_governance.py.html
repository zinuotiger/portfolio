15 @field_validator("ranges")
16 @classmethod
17 def _ranges_sane(cls, v: list[list[int]]) -> list[list[int]]:
18 """区间必须 [lo, hi](lo≤hi、lo≥1)且两两不重叠。"""
19 ordered: list[list[int]] = []
20 for pair in v:
21 if len(pair) != 2:
22 raise ValueError(f"区间须为 [lo, hi] 二元组:{pair}")
23 lo, hi = pair
24 if lo < 1:
25 raise ValueError(f"seq 从 1 起,lo 非法:{pair}")
26 if lo > hi:
27 raise ValueError(f"lo 不得大于 hi:{pair}")
28 ordered.append(pair)
29 ordered.sort(key=lambda p: p[0])
30 for (_, prev_hi), (cur_lo, _) in zip(ordered, ordered[1:]):
31 if cur_lo <= prev_hi:
32 raise ValueError("ranges 区间不得重叠")
33 return v

174class EvidenceArchivedPayload(_PayloadBase):
175 """evidence.archived:治理证据归档事实(M6;S5-1)。
176
177 **只存引用,不复制事件内容**(INV-G4 / INV-E1):
178
179 - ``evidence_id``:工件主键;
180 - ``claim``:被支持的断言;
181 - ``refs``:指向**真源**的引用列表,每项 ``{"kind": <enum>, "locator": str}``。
182 顶层 ``extra="forbid"`` 不约束嵌套 dict,故此处用 ``field_validator`` 把
183 内层严格性补回(键集恰为两键 + kind 值域 + locator 非空);
184 - ``artifact_path``:可选**外部附件**引用——**不承载治理事实**(不得作为
185 治理判定输入;缺失/损坏不影响任何不变量)。
186
187 **通道 = 普通(攒批),非强同步**(EVENT-SCHEMA §3.6 冻结):证据是**索引**,
188 丢失可由既有事件**再派生**(INV-E3)。
189 """
190 evidence_id: str = Field(min_length=1)
191 claim: str = Field(min_length=1)
192 refs: list = Field(default_factory=list)
193 artifact_path: Optional[str] = None

195 @field_validator("refs")
196 @classmethod
197 def _check_refs(cls, v: Any) -> Any:
198 for item in v or ():
199 if not isinstance(item, dict) or set(item) != {"kind", "locator"}:
200 raise ValueError("refs 每项必须是 {kind, locator} 且仅此两键")
201 if item.get("kind") not in _EVIDENCE_REF_KINDS:
202 raise ValueError(f"refs.kind 必须是 {_EVIDENCE_REF_KINDS} 之一")
203 loc = item.get("locator")
204 if not isinstance(loc, str) or not loc:
205 raise ValueError("refs.locator 必须为非空字符串")
206 return v
