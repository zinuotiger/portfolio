132def register_event_type(type_: str, model: type[BaseModel], *,
133 transient: bool = False) -> None:
134 """词表注册(框架 70 类型启动期注册;插件运行时注册 plugin.<id>.<name>)。
135
136 重复注册拒绝(EVT-102,§3.8 只增不改):破坏性变更 = 新类型名,旧类型冻结。
137 """
138 if type_ in _EVENT_REGISTRY:
139 raise_code("EVT-102", type_=type_, detail="重复注册,事件类型只增不改")
140 _EVENT_REGISTRY[type_] = model
141 if transient:
142 _TRANSIENT.add(type_)

169def summarize_validation(e: ValidationError) -> str:
170 """ValidationError → 字段明细文本(如 "content: String should have at least 1 character")。
171
172 只取 loc+msg,绝不携带输入值(input 可能含敏感原文)。
173 """
174 parts: list[str] = []
175 for err in e.errors():
176 loc = ".".join(str(x) for x in err.get("loc", ()))
177 msg = str(err.get("msg", ""))
178 parts.append(f"{loc}: {msg}" if loc else msg)
179 return "; ".join(parts)[:2000]

182def validate_payload(type_: str, payload: dict) -> dict:
183 """payload 强类型二次校验(EVENT-SCHEMA §2.2 第 3 步)。
184
185 成功返回规范化 dict(model_dump);失败 → EVT-100 带字段明细。
186 """
187 model = payload_model_for(type_)
188 if model is None:
189 raise_code("EVT-102", type_=type_)
190 try:
191 return model(**payload).model_dump()
192 except ValidationError as e:
193 # 未注册已在上面抛 EVT-102;此处只可能是字段非法 → EVT-100
194 raise_code("EVT-100", type_=type_, detail=summarize_validation(e))
