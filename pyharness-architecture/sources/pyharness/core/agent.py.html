177class Agent:
178 """会话实体(F007/F064):一会话一 agent 句柄 + ctx 门面 + 生命周期管理。
179
180 状态机(§specs/agent.py.md):
181 INIT ─enter()→ READY ─submit()→ BUSY ─run 结束(loop idle)→ READY
182 READY/BUSY ─close()→ STOPPING →(清理完成)→ CLOSED(终局)
183 CLOSED 后 submit/enter → BUSY 显式拒(不排队不静默)
184
185 字段:agent_id/session_id(1:1 不可换)、state、ctx(Ctx 门面)、caps(已挂载
186 命名空间 → iface 束)、config(Settings 只读)、headless(自动关会话判定)、
187 _session_lock(一会话一 agent 互斥:enter 抢占,close 释放)、_created_event
188 (created 首事件守卫,EVT-106)。
189 """
190
191 def __init__(self, agent_id: str, session_id: str,
192 state: str = "init", config: Any = None,
193 caps: Optional[dict[str, Any]] = None, *,
194 headless: Optional[bool] = None) -> None:
195 if state not in AGENT_STATES: # 构造期即锁死合法态(防非法字面量漂移)
196 _raise_busy(state=state, detail="非法生命周期初值")
197 self.agent_id: str = agent_id
198 self.session_id: str = session_id
199 self.state: str = state
200 self.config: Any = config # CFG 启动只读快照(Settings)
201 self.caps: dict[str, Any] = dict(caps or {}) # 已挂载能力命名空间
202 # headless 决定 run 结束是否自动 close;显式注入优先,否则回落配置
203 self.headless: bool = (bool(headless) if headless is not None
204 else _resolve_headless(config, None))
205 self.ctx: Ctx = None # 门面:create_agent 硬接线注入
206 self._spine: Any = None # 装配束引用(active_agents 释放)
207 self._session_lock: bool = False # 一会话一 agent 互斥(enter 抢占)
208 self._created_event: bool = False # created 已落守卫(EVT-106)
209 self._registered: bool = False # registry 插件注册记账(幂等注销)

211 # ========================================================== 会话占位激活
212 async def enter(self) -> None:
213 """会话占位激活:引导 created(新会话 seq=1)→ 注册 → 订阅 → ready。
214
215 幂等:已 ready/busy 直接返回;stopping/closed 拒(BUSY,建议新建 agent)。
216 失败回滚:释放 active_agents 占位与注册表项,保持可重试(enter 失败则
217 回滚释放,spec 工厂注释语义)。
218 """
219 if self.state in ("ready", "busy"):
220 return # 幂等:已激活直接返回
221 if self.state in ("stopping", "closed"):
222 _raise_busy(advice="句柄已关闭,请 create 新 agent")
223 try:
224 # 新会话引导:首事件 seq=1(EVT-106 守卫由 session.append 把关);
225 # 会话已有事件(恢复回放)则 created 已在日志,跳过追加
226 if (not self._created_event
227 and self.ctx.session.stats()["seq"] == 0):
228 await self.ctx.session.append(
229 "session.created",
230 {"title": "", "model": _created_model(self.config)},
231 actor="system", sync=True)
232 self._created_event = True
233 # F003:注册自身,可被外壳/宿主寻址(plugin:{agent_id})
234 self.ctx.registry.register("plugin", self.agent_id, self)
235 self._registered = True
236 # 订阅本会话事件流(owner=agent_id,close 时按属主精确摘除)
237 self.ctx.bus.subscribe(f"session:{self.session_id}:*",
238 self._on_bus_event, owner=self.agent_id)
239 self.state = "ready"
240 self._session_lock = True # 抢占会话互斥(占用占位)
241 log.info("agent ready", agent_id=self.agent_id,
242 session=self.session_id)
243 except BaseException:
244 self._rollback_enter()
245 raise

247 def _rollback_enter(self) -> None:
248 """enter 失败回滚:注销已注册插件项 + 释放会话占位(半激活 > 僵尸)。"""
249 if self._registered:
250 try:
251 self.ctx.registry.unregister("plugin", self.agent_id)
252 except PyHError as e:
253 if e.code != "TLB-801":
254 log.error("enter rollback unregister failed",
255 code=e.code)
256 self._registered = False
257 self._release_occupancy()

259 def _release_occupancy(self) -> None:
260 """释放会话占位:active_agents 摘除 + 互斥锁复位(close/回滚共用)。"""
261 self._session_lock = False
262 aa = getattr(self._spine, "active_agents", None)
263 if aa is not None and aa.get(self.session_id) is self:
264 aa.pop(self.session_id, None)

266 # ========================================================== 外壳唯一入口
267 async def submit(self, text: str, *,
268 meta: Optional[dict] = None) -> Any:
269 """外壳唯一写口:user.message 强同步落盘 → loop.wake(idle 拉起/入队)。
270
271 三壳(CLI/桌面/ACP)平权同路径,无特权通道;写操作与用户输入同级过
272 session.append 校验链。headless=True 且 run 同步结束时自动 close。
273 返回 loop.wake 的结果(RunResult | None;RunResult 属 agent_loop 模块,
274 此处鸭子返回)。meta 为外壳透传参数(当前规范未消费,保留签名)。
275
276 异常:BUSY(未 enter/已关闭)、EVT-100(空消息)、PERS-202(强同步落盘
277 失败,repair 后重试——输入未生效不丢话)。
278 """
279 if self.state == "init": # 状态机:enter 后方可 submit
280 _raise_busy(advice="agent 未激活:先 await enter() 再 submit")
281 if self.state in ("stopping", "closed"):
282 _raise_busy(advice="会话已关闭,请新开会话")
283 if not text or not text.strip(): # 空消息:信封层前置拒绝
284 raise_code("EVT-100", field="content", advice="消息不能为空")
285 self.state = "busy" # busy = loop running 透传
286 try:
287 # 强同步①:user.message 落盘成功才继续(崩溃一致性锚点,§3.6)
288 env = await self.ctx.session.append(
289 "user.message", {"content": text}, actor="user", sync=True)
290 result = await self.ctx.loop.wake(env) # idle→run;running→入队
291 if self.headless and result is not None: # headless:run 结束即关会话
292 await self.close(reason=getattr(result, "reason", "complete"))
293 return result
294 finally:
295 # 交互模式:run 自然结束(loop 回 idle)且会话保持 → 回 READY
296 if (self.state == "busy"
297 and getattr(self.ctx.loop, "state", None) == "idle"):
298 self.state = "ready"

300 # ========================================================== 会话关闭
301 async def close(self, reason: str = "idle") -> None:
302 """会话关闭(session.finished 唯一归属;幂等)。
303
304 流程:stopping 置位 → 在途 run 声明式取消(cancel reason=close)→
305 session.finished 强同步落盘(EVT-104 单次;失败回滚 state 供重试)→
306 终局清理(close_marker/摘订阅/注销插件句柄/逆序 detach 全部 caps,
307 尽力而为)→ 释放会话互斥 → closed。
308
309 异常:EVT-104(日志已有 finished = 绕过本路径的写口,INV-01 追查)、
310 PERS-202(finished 落盘失败,state 回滚,repair 后重试可重入)。
311 """
312 if self.state in ("stopping", "closed"):
313 return # 幂等:双触发(外壳/repair)安全
314 prev, self.state = self.state, "stopping"
315 # 1) 有在途 run → 先声明式取消(协作式,不杀进程)。S1-02:原判据含 "paused",
316 # 但 loop 无该态(ADR-014),收敛为唯一在途态 "running"。
317 if getattr(self.ctx.loop, "state", None) == "running":
318 await self.ctx.loop.cancel(reason="close")
319 # 2) finished 唯一写路径:失败回滚 state → close 可重入(见偏离 6)
320 try:
321 await self.ctx.session.append("session.finished",
322 {"reason": reason},
323 actor="system", sync=True)
324 except BaseException:
325 self.state = prev
326 raise
327 # 3) 终局清理:单步失败记日志不阻断终局(半卸 > 僵尸)
328 try:
329 self.ctx.session.close_marker() # SessionLog._closed=True
330 self.ctx.bus.unsubscribe_all(owner=self.agent_id) # 摘订阅
331 if self._registered: # 释放句柄寻址
332 try:
333 self.ctx.registry.unregister("plugin", self.agent_id)
334 except PyHError as e: # TLB-801:已摘(双触发/repair)
335 if e.code != "TLB-801":
336 raise
337 finally:
338 self._registered = False
339 await self.ctx.host.close() # 逆序 detach 全部 caps
340 except PyHError as e:
341 log.error("agent close cleanup failed", agent_id=self.agent_id,
342 code=e.code, exc=str(e))
343 finally:
344 self._release_occupancy() # 释放会话互斥(可新开)
345 self.state = "closed" # 终局
346 log.info("agent closed", agent_id=self.agent_id,
347 session=self.session_id, reason=reason)

349 # ========================================================== 能力挂载面
350 async def attach_capability(self, ns: str, iface: Any) -> None:
351 """能力挂载(announce 语义消费侧入口):白名单校验 → 挂载 → announce。
352
353 ns 越出六命名空间白名单 → TLB-802;ns 与脊柱注入成员重名(遮蔽 agent
354 自身依赖)→ TLB-802(INV-08 cap 不可覆盖);重复挂载 → TLB-801。
355 挂载成功后 ctx.{ns} 立即可消费,并 announce 留痕广播(DIS-SEAM §2.4)。
356 """
357 if ns not in Ctx.NAMESPACES: # 白名单越界(TLB-802)
358 raise_code("TLB-802", what=f"ctx.{ns}",
359 advice="脊柱命名空间不接受插件挂载(INV-08);"
360 "脊柱子系统注册走 BUS-002")
361 if ns in self.caps:
362 raise_code("TLB-801", what=f"ctx.{ns}",
363 detail=f"命名空间 {ns} 已挂载:先 detach_capability 再挂")
364 if hasattr(self.ctx, ns): # 遮蔽脊柱成员 = 破坏依赖
365 raise_code("TLB-802", what=f"ctx.{ns}",
366 advice="命名空间与脊柱注入成员重名,cap 不可覆盖(INV-08)")
367 self.caps[ns] = iface # iface = Definition + Provider 束
368 setattr(self.ctx, ns, iface) # 挂载后 ctx.{ns} 立即可消费
369 await self.announce(ns) # registry.updated 留痕广播
370
371 async def announce(self, ns: str) -> Optional["Envelope"]:
372 """能力就绪宣布:registry.updated 广播 + expose_to_llm 进 tools 注册表。
373
374 无此能力(attach 前调用)→ 静默返回 None。留痕为瞬时总线广播(session
375 append 拒瞬时事件,见偏离 3),无落盘信封可返 → 恒返回 None。
376 """
377 iface = self.caps.get(ns)
378 if iface is None:
379 return None # 无此能力:静默
380 if getattr(iface, "expose_to_llm", False): # 声明对 LLM 可见
381 self.ctx.tools.register_definition(iface.definition)
382 await self.ctx.bus.emit( # F003 语义留痕(瞬时)
383 "registry.updated",
384 {"op": "attach", "kind": "capability", "key": ns})
385 return None
386
387 async def detach_capability(self, ns: str) -> None:
388 """能力摘除(幂等):注销工具定义 → ctx 摘除 → registry.updated(op=del)。
389
390 未挂载 → 直接返回;单步失败内部消化并告警留痕,继续摘除剩余步骤
391 (半卸 > 僵尸,spec:detach 抛错仍继续,失败标记 broken)。
392 """
393 if ns not in self.caps:
394 return # 幂等:已 detached
395 iface = self.caps.pop(ns)
396 try:
397 if getattr(iface, "expose_to_llm", False):
398 self.ctx.tools.unregister_definition(iface.definition.name)
399 if hasattr(self.ctx, ns):
400 delattr(self.ctx, ns) # ctx 门面摘除
401 await self.ctx.bus.emit( # op=del 留痕(瞬时)
402 "registry.updated",
403 {"op": "del", "kind": "capability", "key": ns})
404 except Exception as e: # 半卸 > 僵尸:告警不中断
405 log.error("cap detach failed", ns=ns, exc=e)
406 try:
407 await self.ctx.bus.emit("bus.backpressure",
408 {"ns": ns, "broken": True})
409 except Exception: # noqa: BLE001 告警自身失败
410 log.error("cap detach alert failed", ns=ns)

444# ------------------------------------------------------------------ 工厂
445def create_agent(session_id: str, spine: Any, cfg: Any) -> Agent:
446 """会话实体工厂:脊柱硬接线 + ctx 装配 + 会话占位登记(纯构造,不 activate)。
447
448 一会话一 agent(DIS §2.6-1):active_agents 已有同会话 agent → BUSY(防双
449 loop 竞争 seq)。构造 state=init,必须经 await agent.enter() 激活方可
450 submit;enter 失败则回滚释放占位。
451
452 异常:BUSY(同会话已有 active agent / state 非法)、TLB-801(enter 阶段
453 registry 重名注册,由 enter 上抛)。
454
455 参数:spine = 8 模块束(session/bus/registry/scope/tools/llm/loop/… +
456 active_agents 占位表);cfg = 分层配置 Settings。
457 """
458 aa = getattr(spine, "active_agents", None)
459 if aa is None: # 防御:装配束未建占位表 → 就地补建
460 aa = {}
461 spine.active_agents = aa
462 if session_id in aa:
463 _raise_busy(session_id=session_id,
464 advice="该会话已有 active agent,请复用句柄或新开会话")
465 ag = Agent(agent_id=uuid.uuid4().hex, session_id=session_id,
466 state="init", config=cfg, caps={},
467 headless=_resolve_headless(cfg, spine))
468 # ctx 装配:脊柱 8 模块按拓扑序注入(依赖环不可能,INV-08);sys/storage/
469 # persistence 允许缺席(阶段装配期逐模块落地,getattr 兜底 None)
470 ag.ctx = Ctx(session=spine.session, scope=getattr(spine, "scope", None),
471 tools=getattr(spine, "tools", None),
472 llm=getattr(spine, "llm", None),
473 loop=getattr(spine, "loop", None),
474 sys=getattr(spine, "sys", None),
475 storage=getattr(spine, "storage", None),
476 persistence=getattr(spine, "persistence", None))
477 ag.ctx.bus = spine.bus # 事件总线(留痕/订阅出口)
478 ag.ctx.registry = spine.registry # 注册表(F003 寻址)
479 ag.ctx.config = cfg
480 ag.ctx.platform_runtime = getattr(spine, 'platform_runtime', None)
481 # 出口脱敏单口(F016/INV-09):tool_fs / tool_web / spill 三处读 ``ctx.redact``,
482 # 但 2026-09-21 R10 前**全库无注入点** ⇒ 生产恒缺失 ⇒ fs/web 出口静默降级为
483 # 不打码(实测 read_file 输出的 `API_KEY=sk-AAAA…` 原文进事件与 LLM 上下文)。
484 # 唯一注入源 = `config.redactor_of(cfg)`(含 log.redact_enabled 开关语义)。
485 from pyharness.config import redactor_of
486 ag.ctx.redact = redactor_of(cfg)
487 # 通道身份(GAP-11):外壳在装配期经 ``EngineSpine.channel`` 声明,此处落到
488 # agent ctx——治理层 ``authorize()`` 收到的是**本 ctx**,不是外壳自己的 ctx,
489 # 故该跳缺失会让每次决策静默降级为 SYSTEM 主体。未声明(= 哨兵)时**不写属性**,
490 # 由 ``principal_of`` 见属性缺失即 APR-503 fail-closed(绝不静默降级)。
491 _chan = getattr(spine, "channel", CHANNEL_UNDECLARED)
492 if _chan is not CHANNEL_UNDECLARED:
493 ag.ctx.channel = _chan
494 ag.ctx.guard = getattr(spine, "guard", None) # guard 链(executor 关3,F031 兜底)
495 ag.ctx.governance = getattr(spine, "governance", None) # 治理层单实例(ADR-018;S2-3)
496 ag.ctx.approval = getattr(spine, "approval", None) # 审批(executor 转审批 F015)
497 ag.ctx.counters = getattr(spine, "counters", None) # F032 预算闸数据源(report_usage 写)
498 ag.ctx.sysprompt = getattr(spine, "sysprompt", None) # F010 装配器(agent 出网前挂)
499 ag.ctx.compactor = getattr(spine, "compactor", None) # F058 压缩器(请求边界触发)
500 ag.ctx.goals = getattr(spine, "goals", None) # F047 目标管理器(goal.* 工具 + sysprompt 段)
501 ag.ctx.todos = getattr(spine, "todos", None) # #32 待办管理器(todo.* 工具)
502 ag.ctx.ask = getattr(spine, "ask", None) # #38 反问宿主(user.ask 工具)
503 ag.ctx.skills = getattr(spine, "skills", None) # F073 技能管理器(skill.load + 目录段)
504 ag.ctx.plan = getattr(spine, "plan", None) # F045/F046 计划模式
505 ag.ctx.schedule = getattr(spine, "schedule", None) # F048 调度器
506 ag.ctx.jobs = getattr(spine, "jobs", None) # F051 后台任务
507 ag.ctx.subagent = getattr(spine, "subagent", None) # F049 子 Agent
508 ag.ctx.search_backend = getattr(spine, "search_backend", None) # F037
509 ag.ctx.streaming = bool(getattr(spine, "streaming", False)) # F027 流式开关
510 ag.ctx.budget = getattr(spine, "budget", None) # F032 只读预算门面(仪表盘/报表)
511 ag.ctx.session_query = getattr(spine, "fts", None) # F057 索引句柄(懒激活后回填)
512 ag.ctx.tool_registry = getattr(spine, "tool_registry", None) # 工具注册表(F057 announce 注册面)
513 ag.ctx.host = _CapabilityHost(ag) # 关闭时逆序 detach 全部 caps
514 ag._spine = spine # 释放占位需回写装配束
515 aa[session_id] = ag # 占位登记(enter 失败回滚释放)
516 return ag
