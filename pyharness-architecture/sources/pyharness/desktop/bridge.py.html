22class DesktopBridge:
23 """pywebview js_api:webview 线程方法 → 引擎事件循环跨线程投递(run_coroutine_threadsafe)。
24
25 暴露 window.submit(text)/approve(aid,decision)/replay_frame(sid,from_seq)/
26 list_sessions();方法薄壳无业务,返回 JSON 字符串(错误 = code 体 ADR-011)。桥无特权:
27 裁决仍过 guard/预算引擎链(单调性高于人类意志)。_pending_calls 登记在途调用供审计。
28 """

30 def __init__(self, app: DesktopApp) -> None:
31 self.app = app
32 self._pending_calls: dict[str, float] = {} # call_id → 发起时刻(在途登记)
33
34 def _run(self, coro: Any) -> str:
35 """跨线程投递并阻塞取结果(≤60s);PyHError → code 体;超时 → BUSY(不悬挂)。"""
36 app = self.app
37 loop = app.loop
38 if loop is None or not loop.is_running():
39 if inspect.iscoroutine(coro):
40 coro.close() # 未投递协程显式关闭,防 'never awaited' 告警
41 return json.dumps({"code": "BUSY",
42 "advice": "引擎服务未就绪(loop 未运行),稍后再试"},
43 ensure_ascii=False)
44 call_id = uuid.uuid4().hex[:8]
45 self._pending_calls[call_id] = time.monotonic()
46 fut = None
47 try:
48 fut = asyncio.run_coroutine_threadsafe(coro, loop)
49 data = fut.result(timeout=BRIDGE_TIMEOUT_S)
50 return json.dumps(_jsonable(data), ensure_ascii=False)
51 except PyHError as e: # 引擎错:远端只回 code+advice
52 advice = (e.ctx.get("advice") or e.ctx.get("hint") or e.spec.advice)
53 return json.dumps({"code": e.code, "advice": str(advice)},
54 ensure_ascii=False)
55 except (asyncio.TimeoutError, TimeoutError): # 引擎忙:本地错误体(提示稍后再试)
56 if fut is not None:
57 fut.cancel() # 不留下继续消费/重复执行的协程
58 return json.dumps({"code": "BUSY", "advice": "引擎繁忙,稍后再试"},
59 ensure_ascii=False)
60 except Exception as exc: # noqa: BLE001 未预期兜底
61 log.error("bridge 调用未预期异常", exc_info=True)
62 return json.dumps({"code": "CYC-999",
63 "advice": f"{type(exc).__name__}: 引擎内部错误"},
64 ensure_ascii=False)
65 finally:
66 self._pending_calls.pop(call_id, None)
