15 def _missed_window(self, job: ScheduleJob, now_dt: datetime) -> bool:
16 """宕机窗判定:持久化 next_fire_at 已过(窗口落进宕机期)→ 错过。"""
17 return (job.next_fire_at is not None and job.next_fire_at <= now_dt)

19 def _missed_count(self, job: ScheduleJob, now_dt: datetime) -> int:
20 """宕机窗内错过的触发机会数(合并计数;PRD:只记 missed 不补跑)。
21
22 cron:窗口 [next_fire_at, now_dt] 内命中表达式且到点的分钟数(上限 10 万分钟
23 防极端长宕机死循环);interval:过窗周期数;at:1(一次性)。
24 """
25 nxt = job.next_fire_at
26 if nxt is None:
27 return 0
28 if job.kind == "at":
29 return 1
30 spec = self._spec_of(job)
31 if job.kind == "interval":
32 span = (now_dt - nxt).total_seconds()
33 return max(1, int(span // spec.seconds) + 1)
34 t = nxt.replace(second=0, microsecond=0)
35 cap = now_dt.replace(second=0, microsecond=0)
36 count = 0
37 guard = 0
38 while t <= cap and guard < 100_000: # 100k 分钟 ≈ 70 天窗口
39 if self._cron_match(spec, t):
40 count += 1
41 t += timedelta(minutes=1)
42 guard += 1
43 return max(1, count)

45 async def recover(self, ctx: Any = None, *,
46 now_dt: Optional[datetime] = None) -> int:
47 """崩溃恢复(F048 …/恢复):回放 schedule.* 重建注册表;宕机期错过的触发
48 只记 schedule.missed 不补跑;窗口重臂后启动分钟 ticker;返回 missed 总数。
49
50 窗口未到(next_fire_at 仍未来)→ 保持快照原样续跑(与崩溃前一致);
51 窗口已过 → 记 missed 并把 last_fired 清零、从 now 重臂(偏离 3)。
52 """
53 now_dt = now_dt or self._now()
54 sess = self._session_ctx(ctx)
55 # 事件重建注册表(INV-01):重建含 trigger 派生的 last_fired_at/前推窗口
56 rebuilt = type(self)._rebuild_from(self._iter_session_events())
57 self._jobs = rebuilt._jobs
58 missed_total = 0
59 for j in self._jobs.values():
60 if j.paused: # 暂停中:不核算错过
61 continue
62 if self._missed_window(j, now_dt): # 宕机窗内错过 → 只留痕不补跑
63 await sess.append("schedule.missed", {
64 "job": j.name, "missed": self._missed_count(j, now_dt),
65 "since": iso(j.next_fire_at), "until": iso(now_dt)},
66 actor="system")
67 missed_total += 1
68 spec = self._spec_of(j)
69 if j.kind == "interval":
70 # 错过窗已消费:显式从 now 重臂(不依赖 stale last_fired,不补跑)
71 j.next_fire_at = now_dt + timedelta(seconds=spec.seconds)
72 else:
73 j.next_fire_at = self.next_fire(j, now_dt)
74 if j.kind == "at" and j.next_fire_at is None:
75 continue # 一次性 at 已彻底过期:由 remove 清理
76 # 窗口未到:保持快照(next_fire_at 未来),继续未来触发
77 # BUG-1 自愈:旧版扫描窗为 24h,使下次触发在 24h 之外的合法 cron(工作日/
78 # 每周/每月)落盘 next_fire_at=null 并永不触发。这类 job 无触发历史,回放
79 # 不会重算窗口 → 在此按新口径重算并显式落 schedule.updated(事件即真源)。
80 for j in self._jobs.values():
81 if j.paused or j.kind != "cron" or j.next_fire_at is not None:
82 continue
83 try:
84 nxt = self.next_fire(j, now_dt)
85 except PyHError as e: # 真无触发点(如 2/30 的历史遗留)
86 await sess.append("system.error", {
87 "code": e.code,
88 "hint": f"schedule:{j.name} 无未来触发点,已跳过"},
89 actor="system")
90 continue
91 await sess.append("schedule.updated", {
92 "name": j.name, "paused": False, "next_fire_at": iso(nxt)},
93 actor="system")
94 j.next_fire_at = nxt
95
96 if self._auto_ticker:
97 self._ensure_ticker(ctx)
98 return missed_total

100 @classmethod
101 def _rebuild_from(cls, events: list[Any]) -> "Scheduler":
102 """事件回放重建(INV-01,内部):顺序重放 schedule.registered/updated/removed
103 重建 _jobs;孤儿 updated/removed 防御跳过;补扫 schedule.trigger 重建
104 last_fired_at 并前推 next_fire_at(偏离 2:missed 只计宕机窗)。
105
106 返回全新注册表实例(类方法,不触碰 self 状态)。
107 """
108 m = cls() # 清空后按事件重建(无 session:只读)
109 for e in events:
110 t = getattr(e, "type", "")
111 p = getattr(e, "payload", {}) or {}
112 if t == "schedule.registered":
113 m._jobs[p["name"]] = ScheduleJob(
114 name=p["name"], kind=p["kind"], expr=p["expr"],
115 template={"intent": p["template"]["intent"]},
116 is_risky=bool(p.get("is_risky", False)),
117 paused=bool(p.get("paused", False)),
118 next_fire_at=parse_iso(p.get("next_fire_at")),
119 created_seq=getattr(e, "seq", 0))
120 elif t == "schedule.updated": # 暂停/恢复状态迁移
121 j = m._jobs.get(p.get("name"))
122 if j is None:
123 log.warning("orphan schedule.updated seq=%s", getattr(e, "seq", "?"))
124 continue # 孤儿(坏日志):防御跳过
125 j.paused = bool(p.get("paused", j.paused))
126 nxt = parse_iso(p.get("next_fire_at"))
127 if nxt is not None:
128 j.next_fire_at = nxt
129 elif t == "schedule.removed": # 移除留痕
130 m._jobs.pop(p.get("name"), None)
131 # 第二遍:以最后一次 trigger.fired_at 为锚重建 last_fired_at/前推 next_fire_at
132 last_trigger: dict[str, datetime] = {}
133 for e in events:
134 p = getattr(e, "payload", {}) or {}
135 if getattr(e, "type", "") == "schedule.trigger":
136 dt = parse_iso(p.get("fired_at"))
137 if dt is not None and (dt > last_trigger.get(p.get("job"), datetime.min)):
138 last_trigger[p.get("job")] = dt
139 for name, job in m._jobs.items():
140 fired = last_trigger.get(name)
141 if fired is None or job.kind in ("at",) or job.paused:
142 continue # 无触发历史/一次性/暂停:快照即权威
143 job.last_fired_at = fired
144 job.spec = m._parse_spec(job.kind, job.expr, require_future=False)
145 if job.kind == "interval":
146 # 确定性前推:上一触发 + 间隔(宕机窗 = 上一窗之后)
147 job.next_fire_at = fired + timedelta(seconds=job.spec.seconds)
148 else:
149 # cron:fired 之后最近的匹配分钟
150 job.next_fire_at = m.next_fire(job, fired)
151 return m # 内存态 ≡ 日志派生态(INV-01)

182 async def _ticker(self, ctx: Any = None) -> None:
183 """常驻协程:睡到下一分钟边界后 _tick(F048 最小粒度 1 分钟);单轮异常不
184 退出泵(记 system.error 后继续下一轮)。"""
185 while not self._stopped:
186 nxt = (self._now() + timedelta(minutes=1)).replace(second=0, microsecond=0)
187 try:
188 await self._sleep(max(1.0, (nxt - self._now()).total_seconds()))
189 except asyncio.CancelledError:
190 raise # 外部停止:让路
191 try:
192 await self._tick(ctx) # 每分钟检查一次(F048)
193 except asyncio.CancelledError:
194 raise
195 except Exception as e: # noqa: BLE001 单轮异常不杀泵
196 log.exception("schedule tick failed: %s", e)
197 try:
198 await self._session_ctx(ctx).append("system.error", {
199 "code": "CYC-999", "hint": "schedule tick 异常已隔离"},
200 actor="system")
201 except Exception: # noqa: BLE001 留痕失败不阻断泵存活
202 log.exception("schedule tick error event failed")
