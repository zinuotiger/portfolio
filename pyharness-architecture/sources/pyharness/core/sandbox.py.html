64def safe_path(root: Path, name: str) -> Path:
65 """One strict relative path boundary, including symlinks and junctions."""
66 if not isinstance(name, str) or not name or '\\' in name or ':' in name or '\x00' in name:
67 fail('path_rejected')
68 p = PurePosixPath(name)
69 if p.is_absolute() or any(x in ('.', '..') for x in name.split('/')):
70 fail('path_rejected')
71 _reject_link(root, 'workspace root must not be a link')
72 try:
73 base = root.resolve()
74 except (OSError, ValueError):
75 fail('path_rejected')
76 dest = root.joinpath(*p.parts)
77 current = root
78 for part in p.parts:
79 if part.endswith(('.', ' ')) or re.fullmatch(r'(?i)(con|prn|aux|nul|com[1-9]|lpt[1-9])(?:\..*)?',part):
80 fail('path_rejected','reserved path component')

83 try:
84 contained = dest.resolve().is_relative_to(base)

485 async def start_process(self, ident, command):
486 self.workspace(ident)
487 if sum(owner==ident and not task.done() for owner,task in self.processes.values())>=8:
488 fail('sandbox_process_limit')
489 token='process-'+uuid.uuid4().hex
490 task=asyncio.create_task(self.execute(ident,command))
491 self.processes[token]=(ident,task)
492 # Read exceptions even when the caller never polls the process.
493 task.add_done_callback(lambda t: None if t.cancelled() else t.exception())
494 return {'pid':token,'running':True}

496 def process_status(self, ident, token):
497 item=self.processes.get(token)
498 if item is None or item[0]!=ident:
499 fail('sandbox_process_not_found')
500 task=item[1]
501 if not task.done():
502 return {'pid':token,'running':True}
503 if task.cancelled():
504 return {'pid':token,'running':False,'cancelled':True}
505 if task.exception():
506 return {'pid':token,'running':False,'error_code':'sandbox_process_failed'}
507 return {'pid':token,'running':False,**task.result()}

509 async def kill_process(self, ident, token):
510 self.process_status(ident,token)
511 task=self.processes[token][1]
512 if not task.done():
513 task.cancel()
514 await asyncio.gather(task,return_exceptions=True)
515 # Docker cancellation terminates the whole run-owned container.
516 if self.records[ident].backend in ('isolated','remote'):
517 await self.destroy(ident)
518 return self.process_status(ident,token)

520 async def destroy(self, ident):
521 record = self.records.get(ident)
522 if record is None:
523 fail('sandbox_not_found')
524 if record.status == 'destroyed':
525 return record
526 try:
527 tasks=[task for owner,task in self.processes.values() if owner==ident and not task.done()]
528 for task in tasks:
529 task.cancel()
530 if tasks:
531 await asyncio.gather(*tasks,return_exceptions=True)
532 await self.backends[record.backend].destroy(record)
533 root = safe_path(self.root, ident)
534 if root.exists():
535 shutil.rmtree(root)
536 record.status = 'destroyed'
537 except BaseException:
538 record.status = 'cleanup_failed'
539 record.error_code = 'sandbox_destroy_failed'
540 raise
541 return record
