38class Registry:
39 """注册表:三类索引 + 保留名护栏 + registry.updated 留痕(F003)。"""
40
41 RESERVED = RESERVED
42 KINDS: tuple[str, ...] = ("plugin", "tool", "capability")
43
44 def __init__(self, bus: "EventBus"):
45 self._index: dict[str, dict[str, Any]] = {k: {} for k in self.KINDS}
46 self._bus = bus
47
48 # ------------------------------------------------------------ 内部护栏
49 def _check_kind(self, kind: str) -> None:
50 """kind 越界 → TLB-801(非法注册),禁 KeyError 裸崩。"""
51 if kind not in self.KINDS:
52 raise_code("TLB-801", kind=kind, detail="kind 须为 plugin/tool/capability")
53
54 # ------------------------------------------------------------ 写入
55 def register(self, kind: str, key: str, obj: Any) -> None:
56 """三类索引写入(plugin/tool/capability);留痕 registry.updated。"""
57 self._check_kind(kind)
58 if kind == "plugin" and key in self.RESERVED:
59 raise_code("BUS-002", key=key,
60 hint="脊柱八模块/子系统名为保留名,不可注册为插件(原则 5)")
61 if key in self._index[kind]:
62 raise_code("TLB-801", kind=kind, key=key,
63 detail="同 kind 同 key 重复注册(改 = 注销重注册,无脏数据)")
64 self._index[kind][key] = obj
65 self._bus.emit_sync("registry.updated", # F003 瞬时仅内存留痕
66 {"op": "add", "kind": kind, "key": key})
67
68 def lookup(self, kind: str, key: str) -> Any:
69 """O(1) 查询;key 不存在 → 结构化错误(不静默返回 None, F003 边界)。"""
70 self._check_kind(kind)
71 if key not in self._index[kind]:
72 if kind == "tool":
73 raise_code("TLB-802", kind=kind, key=key, hint="查工具名拼写与注册表")
74 raise_code("TLB-801", kind=kind, key=key,
75 detail="key 未注册/已卸载(lookup),不静默返回 None")
76 return self._index[kind][key]
77
78 def unregister(self, kind: str, key: str) -> None:
79 """摘除索引并广播 registry.updated(op=del);注销不存在 key 不静默。"""
80 self._check_kind(kind)
81 if key not in self._index[kind]:
82 raise_code("TLB-801", kind=kind, key=key,
83 detail="unregister-unknown:注销不存在的 key 不静默(幂等由调用方先 lookup)")
84 del self._index[kind][key]
85 self._bus.emit_sync("registry.updated", # op=del 留痕
86 {"op": "del", "kind": kind, "key": key})
87
88 def __contains__(self, pair: tuple[str, str]) -> bool:
89 """(kind, key) 成员判定(内部/测试辅助)。"""
90 kind, key = pair
91 return kind in self._index and key in self._index[kind]
