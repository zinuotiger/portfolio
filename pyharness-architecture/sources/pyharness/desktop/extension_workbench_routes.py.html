11def operator_check(desktop, request):
12 """Return a live authority check, including after long host-process awaits."""
13 original = operator_principal(request)
14 def check():
15 if original:
16 account = store(desktop).authenticate(read_cookie(request, COOKIE))
17 if (not account or account['role'] != 'operator' or account['tenant'] != 'default'
18 or account['user_id'] != original['user_id']):
19 raise HTTPException(403, 'operator_required')
20 elif not desktop._operator_ok(request):
21 raise HTTPException(403, 'operator_required')
22 check()
23 return check

26async def bounded_body(request, limit):
27 chunks = bytearray()
28 async for chunk in request.stream():
29 if len(chunks) + len(chunk) > limit:
30 fail('diagnostic_request_limit')
31 chunks.extend(chunk)
32 value = json.loads(chunks)
33 if type(value) is not dict:
34 fail('invalid_diagnostic_request')
35 return value
