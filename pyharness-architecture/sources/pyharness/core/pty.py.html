1"""pyharness/core/pty.py — 真实伪终端(PTY)执行后端(F053)。
2
3一句话职责:把命令放进**真 PTY** 里跑,使子进程看到 isatty()=True 的终端语义
4(颜色/TTY 判定/行编辑/交互式 CLI 可用)。这是 proc.py 行式管道(stdin 写/stdout
5读)的**升级替代**,不是降级。
6
7后端选型(实测决定,诚实记录):
8- Windows:`pywinpty`(ConPTY 的成熟封装,可选依赖 extra `pty`)。
9 ⚠️ 为什么不自己 ctypes 直连 CreatePseudoConsole:2026-09-12 本机实测,按
10 MSDN 流程(CreatePipe → CreatePseudoConsole → STARTUPINFOEXW +
11 PROC_THREAD_ATTRIBUTE_PSEUDOCONSOLE → CreateProcessW)自建实现能创建并启动
12 子进程,但 **输出端 PeekNamedPipe/ReadFile 恒为 0 字节**(7 种句柄/线程/轮询
13 变体全试过);同环境 pywinpty 一次跑通并返回带控制序列的输出。故采用成熟封装,
14 不再重复造 ConPTY 轮子(依赖缺失 → TLB-807,提示 pip install pywinpty)。
15- POSIX:标准库 `pty.openpty`(零依赖)。
16
17与 proc.py 的分工:
18- proc.py : 后台会话式进程(行式,可 status/kill 管理),无 TTY 语义;
19- pty.py(本): 一次性 PTY 执行(可喂 stdin/超时/收集输出),有 TTY 语义;
20- 两者共同点:cwd 锁在会话沙箱目录、env 走 proc.sandbox_env 白名单(不继承
21 API key/代理)、墙钟上限——**都是误操作约束,不是安全沙箱**;安全主闸仍是
22 exec 域的 danger=high → 审批(F015)。
23"""

48def _tree_kill(pid: Optional[int], fallback: Any) -> None:
49 """树级终止(复用 proc._kill_pid_tree:Windows taskkill /T /F、POSIX killpg)。
50
51 CND-05/契约`SECURITY §4`+`CFG F052`:"超时杀进程树防孤儿"——PTY 的 shell 常有
52 子孙进程,只杀直接进程会遗留孤儿(载体未真正驱逐)。PID 缺失/异常 → 回退 fallback
53 (直接进程,kill/terminate 尽力收尾)。
54 """
55 if pid:
56 try:
57 _proc._kill_pid_tree(int(pid))
58 return
59 except Exception: # noqa: BLE001 收尾尽力
60 pass
61 try:
62 fallback()
63 except Exception: # noqa: BLE001 收尾尽力
64 pass

68class _WinPtySession:
69 """pywinpty(ConPTY)会话:后台线程读、非阻塞轮询取缓冲。"""
70
71 def __init__(self, command: Union[str, list], *, cwd: Path, env: dict,
72 cols: int = DEFAULT_COLS, rows: int = DEFAULT_ROWS) -> None:
73 self.command = command
74 self.cwd = cwd
75 self.env = env
76 self.cols = cols
77 self.rows = rows
78 self._p: Any = None
79 self._parts: list[str] = []
80 self._lock = threading.Lock()
81 self._eof = threading.Event()
82 self._reader: Optional[threading.Thread] = None
83 self._closed = False
84
85 def start(self) -> None:
86 _unavailable()
87 try:
88 from winpty import PtyProcess # 可选依赖 extra `pty`
89 except ImportError as e:
90 raise_code("TLB-807", module="pty",
91 hint=f"pywinpty 未安装({type(e).__name__})",
92 advice="pip install pywinpty 后重试;或改用 exec.shell_run")
93 self._p = PtyProcess.spawn(self.command, cwd=str(self.cwd),
94 env=self.env or None,
95 dimensions=(self.rows, self.cols))
96 self._reader = threading.Thread(target=self._pump, name="pty-reader",
97 daemon=True)
98 self._reader.start()
99
100 def _pump(self) -> None:
101 """后台读:pywinpty.read 阻塞到有数据/EOF;缓冲到内存(上限截断)。"""
102 total = 0
103 while True:
104 try:
105 data = self._p.read(READ_CHUNK)
106 except (EOFError, OSError): # EOF/管道关闭
107 break
108 except Exception as e: # noqa: BLE001 读面异常即收尾
109 log.debug("pty read stopped: %s", type(e).__name__)
110 break
111 if not data:
112 break
113 with self._lock:
114 if total < MAX_OUTPUT_CHARS:
115 self._parts.append(data)
116 total += len(data)
117 self._eof.set()
118 # EOF 后在同线程阻塞收尸:把 exitstatus 落定,后续 exit_code() 读面零等待
119 # (否则 pywinpty 的 isalive/exitstatus 会让每次调用白等数秒,实测 ~4s)
120 try:
121 self._p.wait()
122 except Exception: # noqa: BLE001 收尸失败不影响输出
123 pass
124
125 # -------------------------------------------------- 交互
126 def write(self, text: str) -> None:
127 if self._p is None or self._closed:
128 return
129 try:
130 self._p.write(text)
131 except Exception as e: # noqa: BLE001 管道已断
132 log.debug("pty write failed: %s", type(e).__name__)
133
134 def read(self, timeout: float) -> str:
135 """等 timeout 秒;返回自上次调用起累计的新增输出(非阻塞取缓冲)。
136
137 流已关闭(_eof)且缓冲为空 → 立即返回 ""(不再空等,收尾更快)。
138 """
139 deadline = time.monotonic() + max(0.0, timeout)
140 while time.monotonic() < deadline:
141 with self._lock:
142 if self._parts:
143 data = "".join(self._parts)
144 self._parts.clear()
145 return data
146 if self._eof.is_set():
147 break
148 time.sleep(POLL_S)
149 with self._lock:
150 data = "".join(self._parts)
151 self._parts.clear()
152 return data
153
154 def finished(self) -> bool:
155 """端到端结束判定:PTY 流已关闭(EOF)或进程不可寻 — 收尾优先信号。
156
157 pywinpty 的 isalive() 在子进程退出后可能有秒级滞后(实测 ~4s),故以
158 reader 线程观察到的 EOF 为准,避免每次调用白等。
159 """
160 if self._eof.is_set():
161 return True
162 try:
163 return not self._p.isalive() if self._p is not None else True
164 except Exception: # noqa: BLE001 读态失败保守为未结束
165 return False
166
167 def exit_code(self) -> Optional[int]:
168 if self._p is None:
169 return None
170 try:
171 if self._p.isalive():
172 return None
173 except Exception: # noqa: BLE001 读态失败视为未知
174 return None
175 st = getattr(self._p, "exitstatus", None)
176 if st is None: # EOF 收尸尚未落定:不阻塞调用方
177 return None
178 return int(st)
179
180 def wait(self, timeout: float) -> Optional[int]:
181 deadline = time.monotonic() + max(0.0, timeout)
182 while True:
183 rc = self.exit_code()
184 if rc is not None:
185 return rc
186 if time.monotonic() >= deadline:
187 return None
188 time.sleep(POLL_S)
189
190 def kill(self) -> None:
191 if self._p is None:
192 return
193 _tree_kill(getattr(self._p, "pid", None),
194 lambda: self._p.terminate(force=True))
195
196 def close(self) -> None:
197 if self._closed:
198 return
199 self._closed = True
200 if self._p is not None:
201 _tree_kill(getattr(self._p, "pid", None),
202 lambda: self._p.terminate(force=True))
203 try:
204 self._p.close(force=True)
205 except Exception: # noqa: BLE001
206 pass
207 if self._reader is not None and self._reader.is_alive():
208 self._reader.join(timeout=1.0)

212class _PosixPty:
213 """POSIX 伪终端(pty.openpty;与 Windows 分支同接口,非阻塞轮询)。"""
214
215 def __init__(self, command: Union[str, list], *, cwd: Path, env: dict,
216 cols: int = DEFAULT_COLS, rows: int = DEFAULT_ROWS) -> None:
217 self.command = command
218 self.cwd = cwd
219 self.env = env
220 self.cols = cols
221 self.rows = rows
222 self._master: Optional[int] = None
223 self._proc: Optional[subprocess.Popen] = None
224 self._closed = False
225
226 def start(self) -> None:
227 _unavailable()
228 import pty
229 master, slave = pty.openpty()
230 try:
231 import fcntl
232 import struct
233 import termios
234 fcntl.ioctl(slave, termios.TIOCSWINSZ,
235 struct.pack("HHHH", self.rows, self.cols, 0, 0))
236 os.set_blocking(master, False) # 非阻塞轮询(同 Windows 语义)
237 except Exception: # noqa: BLE001 尺寸/非阻塞尽力
238 pass
239 self._master = master
240 argv = (shlex.split(self.command, posix=True)
241 if isinstance(self.command, str) else list(self.command))
242 self._proc = subprocess.Popen(
243 argv, cwd=str(self.cwd), env=self.env or None,
244 stdin=slave, stdout=slave, stderr=slave, close_fds=True,
245 start_new_session=True) # 独立会话/进程组:killpg 方可安全树级终止(CND-05)
246 os.close(slave)
247
248 def write(self, text: str) -> None:
249 if self._master is None or self._closed:
250 return
251 try:
252 os.write(self._master, text.encode("utf-8", "replace"))
253 except OSError as e:
254 log.debug("pty write failed: %s", e)

256 def _drain(self) -> str:
257 if self._master is None:
258 return ""
259 out = bytearray()
260 while True:
261 try:
262 chunk = os.read(self._master, READ_CHUNK)
263 except (BlockingIOError, OSError):
264 break
265 if not chunk:
266 break
267 out.extend(chunk)
268 if len(out) >= MAX_OUTPUT_CHARS:
269 break
270 return bytes(out).decode("utf-8", "replace")
271
272 def read(self, timeout: float) -> str:
273 deadline = time.monotonic() + max(0.0, timeout)
274 parts: list[str] = []
275 while time.monotonic() < deadline:
276 chunk = self._drain()
277 if chunk:
278 parts.append(chunk)
279 continue
280 if self.exit_code() is not None:
281 parts.append(self._drain())
282 break
283 time.sleep(POLL_S)
284 return "".join(parts)

319 pass

335def open_pty(command: Union[str, list], *, cwd: Path, env: Optional[dict] = None,
336 cols: int = DEFAULT_COLS, rows: int = DEFAULT_ROWS) -> Any:
337 """开一个真 PTY 会话(两平台同接口);不支持 → TLB-807。"""
338 _unavailable()
339 if not pty_supported():
340 raise_code("TLB-807", module="pty",
341 hint=("pywinpty 未安装(Windows 真 PTY 后端缺失)"
342 if os.name == "nt" else f"平台无 PTY 支持(os={os.name})"),
343 advice="pip install pywinpty(或 extras: pip install '.[pty]');"
344 "或改用 exec.shell_run 行式执行")
345 from pyharness.core import proc as proc_mod
346 cwd = Path(cwd)
347 env = env if env is not None else proc_mod.sandbox_env(cwd)
348 cls = _WinPtySession if os.name == "nt" else _PosixPty
349 sess = cls(command, cwd=cwd, env=env, cols=cols, rows=rows)
350 sess.start()
351 return sess

354def run_in_pty(command: Union[str, list], *, cwd: Path,
355 env: Optional[dict] = None, input_text: str = "",
356 timeout_s: int = 60, cols: int = DEFAULT_COLS,
357 rows: int = DEFAULT_ROWS) -> dict:
358 """一次性 PTY 执行(同步;Provider 经 asyncio.to_thread 调用)。
359
360 语义:开 PTY → (可选)喂 stdin → 收集输出直到退出或墙钟超时 → 超时则 kill
361 并标 timed_out=True。env 缺省走 proc.sandbox_env(白名单,cwd 锁沙箱目录)。
362
363 返回 {output, exit_code, timed_out, tty, truncated, elapsed_ms}。
364 `tty=True` 表示走的是真 PTY 后端(探针 scripts/probe_pty.py 用子进程
365 isatty 探测做硬验证)。
366 """
367 started = time.monotonic()
368 sess = open_pty(command, cwd=Path(cwd), env=env, cols=cols, rows=rows)
369 parts: list[str] = []
370 timed_out = False
371 truncated = False
372 total = 0
373 # PTY 的 Enter 是 CR(\r);统一归一,避免调用方用 \n 时"输入没反应"
374 feed = input_text.replace("\r\n", "\r").replace("\n", "\r") if input_text else ""
375 try:
376 if feed:
377 time.sleep(0.15) # 等 shell 就绪再喂(避免丢输入)
378 sess.write(feed)
379 deadline = started + max(1, int(timeout_s))
380 while True:
381 remaining = deadline - time.monotonic()
382 if remaining <= 0:
383 timed_out = True
384 break
385 chunk = sess.read(min(0.3, remaining))
386 if chunk:
387 parts.append(chunk)
388 total += len(chunk)
389 if total >= MAX_OUTPUT_CHARS:
390 truncated = True
391 break
392 continue
393 if sess.finished():
394 parts.append(sess.read(0.3)) # 取尽残留
395 break
396 if timed_out:
397 sess.kill()
398 sess.wait(GRACE_KILL_S)
399 finally:
400 exit_code = sess.exit_code()
401 if exit_code is None:
402 exit_code = sess.wait(0.5)
403 sess.close()
404 return {"output": "".join(parts), "exit_code": exit_code,
405 "timed_out": timed_out, "tty": True, "truncated": truncated,
406 "elapsed_ms": int((time.monotonic() - started) * 1000)}
