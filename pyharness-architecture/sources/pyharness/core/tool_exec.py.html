1"""pyharness/core/tool_exec.py — exec.* 受限进程工具族(#28/#29/#35/#58 Consumer 面)。
2
3工具:
4- exec.shell_run : 在会话工作目录运行 shell 命令(cmd /c;一次性,输出配额内取全文)
5- exec.python_run: 在会话工作目录运行 Python 代码(sys.executable -c)
6- exec.pty : 在**真伪终端(ConPTY/pty)**里运行命令(可喂 stdin;TTY 语义)
7- proc.start/status/kill: 会话式子进程(行式 stdin/stdout,PTY 降级见 proc.py)
8
9安全模型(诚实口径):
10- danger=high → guard g-danger 转审批(F015):LLM 每次调 exec 都要人批准,
11 "LLM 能跑代码但永远过不了人这关"= 审批是本模块的安全主闸;
12- strict 沙箱档下 exec.* 不可见(域不在 workspace/allowlist/self)→ 演示用
13 standard 档(security.policy.preset=standard)并配审批;
14- cwd/env/超时/输出配额只是误操作约束,**不是安全沙箱**:命令仍可访问
15 绝对路径、网络和宿主进程。若需抵御恶意代码,必须使用 OS 级隔离。
16"""

25def _sandbox_dir(ctx: Any):
26 """会话工作目录:workspace_root/sandbox(fail-closed:无根 → CYC-999)。
27
28 workspace_root 本身已是**会话专属根**(F055,{workspaces_dir}/{sid},见
29 scope.session_workspace)——2026-09-21 修:此前再拼一层 /{sid} 是"根=共享
30 workspaces_dir"时代的遗留,会把沙箱塞进 {sid}/{sid}/sandbox 双嵌套。
31 """
32 from pathlib import Path
33 scope = getattr(ctx, "scope", None)
34 ws = None
35 if scope is not None:
36 ws = getattr(scope.policy, "workspace_root", None) if getattr(
37 scope, "policy", None) else getattr(scope, "workspace_root", None)
38 if not ws:
39 raise_code("CYC-999", module="exec",
40 hint="workspace_root 未装配:沙箱目录无法定位(fail-closed)")
41 return Path(str(ws)) / "sandbox"

68async def exec_shell(args: dict, ctx: Any) -> str:
69 """exec.shell_run Provider:审批门控的一次性 shell。"""
70 _require_exec_visible(ctx)
71 command = str(args.get("command") or "").strip()
72 if not command:
73 raise_code("EVT-100", field="command", hint="缺 command")
74 from pyharness.core import proc
75 r = await proc.wait_result_async(_sandbox_dir(ctx), command,
76 timeout_s=_wallclock(ctx))
77 from pyharness.core.provider_result import ProviderOutcome
78 return ProviderOutcome(r)

94async def exec_pty(args: dict, ctx: Any) -> str:
95 """exec.pty Provider:审批门控的**真 PTY**执行(ConPTY;可喂 stdin)。
96
97 同步 PTY 循环经 asyncio.to_thread 跑线程池(不阻塞事件循环);超时/配额由
98 pty.run_in_pty 负责,输出超长走 spill 由 executor 关4 处理。
99 """
100 _require_exec_visible(ctx)
101 command = str(args.get("command") or "").strip()
102 if not command:
103 raise_code("EVT-100", field="command", hint="缺 command")
104 import asyncio
105
106 from pyharness.core import pty as pty_mod
107 timeout_s = int(args.get("timeout_s") or _wallclock(ctx))
108 timeout_s = max(1, min(timeout_s, 600))
109 result = await asyncio.to_thread(
110 pty_mod.run_in_pty, command, cwd=_sandbox_dir(ctx), env=None,
111 input_text=str(args.get("input") or ""), timeout_s=timeout_s)
112 head = (f"[PTY] exit={result['exit_code']} tty={result['tty']} "
113 f"timed_out={result['timed_out']} {result['elapsed_ms']}ms")
114 body = result["output"].strip()
115 if result["truncated"]:
116 body += "\n...(输出被截断)"
117 return f"{head}\n{body}" if body else head

120async def proc_start(args: dict, ctx: Any) -> str:
121 """proc.start Provider:审批门控的会话式后台进程(行式)。"""
122 _require_exec_visible(ctx)
123 command = str(args.get("command") or "").strip()
124 if not command:
125 raise_code("EVT-100", field="command", hint="缺 command")
126 from pyharness.core import proc
127 sid = getattr(getattr(ctx, "session", None), "sid", "anon")
128 sess = proc.start_session(sid, _sandbox_dir(ctx), command,
129 timeout_total_s=_wallclock(ctx))
130 return f"已启动会话 {sess.token}(pid={sess.pid});用 proc.status 查输出"

133async def proc_status(args: dict, ctx: Any) -> str:
134 """proc.status Provider:读会话尾部输出/退出码。"""
135 from pyharness.core import proc
136 sid = getattr(getattr(ctx, "session", None), "sid", "anon")
137 st = proc.find_session(sid, args.get("pid")).status()
138 parts = [f"pid={st['pid']} token={st['token']} "
139 f"running={st['running']} exit={st['exit_code']}"]
140 if st["tail"]:
141 parts.append("尾部输出:\n" + "\n".join(st["tail"]))
142 return "\n".join(parts)

207def register(registry: Any) -> list[str]:
208 """五要素登记 + Provider 绑定(engine 装配面调用)。"""
209 for name, desc, timeout_s, schema in _DEFS:
210 registry.register_tool(
211 ToolDefinition(name=name, danger=_DANGER[name], description=desc,
212 schema=schema, timeout_s=timeout_s, owner="builtin",
213 ctx_path="exec", version="1.0.0"),
214 provider=_PROVIDERS[name])
215 return [d[0] for d in _DEFS]
