112async def _cmd_schedule(sh: ShellCtx, positional: list[str], flags: dict) -> int:
113 """schedule list / add/remove/pause/resume(--session <sid>)。"""
114 from pyharness.core.schedule import Scheduler
115 from pyharness.persistence import session_log_paths
116 sessions_dir = _sessions_dir_ctx(sh)
117 action = (positional or ["list"])[0]
118 if action == "list":
119 jobs: dict[str, dict] = {}
120 if sessions_dir.is_dir():
121 for f in session_log_paths(sessions_dir):
122 sid = f.stem
123 try:
124 with _session_lines(f) as fh: # 会话级真源(段 + 主文件)
125 for line in fh:
126 line = line.strip()
127 if not line:
128 continue
129 try:
130 ev = json.loads(line)
131 except json.JSONDecodeError:
132 continue
133 p = ev.get("payload") or {}
134 name = p.get("name") or p.get("job")
135 t = ev.get("type", "")
136 key = f"{sid}:{name}"
137 if t == "schedule.registered" and name:
138 jobs[key] = {"name": name, "sid": sid,
139 "kind": p.get("kind"),
140 "expr": p.get("expr"),
141 "paused": bool(p.get("paused")),
142 "next_fire_at": p.get("next_fire_at")}
143 elif t == "schedule.updated" and name and key in jobs:
144 jobs[key]["paused"] = bool(p.get("paused",
145 jobs[key]["paused"]))
146 elif t == "schedule.removed" and name:
147 jobs.pop(key, None)
148 except Exception: # noqa: BLE001 坏文件跳过
149 continue
150 ordered = sorted(jobs.values(), key=lambda r: (r["sid"], r["name"]))
151 if flags.get("json"):
152 _json_line({"jobs": ordered})
153 return 0
154 if not ordered:
155 print("无定时任务")
156 return 0
157 for j in ordered:
158 state = "暂停" if j["paused"] else "运行"
159 print(f"{j['sid']}:{j['name']} {j['kind']} {j['expr']} {state}"
160 f" 下次 {j.get('next_fire_at')}")
161 return 0
162 if action not in ("add", "remove", "pause", "resume"):
163 raise_code("EVT-100", advice=f"未知 schedule 子命令:{action}",
164 hint="可用:list | add <name> <kind> <expr> <intent> | "
165 "remove|pause|resume <name>")
166 sid = flags.get("session")
167 if not sid:
168 raise_code("EVT-100", advice="管理定时任务需要 --session <sid>",
169 hint="用法:pyharness schedule add ... --session <sid>")
170 from pyharness.core.session import open_session
171 from pyharness.persistence import flush_kwargs_of, open_store
172 sid = str(sid)
173 if not (sessions_dir / f"{sid}.jsonl").exists():
174 raise_code("EVT-106", session_id=sid, hint="目标会话不存在")
175 store = open_store(sid, dir=sessions_dir,
176 **flush_kwargs_of(sh.ctx.settings))
177 log_ = await open_session(sid, store)
178 sched = Scheduler.rebuild_for_session(log_)
179 try:
180 if action == "add":
181 if len(positional) < 4:
182 raise_code("EVT-100", advice="add 缺参数",
183 hint="用法:schedule add <name> <kind> <expr> <intent>")
184 intent = " ".join(positional[4:])
185 await sched.register(positional[1], positional[2],
186 positional[3], {"intent": intent})
187 elif action in ("remove", "pause", "resume"):
188 if len(positional) < 2:
189 raise_code("EVT-100", advice=f"{action} 需要 name",
190 hint=f"用法:schedule {action} <name> --session <sid>")
191 fn = {"remove": sched.remove, "pause": sched.pause,
192 "resume": sched.resume}[action]
193 await fn(positional[1])
194 finally:
195 store.flush()
196 store.close()
197 msg = {"ok": True, "action": action, "session_id": sid,
198 "name": positional[1] if len(positional) > 1 else None}
199 if flags.get("json"):
200 _json_line(msg)
201 else:
202 print(f"schedule {action} 完成(sid={sid})")
203 return 0

211 hint="用法:pyharness plan <目标>")

218 pm = PlanManager()

221 "expires_at": str(p.expires_at),
222 "steps": [{"action": s.action, "tool": s.tool,

224 for s in p.steps]}
225 if flags.get("json"):
226 _json_line(summary)
227 else:
228 print(f"方案 {p.id}: {p.goal}(selfcheck={'ok' if p.selfcheck_ok else 'fail'})")
229 for i, s in enumerate(p.steps, 1):
230 print(f" {i}. [{s.risk}] {s.tool}: {s.action} → {s.expected}")
231 if sh.headless or sh.ctx.channel is None:
232 print("headless 不自动批准;方案已落盘,可用桌面/交互 chat 后续确认",
233 file=sys.stderr)
234 return 0
235 try:
236 ans = (await asyncio.to_thread(
237 input, f"批准方案 {p.id}? [y/N]: ")).strip().lower()
238 except (EOFError, KeyboardInterrupt):
239 ans = "n"
240 if ans not in ("y", "yes"):
241 await pm.plan_reject(p.id, who="cli", reason="cli-rejected", ctx=sh.ctx)
242 print("已拒绝")
243 return 0
244 await pm.plan_approve(p.id, who="cli", ctx=sh.ctx)
245 tasks = list(getattr(pm, "_exec_tasks", set()) or ())
246 if tasks:
247 await asyncio.gather(*tasks, return_exceptions=True)
248 final = pm.get(p.id)
249 status = final.status if final is not None else "?"
250 print(f"方案完成状态: {status}")
251 return 0 if status == "completed" else 1

253async def _cmd_workflow(sh: ShellCtx, positional: list[str], flags: dict, *, new_session, attach_engine) -> int:
254 """workflow <步骤1> [步骤2 …]:当前(或新建)会话内顺序编排(#45 真链)。
255
256 每步经队列提交真实 AgentLoop(与 chat 同一条 runner seam),workflow.step
257 事件留痕(started/done/failed);stop_on_fail=True = 首败即停。CLI 此前无
258 workflow 入口(管理面只在桌面壳),本条补齐。
259 """
260 from pyharness.core.workflow import WorkflowRunner, queue_submit_adapter
261 steps = [s for s in positional if str(s).strip()]
262 if not steps:
263 raise_code("EVT-100", advice="workflow 需要至少一个步骤文本",
264 hint='用法:pyharness workflow "步骤1" "步骤2" …')
265 if sh.ctx.session is None:
266 sh.ctx.session = await new_session(sh.ctx.settings, sh.ctx.bus)
267 _wire_queue(sh.ctx)
268 await attach_engine(sh)
269 queue = getattr(sh.ctx, "task_queue", None)
270 log_ = sh.ctx.session
271 if queue is None or log_ is None:
272 raise_code("CYC-999", cmd="workflow",
273 hint="会话或任务队列未装配(引擎未接)")
274 runner = WorkflowRunner(log_, queue_submit_adapter(log_, queue), name="cli")
275 results = await runner.run(steps, stop_on_fail=True)
276 ok = bool(results) and all(r["ok"] for r in results)
277 sid = str(getattr(log_, "sid", ""))
278 if flags.get("json"):
279 _json_line({"sid": sid, "ok": ok, "results": results})
280 else:
281 print(f"工作流 {len(results)} 步(会话 {sid}):")
282 for r in results:
283 mark = "✅" if r["ok"] else "❌"
284 print(f" {mark} [{r['index']}] {str(r['intent'])[:60]} → "
285 f"{str(r['summary'])[:100]}")
286 return 0 if ok else 1
