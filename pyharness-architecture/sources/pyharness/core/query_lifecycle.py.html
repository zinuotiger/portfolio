15 async def enter(self, ctx: Any = None) -> None:
16 """开库建表 + 挂订阅(spec 伪码;幂等:已装载直接返回)。
17
18 - 开 SQLite(WAL;文件库父目录 mkdir 0o700,库文件 chmod 0o600 尽力而为);
19 - CREATE VIRTUAL TABLE IF NOT EXISTS events_fts(7 列,actor 为 UNINDEXED
20 冗余列,偏离 1)+ 伴生唯一表 fts_rows(行键幂等闸,偏离 2);
21 - 阈值经 ctx.config/cfg 的 storage.fts.* 覆盖(构造显式参数优先);
22 - ctx.bus 接线:按 _INDEX_TYPES 逐类型订阅 → on_event(owner=**本实例属主**),
23 并启动 index_batch_ms 攒批定时器;
24 - 失败(磁盘/权限/旧 SQLite 无 FTS5)→ PERS-202,回 detached 不留半态。
25 """
26 if self._db is not None:
27 return # 幂等
28 holder = _cfg_holder(ctx)
29 if not self._explicit["index_batch_ms"]:
30 self.index_batch_ms = int(_cfg(holder, "storage.fts.index_batch_ms",
31 self.index_batch_ms))
32 if not self._explicit["query_timeout_s"]:
33 self.query_timeout_s = float(_cfg(holder, "storage.fts.query_timeout_s",
34 self.query_timeout_s))
35 if not self._explicit["result_limit"]:
36 self.result_limit = int(_cfg(holder, "storage.fts.result_limit",
37 self.result_limit))
38 db: Optional[sqlite3.Connection] = None
39 try:
40 if str(self._db_path) != ":memory:":
41 self._db_path.parent.mkdir(parents=True, exist_ok=True, mode=0o700)
42 db = sqlite3.connect(str(self._db_path), check_same_thread=False)
43 db.execute("PRAGMA journal_mode=WAL") # :memory: 返回 memory 无副作用
44 db.execute("PRAGMA busy_timeout=3000")
45 db.execute("CREATE VIRTUAL TABLE IF NOT EXISTS events_fts USING fts5("
46 "session_id UNINDEXED, seq UNINDEXED, type UNINDEXED, "
47 "ts UNINDEXED, actor UNINDEXED, title, content)")
48 # 伴生唯一表:行键 (session_id, seq) 幂等闸(FTS5 无唯一约束,偏离 2)
49 db.execute("CREATE TABLE IF NOT EXISTS fts_rows("
50 "session_id TEXT NOT NULL, seq INTEGER NOT NULL, "
51 "PRIMARY KEY (session_id, seq)) WITHOUT ROWID")
52 db.commit()
53 if str(self._db_path) != ":memory:":
54 try:
55 os.chmod(self._db_path, 0o600) # 权限 600(SECURITY §8.3)
56 except OSError:
57 pass # Windows ACL 尽力而为(chmod 主防线在文件库位置)
58 except (sqlite3.Error, OSError) as e: # 磁盘/权限/旧 SQLite 无 FTS5
59 if db is not None:
60 db.close()
61 raise_code("PERS-202", op="fts_enter", path=str(self._db_path),
62 hint=f"FTS 索引开库/建表失败: {e}",
63 advice="修存储或换带 FTS5 的 SQLite;重试 enter")
64 self._db = db
65 # 总线订阅(持久事件流;瞬时事件 llm.chunk 不在 _INDEX_TYPES,天然不过滤)
66 bus = getattr(ctx, "bus", None) if ctx is not None else None
67 subscribe = getattr(bus, "subscribe", None) if bus is not None else None
68 if callable(subscribe):
69 for t in _INDEX_TYPES:
70 subscribe(t, self.on_event, owner=self._owner)
71 self._bus = bus
72 try: # 200ms 攒批定时器兜底
73 self._sched_task = asyncio.get_running_loop().create_task(
74 self._scheduler())
75 except RuntimeError: # 无事件循环(同步上下文)
76 self._sched_task = None
77 log.info("session fts entered db=%s batch_ms=%d timeout_s=%s limit=%d",
78 self._db_path, self.index_batch_ms, self.query_timeout_s,
79 self.result_limit)

81 async def announce(self, ctx: Any = None) -> None:
82 """对外宣布(F049 同类五步):注册 LLM 可见工具 session.fts_query + 挂 ctx.session.fts。
83
84 失败(重名/保留名 TLB-801)回滚回 detached 无残留。仓库当前阶段无 F064 消费
85 方与全局装配面:ctx 缺 tools/registry 时仅记 info 不阻断(偏离 5)。
86 """
87 if ctx is None:
88 return
89 self._announce_ctx = ctx
90 try:
91 registry = None
92 for attr in ("tool_registry", "tools", "registry"):
93 cand = getattr(ctx, attr, None)
94 if cand is not None and callable(getattr(cand, "register_tool", None)):
95 registry = cand
96 break
97 if registry is not None:
98 # 查重:lookup 命中 = 已注册;抛 TLB-802 = 未注册(静默跳过登记)
99 has = getattr(registry, "has", None)
100 already = has(TOOL_NAME) if callable(has) else False
101 lookup = getattr(registry, "lookup", None)
102 if not callable(has) and callable(lookup):
103 try:
104 lookup(TOOL_NAME)
105 already = True
106 except PyHError as e:
107 if e.code != "TLB-802":
108 raise # 其他注册表故障如实上抛
109 if not already:
110 registry.register_tool(self._tool_definition(),
111 provider=FtsQueryHandle())
112 self._tool_registered = True
113 if not self._mounted: # 幂等:已挂载不重挂(防二次
114 self._mounted = self._mount(ctx) # announce 把 _mounted 清零)
115 log.info("session fts announced tool=%s mounted=%s", TOOL_NAME,
116 self._mounted)
117 except BaseException:
118 self._rollback_announce() # 半载回滚:无脏注册表/挂载
119 raise

144 def _mount(self, ctx: Any) -> bool:
145 """挂载定位器 ctx.session.fts:有 locator.mount 用 locator;否则直挂属性。"""
146 locator = getattr(ctx, "locator", None)
147 mount = getattr(locator, "mount", None)
148 if callable(mount):
149 try:
150 mount("ctx.session.fts", self)
151 return True
152 except Exception as e: # noqa: BLE001
153 _log_warn("fts announce locator.mount 失败: %s", e)
154 return False
155 sess = getattr(ctx, "session", None)
156 if sess is not None and not hasattr(sess, "fts"):
157 try:
158 setattr(sess, "fts", self) # 阶段 fallback(直挂属性)
159 return True
160 except Exception as e: # noqa: BLE001
161 _log_warn("fts announce 挂载 ctx.session.fts 失败: %s", e)
162 return False

179 async def detach(self, ctx: Any = None) -> None:
180 """摘除(spec 伪码):停定时器 → flush 未落批 → 退订 → 摘工具/挂载 → 关库。
181
182 幂等(已摘直接返回);逐步骤容错,失败记日志继续摘,不抛(半卸 > 僵尸)。
183 """
184 if self._db is None:
185 return # 幂等
186 # ① 停 200ms 定时器(取消 + 等待退出;flush 中断已由回填纪律兜底)
187 task = self._sched_task
188 self._sched_task = None
189 if task is not None and not task.done():
190 task.cancel()
191 try:
192 await task
193 except (asyncio.CancelledError, Exception): # noqa: BLE001
194 pass
195 # ② 清空攒批(失败仅告警,继续摘)
196 try:
197 await self.flush()
198 except PyHError as e:
199 _log_warn("fts detach flush 失败 code=%s(缓冲已回填,rebuild 可自愈)", e.code)
200 # ③ 摘订阅(owner 精确)
201 if self._bus is not None:
202 unsub = getattr(self._bus, "unsubscribe_all", None)
203 if callable(unsub):
204 try:
205 unsub(self._owner)
206 except Exception as e: # noqa: BLE001
207 _log_warn("fts detach 退订失败: %s", e)
208 self._bus = None
209 # ④ 摘工具/挂载(announce 痕迹)
210 c = ctx if ctx is not None else self._announce_ctx
211 if c is not None:
212 try:
213 if self._tool_registered:
214 self._unregister_tool(c)
215 if self._mounted:
216 self._unmount(c)
217 except Exception as e: # noqa: BLE001
218 _log_warn("fts detach 摘能力失败: %s", e)
219 self._mounted = False
220 self._tool_registered = False
221 self._announce_ctx = None
222 # ⑤ 关库置 None(幂等)
223 try:
224 self._db.close()
225 except Exception as e: # noqa: BLE001
226 _log_warn("fts detach 关库失败: %s", e)
227 self._db = None
228 log.info("session fts detached")

267 async def _scheduler(self) -> None:
268 """200ms 定时落库(index_batch_ms;spec 攒批兜底):周期 flush,失败告警续跑。"""
269 while True:
270 await asyncio.sleep(self.index_batch_ms / 1000.0)

272 await self.flush()
273 except PyHError as e:
274 _log_warn("fts 定时落库失败 code=%s(下次重试;rebuild 可自愈)", e.code)

276 log.exception("fts 定时落库异常(下次重试)")
