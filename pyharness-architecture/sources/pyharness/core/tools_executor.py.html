152 """工具执行管道总闸(统一 Consumer;DIS-SEAM §2.3)。
153
154 构造注入 registry(契约/模型/Provider 索引;可 None = 自建空注册表)。
155 scope/guard/approval/session/storage 一律经 execute 的 ctx 参数注入
156 (ctx.session/ctx.scope/ctx.guard/ctx.approval/ctx.storage.spill)——与
157 scope/guard/approval 模块"注入式装配"先例一致,装配层接线而非本模块持依赖。
158
159 实例状态(均会话内单调/轮内可重置,单事件循环无需锁):

230 def parse_tool_call(self, raw: Any, parent_seq: Optional[int] = None) -> ToolCall:
231 """LLM 原始 tool_calls 单条 → ToolCall(spec 伪代码权威)。
232
233 缺 id 补生成(uuid4 前 8 位,审计仍可关联);name 非法/缺失 → TLB-802
234 (幻觉名入口拦截,NAME_RE 锚定);arguments 非 JSON/非对象 → TLB-803
235 (解析失败不猜测、不透传执行)。兼容 OpenAI 嵌套(function.name)与
236 llm.response 事件扁平(id/name/arguments)两种 wire 形态。
237 """
238 if not isinstance(raw, dict):
239 raise_code("TLB-803", tool=PARSE_FAIL_NAME,
240 detail=f"tool_calls 元素须为对象,实为 {type(raw).__name__}",
241 advice="按响应结构修正重发;零执行(INV-06)")
242 tid = raw.get("id") or uuid.uuid4().hex[:8]
243 fn = raw.get("function") or {}
244 if not isinstance(fn, dict):
245 fn = {}
246 name = fn.get("name") or raw.get("name")
247 if not isinstance(name, str) or not name or not NAME_RE.fullmatch(name):
248 raise_code("TLB-802", tool=name, rule=NAME_RE.pattern,
249 advice="工具名非法/缺失(幻觉名入口拦截);查 schemas_for 名单")
250 args_raw = fn.get("arguments") if fn.get("arguments") is not None \
251 else raw.get("arguments")
252 try:
253 args = json.loads(args_raw if isinstance(args_raw, str) and args_raw.strip()
254 else args_raw or "{}")
255 except (json.JSONDecodeError, TypeError) as e:
256 raise_code("TLB-803", tool=name,
257 detail=f"arguments 非法 JSON:{e}",
258 advice="按 JSON 格式修正;未执行任何操作")
259 if not isinstance(args, dict):
260 raise_code("TLB-803", tool=name,
261 detail=f"arguments 须为 JSON 对象,实为 {type(args).__name__}",
262 advice="按参数对象格式修正;未执行任何操作")
263 return ToolCall(name=name, raw_args=args, call_id=tid, parent_seq=parent_seq)

284 async def execute_tool_calls(self, calls: list, ctx: Any, *,
285 parent_seq: Optional[int] = None) -> list[ExecResult]:
286 """每轮 LLM 返回的多个 tool_calls 逐条解析执行(spec 伪代码权威)。
287
288 默认串行(保序可审计);解析失败/参数校验失败/guard 拒绝按工具名计连败,
289 ≥2 → 终止本轮后续调用(F026);单条失败各自落 tool.error/guard 事件回喂,
290 不抛到循环(本函数只返回结果列表)。
291 """
292 self.reset_turn_failures()
293 results: list[ExecResult] = []
294 for raw in calls:
295 try:
296 call = self.parse_tool_call(raw, parent_seq)
297 except PyHError as e: # 解析失败:连败计数
298 name = self._attempted_name(raw)
299 self.mark_turn_failure(name)
300 tid = raw.get("id") if isinstance(raw, dict) else None
301 results.append(await self._on_error(
302 ctx, None, e.code, self._parse_fail_message(e, name),
303 name=name, call_id=tid or uuid.uuid4().hex[:8],
304 parent_seq=parent_seq))
305 if self._check_fail_streak(name):
306 break # 连败 2 次终止该轮(F026)
307 continue
308 r = await self.execute(call, ctx) # 全管道纪律在 execute 内
309 results.append(r)
310 key = call.name
311 if not r.ok and r.summary.startswith(("参数校验", "guard")):
312 self.mark_turn_failure(key)
313 if self._check_fail_streak(key):
314 break # TLB-803/GRD 连败 2 → 终止轮
315 return results

335 # ------------------------------------------------------------ 执行管道主函数
336 async def execute(self, call: Any, ctx: Any) -> ExecResult:
337 """四关强制编排(spec 伪代码权威;详见模块 docstring 四关表)。
338
339 任何失败收敛为 tool.error 回喂(ExecResult.ok=False),成功恰一条
340 tool.result;guard/审批拒绝走各自事件不留 result(拒绝零副作用,审计以
341 guard.rejected/approval.* 为证)。
342 """
343 call = self._coerce_call(call) # 鸭子 → guard.ToolCall
344 if call.call_id in self._rejected: # GRD-402:已裁决禁重放
345 raise_code("GRD-402", call_id=call.call_id,
346 advice="该调用已裁决(拒绝),禁止重复执行")
347 self._require_wiring(ctx) # session/scope/guard 必接线
348
349 # ---- 关1a 契约(TLB-802 → tool.error 回喂,零执行)
350 try:
351 defn = self._r.lookup(call.name)
352 except PyHError as e:
353 return await self._on_error(ctx, call, e.code, e.to_model_message())
354 call = replace(call, defn=defn) # 挂契约(guard 读 danger 面)
355 runtime = getattr(ctx, 'platform_runtime', None)
356 validate_path = getattr(runtime, 'validate_file_call', None)
357 if callable(validate_path):
358 error = validate_path(call.name, call.raw_args, ctx)
359 if error:
360 result = await self._on_error(ctx, call, 'TLB-803', json.dumps(error, ensure_ascii=False))
361 if not error['recoverable']:
362 raise PyHError('platform_workspace_path_denied')
363 return result
364 # ---- 关1b 先验后跑(TLB-803 → 参数校验失败 summary;本函数立即返回,
365 # 后续任何一行不执行 —— 畸形参数零调用,INV-06)
366 try:
367 args = self._r.validate_args(call.name, call.raw_args)
368 except PyHError as e:
369 detail = e.ctx.get("detail") or e.ctx.get("advice") or e.message
370 return await self._on_error(
371 ctx, call, e.code,
372 f"参数校验失败(TLB-803):{detail}"
373 if e.code == "TLB-803" else e.to_model_message())
374 call = replace(call, args=args) # guard/审批读校验后实参
375
376 # ---- tool.call 双份存档(raw_args 模型原话 vs args 实际执行;INV-06)
377 call_event = await self._append(ctx, "tool.call",
378 {"name": call.name, "args": dict(args),
379 "raw_args": dict(call.raw_args or {}),
380 "call_id": call.call_id},
381 actor="tool", trace=self._trace(call))
382 # ---- 关2 治理授权(S3-2-2:唯一治理入口):scope 前置 + guard 单调链 +
383 # Decision + decision.issued。scope 前置的运行时**唯一所有者**是
384 # GuardChain._evaluate_full(职责上提,非重复计算);executor 不再自算
385 # scope.can_use,也不再直呼 ctx.guard.evaluate。
386 # S4-P1-1:授权↔执行绑定摘要(既有 `_approval_binding` 语义,**复用不重写**)
387 # 注入为 `Decision.inputs_digest`。治理层不得 import 本模块(ADR-018:308),
388 # 故由执行侧计算后经 `authorize(inputs_digest=...)` 传入——与 `chain_factory`
389 # 注入先例同型(装配/执行侧适配,治理层只持有)。全流程只算一次。
390 digest = _approval_binding(call, args)
391 d = await _authorize_governance(ctx, call, inputs_digest=digest)
392 if d == "reject": # 终局拒(含 scope-hidden)
393 self._rejected.add(call.call_id) # 广义防重放(偏离 11)
394 return ExecResult(ok=False, summary="guard 拒绝,未执行")
395 # ---- 关2.5 审批(F015;danger≥high 由链出 approval 决策)
396 if d == "approval":
397 round_ = await self._approval_round(ctx, call, args, prior=d,
398 binding=digest)
399 if not round_.executed:
400 return ExecResult(ok=False, summary=round_.message) # 非放行 → 不执行
401 # ---- 关3 Provider 执行(线程池 + 超时掐断 TLB-805)
402 timeout = int(defn.timeout_s) or DEFAULT_TOOL_TIMEOUT
403 started = False
404 elapsed_ms = 0.0
405 try:
406 self._r.mark_running(call.name) # registry 注销闸联动
407 t0 = time.perf_counter()
408 from pyharness.core.invocation_binding import InvocationBinding, current_invocation
409 token = current_invocation.set(InvocationBinding(call.call_id,
410 getattr(call_event, 'seq', 0), call.name, digest))
411 try:
412 provider_task = asyncio.create_task(self._run_provider(defn,args,ctx))
413 finally:
414 current_invocation.reset(token)
415 fut = asyncio.create_task(asyncio.wait_for(provider_task, timeout=timeout))
416 self._active = fut
417 started = True # Provider 已进入 await 点
418 try:
419 raw = await fut
420 finally:
421 # Cancellation may arrive before wait_for first runs. The owned
422 # provider Task must still be cancelled and awaited in that window.
423 if not provider_task.done():
424 provider_task.cancel()
425 await asyncio.gather(provider_task,return_exceptions=True)
426 if self._active is fut:
427 self._active = None
428 elapsed_ms = (time.perf_counter() - t0) * 1000.0
429 except asyncio.TimeoutError: # 掐断;进程类杀树归
430 return await self._on_error( # Provider(TO-301),见 docstring
431 ctx, call, "TLB-805", f"执行超时({timeout}s):已停止等待;"
432 "工具可能仍在运行,已发生副作用未撤销")
433 except asyncio.CancelledError: # F025:取消不吞
434 if started: # 已发生副作用如实写 partial
435 await self._append(ctx, "tool.result",
436 {"name": call.name, "call_id": call.call_id,
437 "ok": False, "truncated": True,
438 "summary": "已取消等待:工具可能仍在运行,"
439 "已发生副作用未撤销(partial)"},
440 actor="tool", trace=self._trace(call))
441 raise # 按取消协议 re-raise
442 except PyHError as e: # Provider 结构化错误
443 return await self._on_error(ctx, call, e.code, e.to_model_message())
444 except Exception as e: # noqa: BLE001 Provider 运行异常
445 # 回喂 LLM 的 tool.error 不含异常原文(str(e) 可能带内部路径/实现细节,
446 # P2 收紧):只报类型名判断失败性质;完整堆栈留本地日志,审计以日志为准。
447 log.exception("tool provider failed name=%s call_id=%s",
448 call.name, call.call_id)
449 return await self._on_error(
450 ctx, call, "TLB-805", f"Provider 运行异常:{type(e).__name__}")
451 finally:
452 self._r.mark_idle(call.name)
453 # ---- 关4 结果检查 + finalize(输出 schema/spill/摘要 → tool.result)
454 return await self._finalize(ctx, call, defn, raw, elapsed_ms=elapsed_ms)

457 def _coerce_call(self, call: Any) -> ToolCall:
458 """鸭子 ToolCall(llm.ToolCall/SimpleNamespace)→ guard.ToolCall 规范化。
459
460 call_id 缺省(uuid4 前 8 位)与解析层补 id 同语义——事件配对/审计仍可关联;
461 名字缺失/非 str 不在此拦(registry lookup 会 TLB-802,事件 name 回落 "?")。
462 """
463 if isinstance(call, ToolCall):
464 return call
465 name = getattr(call, "name", None)
466 raw_args = getattr(call, "raw_args", None)
467 cid = getattr(call, "call_id", None) or getattr(call, "id", None)
468 return ToolCall(
469 name=name if isinstance(name, str) else str(name) if name else "",
470 raw_args=dict(raw_args) if isinstance(raw_args, dict) else {},
471 call_id=str(cid) if cid else uuid.uuid4().hex[:8],
472 parent_seq=getattr(call, "parent_seq", None),
473 defn=getattr(call, "defn", None),
474 args=getattr(call, "args", None))

476 def _require_wiring(self, ctx: Any) -> None:
477 """执行前置装配检查(fail-closed):无 session/scope/guard = 非法执行路径。
478
479 缺任一 → CYC-999 上抛(装配期 bug 证据),绝不带缺件执行(INV-04:无
480 guard.evaluated 的执行非法;事件留痕是审计硬前提)。
481 """
482 if getattr(ctx, "session", None) is None:
483 raise_code("CYC-999", module="tools_executor",
484 hint="ctx.session 未接线:事件落点缺失,拒绝执行(INV-04)")
485 if getattr(ctx, "scope", None) is None:
486 raise_code("CYC-999", module="tools_executor",
487 hint="ctx.scope 未接线:scope 前置/guard 求值不可执行")
488 if getattr(ctx, "guard", None) is None:
489 raise_code("CYC-999", module="tools_executor",
490 hint="ctx.guard 未接线:无 guard.evaluated 的执行非法"
491 "(F031 自检兜底)")
492 if getattr(ctx, "governance", None) is None:
493 raise_code("CYC-999", module="tools_executor",
494 hint="ctx.governance 未接线:关2 唯一治理入口缺失,"
495 "拒绝执行(S3-2-2)")

511 async def _on_error(self, ctx: Any, call: Any, code: str, message: str, *,
512 name: Optional[str] = None, call_id: Optional[str] = None,
513 parent_seq: Optional[int] = None) -> ExecResult:
514 """统一错误出口:append tool.error(code/message ≤2000)并回 ExecResult。
515
516 code 原样透传(ERR §5.2);call=None(解析失败尚无 ToolCall)时 name/
517 call_id 取实参或回落占位(偏离 4),保证事件结构合法可审计。失败即
518 单条 tool.error——与 tool.result 互斥,每调用恰一条终局事件。
519 """
520 ename = name if name is not None else (getattr(call, "name", None)
521 if call is not None else None)
522 ecid = call_id if call_id is not None else (getattr(call, "call_id", None)
523 if call is not None else None)
524 if not isinstance(ename, str) or not ename:
525 ename = PARSE_FAIL_NAME
526 if not isinstance(ecid, str) or not ecid:
527 ecid = uuid.uuid4().hex[:8]
528 msg = str(message or code or "工具执行失败")[:2000]
529 seq = parent_seq if parent_seq is not None else (
530 getattr(call, "parent_seq", None) if call is not None else None)
531 trace = {"parent_seq": seq} if seq is not None else None
532 await self._append(ctx, "tool.error",
533 {"name": ename, "call_id": ecid,
534 "code": code or "CYC-999", "message": msg},
535 actor="tool", trace=trace)
536 return ExecResult(ok=False, summary=msg)
