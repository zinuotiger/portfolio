73class TaskQueue:
74 """会话级 FIFO 顺序任务队列(F043/F044 编排地基)。
75
76 私有字段(spec 清单):_q(waiting deque,尾插头弹)、_running(执行中任务,至多一个)、
77 _pause_reasons(Counter,计数>0 即 suspended)、_pump_task(泵单例句柄)、_futures
78 (task_id → 终态 Future,wait_for 填充);实现附加:_done(终态 TaskResult 缓存,幂等
79 等待)、_exec(当前执行子任务,取消目标)、_segments(已开段 task_id 集合,段配对闸)、
80 _cancel_pending(取消请求在子任务创建前到达的传递旗标)、_id_counter(t-N 单调源)。
81 """
82
83 def __init__(self, session: Any, *, runner: Any = None,
84 max_queue: int = DEFAULT_MAX_QUEUE) -> None:
85 """构造队列。
86
87 session = SessionLog 事件源(append 唯一写口);runner = 执行器,须提供 async
88 run_for_task(task)(阶段 4 agent_loop 接口 seam,见偏离 1),可选 cancel_current
89 (task_id);max_queue = waiting 队深上限(默认 32,CFG 锁死;满 → QUE-001)。
90 """
91 self._session = session
92 self._runner = runner
93 self._max_queue: int = int(max_queue)
94 self._q: Deque[Task] = deque()
95 self._running: Optional[Task] = None
96 self._pause_reasons: Counter[str] = Counter()
97 self._pause_lock = asyncio.Lock() # pause/resume 的落盘与唤醒不可交错
98 self._resume_evt = asyncio.Event() # 泵挂起唤醒(非忙等)
99 self._enqueue_ready = asyncio.Event() # 入队事件写完后才允许泵消费
100 self._pump_task: Optional[asyncio.Task] = None # 泵单例句柄(防多泵)
101 self._exec: Optional[asyncio.Task] = None # 执行子任务(取消目标)
102 self._futures: dict[str, asyncio.Future] = {} # 在途等待(task_id → Future)
103 self._done: dict[str, TaskResult] = {} # 终态缓存(重复等待幂等)
104 self._segments: set[str] = set() # 已开段 task_id(配对闸,F044)
105 self._cancel_pending: Optional[str] = None # 取消旗标(见 _run_task)
106 self._terminal_lock = asyncio.Lock()
107 self._terminals: dict[str, str] = {}
108 self._closing: bool = False # 收尾闸(shutdown 幂等位)
109 history = getattr(session, 'events_after', None)
110 self._used_ids = {e.payload['task_id'] for e in history(0)
111 if e.type.startswith('task.') and e.payload.get('task_id')} if callable(history) else set()
112 numbers = [int(ident[2:]) for ident in self._used_ids
113 if ident.startswith('t-') and ident[2:].isdigit() and len(ident) < 22]
114 self._id_counter = itertools.count(max(numbers, default=0) + 1)

116 # ============================================================ 内部辅助
117 def _suspended(self) -> bool:
118 """挂起判定:任一原因计数 > 0 即 suspended(重复挂起合并)。"""
119 return bool(self._pause_reasons)

121 def _next_id(self) -> str:
122 """自动 task_id:t-N 单调(self._seq.next() 语义,itertools 计数实现)。"""
123 while True:
124 ident = f"t-{next(self._id_counter)}"
125 if ident not in self._used_ids:
126 return ident

128 @staticmethod
129 def _ms_since(t0: float) -> int:
130 """耗时毫秒(perf_counter 差,仅计量,不参与事件字段)。"""
131 return int((time.perf_counter() - t0) * 1000)

133 def _settle(self, task_id: str, result: TaskResult) -> None:
134 """终态记账:结果入 _done 缓存 + 填充在途 Future(如有等待者)。
135
136 与终态事件 append 同点调用(每任务至多一个终态);_done 供 wait_for 幂等读取,
137 _futures 记录在途等待(settle 后移除,防泄漏;等待者持本地引用不受影响)。
138 """
139 if task_id in self._done:
140 return
141 self._done[task_id] = result
142 fut = self._futures.pop(task_id, None)
143 if fut is not None and not fut.done():
144 fut.set_result(result)

146 async def _request_cancel(self, task_id: str) -> None:
147 """请求取消执行中任务:先给 runner 协作通道,再兜底取消执行子任务。
148
149 runner.cancel_current(task_id) 存在则先调用(同步/异步兼容,失败记日志不阻断);
150 随后无论 runner 是否已处理,都向 self._exec(包裹 run_for_task 的子任务)注入
151 协作取消——F025 语义:取消信号必达执行协程,由 agent_loop 自身 re-raise 不吞。
152 """
153 runner = self._runner
154 cc = getattr(runner, "cancel_current", None) if runner is not None else None
155 if callable(cc):
156 try:
157 res = cc(task_id)
158 if inspect.isawaitable(res):
159 await res
160 except Exception: # noqa: BLE001 协作钩子失败不阻断兜底取消
161 log.exception("runner.cancel_current failed task=%s", task_id)
162 child = self._exec
163 if child is not None and not child.done():
164 child.cancel() # 兜底传播:不依赖 runner 实现细节

166 # ============================================================ 入队(F043)
167 async def submit(self, intent: str, *, meta: Optional[dict] = None,
168 task_id: Optional[str] = None) -> str:
169 """用户意图尾插入队并返回 task_id(F043)。
170
171 空意图 EVT-100 拒;队深(仅 waiting)≥ max_queue 抛 QUE-001;写 task.enqueued
172 {pos};泵未在跑(或已退出)则启动泵单例——入队后即由队列接管执行。
173 """
174 if not intent or not intent.strip(): # 空意图拒入队(白名单前置)
175 raise_code("EVT-100", hint="空任务意图,submit 拒绝")
176 if self._closing: # R12-3 三阶:收尾后拒新(不留悬挂等待者)
177 # 收尾后泵已退场(挂起态也直接退场)⇒ 入队任务永不执行、其等待者悬挂。
178 # 用队列自己的功能码 QUE-001(拒新);**直接构造**(该码未登记,raise_code
179 # 会改写为 CYC-999,见模块偏离 5),ctx.reason 区分"满"与"已收尾"。
180 raise PyHError("QUE-001", ctx={
181 "reason": "closed", "queue": "closed",
182 "advice": "会话队列已收尾(shutdown 后不再接新任务)"})
183 if len(self._q) >= self._max_queue: # 队深上限(F043 边界:满拒)
184 raise PyHError("QUE-001", ctx={"advice":
185 f"队列已满(≥{self._max_queue}),请等当前任务结束或取消"})
186 t = Task(id=task_id or self._next_id(), # 显式 task_id 供 plan/schedule 溯源
187 intent=intent, meta=meta)
188 if t.id in self._used_ids:
189 raise_code('EVT-100', hint='Task ID has already been used in this session')
190 # Reserve before the first await, including for concurrent submissions.
191 self._used_ids.add(t.id)
192 self._q.append(t) # FIFO:尾插
193 try:
194 env = await self._session.append(
195 "task.enqueued", {"task_id": t.id, "pos": len(self._q)}, actor="system")
196 t.enqueued_seq = env.seq
197 except BaseException as exc:
198 # 拒写/取消不可遗留可执行任务；其容量预留也必须释放。
199 # 泵等待 enqueued_seq，故不会在此 await 窗口提前执行。
200 if t in self._q:
201 self._q.remove(t)
202 code = "cancelled" if isinstance(exc, asyncio.CancelledError) else \
203 getattr(exc, "code", "CYC-999")
204 self._settle(t.id, TaskResult(ok=False, code=code, summary=code))
205 try:
206 await self._terminal_event("task.failed", {
207 "task_id": t.id, "reason": "cancelled" if code == "cancelled"
208 else "error", "error": code})
209 except Exception:
210 log.warning("入队失败终态无法落盘 task=%s", t.id, exc_info=True)
211 raise
212 finally:
213 self._enqueue_ready.set()
214 # 泵单例(防重入):running 非空或泵仍在跑(含 done 的旧句柄)不重启
215 if self._running is None and (self._pump_task is None
216 or self._pump_task.done()):
217 self._pump_task = asyncio.create_task(self._pump())
218 return t.id

220 # ============================================================ 泵(F043)
221 async def _pump(self) -> None:
222 """单飞泵:唯一执行放行口(F043 核心)。
223
224 暂停期不弹任务(等裁决不饿死,resume 置位唤醒);队空即退场由下次 submit 唤醒;
225 否则队首 popleft 置 running 并执行,finally 清位使队首自动接续。任务级取消
226 (CancelledError)已在 _run_task 归一化事件化,泵吸收后继续;其余未预期异常
227 记堆栈仅本地(保泵活,队列继续);泵自身被外部取消则让路(不在此吞)。
228 """
229 while True:
230 if self._suspended(): # 暂停期:只停消费,不丢队不超时
231 if self._closing: # 收尾中:挂起无意义,直接退场
232 # R12-3 二阶:审批真挂起后,"挂起态收尾"成为常态;若在此继续等
233 # `_resume_evt`,shutdown 会走 5s 超时兜底(会话收尾平白多等 5 秒)。
234 # **不清** `_pause_reasons`:让审批侧(close 后 detach → cancel_all)
235 # 照常补一条 `queue.resumed`,事件对保持配对。
236 self._pump_task = None
237 return
238 await self._resume_evt.wait() # resume 时置位唤醒(非忙等)
239 self._resume_evt.clear()
240 continue
241 if not self._q: # 队空:泵退场;下次 submit 重启泵
242 self._pump_task = None
243 return
244 if not self._q[0].enqueued_seq:
245 await self._enqueue_ready.wait()
246 self._enqueue_ready.clear()
247 continue
248 t = self._q.popleft()
249 self._running = t # 队首接跑
250 try:
251 await self._run_task(t) # 执行(内部含段锚与终态事件)
252 except asyncio.CancelledError:
253 # 泵自身被外部取消:不伪造任务终态,让路(任务级取消在 _run_task 内已吸收)
254 raise
255 except Exception: # 泵自身兜底:单任务失败已事件化
256 log.exception("task pump crash task=%s", t.id) # CYC-999 不吞但保泵活
257 finally:
258 self._running = None # 释放 → while 自动接队首(F043)

260 # ============================================================ 单任务执行
261 async def _run_task(self, t: Task) -> None:
262 """单任务生命周期执行器(F043/F044)。
263
264 started → segment.start(强同步,经 open_segment) → runner.run_for_task
265 (执行子任务,可独立取消)→ 终态事件(completed/failed,每任务至多一个)→
266 segment.end 配对关闭。失败分级:PyHError 原码 / 未预期 CYC-999 / 取消
267 cancelled(归一化后泵吸收,接队首);终态落事件即 _settle(唤醒 wait_for)。
268 """
269 t0 = time.perf_counter()
270 try:
271 await self._session.append("task.started", {"task_id": t.id},
272 actor="system")
273 # 强同步段锚:先落盘再执行,崩溃后回放段不悬空(F044)
274 start_seq = await self.open_segment(t.id)
275 except PyHError as e: # 会话已终态/日志拒写:任务不得悬挂
276 # 2026-09-21:此前 started/段锚在 try **之外**,会话若已终态(EVT-104)
277 # 会直接抛出 → 泵记 crash、`_settle` 被跳过 → **等待者永久悬挂**。
278 log.warning("任务无法开始(会话终态/日志拒写)task=%s code=%s",
279 t.id, e.code)
280 self._settle(t.id, TaskResult(ok=False, code=e.code, summary=e.code,
281 duration_ms=0.0))
282 return # 未开段:无需关段
283 # 执行子任务:协作取消注入点(F025,取消不伤泵);runner 内部事件带 task_id 由
284 # agent_loop 自绑(本模块只引用 segment 围栏,不越权写执行期事件)
285 child = asyncio.create_task(self._run_runner(t))
286 self._exec = child
287 try:
288 # 取消请求在子任务创建前已到达(如 started/segment.start 落盘窗口)→ 补取消
289 if self._cancel_pending == t.id:
290 self._cancel_pending = None
291 child.cancel()
292 result = await child
293 reason = getattr(result, "reason", None)
294 explicit_ok = reason == "complete" or (isinstance(result, TaskResult) and result.ok)
295 if not explicit_ok:
296 code = getattr(result, "error_code", None) or getattr(result, "code", None) or {
297 "cancelled": "cancelled", "budget": "LLM-305", "max_turns": "CYC-999",
298 "stall": "CYC-999"}.get(reason, "CYC-999")
299 await self._terminal_event("task.failed", {"task_id": t.id,
300 "reason": "cancelled" if reason == "cancelled" else "error", "error": code})
301 self._settle(t.id, TaskResult(ok=False, code=code, summary=code,
302 duration_ms=self._ms_since(t0)))
303 return
304 await self._terminal_event("task.completed",
305 {"task_id": t.id, "reason": "ok"})
306 self._settle(t.id, TaskResult(ok=True, duration_ms=self._ms_since(t0)))
307 except asyncio.CancelledError:
308 if child.cancelled(): # 取消源 = 执行子任务(API cancel)
309 # F025:归一化 failed(cancelled);事件 + settle 双通道表达"取消不吞"
310 await self._terminal_event("task.failed", {"task_id": t.id,
311 "reason": "cancelled", "error": "cancelled"})
312 self._settle(t.id, TaskResult(ok=False, code="cancelled",
313 summary="cancelled", duration_ms=self._ms_since(t0)))
314 # 不向泵 re-raise:取消已事件化,泵自动接队首(GWT-F043-02,见偏离 2)
315 else:
316 raise # 泵自身被外部取消:让路
317 except PyHError as e: # 业务失败:事件化并继续队列(F043)
318 await self._terminal_event("task.failed", {"task_id": t.id,
319 "reason": "error", "error": e.code})
320 self._settle(t.id, TaskResult(ok=False, code=e.code, summary=e.code,
321 duration_ms=self._ms_since(t0)))
322 except BudgetExhausted: # F032 预算硬闸:保预算终态语义
323 # 常态下 agent_loop 自行收敛 BudgetExhausted(reason=budget)不会到此处;
324 # 本分支是防御路径(预算在循环 try 之外被抛/执行器直抛)。无此分支则
325 # 落入下面的兜底 → error:CYC-999,丢掉 budget(ERR.md 登记 LLM-305)。
326 await self._terminal_event("task.failed", {"task_id": t.id,
327 "reason": "budget", "error": "LLM-305"})
328 self._settle(t.id, TaskResult(ok=False, code="LLM-305",
329 summary="budget", duration_ms=self._ms_since(t0)))
330 except Exception as e: # 未预期:归 CYC-999,堆栈仅本地
331 log.exception("task failed unexpected task=%s", t.id)
332 await self._terminal_event("task.failed", {"task_id": t.id,
333 "reason": "error", "error": "CYC-999"})
334 self._settle(t.id, TaskResult(ok=False, code="CYC-999",
335 summary=f"{type(e).__name__}: {e}"[:200], duration_ms=self._ms_since(t0)))
336 finally:
337 self._exec = None
338 await self._close_segment_guarded(t.id, start_seq) # 配对关闭(F058)

340 async def _terminal_event(self, type_: str, payload: dict) -> None:
341 async with self._terminal_lock:
342 ident = payload['task_id']
343 if ident in self._terminals:
344 return
345 await self._write_terminal_event(type_, payload)
346 self._terminals[ident] = type_

348 async def _write_terminal_event(self, type_: str, payload: dict) -> None:
349 """终态事件写入(**会话已终态时不吞结算**)。
350
351 会话可在任务在途时收尾(`session.finished` 落盘 + 终态位置位)⇒ 本任务的
352 终态事件被 EVT-104 拒写。拒写只应损失"事件留痕",**不得**让 `_settle` 被
353 跳过(等待者永久悬挂)。失败终态也允许在磁盘故障时结算失败，但成功终态
354 的 PERS-202 必须上抛，转为失败，禁止把落盘故障报告为成功。
355 """
356 try:
357 await self._session.append(type_, payload, actor="system")
358 except PyHError as e:
359 if type_ == "task.failed" and e.code == "PERS-202":
360 log.warning("失败终态无法落盘 task=%s code=%s;仍向等待者报告失败",
361 payload.get("task_id"), e.code)
362 return
363 if e.code == "EVT-104":
364 log.warning("终态事件被拒(会话已终态)type=%s task=%s;"
365 "结算继续(以 _done 缓存为准)",
366 type_, payload.get("task_id"))
367 return
368 raise

370 async def _close_segment_guarded(self, task_id: str, start_seq: int) -> None:
371 """终态会话/磁盘故障关段失败时告警并释放内存段标记，不伪造落盘成功。
372
373 此前 `close_segment` 直调 append ⇒ 会话已终态时在 finally 中抛出,泵记
374 "task pump crash" 且段锚永久不闭合(证据生产者不触发 ⇒ 该任务零证据)。
375 """
376 try:
377 await self.close_segment(task_id, start_seq)
378 except PyHError as e:
379 if e.code in {"EVT-104", "PERS-202"}:
380 self._segments.discard(task_id)
381 log.warning("段锚未落盘闭合 task=%s code=%s", task_id, e.code)
382 return
383 raise

385 async def _run_runner(self, t: Task) -> Any:
386 """执行器调用(子任务体):runner 须可调用 run_for_task(task)。
387
388 未装配 runner 时按 CYC-999 快速失败(防 S-1 伪造已执行);接受对象方法或
389 裸可调用(同步/异步兼容,INV-08:执行器经构造注入,本模块不 import 外围)。
390 """
391 runner = self._runner
392 if runner is None:
393 raise_code("CYC-999", hint="队列未装配执行器(run_for_task),任务拒绝执行")
394 fn = getattr(runner, "run_for_task", runner)
395 if not callable(fn):
396 raise_code("CYC-999", hint=f"执行器不可调用: {type(runner).__name__}")
397 res = fn(t)
398 if inspect.isawaitable(res):
399 return await res
400 return res

407 async def _pause_owned(self, reason: str, *, by: str) -> None:
408 """队列挂起(F015 联动):审批等待/预算暂停时停消费——只停弹任务,运行中任务
409 不受影响,等待不超时饿死。
410
411 重复挂起合并:已挂期间任何新 pause(同/异原因)仅计数累加,不发事件;仅挂起
412 状态"从无到有"(全部计数 0→1)才写 queue.suspended——先 suspended 后 resumed
413 事件序,resume 全解除才写 resumed(EVENT-SCHEMA §3.5.3,见偏离 3)。
414 """
415 if not reason or not reason.strip(): # 空原因拒(防计数污染挂起态)
416 raise_code("EVT-100", hint="pause 原因不能为空(approval-pending/budget 等)")
417 if self._suspended(): # 已挂:仅计数合并,event 不重复(EVT 词表)
418 self._pause_reasons[reason] += 1
419 return
420 self._pause_reasons[reason] += 1 # 首挂起(0→1):发事件 + 泵阻塞
421 self._resume_evt.clear() # 在落盘 await 前关闭消费闸
422 await self._session.append("queue.suspended",
423 {"reason": reason, "by": by}, actor="system")
424
425 async def resume(self, reason: str, *, by: str = "system") -> None:
426 async with self._pause_lock:
427 await self._resume_owned(reason, by=by)
428
429 async def _resume_owned(self, reason: str, *, by: str) -> None:
430 """解除一个挂起原因:计数减一;仍有其他原因保持挂起;全部解除才写
431 queue.resumed 并唤醒泵。未挂起的 reason 幂等无害(无事件)。
432
433 by 参数保留签名对称;queue.resumed payload 模型无 by 字段(偏离 4),事件只落 reason。
434 """
435 if self._pause_reasons.get(reason, 0) <= 0: # 未挂起的 reason:幂等,无事件
436 return
437 if self._pause_reasons[reason] > 1: # 该原因多层挂起:仅减计数
438 self._pause_reasons[reason] -= 1
439 return
440 if len(self._pause_reasons) > 1: # 仍有其他原因 → 维持挂起
441 del self._pause_reasons[reason]
442 return
443 await self._session.append("queue.resumed",
444 {"reason": reason}, actor="system") # 全解除才写 resumed(偏离 4)
445 del self._pause_reasons[reason] # 落盘失败时保留暂停及重试能力
446 self._resume_evt.set() # 唤醒泵(先 resumed 后放行)

448 # ============================================================ 取消(F025)
449 async def shutdown(self, *, reason: str = "close") -> None:
450 """会话收尾**前置**排空(幂等):取消在途任务、等泵退场、兜底结算。
451
452 为什么必须在 `session.finished` **之前**:任务终态事件与 `segment.end` 都是
453 append,会话终态位一旦置位即 EVT-104 拒写 ⇒ 在途任务**丢终态、丢段锚、
454 等待者悬挂**(2026-09-21 实测复现:`task.completed=0/segment.end=0/wait_for
455 3s 未返回`)。取消路径本身已归一化为 `task.failed{cancelled}` + `finally`
456 关段,故"先取消并等泵退场"即得完整终局。
457
458 超时/异常路径不放弃:仍在途等待者统一按 `cancelled` 结算(绝不悬挂)。
459 """
460 task = getattr(self, '_shutdown_task', None)
461 if task is None:
462 self._closing = True
463 task = self._shutdown_task = asyncio.create_task(self._shutdown_owned(reason))
464 await asyncio.shield(task)
465
466 async def _shutdown_owned(self, reason):
467 # ① 队内等待任务:直接摘除并记 failed(cancelled)(不执行 —— 收尾不是"继续跑")
468 for t in list(self._q):
469 try:
470 await self.cancel(t.id, by=reason)
471 except PyHError as e:
472 log.warning("queue shutdown:摘除等待任务失败 code=%s task=%s",
473 e.code, t.id)
474 # ② 在途任务:取消 → 归一化 failed(cancelled) + finally 关段
475 running = self._running
476 if running is not None:
477 try:
478 await self.cancel(running.id, by=reason)
479 except PyHError as e: # 审计失败不阻断排空
480 log.warning("queue shutdown:取消在途任务失败 code=%s", e.code)
481 # ②' **先唤醒泵**(R12-3 二阶):队列若处于挂起态(如等审批),泵阻塞在
482 # `_resume_evt.wait()` ⇒ 直接 await 会走 5s 超时兜底才退场(会话收尾平白
483 # 多等 5 秒)。置位唤醒后泵醒来即见空队退场。
484 self._resume_evt.set()
485 self._enqueue_ready.set() # 也唤醒等待入队提交完成的泵
486 pump = self._pump_task
487 if pump is not None and not pump.done():
488 try:
489 await asyncio.wait_for(asyncio.shield(pump), timeout=5.0)
490 except asyncio.TimeoutError:
491 if not pump.cancelling(): pump.cancel()
492 await asyncio.gather(pump, return_exceptions=True)
493 except asyncio.CancelledError:
494 raise
495 except Exception:
496 log.warning("queue shutdown: pump failed", exc_info=True)
497 # 兜底:任何仍未结算的等待者不得悬挂(泵被外部取消/超时/崩溃路径)
498 for tid in list(self._futures):
499 self._settle(tid, TaskResult(ok=False, code="cancelled",
500 summary="session-close"))
501
502 async def cancel(self, task_id: str, *, by: str = "system") -> bool:
503 """任务取消(F025):running → system.cancelled 审计 + 传播取消,队首自动接;
504 waiting → 直接摘除并记 failed(cancelled);不存在/已终态 → 幂等 False。
505 """
506 if self._running is not None and self._running.id == task_id:
507 if self._cancel_pending == task_id:
508 return False
509 self._cancel_pending = task_id
510 await self._request_cancel(task_id)
511 # Cancellation must survive any audit failure; the error remains visible.
512 await self._session.append("system.cancelled",
513 {"what": f"task:{task_id}", "reason": by}, actor="system")
514 return True
515 for i, w in enumerate(self._q): # waiting:摘除,无需通知循环
516 if w.id == task_id:
517 del self._q[i]
518 try:
519 await self._session.append("task.failed", {"task_id": task_id,
520 "reason": "cancelled"}, actor="system")
521 finally:
522 self._settle(task_id, TaskResult(ok=False, code="cancelled",
523 summary="cancelled"))
524 return True
525 return False # 已终态/不存在:幂等 False(防重放)

538 # ============================================================ 终态等待
539 async def wait_for(self, task_id: str, *,
540 timeout: Optional[float] = None) -> TaskResult:
541 """外部等待任务终态(F046/jobs 复用):终态缓存/Future 填充后返回 TaskResult;
542 timeout=None 永久等(审批挂起不饿死),超时抛 BUSY 显式拒——等待不影响任务本身
543 (shield)。未知 task_id 显式 EVT-100(防永久悬挂,偏离 6)。
544 """
545 done = self._done.get(task_id) # 已终态:直接返回(幂等)
546 if done is not None:
547 return done
548 known = (self._running is not None and self._running.id == task_id) \
549 or any(w.id == task_id for w in self._q) or task_id in self._futures
550 if not known: # 从未入队/已结算但缓存被清理
551 raise_code("EVT-100", task_id=task_id,
552 hint="wait_for 目标任务不存在(未入队或非本队列任务)")
553 fut = self._futures.get(task_id)
554 if fut is None: # 每 task_id 一个终态 Future
555 fut = asyncio.get_running_loop().create_future()
556 self._futures[task_id] = fut
557 try:
558 if timeout is None:
559 await asyncio.shield(fut) # 等待可被外部取消,任务不受牵连
560 else:
561 await asyncio.wait_for(asyncio.shield(fut), timeout)
562 except asyncio.TimeoutError:
563 raise PyHError("BUSY", ctx={"hint": f"等待任务 {task_id} 超时",
564 "advice": "查任务是否仍 running/挂起"})
565 return fut.result() # TaskResult(ok/code/summary)

568 async def open_segment(self, task_id: str) -> int:
569 """开段:写 segment.start(强同步)返回其 seq(F044 验收直译)。
570
571 段锚先落盘再执行(崩溃后回放段不悬空);同一 task_id 嵌套开段 → EVT-100

573 """
574 if task_id in self._segments:
575 raise_code("EVT-100", task_id=task_id,
576 hint="段已打开:segment.start/end 必须一一配对,禁嵌套开段")
577 env = await self._session.append("segment.start",
578 {"task_id": task_id}, actor="system", sync=True) # 强同步:段锚先落
579 self._segments.add(task_id)
580 return env.seq
