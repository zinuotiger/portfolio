53 def owns_initializing(self, approval_id: int, *, sid: Optional[str] = None) -> bool:
54 """Route a published requested event while registration still holds the lock.

56 The async decision revalidates pending under that same lock. This is only

106 def _spawn(self, coro: Any) -> bool:
107 """异步协程排程(fire-and-forget):失败只记日志;任务登记防 GC。"""
108 if not inspect.isawaitable(coro):
109 coro = coro()
110 try:
111 loop = asyncio.get_running_loop()
112 except RuntimeError:
113 logger.warning("approval 无运行中事件循环,协程未投递: %s",
114 getattr(coro, "__name__", coro))
115 return False
116 task = loop.create_task(coro)
117 self._tasks.add(task)
118 task.add_done_callback(self._tasks.discard)
119 task.add_done_callback(self._task_error)
120 return True
