156def _inventory_tree(root,view,facts,runtime):
157 """Native Git sees owned placeholders and validated bounded rule copies only.
158
159 This is a disposable metadata view, never a captured source/context. Ordinary
160 candidate bytes are not opened. The caller owns and removes the outer view.
161 """
162 mirror=view/'candidate-tree';mirror.mkdir()
163 entries=sorted(((key[6:],kind) for key,kind in facts.items() if key.startswith('entry:')),

170 raw=_read_ignore_rule(root,name,rule,runtime) if rule is not None else b''
