59 @admitted
60 async def schedule_action(self, sid: str, action: str, *,
61 name: str = "", kind: str = "cron",
62 expr: str = "", intent: str = "",
63 is_risky: Optional[bool] = None) -> dict:
64 self._ensure_open()
65 action = str(action or "").strip().lower()
66 name = str(name or "").strip()
67 log_ = await self.require_session(sid)
68 _, spine = await self.public_spine_for(sid)
69 ctx = SimpleNamespace(session=getattr(spine, "session", None),
70 task_queue=getattr(spine, "task_queue", None))
71 if action == "add":
72 expr = str(expr or "").strip()
73 intent = str(intent or "").strip()
74 if not name or not expr or not intent:
75 raise_code("EVT-100", advice="schedule add 需要 name/expr/intent")
76 await self.queue_for(sid, log_)
77 await spine.schedule.register(
78 name, str(kind or "cron").strip().lower(), expr,
79 {"intent": intent}, is_risky=is_risky, ctx=ctx)
80 elif action == 'edit':
81 await spine.schedule.edit(name,kind,expr,intent,ctx=ctx)
82 elif action == 'run_now':
83 await spine.schedule.run_now(name,ctx=ctx)
84 elif action in {"pause", "resume", "remove"}:
85 if not name:
86 raise_code("EVT-100", field="name", advice=f"schedule {action} 需要 name")
87 await getattr(spine.schedule, action)(name, ctx=ctx)
88 else:
89 raise_code("EVT-100", field="action",
90 advice="action 须为 add/pause/resume/remove")
91 return {"ok": True, "action": action, "name": name}

112 row.update(status='completed' if event.type=='subagent.joined' else 'failed',
113 finished_at=event.ts,summary=summary(event.payload.get('summary','')),error_code=None)
114 return {"status": status, 'history': history}

170 raise_code('GRD-403', reason='team_memory_read_only')
171 if op!='keys' and (not isinstance(key,str) or not re.fullmatch(r'[A-Za-z0-9_.-]{1,64}',key)):
172 raise_code('EVT-100',field='memory.key')

174 raise_code('EVT-100',field='memory.value')
175 if not hasattr(self,'_team_memory'):
