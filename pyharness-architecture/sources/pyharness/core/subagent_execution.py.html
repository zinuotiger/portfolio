15 async def spawn(self, spec: SubagentSpec, ctx: Any = None) -> str:
16 """F049 核心入口:四道约束闸(深度/工具子集在册/并发/预算)全过 → 建独立
17 子会话 → 父会话落 subagent.spawned → 登记 ChildHandle 并起 _run_child;
18 返回 sub_id(join 的合法凭据,结果只可 join 一次)。
19
20 异常:BUSY(深度 >3/层级不递增/并发满)、TLB-802(工具子集含未注册名)、
21 EVT-100(spec 非法)、EVT-104(父会话已 finished——session.append 校验链拒)。
22 任一步失败:并发闸已占则释放,零残留(无脏登记/无孤儿会话引导)。
23 """
24 self._ensure_session()
25 self._validate_spec(spec)
26 if spec.depth > self._max_depth: # ① 递归闸上界 ≤3(PRD F049)
27 raise PyHError("BUSY", ctx={
28 "hint": f"子任务深度 {spec.depth} 超上限 {self._max_depth}",
29 "advice": "子任务不得再嵌套超过 3 层;考虑平铺任务或合入父意图"})
30 if spec.depth < self._depth + 1: # 递归闸递增链(偏离 5)
31 raise PyHError("BUSY", ctx={
32 "hint": f"子任务深度 {spec.depth} 未达本层 {self._depth} + 1",
33 "advice": "深度必须逐层 +1(平级/降级派发会绕过递归闸)"})
34 # ② 工具子集必须已注册(幻觉名拦截;tools 未接线 → 日志降级跳过)
35 tools = self._tools
36 if ctx is not None:
37 tools = tools or getattr(ctx, "tools", None)
38 if tools is not None:
39 has = getattr(tools, "has", None)
40 for t in spec.tools_subset or []:
41 if has is not None and not has(t):
42 raise_code("TLB-802", tool=t,
43 advice="tools_subset 含未注册工具;核对注册表,删幻觉名")
44 # ④ 预算 = 父 × spec.budget_ratio(计入父任务:charge_hook 注入位,偏离 3)
45 parent_scope = self._parent_scope
46 if ctx is not None:
47 parent_scope = getattr(ctx, "scope", None) or parent_scope
48 limits = None
49 if parent_scope is not None:
50 limits = _scale_limits(getattr(parent_scope, "limits", None),
51 float(spec.budget_ratio))
52 # ③ 并发闸(满 → BUSY;随后任何失败路径都释放,零残留)
53 await self._acquire_slot()
54 sub_id = None
55 try:
56 sub_id = self._child_id_factory()
57 child = self._session_factory(sub_id)
58 child_scope = None
59 factory = self._child_scope_factory
60 if callable(factory):
61 child_scope = factory(parent_scope, sub_id, child, limits,
62 spec.deny_extra)
63 if parent_scope is not None and child_scope is None:
64 raise PyHError('GRD-403', ctx={'reason': 'child_scope_unavailable'})
65 if child_scope is not None and spec.tools_subset is not None:
66 permitted = set(spec.tools_subset)
67 inherited = getattr(child_scope.policy, 'allowed_tools', None)
68 child_scope.policy.allowed_tools = permitted if inherited is None else permitted & inherited
69 anchor = self._parent_anchor(ctx)
70 # 子会话引导(独立 seq=1;真实 JSONL 由装配注 persistence)
71 await self._bootstrap_child(child, sub_id)
72 # 父会话只记委派事实(普通落盘;actor=agent,EVENT-SCHEMA §3.5.3)
73 env = await self._session.append(
74 "subagent.spawned",
75 {"sub_id": sub_id, "parent_seq": anchor, "task": spec.task},
76 actor="agent", origin=SUBA_OWNER)
77 h = ChildHandle(sub_id=sub_id, parent_sid=self._session.sid,
78 parent_seq=env.seq, spec=spec, session=child,
79 state="spawning",
80 result=asyncio.get_running_loop().create_future(),
81 started_ts=time.time(),
82 budget_limits=limits, scope=child_scope,
83 slot=True)
84 h.env = ChildContext(sub_id=sub_id, session=child, spec=spec,
85 scope=child_scope, budget_limits=limits,
86 parent_sid=self._session.sid)
87 self._children[sub_id] = h
88 h.task = asyncio.create_task(self._run_child(h))
89 h.state = "running"
90 log.info("subagent spawned sub=%s parent=%s depth=%d task=%.60s",
91 sub_id, self._session.sid, spec.depth, spec.task)
92 return sub_id
93 except BaseException as primary:
94 self._release_slot()
95 if sub_id is not None:
96 self._children.pop(sub_id, None)
97 cleanup = getattr(self._runner, 'close_child', None)
98 if callable(cleanup):
99 try:
100 result = cleanup(sub_id)
101 if inspect.isawaitable(result): await result
102 except BaseException as secondary:
103 primary.add_note(f'child rollback failed: {secondary!r}')
104 raise

106 async def _invoke_child(self, h: ChildHandle) -> Any:
107 """子执行器调用 seam(偏离 7):runner.run_child(env) 异步/同步兼容。
108
109 未装配 runner → CYC-999 快速失败(防 S-1"伪造已执行":宁失败不静默假完成,
110 task_queue._run_runner 同款)。F007 唯一 LLM 入口纪律(INV-02)由 runner
111 内部保证——子任务任何 LLM/工具动作都只能发生在 runner 提供的子循环内。
112 """
113 runner = self._runner
114 if runner is None:
115 raise_code("CYC-999", hint="SubagentManager 未装配子会话执行器",
116 advice="装配层注入 agent_loop 适配 runner(INV-02 纪律)")
117 fn = getattr(runner, "run_child", runner)
118 if not callable(fn):
119 raise_code("CYC-999", hint=f"子执行器不可调用: {type(runner).__name__}")
120 res = fn(h.env)
121 if inspect.isawaitable(res):
122 res = await res
123 if self._charge_hook is not None: # 子用量计入父任务(偏离 3)
124 try:
125 r = self._charge_hook(h.env)
126 if inspect.isawaitable(r):
127 await r
128 except Exception as exc: # noqa: BLE001 记账失败不阻断回收
129 log.warning("charge_hook 失败 sub=%s: %s", h.sub_id, exc)
130 return res

132 async def _run_child(self, h: ChildHandle) -> None:
133 """子会话执行与回收(F049 核心):委派意图交子执行器(独立 session 上跑,
134 不占用父会话 running)→ 终态摘要(≤2KB)→ subagent.joined;PyHError/取消 →
135 subagent.failed / cancelled;finally 释放并发闸 + 摘登记(已回收即摘)。
136
137 取消分支不吞:re-raise CancelledError 沿 await 链传播(F025);终态审计事件
138 由 cancel()/_cancel_all_children() 统一落(见规格 _run_child 注释)。
139 """
140 try:
141 out = await self._invoke_child(h) # F007 子循环(seam)
142 summary = summarize(out, self._summary_max_chars) # ≤2KB(数据身份)
143 if h.spec.notify: # 回收写父会话(普通落盘)
144 await self._write_event(
145 "subagent.joined",
146 {"sub_id": h.sub_id, "summary": summary},
147 actor="agent", trace={"parent_seq": h.parent_seq})
148 h.summary, h.state = summary, "done"
149 if h.result is not None and not h.result.done():
150 h.result.set_result(summary)
151 except asyncio.CancelledError: # F025 取消传播:不吞
152 h.state = "cancelled"
153 if h.result is not None and not h.result.done():
154 h.result.cancel() # 唤醒 join 等待者(取消态)
155 raise
156 except PyHError as e: # 结构化失败:原码透传
157 h.error_code, h.state = e.code, "failed"
158 if h.spec.notify:
159 await self._write_event(
160 "subagent.failed",
161 {"sub_id": h.sub_id,
162 "summary": summarize(
163 f"[{e.code}] {e.spec.advice}", self._summary_max_chars)},
164 actor="agent", trace={"parent_seq": h.parent_seq})
165 if h.result is not None and not h.result.done():
166 h.result.set_exception(e) # join 上抛原 PyHError
167 except Exception as e: # 未预期兜底:CYC-999
168 err = wrap_unexpected(e, "subagent")
169 h.error_code, h.state = err.code, "failed"
170 if h.spec.notify:
171 await self._write_event(
172 "subagent.failed",
173 {"sub_id": h.sub_id,
174 "summary": summarize(
175 f"[{err.code}] {err.spec.advice}", self._summary_max_chars)},
176 actor="agent", trace={"parent_seq": h.parent_seq})
177 if h.result is not None and not h.result.done():
178 h.result.set_exception(err)
179 finally:
180 self._finalize(h) # 释放闸 + 摘登记(幂等)

182 def _finalize(self, h: ChildHandle) -> None:
183 """句柄收尾(幂等):并发闸归还(防双放:slot 标记) + 从在途表摘除。"""
184 if h.slot:
185 h.slot = False
186 self._release_slot()
187 if self._children.get(h.sub_id) is h:
188 self._children.pop(h.sub_id, None)

197 def _historic_result(self, sub_id):
198 """Only owned child IDs may be recovered from this parent's event log."""
199 events = list(self._ensure_session().events_after(0))
200 if not any(e.type == 'subagent.spawned' and e.payload.get('sub_id') == sub_id for e in events):
201 return None
202 for event in reversed(events):
203 p = event.payload
204 if p.get('sub_id') == sub_id:
205 if event.type == 'subagent.joined':
206 return ('done', p['summary'], None)
207 if event.type == 'subagent.failed':
208 import re
209 match = re.match(r'\[([A-Z]+-\d+)\]', p.get('summary', ''))
210 return ('failed', None, match[1] if match else 'CYC-999')
211 if event.type == 'system.cancelled' and p.get('what') == 'subagent:'+sub_id:
212 return ('cancelled', None, None)
213 reader = getattr(self._runner, 'completed_result', None)
214 return reader(sub_id) if callable(reader) else None

216 async def join(self, sub_id: str, *, timeout: Optional[float] = None,
217 ctx: Any = None) -> Optional[str]:
218 """主循环 await 子任务终态取回摘要:成功 → summary(≤2KB);失败 → raise 原
219 PyHError(回喂可行动文本);timeout → None(调用方决定再等/取消/先干别的);
220 Completed results support repeated join; evicted entries use owned history.
221 """
222 h = self._children.get(sub_id)
223 if h is None:
224 result = self._completed.get(sub_id) or self._historic_result(sub_id)
225 if result is not None:
226 state, summary, code = result
227 if state == 'cancelled':
228 raise asyncio.CancelledError
229 if state == 'failed':
230 raise PyHError(code or 'CYC-999')
231 return summary
232 raise_code("EVT-101", sub_id=sub_id,
233 hint="sub_id 不存在或没有可恢复的终态",
234 advice="使用本会话 spawn 返回的子任务 ID")
235 if h.state in ("spawning", "running"):
236 h.state = "joining" # 等待期状态(ChildState 词表)
237 try:
238 # shield:超时只取消本等待,不毒化 h.result 内部 Future——否则超时后
239 # 再次 join 读到 cancelled future 直接抛 CancelledError、结果永久丢失
240 # (task_queue.wait_for 同款处理,P2 修复)。
241 return await asyncio.wait_for(asyncio.shield(h.result), timeout)
242 except asyncio.TimeoutError:
243 return None # 调用方:再等/取消/先干别的

245 async def cancel(self, sub_id: str, *, by: str = "system",
246 reason: str = "user-cancel") -> bool:
247 """取消在途子任务并归一化:system.cancelled 审计(父会话)→ 任务 cancel 传播
248 → await 其清理(终态由子循环自记 state=cancelled)→ 取消注入早于任务启动
249 (停 spawning)则补漏终态;已终态/不存在 → 幂等 False。
250 """
251 h = self._children.get(sub_id)
252 if h is None or h.state in _TERMINAL:
253 return False # 已终态/不存在 → 幂等 False
254 if h.task is not None and not h.task.done():
255 h.task.cancel() # 沿 await 链传播 CancelledError
256 await self._write_event( # 声明式取消审计(父会话)
257 "system.cancelled",
258 {"what": f"subagent:{sub_id}", "reason": reason},
259 actor="system")
260 if h.task is not None:
261 await asyncio.gather(h.task, return_exceptions=True) # 等清理完成
262 if self._children.get(sub_id) is h and h.state not in _TERMINAL:
263 h.state = "cancelled" # 取消早于任务启动:补漏终态
264 self._finalize(h) # 释放闸 + 摘登记(防泄漏)
265 return True

277 async def _cancel_all_children(self, *, reason: str) -> None:
278 """child-first 清理主体:先逐子在途任务声明式取消(审计 + cancel),await
279 全部退出,再补漏(取消注入早于任务启动的 spawning 句柄落 subagent.failed
280 终态事件 + 释放闸);摘净后才交还调用方(结构上杜绝"父已终态、子还在跑/
281 回写 EVT-104")。
282 """
283 await stop_admissions(self)
284 hs = list(self._children.values())
285 if not hs:
286 return
287 errors = []
288 for h in hs:
289 if h.state not in _TERMINAL and h.task is not None and not h.task.done():
290 h.task.cancel()
291 for h in hs:
292 if h.state in _TERMINAL:
293 continue
294 try:
295 await self._write_event('system.cancelled',
296 {'what': f'subagent:{h.sub_id}', 'reason': reason}, actor='system')
297 except BaseException as exc:
298 errors.append(exc)
299 await asyncio.gather(*(h.task for h in hs if h.task is not None), return_exceptions=True)
300 for h in hs:
301 try:
302 if h.state not in _TERMINAL:
303 await self._write_event('subagent.failed',
304 {'sub_id':h.sub_id, 'summary':f'parent closed ({reason})'},
305 actor='agent', trace={'parent_seq':h.parent_seq})
306 h.state = 'cancelled'
307 except BaseException as exc:
308 errors.append(exc)
309 finally:
310 self._finalize(h)
311 if errors:
312 raise BaseExceptionGroup('child cancellation audit failed', errors)
