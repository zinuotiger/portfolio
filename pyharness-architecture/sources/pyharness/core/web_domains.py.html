8def _normalize_hostname(raw: Any) -> Optional[str]:
9 """URL/域名 → 可比对归一 hostname(spec 伪码口径,与 guard g5 同语义)。
10
11 归一化:小写 → 去尾点 → 去 userinfo/port/路径(urlsplit hostname)→ IDNA
12 编码(国际化域名/编码混淆失效);非 http(s) scheme/解析失败 → None(调用方拒)。
13 """
14 s = str(raw or "").strip()
15 if not s:
16 return None
17 try:
18 if "://" not in s: # 裸域名/带 port 域名补 scheme(仅解析)
19 s = "http://" + s
20 parts = urlsplit(s)
21 if parts.scheme.lower() not in ("http", "https"):
22 return None
23 host = parts.hostname
24 except ValueError:
25 return None
26 if not host:
27 return None
28 host = host.lower().rstrip(".")
29 if not host:
30 return None
31 try:
32 host = host.encode("idna").decode("ascii") # IDNA 归一(逐标签)
33 except UnicodeError:
34 return None # 编码无法归一 → 不可比对 → 拒
35 return host

37def _domain_allowed(url: str, ctx: Any) -> Optional[str]:
38 """域名归一比对(工具内第二道,F038;spec 伪码权威)。
39
40 命中 allowlist(精确匹配,归一后比较;子域/IP/编码混淆不隐含放行)返回归一
41 域名;未命中/非法 URL 返回 None,由调用方映射 POL-NET-1。
42 """
