13def _skill_cmd(positional: list[str], flags: dict) -> int:
14 """skill [name]:列本地技能包目录(离线零装配)或打印某技能正文。
15
16 技能库 = 仓库 skills/(内置示例)+ settings.skills.dir(用户目录);正文经
17 SkillManager.load 截断保护(超长截断)。CLI 此前只能由 LLM 调 skill.load
18 看技能,人看不到——本条给人一个只读入口(与桌面技能页同源数据)。
19 """
20 from pyharness.core.skill import SkillManager
21 settings = _load_settings(flags.get("config"))
22 roots = [Path(__file__).resolve().parents[1] / "skills"]
23 extra = getattr(getattr(settings, "skills", None), "dir", None)
24 if extra:
25 roots.append(Path(str(extra)).expanduser())
26 mgr = SkillManager(roots)
27 name = " ".join(positional).strip()
28 if not name:
29 items = mgr.list()
30 data = {"roots": [str(r) for r in roots], "count": len(items),
31 "skills": items}
32 if flags.get("json"):
33 _json_line(data)
34 else:
35 print(f"技能库({len(items)} 个):{', '.join(str(r) for r in roots)}")
36 for it in items:
37 print(f" - {it['name']}: {it['description']}")
38 return 0
39 skill = mgr.load(name) # 未知名 → SKL-901(带 advice)
40 if flags.get("json"):
41 _json_line(skill)
42 else:
43 print(f"# {skill['name']}\n{skill['description']}\n目录: {skill['dir']}\n")
44 print(skill["body"])
45 return 0

56def _scan_usage(sessions_dir: Path) -> dict:
57 """会话日志只读扫描(离线派生):逐**会话**数事件、尾部终态与 llm.usage 成本聚合。
58
59 单文件不可读(被占用/权限)只跳过并告警,**不中止整条离线命令**——与
60 ``_scan_unhealthy`` 同款容错(RT-02:此前一个坏文件使 budget/stats 整体
61 以 CYC-999 退出)。``sessions`` 仍记扫描面(含被跳过者),聚合值只含可读者。
62
63 2026-09-21 R14-3:改为按**会话**聚合(``session_data_paths`` = 轮转段 + 主文件,
64 每事件恰计一次)。此前按 ``glob("*.jsonl")`` **逐文件**计,而 ``{sid}.N.jsonl`` /
65 ``{sid}.corrupt-*`` / ``{sid}.quarantine-*`` 全被当作独立会话 ⇒ 修复备份是主文件的
66 **字节副本** ⇒ 事件与成本**重复计**,隔离档的坏行也计入事件数(实测:1 会话 → 报
67 "3 个会话 / 3 事件")。终态只取**主文件**尾部(段不承载终态)。
68 """
69 from pyharness.persistence import session_data_paths, session_log_paths
70 sessions = session_log_paths(sessions_dir)
71 total_events = 0
72 finished = 0
73 cost = 0.0
74 for main in sessions:
75 tail_type = ""
76 n = 0
77 try:
78 for f in session_data_paths(main):
79 with open(f, encoding="utf-8", errors="replace") as fh:
80 for line in fh:
81 line = line.strip()
82 if not line:
83 continue
84 n += 1
85 try:
86 ev = json.loads(line)
87 except json.JSONDecodeError:
88 continue # 坏行不计(repair 域)
89 if f == main: # 终态只见于主文件尾部(按值比较:
90 tail_type = ev.get("type", "") # Path(p) 非同对象)
91 if ev.get("type") == "llm.usage":
92 cost += float(
93 (ev.get("payload") or {}).get("cost_est") or 0.0)
94 except OSError as e: # 单会话不可读:跳过并告警,不中止
95 log.warning("budget 扫描跳过不可读会话 sid=%s why=%s", main.stem, e)
96 continue
97 total_events += n
98 if tail_type == "session.finished":
99 finished += 1
100 return {"sessions": len(sessions), "events": total_events,
101 "finished": finished, "cost_est": round(cost, 4)}

103def _budget_cmd(flags: dict) -> int:

117 }

119 _json_line(data)
120 return 0
121 lim = b.monthly.limit_yuan
122 limit_txt = f"¥{lim:.2f}(未启用)" if lim <= 0 else f"¥{lim:.2f}"

145async def repair_cmd(sh: ShellCtx, sid: Optional[str] = None) -> int:
146 """repair 子命令桥(F060):对 sid(缺省 = 自动扫描首损)执行修复管线;报告输出;
147 不可修复 → 明确报错 + 备份留存(PERS-201 上抛)。"""
148 from pyharness.repair import auto_scan, repair_session # 惰性(偏离 7)
149 if not sid:
150 # R11-2:带上 db_path ⇒ 索引落后对账一并生效(此前该对账无提供者,恒不启用)
151 _db = str(getattr(getattr(getattr(sh.ctx, "settings", None), "storage", None),
152 "db_path", "") or "")
153 hits = await auto_scan(sh.ctx.storage.sessions_dir,
154 db_path=_db or None)
155 # R14-7:与启动自检同一候选筛选点(跳过活会话)—— 否则唯一"待修"是活会话时
156 # 整条命令直接抛 PERS-202,且够不到真正损坏的会话。
157 sid = _repairable_unhealthy(hits, sh.ctx.storage.sessions_dir)
158 if not sid:
159 print("[repair] 无损坏会话")
160 return 0
161 interactive = sys.stdin.isatty()
162 report = await repair_session(sh.ctx, sid, interactive=interactive)
163 if sh.flags.get("json"):
164 _json_line(_report_dict(report, sid))
165 else:
166 print(f"[repair] {sid}: fixed={_report_field(report, 'fixed')} "
167 f"lost={_report_field(report, 'lost', 0)} "
168 f"backup={_report_field(report, 'backup_path', '')} "
169 f"quarantined={_report_field(report, 'quarantined', 0)}")
170 return 0
