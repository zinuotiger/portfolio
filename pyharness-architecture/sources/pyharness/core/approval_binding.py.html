124 def enable_trust(self, on: bool, *, by: str) -> None:
125 """信任名单开关(默认关;仅交互会话可开;headless 调用直接拒绝 APR-501)。"""
126 self._require_human(by)
127 if on and (self._headless_of(None) or self._channel not in CHANNELS):
128 raise_code("APR-501", op="enable_trust",
129 hint="headless/无交互通道:信任名单仅交互 cli/web/acp 可开,"
130 "headless 永不生效")
131 self._enabled = bool(on)
132 self._emit_trust(_TRUST_SET, {"on": self._enabled, "by": by})
133 logger.info("approval 信任名单 %s(by=%s)",
134 "启用" if self._enabled else "关闭", by)

136 def clear_trust(self, tool: Optional[str] = None) -> int:
137 """清除信任(全会话或单工具;用户可手动撤销)→ 返回清除条数。"""
138 if tool is None:
139 count = len(self._trust)
140 self._trust.clear()
141 else:
142 victims = [fp for fp, e in self._trust.items() if e.tool == tool]
143 count = len(victims)
144 for fp in victims:
145 self._trust.pop(fp, None)
146 if count:
147 self._emit_trust(_TRUST_CLEARED, {"tool": tool, "count": count})
148 return count

150 def _trust_hit(self, fp: str, sid: str, ch: str) -> bool:
151 """信任命中:开关开 + 交互通道 + 指纹在表(同会话);命中计数 + 留痕。"""
152 if not self._enabled or ch not in CHANNELS:
153 return False
154 entry = self._trust.get(fp)
155 if entry is None:
156 return False
157 if sid and entry.session_id and entry.session_id != sid:
158 return False # 会话级:他会话信任不生效
159 entry.hits += 1
160 self._emit_trust(_TRUST_HIT, {"tool": entry.tool, "hits": entry.hits})
161 return True # 命中 ≠ 跳过 guard(executor 重入)

163 def _remember(self, req: ApprovalRequest) -> None:

165 if req.danger == "critical":
166 return # critical 类绝不在名单
167 entry = TrustEntry(tool=req.tool, fingerprint=req.fingerprint,
168 by=req.by or "system", added_seq=req.approval_id or 0,
169 session_id=req.session_id)
170 self._trust[req.fingerprint] = entry
171 while len(self._trust) > self._trust_max: # FIFO 淘汰(按加入序)
172 oldest = min(self._trust.values(), key=lambda e: e.added_seq)

174 self._emit_trust(_TRUST_ADDED, {"tool": entry.tool, "by": entry.by,
