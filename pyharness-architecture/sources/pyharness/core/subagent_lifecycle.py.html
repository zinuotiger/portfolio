73 async def announce(self, ctx: Any = None) -> None:
74 """对外宣布(DIS-SEAM §2.4 第②动作,五步):注册工具 → 挂订阅 → mount →
75 registry.updated 留痕;任一步失败回滚已做步骤并回 detached(半装必回滚)。
76
77 已 announce 幂等返回。mount 目标 = ctx.agent 命名空间(装配层未来 locator
78 宿主;未挂 agent 命名空间时日志降级不阻断——工具已注册,主循环仍可经
79 tools.execute 触达)。
80 """
81 if self._announced:
82 return
83 done: list[str] = []
84 try:
85 ctx = ctx or self._announce_ctx
86 bus = self._bus if ctx is None else (getattr(ctx, "bus", None)
87 or self._bus)
88 tools = self._tools if ctx is None else (getattr(ctx, "tools", None)
89 or self._tools)
90 if tools is not None:
91 # ① ③ LLM 可见:register_tool 绑 Provider(spawn+join 适配器,偏离 8)
92 register = getattr(tools, "register_tool", None) or \
93 getattr(tools, "register_definition", None)
94 if callable(register):
95 try:
96 register(self.definition(), provider=SubagentSpawnProvider(self))
97 except TypeError:
98 register(self.definition()) # 无 provider 面的注册表别名
99 done.append("tool")
100 if bus is not None:
101 # ② 挂订阅:session.closing 预关钩子(规范预留,偏离 11)
102 bus.subscribe("session.closing", self._on_session_closing,
103 owner=self._owner)
104 done.append("sub")
105 # ④ mount ctx.agent.subagent(定位器;可缺省降级)
106 mounted = self._mount(ctx, "subagent")
107 if mounted:
108 done.append("mount")
109 if bus is not None:
110 # ⑤ registry.updated 留痕(瞬时事件,仅总线)
111 await bus.emit("registry.updated",
112 {"op": "attach", "kind": "capability",
113 "key": "ctx.agent.subagent"})
114 self._announce_ctx = ctx
115 self._announced = True
116 log.info("subagent capability announced session=%s",
117 getattr(self._session, "sid", None))
118 except BaseException:
119 for step in reversed(done): # 半装必回滚(逆序)
120 try:
121 self._rollback(step, ctx)
122 except Exception: # noqa: BLE001 回滚自身失败不吞原错
123 log.exception("announce 回滚失败 step=%s", step)
124 self._entered = False # 回 detached(可重入重试)
125 raise

149 async def detach(self, ctx: Any = None) -> None:
150 """能力摘除,child-first(DIS-SEAM §2.4 第③动作):先杀光在途子任务并落
151 终态事件,再摘订阅/注销工具/unmount;单步失败记 broken 告警仍继续摘除
152 (半卸 > 僵尸);幂等(重复调用安全)。
153 """
154 if not self._entered and not self._announced and not self._children:
155 return # 幂等:无装载无在途 → 无事可做
156 await stop_admissions(self)
157 cleanup_errors = []
158 # ① child-first:先杀孩子(取消传播 F025 + 终态事件 + 等全部真正退出)
159 try:
160 await self._cancel_all_children(reason="parent-detach")
161 except BaseException as exc:
162 cleanup_errors.append(exc)
163 ctx = ctx or self._announce_ctx
164 bus = self._bus if ctx is None else (getattr(ctx, "bus", None)
165 or self._bus)
166 tools = self._tools if ctx is None else (getattr(ctx, "tools", None)
167 or self._tools)
168 broken: list[str] = []
169 # ② 摘订阅(owner 精确摘除;幂等)
170 if bus is not None:
171 try:
172 bus.unsubscribe_all(owner=self._owner)
173 except Exception as exc: # noqa: BLE001 半卸 > 僵尸
174 log.warning("detach 摘订阅失败: %s", exc)
175 broken.append("sub")
176 # ③ 注销工具(TLB-802 = 已摘,幂等静默)
177 if tools is not None:
178 unreg = getattr(tools, "unregister_definition", None) or \
179 getattr(tools, "unregister", None)
180 if callable(unreg):
181 try:
182 unreg(TOOL_NAME)
183 except PyHError as e:
184 if e.code != "TLB-802":
185 log.warning("detach 注销工具失败 code=%s", e.code)
186 broken.append("tool")
187 except Exception as exc: # noqa: BLE001
188 log.warning("detach 注销工具异常: %s", exc)
189 broken.append("tool")
190 # ④ unmount(ctx.agent.subagent)
191 try:
192 self._unmount(ctx)
193 except Exception as exc: # noqa: BLE001
194 log.warning("detach unmount 失败: %s", exc)
195 broken.append("mount")
196 # ⑤ registry.updated op=del 留痕(尽力而为)
197 if bus is not None and not broken:
198 try:
199 await bus.emit("registry.updated",
200 {"op": "del", "kind": "capability",
201 "key": "ctx.agent.subagent"})
202 except Exception: # noqa: BLE001
203 log.warning("detach registry.updated 留痕失败")
204 self._announced = False
205 self._entered = False
206 self._children.clear()
207 if broken:
208 cleanup_errors.append(RuntimeError("subagent detach incomplete: " + ",".join(broken)))
209 if cleanup_errors:
210 raise BaseExceptionGroup("subagent detach failed", cleanup_errors)

212 def _mount(self, ctx: Any, key: str) -> bool:
213 """locator.mount:把本 manager 挂到 ctx.agent 命名空间的 subagent 键。
214
215 挂载前记录旧值(detach 恢复);ctx.agent 命名空间未建(装配层 locator 未
216 接线)→ 日志降级返回 False(工具面仍可用,不阻断 announce)。
217 """
218 if ctx is None:
219 return False
220 ns = getattr(ctx, "agent", None)
221 if ns is None:
222 log.warning("ctx.agent 命名空间未挂载,ctx.agent.subagent 未 mount"
223 " (tools.execute 面仍可用)")
224 return False
225 self._mount_ns = ns
226 self._mount_prev = getattr(ns, key, None)
227 try:
228 setattr(ns, key, self)
229 return True
230 except Exception as exc: # noqa: BLE001 只读/受限目标
231 log.warning("mount ctx.agent.%s 失败: %s", key, exc)
232 self._mount_ns = None
233 self._mount_prev = None
234 return False

236 def _unmount(self, ctx: Any = None) -> None:
237 """locator.unmount 逆操作:仅当现值仍为本 manager 时摘除并恢复旧值。"""
238 ns = self._mount_ns
239 if ns is None and ctx is not None:
240 ns = getattr(ctx, "agent", None)
241 if ns is None:
242 return
243 key = "subagent"
244 if getattr(ns, key, None) is self:
245 if self._mount_prev is None:
246 try:
247 delattr(ns, key)
248 except Exception: # noqa: BLE001
249 log.debug("subagent namespace unmount failed", exc_info=True)
250 else:
251 try:
252 setattr(ns, key, self._mount_prev)
253 except Exception: # noqa: BLE001
254 log.debug("subagent namespace restore failed", exc_info=True)
255 self._mount_ns = None
256 self._mount_prev = None

262 if ctx is not None:
263 rs = getattr(ctx, "round_seq", None)

265 return rs

267 cur = getattr(loop, "current", None) if loop is not None else None

293 raise_code("EVT-100", field="tools_subset",
294 hint="tools_subset 须为 list[str] 或 None")

297 raise_code("EVT-100", field="tools_subset", tool=t,
298 hint="tools_subset 元素须为工具名字符串")

300 or not isinstance(spec.creds_allowed, list):
301 raise_code("EVT-100", hint="deny_extra/creds_allowed 须为 list")

303 raise_code("EVT-100", field="depth", depth=spec.depth,
304 hint="depth 须为 int(递归层级)")
