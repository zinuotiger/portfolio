188 # ============================================================ 命令分发
189 async def goal(self, args: dict, ctx: Any) -> str:
190 """F047 验收伪代码直译的 op 分发器 → 用户可读 summary。
191
192 args = {"op", "text", "id", "progress", "checked", "claim_done",
193 "status", "note", "task_id", …};ctx = 会话实体(取 ctx.task_id
194 兜底关联,F044)。op 非法 → EVT-100;goal_id 不存在 → EVT-101。
195 """
196 op = args.get("op")
197 if op is None:
198 raise_code("EVT-100", hint="goal 命令缺 op 字段",
199 advice="op ∈ create/update/check/abandon/list")
200 if op not in ("create", "update", "check", "abandon", "list"):
201 # op 合法性先于定位(伪码先 _get 后判 op,会使未知 op 在缺 id 时误报
202 # EVT-101;以异常表"op 非法 → EVT-100"语义优先——见偏离说明)
203 raise_code("EVT-100", hint=f"未知 goal op: {op}",
204 advice="op ∈ create/update/check/abandon/list")
205 gid = args.get("id")
206 if op == "create": # 创建:可关联当前任务 ctx.task_id
207 g = await self.goal_create(
208 args.get("text", ""),
209 task_id=args.get("task_id") or getattr(ctx, "task_id", None))
210 return g.summary()
211 if op == "list":
212 return "\n".join(g.summary() for g in self.list_active()) or "无活动目标"
213 g = self._get(gid) # 其余 op 先定位;不存在 → EVT-101
214 if op == "update": # 改 note/显式切状态(paused 等)
215 await self.goal_update(gid, note=args.get("note"),
216 status=args.get("status"))
217 return g.summary()
218 if op == "check": # 核对:打勾/报进度/声称完成
219 r = await self.goal_check(gid, progress=args.get("progress"),
220 checked=args.get("checked"),
221 claim_done=bool(args.get("claim_done")))
222 return r.reply
223 await self.goal_abandon(gid, reason=args.get("reason", "")) # op=abandon
224 return g.summary()

227 async def goal_create(self, text: str, *, task_id: Optional[str] = None) -> Goal:
228 """新建活动 Goal 并写 goal.created(actor=agent)。
229
230 活动目标已达 active_limit(≤8)→ BUSY 拒(须先 done/abandoned 旧的);
231 text 空白 → EVT-100。事件先落,内存后对齐(INV-01)。
232 """
233 if not text or not text.strip():
234 raise_code("EVT-100", hint="goal 描述为空",
235 advice="提供可核对的目标描述文本")
236 if self._nondone() >= self._active_limit: # 活动目标 ≤8(F047 边界)
237 raise PyHError("BUSY", ctx={
238 "advice": f"活动目标已达 {self._active_limit} 个上限;"
239 f"请先完成/放弃旧目标", "active": self._nondone()})
240 sess = self._require_session()
241 desc = text.strip()
242 gid = self._alloc_gid()
243 # 事件先落:状态机只认事件(INV-01);seq/actor 由框架打
244 env = await sess.append("goal.created",
245 {"goal_id": gid, "desc": desc,
246 "task_id": task_id}, actor="agent")
247 g = Goal(id=gid, desc=desc, status="active", task_id=task_id,
248 created_seq=env.seq)
249 self._goals[gid] = g # 事件落定 → 注册(created_seq=事件 seq)
250 self._touch_activity()
251 return g

253 async def goal_update(self, goal_id: str, *, note: Optional[str] = None,
254 status: Optional[str] = None,
255 progress: Optional[float] = None) -> None:
256 """改 note/显式切状态(paused↔active)/报进度,统一写 goal.updated。
257
258 done 目标不可再 updated(EVENT-SCHEMA 词表校验 → BUSY);status 只收
259 词表值(active/paused/abandoned;done 只能经核对完成写入);无变更项
260 幂等返回不写事件。
261 """
262 g = self._get(goal_id) # 不存在 → EVT-101(前必有 created)
263 if g.status == "done": # completed 后不可再 updated
264 raise PyHError("BUSY", ctx={
265 "hint": f"目标 {goal_id} 已完成,不可再更新",
266 "advice": "完成态不可逆;新需求请新建目标"})
267 if status is not None and status not in ("active", "paused", "abandoned"):
268 raise_code("EVT-100", hint=f"非法 goal 状态 {status}",
269 advice="状态 ∈ active/paused/done/abandoned(词表四值;"
270 "done 只能经核对完成写入)")
271 if note is None and status is None and progress is None:
272 return # 幂等:无变更项不写事件
273 # 全量快照载荷:updated 事件携带目标完整状态(status/note/progress)。原因:
274 # payload 模型 canonical 化恒补齐全部键(缺省 None),rebuild 若按"缺键保留"
275 # 语义会拿显式 None 抹掉旧值——快照语义保证回放逐字段一致(INV-01)。
276 new_status = status if status is not None else g.status
277 new_note = note if note is not None else g.note
278 new_progress = (_clamp01(progress) if progress is not None else g.progress)
279 payload: dict[str, Any] = {"goal_id": goal_id, "status": new_status,
280 "note": new_note, "progress": new_progress}
281 sess = self._require_session()
282 await sess.append("goal.updated", payload, actor="agent")
283 # 事件落定 → 内存对齐到载荷(INV-01:内存 ≡ 事件派生)
284 g.status = new_status
285 g.note = new_note
286 g.progress = new_progress
287 self._touch_activity()

289 async def goal_check(self, goal_id: str, *,
290 progress: Optional[float] = None,
291 checked: Optional[list[str]] = None,
292 claim_done: bool = False) -> GoalCheck:
293 """核对(打勾/报进度/完成声明)。
294
295 checked 项置 done、报 progress;claim_done=True 时先 verify_completion
296 验证据:过核 → 写 goal.completed(actor=system);不过 → 返回追问文本,
297 状态不变(未完成却说完成 → 追问一次,F047 边界,不落 completed)。
298 """
299 g = self._get(goal_id)
300 if g.status == "done": # 幂等:已完成核对直接确认返回
301 return GoalCheck(goal_id, ok=True, claim_done=True, missing=[],
302 reply="该目标已完成")
303 sess = self._require_session()
304 if checked:
305 done_set = set(checked)
306 for c in g.checklist: # 打勾(子目标核对,瞬态工作数据)
307 if c.text in done_set:
308 c.done = True
309 if progress is not None: # 报进度(clamp;claim 前先入内存供核对)
310 g.progress = _clamp01(progress)
311 if not claim_done:
312 await sess.append("goal.updated",
313 {"goal_id": goal_id, "status": g.status,
314 "note": g.note, "progress": g.progress},
315 actor="agent")
316 self._touch_activity()
317 return GoalCheck(goal_id, ok=True, claim_done=False, missing=[],
318 reply=f"已记录进度 {g.progress:.0%}")
319 # claim_done:带进度先补落一条 updated(INV-01:进度变化可回放),再验证据
320 if progress is not None:
321 await sess.append("goal.updated",
322 {"goal_id": goal_id, "status": g.status,
323 "note": g.note, "progress": g.progress},
324 actor="agent")
325 verdict = self.verify_completion(g) # 声称完成 → 先过核对(纯函数)
326 if not verdict.ok: # 证据不足:追问一次,不落 completed
327 self._touch_activity()
328 return GoalCheck(goal_id, ok=False, claim_done=True,
329 missing=verdict.missing,
330 reply=f"你说已完成,但核对项未过:"
331 f"{'、'.join(verdict.missing)};"
332 f"逐条核对(goal check)后再报完成。")
333 await self.mark_completed(g) # 过核 → completed(actor=system)
334 return GoalCheck(goal_id, ok=True, claim_done=True, missing=[],
335 reply="核对通过,目标已完成")

337 def verify_completion(self, goal: Goal) -> CompletionVerdict:
338 """完成核对(纯函数,S-1 防伪闸):progress≥1.0 且 checklist 全 done 且
339 关联 task 无 failed;返回 ok 与 missing 明细。零写副作用,供 goal_check
340 与测试直接调用;ok=False → 追问(不落 completed)。
341 """
342 missing: list[str] = []
343 if goal.progress < 1.0: # 进度未满
344 missing.append(f"进度仅 {goal.progress:.0%}")
345 for c in goal.checklist: # 子目标逐条核对(PRD F047)
346 if not c.done:
347 missing.append(f"未打勾: {c.text}")
348 if goal.task_id and self._task_failed(goal.task_id):
349 missing.append(f"关联任务 {goal.task_id} 未成功结束")
350 return CompletionVerdict(ok=not missing, missing=missing)

376 note = reason or None
377 await sess.append("goal.updated",
378 {"goal_id": goal_id, "status": "abandoned",
379 "note": note, "progress": g.progress},
380 actor="agent") # 快照载荷:model canonical 恒补 progress
381 g.status = "abandoned" # 事件落定 → 内存对齐(INV-01)
382 g.note = note # 与载荷一致(reason 空则清旧 note)
383 self._touch_activity()

385 # ============================================================ 看板/渲染
386 def list_active(self) -> list[Goal]:
387 """活动目标看板源:active+paused,按 created_seq 升序;≤8 由 create 保证。"""
388 return [g for g in sorted(self._goals.values(), key=lambda x: x.created_seq)
389 if g.status in ("active", "paused")] # done/abandoned 不入板

391 def render_goal_segment(self, *, max_tokens: int = 800) -> str:
392 """活动目标板 → 系统提示词段(F010 每次 LLM 调用携带;软锚定不硬限制)。
393
394 无活动目标返回空串(零开销不注入);超长按 max_tokens 截断防超窗。
395 """
396 acts = self.list_active()
397 if not acts:
398 return ""
399 lines = ["<goals> 当前活动目标(软提醒:汇报完成必须先过核对):"]
400 for g in acts:
401 pend = [c.text for c in g.checklist if not c.done] # 待核子目标
402 lines.append(f"- [{g.id}] {g.desc} | 进度 {g.progress:.0%}"
403 f" | 待核:{','.join(pend) if pend else '无'}"
404 + (f" | 关联任务 {g.task_id}" if g.task_id else ""))
405 return "\n".join(lines)[:max_tokens] # 截断防超窗(余量 ≥25% 由 F058 控)

407 # ============================================================ 漂移判定
408 def round_end(self, events: list[Any]) -> None:
409 """agent-loop 每轮末喂本轮事件:轮号 +1 后扫 goal.* / 带 goal 溯源任务。
410
411 轮内出现 goal 关联活动(goal.* 事件或 task.enqueued 携带 meta.goal_id)
412 → _last_goal_round = 本轮(静默轮计数清零);否则静默轮累计。
413 """
414 self._round_no += 1
415 for e in events:
416 t = getattr(e, "type", "")
417 p = getattr(e, "payload", {}) or {}
418 if t.startswith("goal.") or (t == "task.enqueued"
419 and (p.get("meta") or {}).get("goal_id")):
420 self._last_goal_round = self._round_no # 本轮有目标活动
421
422 def drift_reminder(self, *, quiet_rounds: int = 3) -> Optional[str]:
423 """离题提醒(确定性规则):有活动目标但连续 quiet_rounds(≥3)轮无任何
424 goal 关联活动 → 返回提醒文本(注入下一轮系统提示,软提示不阻断);
425 否则 None。会话从未有目标活动 → 不判定不提醒。
426 """
427 if not self.list_active(): # 无活动目标:不判定不提醒
428 return None
429 if self._last_goal_round is None: # 会话从未有目标活动:不判定
430 return None
431 if self._round_no - self._last_goal_round >= quiet_rounds:
432 return ("提示:已连续数轮未推进任何活动目标(goal 板见上)。"
433 "若已偏离原目标,请说明理由或更新/放弃目标,避免跑偏。")
434 return None

436 # ============================================================ 事件溯源重建
437 @classmethod
438 def rebuild_from_events(cls, events: list[Any]) -> "GoalManager":
439 """事件溯源重建(INV-01):顺序重放 goal.created/updated/completed 重建
440 注册表(seq 升序,只读不写回)——崩溃恢复/会话恢复后内存态与日志一致。
441
442 孤儿 updated/completed(前无 created)属日志损坏,由 EVENT-SCHEMA 校验层
443 拦截;此处防御性跳过并告警,重放不中断。created_seq = 事件 seq;
444 g-N 计数器按现存 id 续号(折叠部分旧目标后新建仍单调唯一)。
445 """
446 m = cls() # 全新空注册表(无 session:只读投影)
447 for e in events: # 顺序重放(seq 升序)
448 p = getattr(e, "payload", {}) or {}
449 if e.type == "goal.created":
450 gid = p["goal_id"]
451 m._goals[gid] = Goal(id=gid, desc=p.get("desc") or "",
452 task_id=p.get("task_id"), status="active",
453 created_seq=e.seq)
454 m._bump_gid_counter(gid)
455 elif e.type == "goal.updated":
456 g = m._goals.get(p.get("goal_id"))
457 if g is None:
458 log.warning("orphan goal.updated seq=%s", e.seq)
459 continue # 防御跳过,不中断重放
460 g.progress = p.get("progress", g.progress)
461 g.status = p.get("status", g.status) # paused/abandoned 迁移
462 g.note = p.get("note", g.note)
463 elif e.type == "goal.completed": # actor=system;终态
464 g = m._goals.get(p.get("goal_id"))
465 if g is not None:
466 g.status = "done"
467 return m # 内存态 ≡ 日志派生态(INV-01)
