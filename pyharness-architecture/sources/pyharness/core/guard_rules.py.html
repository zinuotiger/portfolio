79async def g_credential_read_check(
80 call: Any, scope: Any,
81 *, credential_paths: Optional[Iterable[str]] = None,
82 link_resolver: Optional[Callable[[str], str]] = None
83) -> tuple[str, Optional[str]]:
84 """g4 check:读目标命中凭据文件清单 → 拒(POL-CRED-1)。
85
86 命中口径双轨(见偏离 1 注释/模块 docstring):①规范化绝对路径 ∈ 清单;
87 ②文件形态名(basename = credentials.yaml/yml 等,纵深:凭据文件拷进
88 workspace 的任何位置都拦)。清单由装配注入(credentials.yaml 等,F016
89 credentials 模块落地后同源);路径先过 g3 几何(g4 只做命中比较)。
90 """
91 paths = list(credential_paths or ())
92 if not paths: # 未装配清单 → 兜底默认凭据文件
93 paths = [os.path.expanduser(p) for p in DEFAULT_CREDENTIAL_FILES]
94 ws = getattr(_policy_of(scope), "workspace_root", None) or ""
95 target, policy = _resolve_geometry(_fs_target(call), ws,
96 link_resolver=link_resolver)
97 if policy:
98 return ("reject", policy) # 越界读:g3 语义优先(g4 不越权)
99 norm_target = _normcase_path(target)
100 for c in paths:
101 norm_c = _normcase_path(os.path.abspath(os.path.expanduser(str(c))))
102 if norm_target == norm_c:
103 return ("reject", "POL-CRED-1") # 命中清单精确路径
104 base = os.path.basename(norm_target).lower()
105 if base in _CRED_BASENAMES or any(
106 os.path.basename(os.path.normpath(str(c))).lower() == base
107 for c in paths if str(c).strip()):
108 return ("reject", "POL-CRED-1") # 凭据文件形态(名字即证据)
109 return ("allow", None)

115def _normalize_domain(raw: Any) -> str:
116 """域名/URL 归一为可比对 hostname(小写、去尾点、剥 scheme/port/路径)。
117
118 URL 用 urlsplit 取 hostname;裸域名/IP/带端口原样取首段;解析不出 → 空串
119 (空串必不在 allowlist → 拒,禁外发默认)。
120 """
121 s = str(raw or "").strip().lower().rstrip(".")
122 if not s:
123 return ""
124 try:
125 if "://" not in s:
126 s = "http://" + s # scheme 缺失补默认(仅解析用)
127 host = urlsplit(s).hostname
128 return (host or "").lower().rstrip(".")
129 except ValueError:
130 return ""

132async def g_net_outbound_check(call: Any, scope: Any
133 ) -> tuple[str, Optional[str]]:
134 """g5 check:目标域不在 scope.policy.allowed_domains → 拒(POL-NET-1)。
135
136 默认空 = 禁一切外发;字面 hostname 精确匹配(见偏离 8:子域/IP/混淆不隐含
137 放行;通配 "*" 是越权配置,config 层已拒,此处纵深不构成放行)。
138 目标域取参数 url/domain/host(URL 或裸域都归一);无域参数 → 拒(禁外发)。
139 """
140 allowed = getattr(_policy_of(scope), "allowed_domains", None) or set()
141 args = getattr(call, "safe_args", {}) or {}
142 raw = next((args[k] for k in ("url", "domain", "host", "target")
143 if isinstance(args.get(k), str) and args[k].strip()), "")
144 if not raw and call.name == "web.search":
145 # web.search 无 url 参数,g5 无法从 args 取目标 → 取装配期记录的后端 host
146 # (operator 配置的搜索端点;P3:web.search 恒拒)。空=未配后端 → 仍拒。
147 raw = getattr(_policy_of(scope), "search_host", "") or ""
148 host = _normalize_domain(raw)
149 if not host or host not in {str(d).strip().lower().rstrip(".")
150 for d in allowed}:
151 return ("reject", "POL-NET-1")
152 return ("allow", None)

158async def g_exec_check(call: Any, scope: Any) -> tuple[str, Optional[str]]:
159 """g6 check:exec 执行约束(POL-EXEC-1)。
160
161 三条拒绝面(任一命中即拒,零执行):
162 ① strict 沙箱直拒——scope 前置已把 exec.* 判不可见(域外),此处兜底
163 (偏离 6:strict 下 exec 无"scope 显式授权"可言);
164 ② args.shell is True ——禁 shell=True(解释器注入面);
165 ③ cwd 绝对路径越出 workspace ——进程工作目录限沙箱。
166 其余(basic/off + 无 shell + cwd 界内/缺省)→ allow。
167 """
168 pol = _policy_of(scope)
169 if getattr(pol, "sandbox_level", "strict") == "strict":
170 return ("reject", "POL-EXEC-1")
171 args = getattr(call, "safe_args", {}) or {}
172 if args.get("shell") is True:
173 return ("reject", "POL-EXEC-1")
174 cwd = args.get("cwd")
175 if isinstance(cwd, str) and cwd.strip() and os.path.isabs(cwd):
176 ws = getattr(pol, "workspace_root", None) or ""
177 if not _inside_workspace(os.path.normpath(cwd), ws):
178 return ("reject", "POL-EXEC-1")
179 return ("allow", None)

185async def g_overwrite_check(
186 call: Any, scope: Any,
187 *, path_exists: Optional[Callable[[str], bool]] = None
188) -> tuple[str, Optional[str]]:
189 """g7 check:覆写既有文件(含 append 读改写)→ approval(POL-OVW-1)。
190
191 规则:目标不存在(新建)→ allow;目标已存在 → approval(无论 mode=write 整覆
192 还是 mode=append 读改写——append 是 read_text()+整体覆写,同为声明外后门,
193 SECURITY §4.2 A3);目标探测走注入 path_exists(缺省 os.path.exists,测试可
194 替身化),探测失败视同不存在(新建路径)。路径先过 g3 几何,越界轮不到 g7
195 (链序保证)。
196 """
197 args = getattr(call, "safe_args", {}) or {}
198 ws = getattr(_policy_of(scope), "workspace_root", None) or ""
199 target, policy = _resolve_geometry(_path_from_args(args), ws)
200 if policy:
201 return ("reject", policy)
202 probe = path_exists or os.path.exists
203 try:
204 exists = bool(probe(target))
205 except OSError:
206 exists = False # 探测异常 → 视同不存在
207 if not exists:
208 return ("allow", None) # 新建路径:无覆写面
209 return ("approval", "POL-OVW-1") # 已存在:write/append 均转审批
