1"""pyharness/governance/audit.py — 治理审计视图(S5-3a)。
2
3一句话职责:回答**"某一条决策的完整因果链是什么"**与**"哪些调用被拒、依据什么"**。
4``AuditSystem`` 是**纯派生视图(derived view)**——不是新真源,不是缓存,不订阅、
5不写事件、不跨调用保存状态。
6
7依赖方向(ADR-018:308,强制):只 import 标准库、``pyharness.errors`` 与本包;
8**禁止** ``pyharness.core.*`` / ``engine`` / ``bus`` / ``persistence``。会话经
9``ctx.session``/构造注入**鸭子类型**只读使用。
10
11**replay-only(仅重放,本模块的第一纪律)**:
12
13- **不订阅**实时事件 —— 订阅态索引在重启后为空,审计必须**仅凭日志**可用;
14- **不复用** ``EvidenceCollector`` 的索引 —— 其订阅面仅 5 类,缺
15 ``tool.result`` / ``guard.*`` / ``approval.*``;且它按**段**组织,审计按**因果**串联;
16- 每次调用**即时全量重放** ``_read_events(session)``(O(n);v1.0 接受,见 Change Report);
17- 因此:同一日志 ⇒ 同一审计结果(可复现,不依时序、不依订阅先后)。
18
19**与 EvidenceCollector 的边界**:Evidence 回答"**这个任务**有哪些证据"(段轴,
20只覆盖**被归档**的证据);Audit 回答"**这条决策**为什么发生、经过了什么"
21(因果轴,覆盖**全部**决策,含未归档者)。
22
23**边界**：
24
25- **S5-3**：``causal_chain``（因果链）与 ``denied_report``（拒绝清单）—— **零写**。
26- **S5-4**：``reconcile``（三类一致性对账）+ ``legacy_session_audit``（旧面兼容，
27 **经注入**）。``reconcile`` **默认纯只读**（``emit=False``）;仅 ``emit=True``
28 且 findings 非空才追加**一条既有** ``syscheck.fail``（**唯一写点**，不新增事件类型）。
29 ``CACHE-STALE`` **本版未实现**（deferred debt：避免依赖 ``EvidenceCollector`` 内部结构）。
30- **S7**：``TraceabilityMatrix``。
31"""

81 # ------------------------------------------------------------ 因果链
82 def causal_chain(self, decision_id: str, *, ctx: Any = None) -> dict:
83 """由 ``decision_id`` 还原完整因果链(**replay-only**,无副作用)。
84
85 输出键(冻结):``decision_id`` / ``call_id`` / ``request`` / ``policy`` /
86 ``decision`` / ``approval`` / ``supersedes`` / ``receipt`` / ``execution`` /
87 ``denied`` / ``missing``。
88
89 **不臆造**:未命中的环节一律 ``None``,并按其"是否应有"记入 ``missing``;
90 ``decision.issued`` 本身缺失 ⇒ 直接返回并只记该段(**不抛**)。
91 """
92 out: dict = {
93 "decision_id": str(decision_id), "call_id": "", "request": None,
94 "policy": None, "decision": None, "approval": None,
95 "supersedes": None, "receipt": None, "execution": None,
96 "denied": False, "missing": [],
97 }
98 events = self._events(ctx)
99 # ① 定位本决策事实
100 dec_ev = next((e for e in events
101 if getattr(e, "type", None) == "decision.issued"
102 and str((getattr(e, "payload", None) or {})
103 .get("decision_id") or "") == str(decision_id)), None)
104 if dec_ev is None:
105 out["missing"] = ["decision.issued"]
106 return out
107 p = getattr(dec_ev, "payload", None) or {}
108 verdict = str(p.get("verdict") or "")
109 approval_ref = p.get("approval_ref")
110 call_id = self._trace_call_id(dec_ev)
111 out["call_id"] = call_id
112 out["denied"] = verdict == "reject"
113 out["supersedes"] = p.get("supersedes")
114 out["decision"] = {"verdict": verdict, "ts": p.get("ts"),
115 "seq": getattr(dec_ev, "seq", None),
116 "principal": {"kind": p.get("principal_kind"),
117 "id": p.get("principal_id"),
118 "channel": p.get("principal_channel")}}
119 out["policy"] = {"fingerprint": p.get("policy_fingerprint"),
120 "refs": list(p.get("policy_refs") or []),
121 "guards": list(p.get("guard_ids") or [])}
122 # ② 按 call_id 关联请求 / 执行(同一 call_id 可能有多次求值 → 取首个请求)
123 if call_id:
124 tc = next((e for e in events
125 if getattr(e, "type", None) == "tool.call"
126 and str((getattr(e, "payload", None) or {})
127 .get("call_id") or "") == call_id), None)
128 if tc is None:
129 out["missing"].append("tool.call")
130 else:
131 tp = getattr(tc, "payload", None) or {}
132 out["request"] = {"tool": tp.get("name", ""),
133 "seq": getattr(tc, "seq", None)}
134 tr = next((e for e in events
135 if getattr(e, "type", None) == "tool.result"
136 and str((getattr(e, "payload", None) or {})
137 .get("call_id") or "") == call_id), None)
138 if tr is not None:
139 rp = getattr(tr, "payload", None) or {}
140 out["execution"] = {"ok": bool(rp.get("ok")),
141 "seq": getattr(tr, "seq", None)}
142 # ③ 审批段:approval_ref(= approval.requested 的 seq) → 结果事件
143 if approval_ref is not None:
144 ref = int(approval_ref)
145 outcome = next((e for e in events
146 if getattr(e, "type", None) in (
147 "approval.granted", "approval.denied",
148 "approval.timeout")
149 and int((getattr(e, "payload", None) or {})
150 .get("approval_id") or -1) == ref), None)
151 if outcome is None:
152 out["missing"].append("approval")
153 else:
154 op = getattr(outcome, "payload", None) or {}
155 out["approval"] = {
156 "approval_id": ref,
157 "outcome": str(getattr(outcome, "type", "")).rsplit(".", 1)[-1],
158 "by": op.get("by", ""),
159 "seq": getattr(outcome, "seq", None)}
160 # ④ 凭证段:按 decision_id 关联
161 rc = next((e for e in events
162 if getattr(e, "type", None) == "receipt.emitted"
163 and str((getattr(e, "payload", None) or {})
164 .get("decision_id") or "") == str(decision_id)), None)
165 if rc is not None:
166 rp = getattr(rc, "payload", None) or {}
167 out["receipt"] = {"receipt_id": rp.get("receipt_id"),
168 "kind": rp.get("kind"), "digest": rp.get("digest"),
169 "prev_hash": rp.get("prev_hash"),
170 "seq": getattr(rc, "seq", None)}
171 elif _receipt_expected(verdict, approval_ref):
172 out["missing"].append("receipt.emitted")
173 return out

175 # ------------------------------------------------------------ 拒绝报告
176 def denied_report(self, *, session_id: str = "", since_seq: int = 0,
177 ctx: Any = None) -> list[dict]:
178 """被拒调用清单(**replay-only**;按事件 seq 升序)。
179
180 **两类来源必须可区分**(``source`` 字段):
181
182 - ``"guard.rejected"`` —— 规则级拒绝(``INV-05`` 的强同步留痕);
183 - ``"decision.issued"`` —— 治理级拒绝(``verdict=reject``)。
184
185 每条记录键(冻结):``source`` / ``seq`` / ``tool`` / ``call_id`` /
186 ``guard_id`` / ``policy_ref`` / ``decision_id`` / ``reason`` / ``executed``。
187 ``executed`` = 该 ``call_id`` 是否存在配对 ``tool.result``(审计可证
188 "拦了且没执行";``INV-A3``)。
189 """
190 events = self._events(ctx)
191 if session_id:
192 events = [e for e in events
193 if str(getattr(e, "session_id", "") or "") == session_id]
194 executed = {str((getattr(e, "payload", None) or {}).get("call_id") or "")
195 for e in events
196 if getattr(e, "type", None) == "tool.result"}
197 rows: list[dict] = []
198 for e in events:
199 seq = int(getattr(e, "seq", 0) or 0)
200 if seq <= int(since_seq or 0):
201 continue
202 t = getattr(e, "type", None)
203 p = getattr(e, "payload", None) or {}
204 if t == "guard.rejected":
205 cid = self._trace_call_id(e)
206 rows.append({
207 "source": "guard.rejected", "seq": seq,
208 "tool": str(p.get("tool") or ""), "call_id": cid,
209 "guard_id": str(p.get("guard_id") or ""),
210 "policy_ref": str(p.get("policy_ref") or ""),
211 "decision_id": "", "reason": str(p.get("reason") or ""),
212 "executed": bool(cid) and cid in executed})
213 elif t == "decision.issued" and str(p.get("verdict") or "") == "reject":
214 cid = self._trace_call_id(e)
215 rows.append({
216 "source": "decision.issued", "seq": seq,
217 "tool": str(p.get("tool") or ""), "call_id": cid,
218 "guard_id": "", "policy_ref": "",
219 "decision_id": str(p.get("decision_id") or ""),
220 "reason": ",".join(str(r) for r in (p.get("policy_refs") or [])),
221 "executed": bool(cid) and cid in executed})
222 rows.sort(key=lambda r: r["seq"])
223 return rows

228 def _declared_holes(events: list) -> list:
229 """合法空洞声明(**只读**):对账用 ``SEQ-GAP`` 判据。
230
231 2026-09-21 R14-9:判据**唯一来源** ``events.declared_ranges``。此前本处自写一份
232 且**漏掉** ``session.recovered.fixed`` 里的 ``"seq-holes:[…]"`` 词条 ⇒ repair
233 刚声明过的空洞在治理对账里仍被报成 ``SEQ-GAP``(**假一致性发现**:对账把一次
234 合法修复当成治理缺陷)。
235 """
236 out: list = []
237 for e in events:
238 out.extend(declared_ranges(e))

241 async def reconcile(self, *, session_id: str = "", ctx: Any = None,
242 emit: bool = False) -> list[str]:
243 """治理一致性对账(**默认纯只读**)。
244
245 **三类检查**(判据)：
246
247 - ``SEQ-GAP`` —— 事件 seq 中**未被声明区间覆盖**的空洞(复用既有
248 ``events.envelope.check_seq_gap``);
249 - ``NO-GUARD-EVENT`` —— 某 ``call_id`` 有 ``tool.result`` 但**无**同
250 ``call_id``(经 ``trace``)的 ``guard.evaluated``(``INV-04``);
251 - ``NO-RECEIPT-FOR-GRANT`` —— 某 ``approval.granted(approval_id=N)``
252 找不到对应 D2(``approval_ref=N``),**或**找到但其 ``decision_id``
253 无 ``receipt.emitted``(**INV-G2** 的运行时哨兵)。
254
255 ``CACHE-STALE`` **本版未实现**(登记 deferred debt:避免依赖
256 ``EvidenceCollector`` 内部结构)。
257
258 **写点(默认关)**：仅当 ``emit=True`` **且** findings 非空 **且** 会话在位,
259 才追加**一条既有** ``syscheck.fail``(不新增事件类型)。``findings`` 是
260 建议性事实,可由重跑复得,故该事件为**普通攒批**(非强同步)。
261
262 返回:排序后的 findings(**确定性**;空列表 = 一致);无会话 ⇒ ``[]`` 不抛。
263 """
264 events = self._events(ctx)
265 if session_id:
266 events = [e for e in events
267 if str(getattr(e, "session_id", "") or "") == session_id]
268 if not events:
269 return []
270 findings: list[str] = []
271
272 # ① SEQ-GAP
273 seqs = [int(getattr(e, "seq", 0) or 0) for e in events]
274 holes = check_seq_gap(seqs, self._declared_holes(events))
275 if holes:
276 findings.append("SEQ-GAP:" + ",".join(str(h) for h in holes))
277
278 # ② NO-GUARD-EVENT(执行了但没有规则级求值留痕)
279 executed = {str((getattr(e, "payload", None) or {}).get("call_id") or "")
280 for e in events
281 if getattr(e, "type", None) == "tool.result"}
282 guarded = {self._trace_call_id(e) for e in events
283 if getattr(e, "type", None) == "guard.evaluated"}
284 for cid in sorted(c for c in executed if c and c not in guarded):
285 findings.append(f"NO-GUARD-EVENT:call_id={cid}")
286
287 # ③ NO-RECEIPT-FOR-GRANT
288 d2_by_ref: dict[int, str] = {}
289 for e in events:
290 if getattr(e, "type", None) != "decision.issued":
291 continue
292 ref = (getattr(e, "payload", None) or {}).get("approval_ref")
293 if ref is not None:
294 d2_by_ref[int(ref)] = str(
295 (getattr(e, "payload", None) or {}).get("decision_id") or "")
296 receipt_dids = {str((getattr(e, "payload", None) or {})
297 .get("decision_id") or "")
298 for e in events
299 if getattr(e, "type", None) == "receipt.emitted"}
300 grants = {int((getattr(e, "payload", None) or {}).get("approval_id") or -1)
301 for e in events if getattr(e, "type", None) == "approval.granted"}
302 for aid in sorted(g for g in grants if g >= 0):
303 did = d2_by_ref.get(aid)
304 if not did or did not in receipt_dids:
305 findings.append(f"NO-RECEIPT-FOR-GRANT:approval_id={aid}")
306
307 findings.sort() # 确定性(顺序稳定)
308 if emit and findings:
309 sess = (getattr(ctx, "session", None) if ctx is not None
310 else self.session)
311 if sess is not None:
312 r = sess.append("syscheck.fail",
313 {"findings": list(findings),
314 "trigger": "governance.reconcile"},
315 actor="system")
316 if hasattr(r, "__await__"):
317 await r # 真实 SessionLog.append 为 async
318 return findings
