138def _bump_search_calls(ctx: Any) -> int:
139 """web.search 会话计数 +1,返回新计数值(spec: ctx.counters.bump("search_calls"))。
140
141 counters 三形态兼容(dict / 带 bump 的对象 / 属性形态);均不可用 → debug 日志
142 回落 0 读数(偏离 3,零读数=未超限放行,沿用 scope 先例)。计数键 search_calls
143 会话级,新会话清零(次数闸 ≤20 次/会话)。
144 """
145 c = getattr(ctx, "counters", None)
146 if c is None:
147 log.debug("ctx.counters 未接线:搜索次数闸失读(回落 0)")
148 return 0
149 if isinstance(c, dict):
150 nxt = int(c.get("search_calls", 0)) + 1
151 c["search_calls"] = nxt
152 return nxt
153 bump = getattr(c, "bump", None)
154 if callable(bump):
155 try:
156 out = bump("search_calls", 1) # (key, delta) 形态
157 except TypeError:
158 out = bump("search_calls") # (key) 形态(内部自增)
159 if isinstance(out, int):
160 return out
161 cur = getattr(c, "search_calls", None)
162 return int(cur) if cur is not None else 1
163 cur = int(getattr(c, "search_calls", 0) or 0) + 1 # 属性形态
164 try:
165 setattr(c, "search_calls", cur)
166 except Exception: # noqa: BLE001 只读对象:降级
167 log.debug("counters 属性形态不可写 search_calls")
168 return cur

172async def web_search(args: dict, ctx: Any) -> dict:
173 """web.search(F037):次数闸 → F016 取 key → SearchBackend → ≤8KB 摘要返回。
174
175 目标域固定为已授权搜索服务端点(ctx.search_backend 装配注入),不受逐调用
176 allowlist 约束(职责 1 区分依据);受次数闸与凭据闸约束。异常全走 raise_code。
177 """
178 used = _bump_search_calls(ctx) # 会话计数 +1(先计数后判断,伪码序)
179 limit = _cfg_read(ctx, "security.network.web_search_per_session",
180 SEARCH_PER_SESSION)
181 try:
182 limit = int(limit)
183 except (TypeError, ValueError):
184 limit = SEARCH_PER_SESSION
185 if limit == 0 or used > limit: # 0=禁用;超闸零外发拒绝
186 raise_code("GRD-401", reason="search-quota", used=used, limit=limit,
187 policy_ref="PRD F037 / CFG security.network.web_search_per_session",
188 advice=f"本会话搜索次数已达上限({limit});"
189 "改用 web.fetch 直抓已知 URL 或新开会话")
190 backend = getattr(ctx, "search_backend", None)
191 key = _get_secret(ctx, "SEARCH_API_KEY") # F016 单一读取口
192 needs_key = True if backend is None else bool(
193 getattr(backend, "requires_key", True))
194 if needs_key and key is None:
195 raise_code("CRED-701", hint="配置 SEARCH_API_KEY(F016)")
196 if backend is None or not callable(getattr(backend, "search", None)):
197 raise_code("CYC-999", module="tool_web",
198 hint="ctx.search_backend 未接线:搜索服务后端缺失"
199 "(装配 bug),fail-closed 拒绝外发")

203 except (TypeError, ValueError):
204 top_k = 5 # 契约层已拦;直调路径兜底
205 hits = await backend.search(query=args["query"], top_k=top_k, key=key)

209 max_chars = int(max_chars)
210 except (TypeError, ValueError):
211 max_chars = SEARCH_RESULT_CHARS

215def _summarize_search(hits: list, *, max_chars: int) -> dict:
216 """搜索结果摘要截断(spec 伪码权威,偏离 5 字段容错)。
217
218 逐条标题/URL/摘要压进 max_chars(8000)预算,单条字段各自截断(title 200 /
219 url 500 / snippet 600)防单条长文独占预算;装不下整条即停并置 truncated;
220 total 恒 = 后端命中总数(含被预算截掉者)。返回 _SearchOut 字典。
221 """
222 out: list[dict] = []
223 budget = int(max_chars)
224 truncated = False
225 for h in hits or []:
226 if not isinstance(h, dict): # 脏数据防御:非 dict 单条跳过
227 continue
228 item = {"title": str(h.get("title", ""))[:200],
229 "url": str(h.get("url", ""))[:500],
230 "snippet": str(h.get("snippet", ""))[:600]}
231 cost = sum(len(v) for v in item.values())
232 if budget - cost < 0: # 装不下:截断留痕
233 truncated = True
234 break
235 out.append(item)
236 budget -= cost
237 return {"results": out, "truncated": truncated, "total": len(hits or [])}

241class _DuckDuckGoParser(html.parser.HTMLParser):
242 """Small parser for DuckDuckGo's no-JS HTML result page."""
243
244 def __init__(self) -> None:
245 super().__init__(convert_charrefs=True)
246 self.hits: list[dict] = []
247 self._mode: Optional[str] = None
248 self._buf: list[str] = []
249 self._href = ""
250
251 @staticmethod
252 def _is(attrs: list, needle: str) -> bool:
253 return needle in " ".join(
254 str(v) for k, v in attrs if k in ("class", "id") and v)
255
256 def handle_starttag(self, tag: str, attrs: list) -> None:
257 if tag != "a":
258 return
259 if self._is(attrs, "result__a"):
260 href = next((str(v) for k, v in attrs if k == "href" and v), "")
261 parsed = urlsplit(href)
262 qs = parse_qs(parsed.query)
263 href = unquote((qs.get("uddg") or [href])[0])
264 self._mode, self._buf, self._href = "title", [], href
265 elif self._is(attrs, "result__snippet"):
266 self._mode, self._buf = "snippet", []
267
268 def handle_data(self, data: str) -> None:
269 if self._mode:
270 self._buf.append(data)
271
272 def handle_endtag(self, tag: str) -> None:
273 if tag != "a" or not self._mode:
274 return
275 text = " ".join("".join(self._buf).split())
276 if self._mode == "title" and text and self._href:
277 self.hits.append({"title": text, "url": self._href,
278 "snippet": ""})
279 elif self._mode == "snippet" and text and self.hits:
280 if not self.hits[-1]["snippet"]:
281 self.hits[-1]["snippet"] = text
282 self._mode, self._buf, self._href = None, [], ""

311class BingRssBackend:
312 """No-key Bing RSS backend (stable XML, no HTML scraping)."""
313
314 requires_key = False
315
316 def __init__(self, endpoint: str = "https://cn.bing.com/search") -> None:
317 self.endpoint = endpoint

319 async def search(self, *, query: str, top_k: int, key: Optional[str] = None) -> list:
320 url = f"{self.endpoint}?format=rss&{urlencode({'q': query})}"
321 resp = await _http_once(None, url, max_bytes=RAW_HTML_MAX)

333 link = (item.findtext("link") or "").strip()

335 if title or link or snippet:
336 out.append({"title": title, "url": link, "snippet": snippet})
337 if len(out) >= max(1, int(top_k)):
338 break

374 "truncated": True, # 上下文只放摘要 + 引用

408 """单跳原始响应契约(传输 seam 返回型):状态码/头/字节/终址。"""
409
410 status: int
411 headers: dict
412 raw: bytes
413 final_url: str

416async def _http_once(ctx: Any, url: str, *, max_bytes: int) -> _RawResponse:
417 """单跳 GET(不跟随重定向;传输 seam,偏离 1)——真实路径走 httpx。
418
419 连接 10s/总 15s 超时(httpx.Timeout(15, connect=10),F038);3xx/4xx/5xx 只取
420 头不读体(重定向/错误响应无需下载);2xx 限量读流,读满 max_bytes 即 PERS-223
421 (超 2MB 拒,防内存/带宽滥用);httpx 超时/网络错误映射 TLB-805 明确原因。
422 测试注入面:monkeypatch 本函数即可(mock transport,零 HTTP 外发,T-SEC-07)。
423 """
424 timeout = httpx.Timeout(HTTP_TIMEOUT, connect=CONNECT_TIMEOUT)
425 try:
426 # 墙钟总闸:httpx.Timeout 是"每步"超时,慢滴流服务器(m每步都 <15s)可无限拖延;
427 # asyncio.timeout 给整次抓取一个硬上限(P3:tool_web 每步超时→墙钟)。
428 async with asyncio.timeout(HTTP_TIMEOUT):
429 async with httpx.AsyncClient(timeout=timeout,
430 follow_redirects=False) as client:
431 async with client.stream("GET", url) as resp: # 手动控跳
432 headers = {str(k).lower(): str(v)
433 for k, v in resp.headers.items()}
434 if resp.status_code >= 300: # 重定向/错误:不读体
435 return _RawResponse(resp.status_code, headers, b"",
436 str(resp.url))
437 raw = b""
438 async for chunk in resp.aiter_bytes(): # 限量读流
439 raw += chunk
440 if len(raw) > max_bytes: # 超原始上限 → 停并拒
441 raise_code("PERS-223", url=url, got=len(raw),
442 max=max_bytes,
443 advice="页面超过 2MB 抓取上限,换更小目标")
444 return _RawResponse(resp.status_code, headers, raw,
445 str(resp.url))
446 except PyHError:
447 raise
448 except (TimeoutError, httpx.TimeoutException):
449 raise_code("TLB-805", url=url, stage="timeout",
450 advice="抓取超时(15s 墙钟)")
451 except httpx.HTTPError as e:
452 raise_code("TLB-805", url=url, stage="transport",
453 cause=e, advice=f"网络失败:{type(e).__name__}")
454

456def _header(resp: _RawResponse, name: str) -> Optional[str]:
457 """大小写不敏感头读取(fake/真实传输头键形态统一收敛)。"""
458 target = name.lower()
459 for k, v in (resp.headers or {}).items():
460 if str(k).lower() == target:
461 return str(v)
462 return None

465async def http_get(url: str, *, max_bytes: int, ctx: Any,
466 redirects: int = REDIRECT_MAX) -> tuple[bytes, str]:
467 """受限 HTTP 抓取(F038 传输层,spec 伪码权威 + 偏离 1/2)。
468
469 跟随重定向但逐跳对目标域名重验 allowlist(防白名单域名跳黑名单域名),超
470 跳数/域逃逸即拒(GRD-401 POL-NET-1 / TLB-805);4xx/5xx 抛结构化错误带
471 http_status;成功返回 (raw_bytes, final_url=重定向终址)。
472 """
473 if redirects <= 0: # 偏离 2:预算耗尽即停(≤3 跳)
474 raise_code("TLB-805", url=url, redirects=REDIRECT_MAX,
475 advice=_MAX_REDIRECT_ADVICE)
476 if _domain_allowed(url, ctx) is None: # 每跳重验域(含首跳,第二道闸)
477 raise_code("GRD-401", reason="POL-NET-1", url=url,
478 advice="重定向目标域未放行,已停止")
479 resp = await _http_once(ctx, url, max_bytes=max_bytes)
480 status = resp.status
481 if 300 <= status < 400: # 重定向:手动跟随(urljoin 补相对)
482 loc = _header(resp, "location")
483 if not loc:
484 raise_code("TLB-805", url=url, http_status=status,
485 advice="重定向无 Location")
486 return await http_get(urljoin(url, loc), max_bytes=max_bytes,
487 ctx=ctx, redirects=redirects - 1)
488 if status >= 400: # 4xx/5xx:明确原因,工具内不自动重试
489 raise_code("TLB-805", url=url, http_status=status,
490 advice=f"HTTP {status},可稍后重试或换源")
491 return resp.raw, resp.final_url # 2xx:原文 + 终址
