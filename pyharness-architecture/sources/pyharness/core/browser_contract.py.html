25def allowed_origin(value):
26 if (type(value) is not str or not value or len(value) > 300
27 or '\\' in value or any(ch.isspace() or ord(ch) < 32 for ch in value)):
28 raise ValueError('invalid_browser_origin')
29 parsed = urlsplit(value)
30 if (parsed.scheme not in ('http', 'https') or not parsed.hostname
31 or parsed.username or parsed.password or parsed.query or parsed.fragment
32 or parsed.path not in ('', '/') or parsed.hostname.endswith('.')):
33 raise ValueError('invalid_browser_origin')
34 port = parsed.port
35 if port == 0:
36 raise ValueError('invalid_browser_origin')
37 host = parsed.hostname.encode('idna').decode('ascii').lower()
38 if ':' in host:
39 host = '[' + host + ']'
40 default = 80 if parsed.scheme == 'http' else 443
41 return parsed.scheme + '://' + host + (':' + str(port) if port and port != default else '')
