212 async def _queue_for(self, sid: str, log_: Any) -> TaskQueue:
213 """per-session TaskQueue(runner = per-session 真实引擎;真 LLM 已注册)。"""
214 lock = self._queue_locks.setdefault(sid, asyncio.Lock())
215 async with lock:
216 q = self._queues.get(sid)
217 if q is None:
218 runner = await self._engine_runner_for(sid, log_)
219 q = TaskQueue(session=log_, runner=runner,
220 **queue_kwargs_of(getattr(self.ctx, "settings", None)))
221 self._queues[sid] = q
222 spine = self._engines.get(sid)
223 if spine is not None:
224 from pyharness import engine as _eng
225 await _eng.activate_orchestration(spine, task_queue=q)
226 return q

228 async def _engine_runner_for(self, sid: str, log_: Any) -> Any:
229 """per-session 引擎 runner(懒装配):复用 manager 已 open 的 log_/bus,
230 补 loop/scope/llm/空工具表 + 总线落盘订阅 → engine.make_runner。
231
232 无可用 LLM 凭据时在落 user.message 前失败,避免留下“有消息无任务”的半提交。
233 """
234 spine = self._engines.get(sid)
235 from pyharness import engine as _eng
236 from pyharness.engine import make_runner as _eng_make_runner
237 cfg = self.ctx.settings
238 _eng.register_default_llm(cfg) # 幂等;无 key → CRED-701
239 if spine is None:
240 store = getattr(self._surface_mgr(), "_stores", {}).get(sid)
241 spine = _eng.build_runner_components(
242 cfg, log_=log_, bus=getattr(self.ctx, "bus", None),
243 sessions_dir=_sessions_dir_of(self.ctx), store=store,
244 attach_persistence=False, # manager 已订阅落盘;重复订=双写卡死
245 # GAP-11:Web 桌面壳 = human 通道 "desktop"(框架侧声明,不信任
246 # 前端自报);不下沉到 spine 则该会话每次工具授权 APR-503。
247 channel="desktop")
248 self._engines[sid] = spine
249 if not getattr(spine, "_plugins_ready", False):
250 await _eng._preload_plugins(spine) # 示例插件预载(util.now 会话可用)
251 spine._plugins_ready = True
252 # 审批接线:engine ApprovalProvider → ctx.approval(桌面裁决端点
253 # _provider_owning 经 ctx.approval 兜底定位;否则弹窗 A/B 打来 APR-503)
254 if getattr(self.ctx, "approval", None) is None:
255 self.ctx.approval = spine.approval
256 self.ctx._approval_owner_sid = sid # 归属记账:_approval_for 据此判能否复用
257 if getattr(self.ctx, "guard", None) is None:
258 self.ctx.guard = spine.guard
259 self.ctx._guard_owner_sid = sid
260 # 预算门面:按会话记账归属;/api/budget/{sid} 主路径读本会话 spine.budget,
261 # ctx.budget 仅作 fallback 且需 owner==sid(防多会话共享 ctx 串场)
262 self.ctx.budget = spine.budget
263 self.ctx._budget_owner_sid = sid
264 return _eng_make_runner(spine)

299 def _approval_for(self, sid: str, log_: Any) -> ApprovalProvider:
300 """per-session ApprovalProvider;ctx.approval(装配注入,单会话形态)优先。
301
302 多会话各 provider 以 bus=None 构造(偏离 1 注):裁决结果事件仍由 sess.append
303 经会话总线分发到 hub(SSE 可见),provider 侧按 approval.py 偏离 3 直接消费
304 on_verdict——避免共享总线把 A 会话裁决扇到 B provider 触发跨会话 APR-503
305 system.error 噪音(approval.on_verdict 未知 id 兜底路径)。
306 """
307 provider = self._approvals.get(sid)
308 if provider is None:
309 ctx_approval = getattr(self.ctx, "approval", None)
310 owner = getattr(self.ctx, "_approval_owner_sid", None)
311 # 仅当共享 ctx.approval 确属本会话(owner==sid)或未记归属才复用;否则
312 # 按本会话 log_ 建 per-session provider——防第二会话拿到第一会话的审批通道。
313 if ctx_approval is not None and owner in (None, sid):
314 return ctx_approval # 装配注入的全会话 provider(单会话形态)
315 provider = ApprovalProvider(
316 session=log_, bus=None, config=_cfg_of(self.ctx),
317 # R12-3:队列联动(惰性取值 ⇒ 与构造序无关)
318 queue_getter=lambda: self._queues.get(sid))
319 self._approvals[sid] = provider
320 return provider

336 def publish_operator_token(self) -> Optional[Path]:
337 """把**本进程生成**的操作者令牌落盘,供浏览器/外部客户端取用。
338
339 ``R31-2`` 的配套。引导页按 ADR-023 不再向无凭证调用方发放令牌,而
340 ``shell.web.token`` 未配置时令牌是**进程随机**且从不打印 ⇒ 浏览器用户无从
341 取得,该入口等于关闭(实测:改动后浏览器访问 401 且无处可查)。故生成态令牌
342 落盘到 ``<storage.root>/web.token``(与租户令牌 ``tenants/<t>/token`` 同口径)。
343 **不写日志、不进 API 响应**;调用方只应记录**路径**。
344
345 **权限(2026-09-21 UI 验收实测更正)**:代码调用 ``chmod(0o600)``,但
346
347 - **POSIX**:生效,文件为 ``0600``。
348 - **Windows**:``chmod`` **不生效** —— 实测 ``stat.S_IMODE`` 仍为 ``0o666``
349 (NTFS 不存 POSIX 权限位)。实际可读面由**父目录 ACL 继承**决定:
350 ``%USERPROFILE%`` 下的 ``.pyharness`` 默认只含当前用户 + SYSTEM +
351 Administrators(等效"仅属主");若该目录被外部工具**额外授权**给别的组(本机实测存在
352 一条 ``CodexSandboxUsers:(OI)(CI)(RX)``),本文件会**继承**该授权。
353 故 Windows 上**不得**声称"0600";要收紧须显式改 ACL(未做,见 ADR-023
354 的已知边界)。
355
356 配置了 ``shell.web.token`` 时**不落盘** —— 用户已知该值,没必要多一份副本。
357 返回落盘路径;非生成态 / 取不到 ``storage.root`` ⇒ ``None``。
358 """
359 if not (self._api_token and self._api_token_generated):
360 return None
361 root = self._storage_root()
362 if root is None:
363 return None
364 p = root / "web.token"
365 try:
366 root.mkdir(parents=True, exist_ok=True)
367 tmp = p.with_name(p.name + ".tmp")
368 tmp.write_text(self._api_token, encoding="utf-8")
369 tmp.replace(p) # pathlib 原子替换
370 p.chmod(stat.S_IRUSR | stat.S_IWUSR) # POSIX 0600;Windows 无效(见 docstring)
371 except OSError:
372 return None
373 return p
