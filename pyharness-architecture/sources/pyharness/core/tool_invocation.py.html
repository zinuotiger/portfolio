19 async def _run_provider(self, defn: Any, args: dict, ctx: Any) -> Any:
20 """Provider 执行(runner):async handle 直接协程;同步 handle 经独立单工线程池。
21
22 同步函数用**每次调用独立**的 ``ThreadPoolExecutor(max_workers=1)`` 承载
23 (不阻塞事件循环,spec 关3);async handler 走事件循环协作(不占线程池,偏离 7)。
24
25 超时(外层 wait_for 掐断)/取消掐断 → 内部 await 抛 CancelledError → 本函数
26 以 ``shutdown(wait=False)`` **驱逐**该池(线程把副作用在后台跑完),并保留引用
27 防 GC 半途杀线程——该池绝不被复用于后续调用,消除"超时线程回池 + mark_idle
28 后并发重入 → 重叠副作用"(Python 无法安全 kill 运行中线程,只能不复用)。
29 正常完成则 ``shutdown(wait=True)`` 立即收口(线程已结束)。见偏离说明 13。
30 """
31 runtime = getattr(ctx, 'platform_runtime', None)
32 if runtime is not None:
33 async def original():
34 return await self._run_original_provider(defn, args, ctx)
35 return await runtime.execute(defn.name, args, ctx, original)
36 return await self._run_original_provider(defn, args, ctx)

38 async def _run_original_provider(self, defn: Any, args: dict, ctx: Any) -> Any:
39 handle = self._provider_handle(defn.name)
40 if inspect.iscoroutinefunction(handle):
41 return await handle(args, ctx) # wait_for 在外层掐断
42 loop = asyncio.get_running_loop()
43 pool = self._new_provider_pool(defn.name)
44 try:
45 result = await loop.run_in_executor(pool, handle, args, ctx)
46 except BaseException: # 超时/取消/异常:驱逐不复用
47 pool.shutdown(wait=False, cancel_futures=True)
48 self._zombie_pools.add(pool)
49 self._prune_zombies()
50 raise
51 else:
52 pool.shutdown(wait=True) # 正常完成:线程已结束
53 self._zombie_pools.discard(pool)
54 return result

56 def _prune_zombies(self) -> None:
57 """僵尸池裁剪:超时产生的池,线程跑完后自会空转结束;仅当僵尸池累积超过
58 上限时,丢弃其中线程已全部结束的池(仍在跑的保留引用,防 GC 半途回收)。"""
59 if len(self._zombie_pools) < _ZOMBIE_POOL_LIMIT:
60 return
61 keep: set[ThreadPoolExecutor] = set()
62 for pool in self._zombie_pools:
63 try:
64 threads = getattr(pool, "_threads", ())
65 alive = any(getattr(t, "is_alive", lambda: False)()
66 for t in threads)
67 except Exception: # noqa: BLE001 线程查询异常:保守保留
68 alive = True
69 if alive:
70 keep.add(pool)
71 self._zombie_pools = keep

83 """取消在途 Provider 任务(协作式,F025):掐断 wait_for → execute 内
84 CancelledError 分支写 partial + re-raise(已发生副作用不假装回滚)。

86 无在途任务 → 返回 False(幂等)。partial 事件由 execute 的取消处理分支
87 落盘,本函数只负责触发取消(单事件循环单在途,阶段 1 agent-loop 串行)。
88 """
89 fut = self._active
90 if fut is None or fut.done() or fut.cancelling():
91 return False
92 fut.cancel()
93 return True
