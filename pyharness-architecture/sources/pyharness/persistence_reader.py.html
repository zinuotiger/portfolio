15def rotated_segment_paths(main_path: Any) -> list:
16 """该会话的轮转段 ``{sid}.{n}.jsonl``(n 升序)—— 真源的一部分,非独立会话。"""
17 main_path = Path(main_path)
18 pat = re.compile(rf"^{re.escape(main_path.stem)}\.(\d+)\.jsonl$")
19 segs = [p for p in main_path.parent.glob(f"{main_path.stem}.*.jsonl")
20 if pat.match(p.name)]
21 return sorted(segs, key=lambda p: int(pat.match(p.name).group(1)))

23def session_data_paths(main_path: Any) -> list:

32def _iter_replay(paths: list, quarantine: set, *,
33 tolerate_unreadable: bool = False) -> Iterator[Envelope]:
34 """会话真源(按 ``paths`` 给定顺序的多文件)逐事件回放(**唯一回放实现**)。
35
36 空行跳过;坏行(解码失败/JSON 非法/信封校验失败)行号记入 ``quarantine`` 并**继续**
37 —— 回放 = 恢复 = 审计同一路径,绝不因单行损坏中断(§3.8)。行号按文件从 1 起
38 (多文件时可能重号;隔离判定以 repair 主文件扫描为准)。
39
40 ``tolerate_unreadable``:整个文件打不开(权限/被占/是目录)时,``False`` = 上抛
41 (``SessionStore`` 恢复语义:读不到主文件就是读不到基线),``True`` = 告警跳过
42 (``SessionReader`` 只读派生语义:离线命令不该因单个会话不可读而整体失败)。
43
44 2026-09-21 R14-5:从 ``SessionStore.replay`` 抽出,使**只读**读取方
45 (``SessionReader``)复用同一实现 —— 此前只读命令只能借 ``open_store`` 读,而它会取
46 INV-07 写锁 ⇒ 把所有会话锁住。
47 """
48 for p in paths:
49 try:
50 fh = open(p, "rb")
51 except FileNotFoundError:
52 continue # 段/主文件尚未存在:跳过(原语义)
53 except OSError as e:
54 if not tolerate_unreadable:
55 raise
56 log.warning("PERS-201 域:回放源不可读,跳过 file=%s why=%s", p, e)
57 continue
58 with fh:
59 for no, raw in enumerate(fh, 1):
60 line = raw.rstrip(b"\r\n") # 容忍 \r\n 遗留(Windows 兼容)
61 if not line:
62 continue # 空行跳过(轮转残留容忍)
63 try:
64 text = line.decode("utf-8")
65 env = Envelope.model_validate_json(text) # 信封一次校验
66 # 读侧 payload 强校验(P1-4):写侧拒坏 payload、读侧曾漏网——
67 # 缺字段/错类型的行"合法"通过回放,到 reducer/repair 深处才
68 # KeyError 崩。校验失败落入下方 PERS-201 隔离(与写侧同口径)。
69 canonical = validate_payload(env.type, env.payload)
70 if canonical is not env.payload:
71 env = env.model_copy(update={"payload": canonical})
72 yield env
73 except Exception: # noqa: BLE001
74 # 解码坏块/JSON 非法/信封校验失败统一记 PERS-201 隔离
75 quarantine.add(no) # 中部坏行隔离(记跳不中断)
76 log.warning("PERS-201 域:坏行隔离(记跳不中断,删否经 repair) "
77 "file=%s line_no=%s", p, no)

79class SessionReader:
80 """**只读**会话回放源:**不取**会话写锁、不开写句柄、不建文件(R14-5)。
81
82 与 ``SessionStore`` 的差价只在"要不要单写者锁":``SessionStore`` 可能写(append/
83 flush/repair),故必须持 INV-07 锁;只读取真源的命令(如 CLI ``search`` 建索引回放源)
84 却因此把所有会话一并锁住 ⇒ 其他进程开会话撞 PERS-202(实测:`search` 运行期间
85 并发 open 交替得到 OPEN_OK / BLOCKED PERS-202)。
86
87 只暴露 ``replay()``(与 ``SessionStore`` 同形,可直接作 session_query 的注入源);
88 **无任何写面**。并发写者正在追加时可能读到半行 ⇒ 按坏行跳过(读侧不变式)。
89 """

98 """同 ``SessionStore.replay``(轮转段升序 + 主文件;坏行记跳不中断)。

112def _last_newline_before(f, end: int) -> Optional[int]:
113 """在 [0, end) 内从后向前分块查找最近 \\n 的字节偏移;无换行 → None。
114
115 detect_truncation 的半行超窗兜底:尾部窗口内无换行不代表文件前部无完整行
116 ——>4096B 半行曾致返回 0 触发 repair 把含完整数据的文件整段截断(数据事故),
117 故需扩大到窗口前区域定位最后完整行的切点。分块读取,避免整文件载入内存。
118 """
119 pos = end
120 while pos > 0:
121 step = min(pos, _TAIL_WINDOW)
122 pos -= step
123 f.seek(pos)
124 i = f.read(step).rfind(b"\n")
125 if i >= 0:
126 return pos + i
127 return None

129def detect_truncation(path: Path) -> Optional[int]:
130 """截断检测(只读):文件非空且末字节不是 \\n → 返回半行起始字节偏移;否则 None。
131
132 崩溃遗留的尾部半行 = 未完成写的事实,不假装发生:open/回放不修,repair 先备份
133 再截断(F060)。实现按 spec 注释意图读取尾部窗口——伪码 seek(0,2) 后未回退
134 read 为空,实际以 seek(-window, 2) 读窗口再定位最后一个换行;窗口内无换行
135 (半行超窗)时扩大到窗口前区域,仅当整文件都无换行(单条半行文件)才返回 0。
136 """
137 try:
138 if not path.exists() or path.stat().st_size == 0:
139 return None
140 except OSError:
141 return None
142 with open(path, "rb") as f:
143 f.seek(0, 2)
144 size = f.tell()
145 window = min(size, _TAIL_WINDOW)
146 f.seek(-window, 2) # 回退读尾部窗口(修正伪码遗漏)
147 tail = f.read(window)
148 if tail.endswith(b"\n"):
149 return None # 完整行结尾:无截断
150 last_nl = tail.rfind(b"\n")
151 if last_nl >= 0:
152 # 偏移 = 文件大小 - 窗口内最后一个换行之后的字节数(字节级切点,保完整行)
153 return size - (len(tail) - last_nl - 1)
154 # 窗口内无换行:半行超窗。扩大到窗口前区域定位最后完整行;整文件仍无
155 # 换行 = 真·单条半行文件,保留 offset=0 语义;否则禁止误判为 0。
156 prev = _last_newline_before(f, size - window)
157 return (prev + 1) if prev is not None else 0

159def _read_clean(path: Path) -> list[tuple[int, str]]:
160 """整文件解码为 (行号, 文本) 列表;编码级坏块 → 抛 UnicodeDecodeError。
161
162 repair 中部坏行扫描专用:先整读保证\"中部编码坏块 = 不可修复\"可整体暴露
163 (逐行容错会把编码坏块降级成单行隔离,违背 spec PERS-202 处置表)。
164 """
165 out: list[tuple[int, str]] = []
166 with open(path, "rb") as f:
167 for no, raw in enumerate(f, 1):
168 line = raw.rstrip(b"\r\n")
169 if not line:
170 continue
171 out.append((no, line.decode("utf-8"))) # 坏块在此处整体抛出
172 return out
