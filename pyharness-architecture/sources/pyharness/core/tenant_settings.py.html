66def _dpapi(data: bytes, *, protect: bool) -> bytes:
67 if os.name != "nt":
68 raise OSError("DPAPI unavailable")
69 crypt32 = ctypes.windll.crypt32
70 kernel32 = ctypes.windll.kernel32
71 fn = crypt32.CryptProtectData if protect else crypt32.CryptUnprotectData
72 in_blob, keepalive = _blob(data)
73 out_blob = _DataBlob()
74 description = ctypes.c_wchar_p("PyHarness tenant secret")
75 desc_arg = ctypes.byref(description) if protect else None
76 ok = fn(ctypes.byref(in_blob), desc_arg, None, None, None,
77 0x01, ctypes.byref(out_blob)) # CRYPTPROTECT_UI_FORBIDDEN
78 del keepalive
79 if not ok:
80 raise OSError("DPAPI operation failed")
81 try:
82 return ctypes.string_at(out_blob.pbData, out_blob.cbData)
83 finally:
84 kernel32.LocalFree(out_blob.pbData)

128 def api_token(self, tenant_id: str) -> str:
129 """读取本租户令牌;不存在则**生成**(32 字节 hex,原子写 + ``chmod 0600``,后者仅 POSIX 生效)。
130
131 R31-2:per-tenant 鉴权的凭证。**创建路径 = 引导页持操作者令牌时铸/取**
132 (``DesktopApp._index_page``);运维也可直接读取该文件交给外部客户端。
133 **不落日志、不进 API 响应**。
134
135 权限:POSIX 上为 0600;**Windows 上 ``chmod`` 不生效**,可读面由父目录 ACL
136 继承决定(见 ``DesktopApp.publish_operator_token`` 的实测说明)。
137
138 ``default`` 租户**不使用**本机制(沿用全局 ``shell.web.token``,保兼容),
139 故调用方只对非 default 租户取用。
140 """
141 p = self.token_path(tenant_id)
142 try:
143 if p.is_file():
144 tok = p.read_text(encoding="utf-8").strip()
145 if tok:
146 return tok
147 except OSError:
148 pass
149 tok = secrets.token_hex(32)
150 try:
151 p.parent.mkdir(parents=True, exist_ok=True) # 生成=显式行为,可建目录
152 except OSError:
153 pass
154 self._atomic_write(p, tok.encode("utf-8"))
155 try:
156 os.chmod(p, stat.S_IRUSR | stat.S_IWUSR) # POSIX 0600;Windows 无效(见 docstring)
157 except OSError:
158 pass
159 return tok

161 def verify_token(self, tenant_id: str, given: str) -> bool:
162 """常数时间比对;未出示 / 无令牌文件 / 不匹配 → ``False``(**只读**)。
163
164 刻意**不调用** :meth:`api_token`:校验发生在**未认证**的请求路径上,若在那里
165 "读不到就生成",任意人报一个租户名即可凭空造出令牌与目录(且能读到自己的
166 Set-Cookie),整条授权即为空转。故校验只读,生成只走壳/运维。

169 return False
170 p = self.token_path(tenant_id)

172 if not p.is_file():
173 return False

178 return False

186 @staticmethod
187 def _atomic_write(path: Path, data: bytes) -> None:
188 fd, name = tempfile.mkstemp(dir=path.parent, prefix=f'.{path.name}-')
189 tmp = Path(name)
190 try:
191 # mkstemp creates exclusively with mode 0600; verify before bytes exist.
192 if os.name != 'nt':
193 os.fchmod(fd, 0o600)
194 if stat.S_IMODE(os.fstat(fd).st_mode) != 0o600:
195 raise PermissionError('private file permissions unavailable')
196 remaining = memoryview(data)
197 while remaining:
198 n = os.write(fd, remaining)
199 if n <= 0:
200 raise OSError('short private file write')
201 remaining = remaining[n:]
202 os.fsync(fd)
203 closing_fd, fd = fd, -1
204 os.close(closing_fd)
205 os.replace(tmp, path)
206 except BaseException as primary:
207 if fd != -1:
208 try:
209 os.close(fd)
210 except BaseException as secondary:
211 primary.add_note(f'private descriptor cleanup failed: {type(secondary).__name__}')
212 try:
213 tmp.unlink(missing_ok=True)
214 except OSError as secondary:
215 primary.add_note(f'private temporary cleanup failed: {type(secondary).__name__}')
216 raise

249 def _load_secrets(self, tenant_id: str) -> dict[str, str]:
250 doc = self._load_doc(tenant_id)
251 secret_path = self._secret_path(tenant_id, doc)
252 if not secret_path.exists():
253 if doc.get('secret_generation') or any(p.get('has_api_key') and not p.get('api_key_ref')

267 advice="租户密钥文件格式非法")

276 def _commit(self, tenant, doc, values):
277 old = self._secret_path(tenant, self._load_doc(tenant))
278 name = f'model-secrets-{uuid.uuid4().hex}.bin'
279 new = self.tenant_dir(tenant) / name
280 self._save_secrets(tenant, values, path=new)
281 doc['secret_generation'] = name
282 try:
283 self._save_doc(tenant, doc) # sole commit point; old generation unchanged
284 except BaseException as primary:
285 try:
286 new.unlink(missing_ok=True)
287 except OSError as secondary:
288 primary.add_note(f'uncommitted secret cleanup failed: {type(secondary).__name__}')
289 raise
290 pending = False
291 candidates = [old, *self.tenant_dir(tenant).glob('model-secrets-*.bin')]
292 for retired in set(candidates):
293 if retired == new:
294 continue
295 if retired.name != 'model-secrets.bin' and not re.fullmatch(r'model-secrets-[a-f0-9]{32}\.bin', retired.name):
296 continue
297 try:
298 if retired.is_symlink():
299 pending = True
300 continue
301 retired.unlink(missing_ok=True)
302 except OSError:
303 pending = True
304 if pending:
305 log.warning('retired secret generation cleanup pending')
306 return pending

489 @_transactional
490 def set_service_secret(self, tenant_id: str, service_id: str, endpoint: str, value: str) -> str:
491 """Write-only service credential, using the existing atomic secret store."""
492 tenant = normalize_tenant_id(tenant_id)
493 if not re.fullmatch(r'(?:(?:remote_rag|feishu|wechat):[a-z][a-z0-9_]{1,40}|wechat_context:[0-9a-f]{64})', service_id):
494 raise_code('CRED-703', reason='invalid_service_id')
495 if not isinstance(value, str) or len(value) > 8192 or any(c in value for c in '\r\n'):
496 raise_code('CRED-703', reason='invalid_service_secret')
497 doc = self._load_doc(tenant)
498 values = self._load_secrets(tenant)
499 key = 'service:'+service_id
500 if value:
501 values[key] = value
502 doc.setdefault('service_credentials', {})[service_id] = {
503 'endpoint': endpoint, 'credential_revision': uuid.uuid4().hex}
504 else:
505 values.pop(key, None)
506 doc.setdefault('service_credentials', {}).pop(service_id, None)
507 self._commit(tenant, doc, values)
508 return 'service:'+tenant+':'+service_id if value else ''

510 def resolve_service_secret(self, ref: str, tenant_id: str, endpoint: str) -> str:
511 prefix = 'service:'+normalize_tenant_id(tenant_id)+':'
512 if not isinstance(ref, str) or not ref.startswith(prefix):
513 raise_code('CRED-703', reason='service_secret_owner_mismatch')
514 service_id = ref[len(prefix):]
515 with self._transaction(tenant_id):
516 record = self._load_doc(tenant_id).get('service_credentials', {}).get(service_id)
517 if not record or record.get('endpoint') != endpoint:
518 raise_code('CRED-703', reason='service_secret_endpoint_mismatch')
519 value = self._load_secrets(tenant_id).get('service:'+service_id)
520 if not value:
521 raise_code('CRED-701', reason='service_secret_missing')
522 return value

524 def resolve_ref(self, ref: str, *, binding=None) -> str:
525 parts = str(ref or "").split(":", 2)
526 if len(parts) != 3 or parts[0] != "tenant":
527 raise_code("CRED-701", ref=ref, reason="tenant_ref")
528 tenant = normalize_tenant_id(parts[1])
529 profile_id = normalize_profile_id(parts[2])
530 with self._transaction(tenant):
531 profile = self._validated_profile(tenant, profile_id, binding=binding)
532 values = self._load_secrets(tenant)
533 value = str(values.get(profile_id) or '')
534 if value:
535 return value
536 fallback = str(profile.get('api_key_ref') or '')
537 if fallback:
538 from pyharness.core.llm import resolve_secret_ref
539 return resolve_secret_ref(fallback)
540 raise_code('CRED-701', reason='missing', advice='模型档案未配置 API Key')

542 def _validated_profile(self, tenant, profile_id, *, binding=None):
543 profile = next((p for p in self._load_doc(tenant).get("profiles", [])
544 if str(p.get("id")) == profile_id), None)
545 if profile is None:
546 raise_code('CRED-701', reason='missing_profile')
547 fallback = str((profile or {}).get("api_key_ref") or "")
548 if fallback and tenant != 'default':
549 raise_code('CRED-703', reason='host_reference_forbidden')
550 if binding is not None and (profile.get('credential_revision', '') != binding['revision']
551 or profile.get('base_url') != binding['endpoint']):
552 raise_code('CRED-703', reason='stale_model_binding', advice='模型配置已变更;请创建新会话')
553 return profile

555 def binding(self, ref, endpoint):
556 _, tenant, profile = ref.split(':', 2)
557 with self._transaction(tenant):
558 row = self._validated_profile(tenant, profile)
559 if row.get('base_url') != endpoint:
560 raise_code('CRED-703', reason='endpoint_binding_mismatch')
561 return {'revision': row.get('credential_revision', ''), 'endpoint': endpoint}
