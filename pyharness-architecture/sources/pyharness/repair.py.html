109async def repair_session(ctx: Any, sid: str, *, interactive: bool = False,
110 policy: Optional[RepairPolicy] = None) -> RepairReport:
111 """F060 主入口(PRD 验收伪码直译):扫描 → 无损坏空报告(幂等)→ 先备份
112 `.corrupt-{ts}` → 尾部半行截断 → 中部坏行隔离(删留决策按 policy)→ seq 空洞
113 定位告警 → 派生视图整体重建 → 强同步追加 session.recovered → 返回报告。
114
115 异常表:备份/重写落盘失败 → PERS-202(原文件未动或备份留存);会话被 ctx.session
116 持有(sid 匹配,双开)→ PERS-202 拒修;不可修复编码整块由 persistence 层处置。
117 """
118 sdir = _sessions_dir(ctx)
119 path = sdir / f"{sid}.jsonl"
120 if not path.exists(): # 不存在 = 空报告(幂等,新会话前身)
121 return RepairReport(sid=sid)
122 sess = getattr(ctx, "session", None) # 禁双开:活会话文件级重写 = 句柄失同步
123 if sess is not None and getattr(sess, "sid", None) == sid:
124 raise_code("PERS-202", sid=sid,
125 advice="会话正被 ctx.session 持有,禁双开修复(DEP §6);"
126 "先退出该会话进程再 repair")
127 lock_path = session_lock_path(path) # 跨进程独占锁:活会话占用 → PERS-202
128 acquire_session_lock(lock_path)
129 try:
130 return await _repair_locked(ctx, sid, path,
131 interactive=interactive, policy=policy)
132 finally:
133 release_session_lock(lock_path)

136async def _repair_locked(ctx: Any, sid: str, path: Path, *,
137 interactive: bool,
138 policy: Optional[RepairPolicy]) -> RepairReport:
139 """repair_session 的持锁突变段(扫描 → 备份 → 截断/隔离 → 重建 → 声明)。
140
141 调用方(repair_session)已持会话独占锁并经 ctx.session 双开守卫,故本段对目标
142 文件的全部读-改-写处于单写者保护下(跨进程活写者已被锁挡在 repair 之外,不再
143 出现"repair 关句柄整写时他进程活句柄写的字节静默消失")。
144 """
145 health = await scan_session(path, fts_last_seq=_fts_last_seq_provider(ctx))
146 seg_damage = _segment_damage(path) # 轮转段损坏面(只读;R13-5)
147 if health.healthy and not seg_damage: # 二次 repair:无动作(幂等)
148 return RepairReport(sid=sid)
149 pol = policy
150 if pol is None:
151 pol = (interactive_confirm(health) if interactive else RepairPolicy())
152 backup = backup_file(path) # ① 修复前强制备份 .corrupt-{ts}
153 fixed: list[str] = []
154 lost_seqs: list[int] = []
155 kept: list[QuarantineEntry] = []
156 if health.tail_truncated and pol.drop_tail: # ② 尾部半行:截断续跑
157 lost_seqs = _dropped_seqs(path, health.tail_offset)
158 truncate_tail(path, health.tail_offset or 0) # offset=0 = 整文件半行(偏离 2)
159 fixed.append("tail-truncated")
160 if health.bad_lines: # ③ 中部坏行:隔离不自动删
161 kept = await quarantine_lines(path, health.bad_lines, pol)
162 fixed += [f"quarantine-{e.line_no}" for e in kept if e.archived]
163 # ⑤' 轮转段同口径修复(R13-5):它们是真源的一部分,损坏必须可见且可修 ——
164 # 段级同样"修复前强制备份";尾部半行截断、中部坏行隔离(不删、字节日志保真)。
165 for seg, torn_off, bad_lines in seg_damage:
166 label = _seg_label(seg)
167 backup_file(seg) # 段级备份(失败即 PERS-202 中止)
168 if torn_off is not None and pol.drop_tail:
169 lost_seqs += _dropped_seqs(seg, torn_off)
170 truncate_tail(seg, torn_off or 0)
171 fixed.append(f"{label}-tail-truncated")
172 if bad_lines:
173 entries = await quarantine_lines(seg, bad_lines, pol)
174 kept += entries
175 fixed += [f"{label}-quarantine-{e.line_no}"
176 for e in entries if e.archived]
177 holes_alert = locate_holes(health) # ④ seq 空洞:未声明 → 告警 F031 深查
178 if holes_alert and not interactive:
179 fixed.append("seq-holes:" + repr(holes_alert))
180 rebuilt = False
181 if health.index_stale or fixed: # ⑤ 派生视图整体重建(原则 1)
182 rebuilt = await rebuild_derived_views(ctx, sid)
183 if rebuilt and "views-rebuilt" not in fixed:
184 fixed.append("views-rebuilt") # 结构表词条:重建动作入审计
185 seq = await declare_recovered(ctx, sid, fixed, lost_seqs, kept, backup)
186 return RepairReport(sid=sid, fixed=fixed, quarantined=kept,
187 lost=len(lost_seqs), backup_path=backup,
188 holes_alert=holes_alert, recovered_seq=seq)

192def _ask(prompt: str, default: bool = True) -> bool:
193 """交互问询([Y/n] 语义);无输入(EOF/非 tty)→ 安全默认。
194 特殊:prompt 含 '[K]保留' 时按 k 视同 n(用户显式保留 = 拒绝默认动作)。
195 """
196 try:
197 ans = input(prompt).strip().lower()
198 except (EOFError, KeyboardInterrupt):
199 return default
200 if not ans:
201 return default
202 if "[k]" in prompt.lower() and ans in ("k", "keep"):
203 return False # 用户选保留(拒绝截断/隔离默认动作)
204 return not ans.startswith("n")

207def interactive_confirm(health: SessionHealth) -> RepairPolicy:
208 """交互决策(tty):尾部半行 → [T]截断续跑(默认)/[K]保留退出;中部坏行 → 隔离
209 默认(保留原地可选);空洞 → F031 深查提示(不回填)。headless(auto)= 全部安全
210 默认,不悬挂等待。EOFError → 回退 auto 默认(安全侧:截断+隔离,偏离 8 单次问询)。
211 """
212 print(f"[repair] {health.sid} 检测:尾部截断={health.tail_truncated} "
213 f"坏行={len(health.bad_lines)} 空洞={health.holes}")
214 drop_tail = True
215 if health.tail_truncated:
216 ans = _ask("尾部半行未完成事实,已备份。[T]截断续跑(默认) [K]保留退出: ",
217 default=True)
218 if not ans: # 用户选择保留:会话不可续跑
219 drop_tail = False
220 keep_q = True
221 if health.bad_lines:
222 keep_q = _ask(f"隔离 {len(health.bad_lines)} 行坏数据? [Y/n](n=保留原地): ",
223 default=True)
224 if health.holes:
225 print(f"[repair] 存在未声明 seq 空洞 {health.holes} → F031 深查线索;"
226 "空洞不回填,append 恒 max+1")
227 return RepairPolicy(mode="interactive", drop_tail=drop_tail,
228 keep_quarantine=keep_q)

257 archived=bool(e.get("archived", True))))

279def truncate_tail(path: Path, offset: int) -> None:
280 """尾部半行原子截断:截断到 offset(最后完整行末尾)+ fsync。
281
282 半行内未完成事件按 lost 声明(declare_recovered),绝不尝试补全;offset==0 且
283 文件非空(整文件即半行,无任何换行)→ 截到 0(偏离 2:未完成即未发生)。
284 异常:OSError → PERS-202(备份已存在,报错中止,禁止继续)。
285 """
286 if offset is None:
287 return
288 if offset <= 0 and path.stat().st_size == 0:
289 return # 空文件:无半行不动
290 try:
291 with open(path, "r+b") as f:
292 f.truncate(offset) # 截断点 = 最后完整 \\n 后(字节级)
293 f.flush()
294 os.fsync(f.fileno())
295 except OSError as e:
296 raise_code("PERS-202", op="truncate_tail", path=str(path), why=str(e),
297 advice="截断失败(备份已在 .corrupt-{ts});文件可能半截,禁止继续")
298 log.warning("repair: 尾部半行截断于 offset=%d path=%s", offset, path)

301def _append_quarantine(qpath: Path, record: dict) -> bool:
302 """隔离档逐条追加写(自持结构,非 Envelope;≤8KB 防爆已在调用方截断)。
303
304 失败(磁盘/权限)→ 记日志返回 False:调用方把该行保留在主文件(隔离≠丢数据)。

312 log.warning("repair: 隔离档写失败(行保留原地) qpath=%s why=%s", qpath, e)
313 return False

316async def quarantine_lines(path: Path, bad: list[int],
317 pol: RepairPolicy) -> list[QuarantineEntry]:
318 """中部坏行隔离(不删,删留用户定):坏行抽到 `{sid}.quarantine-{ts}.jsonl`
319 (行号/原文/原因,≤8KB/条),好行写临时文件,fsync+rename 原子替换主文件。
320
321 pol.keep_quarantine=False(交互问询选"保留原地")→ 不重写主文件,只返回
322 reason="user-keep" 的只读记录(报警留痕);隔离档写失败 → 该行保留原地且
323 archived=False(数据优先:隔离≠删除,任何丢行都是失败)。
324 异常:临时文件/rename 失败 → PERS-202(已备份存在)。
325 """
326 if not bad:
327 return []
328 badset = set(bad)
329 if not pol.keep_quarantine: # 用户保留原地(不删不隔离)
330 out: list[QuarantineEntry] = []
331 for no, text in _iter_text(path):
332 if no in badset:
333 out.append(QuarantineEntry(
334 line_no=no, raw=(text or "")[:_QUARANTINE_RAW_CAP],
335 reason="user-keep", archived=False))
336 return out
337 qpath = path.with_name(f"{path.stem}.quarantine-{now_ts()}.jsonl")
338 tmp = path.with_name(path.name + f".tmp-{os.getpid()}")
339 kept: list[QuarantineEntry] = []
340 try:
341 with open(tmp, "wb") as out_fh:
342 for no, rawb in _iter_bytes(path):
343 if no not in badset:
344 out_fh.write(rawb) # 好行原样保留(含原行尾,字节往返)
345 continue
346 text = rawb.rstrip(b"\r\n").decode("utf-8", "replace")[:_QUARANTINE_RAW_CAP]
347 entry = QuarantineEntry(line_no=no, raw=text,
348 reason="parse-fail", archived=True)
349 if _append_quarantine(qpath, {"line_no": no, "raw": text,
350 "reason": "parse-fail",
351 "archived": True}):
352 kept.append(entry)
353 else: # 隔离档写失败:行保留原地(不丢数据)
354 out_fh.write(rawb) # 原样(含原行尾)
355 kept.append(QuarantineEntry(line_no=no, raw=text,
356 reason="parse-fail",
357 archived=False))
358 out_fh.flush()
359 os.fsync(out_fh.fileno()) # 数据落盘再换名(无半写,DIS §8)
360 os.replace(tmp, path) # rename 原子替换:旧完整或新完整
361 _fsync_dir(path.parent)
362 except OSError as e:
363 if tmp.exists():
364 try:
365 tmp.unlink() # 失败清理,不留临时残留
366 except OSError:
367 pass
368 raise_code("PERS-202", op="quarantine", path=str(path), why=str(e),
369 advice="隔离重写失败(备份已在 .corrupt-{ts});重跑 repair")
370 if not qpath.exists(): # 全部隔离失败且行已保留:无隔离档
371 log.warning("repair: 坏行全部保留原地(隔离档写失败),见报告 archived=False")
372 log.info("repair: 中部坏行隔离 path=%s archived=%d kept=%d",
373 qpath, sum(1 for e in kept if e.archived), len(kept))
374 return kept
