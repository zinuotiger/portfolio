15def backup_file(path: Path, *, _clock=now_ts, _sync_directory) -> Path:
16 """修复前强制备份:copy2 整文件 → `{sid}.corrupt-{utc_ms}.jsonl`(PRD F060 命名)。
17
18 同 ts 备份已存在(重试/并发)→ 直接返回(幂等,不重复备份);备份权限 600
19 (Windows NTFS chmod 尽力而为,失败只记日志不阻断——权限收紧非修复动作)。
20 异常:OSError → PERS-202(备份优先于一切修复动作,备份失败 = 修复中止)。
21 """
22 dst = path.with_name(f"{path.stem}.corrupt-{_clock()}.jsonl")
23 if dst.exists():
24 return dst
25 try:
26 shutil.copy2(path, dst) # 元数据一并保留
27 except OSError as e:
28 raise_code("PERS-202", op="backup", path=str(path), why=str(e),
29 advice="备份失败,修复中止(备份优先于一切修复动作);查磁盘/权限")
30 try:
31 os.chmod(dst, 0o600)
32 except OSError as e:
33 log.warning("repair: 备份文件权限 600 收紧失败(尽力而为) dst=%s why=%s",
34 dst, e)
35 log.info("repair: 修复前备份 %s", dst)
36 return dst

41def locate_holes(health: SessionHealth) -> list[int]:
42 """seq 空洞定位(声明合法化,纯函数):holes 已被 compacted/recovered 声明
43 覆盖 = 合法(压缩/修复事实)不告警;未声明空洞 → 告警并触发 F031 深查线索
44 (不自动补 seq,append 永远 max+1)。scan 已滤声明,此处按 spec 语义再防御一遍。
45 """
46 if not health.holes:
47 return []
48 undeclared = []
49 for h in health.holes:
50 if any(lo <= h <= hi for lo, hi in health.holes_declared):
51 continue
52 undeclared.append(h)
53 if undeclared:
54 log.warning("repair: 未声明 seq 空洞 %s → F031 深查线索(不回填,append "
55 "恒 max+1) sid=%s", undeclared, health.sid)
56 return undeclared

58def _index_handle(ctx: Any) -> tuple[Any, bool]:
59 """FTS 派生视图句柄解析(**唯一入口**):返回 (句柄, 是否本函数自查构造)。
60
61 解析序(鸭子类型,逐级兜底):
62 ① ``ctx.session_query``(装配注入面;引擎**只注入在 agent ctx** 上);
63 ② 兄弟形态 ``ctx.fts`` / ``ctx.engine_spine.fts``(engine 的 spine 自带);
64 ③ 由 ``cfg.storage.db_path`` **惰性构造**(repair 只需"库在哪" —— 与 R11-2
65 的 `read_max_seq` 同款思路,自有句柄由调用方 detach 收尾)。
66
67 2026-09-21 R14-1 修:此前只认 ①,而 repair 的调用方(CLI 命令 / 桌面)传的是
68 **外壳 ctx**(无 `session_query`)⇒ "⑤派生视图整体重建"步骤**不可达** ⇒ repair
69 修完仍 `index_stale=True`(不收敛)。同 PIT-19("约定被读 ≠ 被注入")。
70 """
71 sq = getattr(ctx, "session_query", None)
72 if sq is not None and callable(getattr(sq, "rebuild", None)):
73 return sq, False
74 for holder in (ctx, getattr(ctx, "engine_spine", None)):
75 cand = getattr(holder, "fts", None) if holder is not None else None
76 if cand is not None and callable(getattr(cand, "rebuild", None)):
77 return cand, False
78 db_path = _db_path_of(ctx)
79 if not db_path:
80 return None, False
81 try:
82 from pyharness.core.session_query import SessionQueryIndex
83 return SessionQueryIndex(db_path=db_path), True
84 except Exception as e: # noqa: BLE001 构造失败=未装配
85 log.warning("repair: 构造 FTS 句柄失败(%s):视图重建跳过 sid=?", e)
86 return None, False

88async def rebuild_derived_views(ctx: Any, sid: str) -> bool:
89 """派生视图整体重建(原则 1):FTS 与 storage KV 全部丢弃重建(重放,非增量修补)。
90
91 句柄经 ``_index_handle`` 解析(装配注入面 → 兄弟形态 → cfg 惰性构造;R14-1);
92 视图未装配 → 日志降级返回 False(派生可弃,装配后自动生效);重建后对账(视图
93 暴露 max_seq(sid) 时):日志末 seq != 视图末 seq → PERS-201(视图落后不静默)。
94 索引写失败 → PERS-202(原日志不受影响,索引=派生可再建)。
95 """
96 sq, owned = _index_handle(ctx)
97 if owned and sq is not None: # 自建句柄须先装载(R14-1)
98 try:
99 await sq.enter(ctx)
100 except Exception as e: # noqa: BLE001 装载失败=视图不可用
101 log.warning("repair: 自建 FTS 句柄装载失败(%s):视图重建跳过 sid=%s",
102 type(e).__name__, sid)
103 sq, owned = None, False
104 rebuild = getattr(sq, "rebuild", None) if sq is not None else None
105 storage = getattr(ctx, "storage", None)
106 kv = None
107 if storage is not None:
108 cand = getattr(storage, "kv_rebuild", None)
109 kv = cand if callable(cand) else None
110 if rebuild is None and kv is None:
111 log.info("repair: 派生视图(FTS/storage)未装配,整体重建跳过 sid=%s", sid)
112 return False
113 store = None
114 view_last: Optional[int] = None
115 try:
116 if rebuild is not None: # FTS 重放源:修复后的主文件
117 store = open_store(
118 sid, dir=_sessions_dir(ctx),
119 **flush_kwargs_of(getattr(ctx, "config", None)
120 or getattr(ctx, "settings", None)))
121 attach = getattr(sq, "attach_source", None)
122 if callable(attach):
123 attach(sid, store)
124 try:
125 await rebuild(session_id=sid)
126 except OSError as e:
127 raise_code("PERS-202", op="views-rebuild", sid=sid, why=str(e),
128 advice="索引为派生副本可再建;真源不受影响,重跑 repair")
129 log.info("repair: FTS 派生视图重建完成 sid=%s", sid)
130 if kv is not None: # storage KV 派生缓存重建(鸭子面)
131 try:
132 kv(sid)
133 except OSError as e:
134 raise_code("PERS-202", op="kv_rebuild", sid=sid, why=str(e),
135 advice="KV 派生缓存可再建;真源不受影响")
136 # 对账读数**必须在句柄有效期内**取(R14-1 二阶):自建句柄在 finally 里
137 # detach 关库(``_db=None``),此后 ``max_seq`` 恒返 0 ⇒ 与日志水位不等 ⇒
138 # **假 PERS-201**(修复自身引出的缺陷:重建明明成功却被判不一致)。
139 mx = getattr(sq, "max_seq", None) if sq is not None else None
140 if callable(mx):
141 try:
142 view_last = mx(sid)
143 except Exception: # noqa: BLE001 — 提供者故障降级
144 log.warning("repair: 视图 max_seq(%s) 查询失败,对账跳过", sid)
145 finally:
146 if store is not None:
147 try:
148 store.close() # 只读回放源句柄收尾
149 except Exception: # noqa: BLE001 — 清理不阻断
150 log.debug("repair: FTS 回放源 close 失败", exc_info=True)
151 if owned and sq is not None: # 本函数自建的句柄:摘除并关库(R14-1)
152 try:
153 det = getattr(sq, "detach", None)
154 if callable(det):
155 r = det(None)
156 if inspect.isawaitable(r):
157 await r
158 except Exception: # noqa: BLE001 — 清理不阻断
159 log.debug("repair: 自建 FTS 句柄 detach 失败", exc_info=True)
160 # 对账判定:日志末 seq == 视图末 seq(读数已在句柄有效期内取得;不一致 → PERS-201)
161 if view_last is not None:
162 # 对账口径:与索引 max_seq 同口径 = **可索引事件**(内容型 insert)末 seq——
163 # 排除 session.recovered 声明(不索引)与一切非内容事件(created/finished/
164 # llm.request/guard.* 等,不产索引行);否则尾部非内容即误抛 PERS-201。
165 # 水位是**会话级**的(R14-1 二阶):重建源 replay 合并读轮转段 ⇒ 对账也须跨段
166 # 取 max(与 scan_session 单源),否则主文件被截空时假 PERS-201。
167 log_last = _session_content_last(_sessions_dir(ctx) / f"{sid}.jsonl")
168 if log_last != view_last:
169 raise_code("PERS-201", op="view-reconcile", sid=sid,
170 log_last=log_last, view_last=view_last,
171 advice="索引对账不一致(视图落后不静默);重跑 repair "
172 "或手动重建派生视图")
173 return True

175async def declare_recovered(ctx: Any, sid: str, fixed: list[str],
176 lost: Optional[list[int]] = None,
177 kept: Optional[list[QuarantineEntry]] = None,
178 backup: Optional[Path] = None) -> Optional[int]:
179 """recovered 事件声明(修复是事件不是抹除):actor=system 强同步 append 到目标
180 会话日志;声明是修复的审计落点与"空洞合法化"依据(persistence._declared_by_
181 compaction 消费 fixed seq-holes/lost,后续 replay 不再告警)。
182
183 载荷按注册词表 SessionRecoveredPayload(fixed 非空/backup/lost,偏离 1):
184 fixed 为空(无修复动作,含"用户保留原地")→ 不追加(幂等,注册载荷 min_length=1);
185 目标会话已终态(finished)/空文件未 created → append 语义 EVT-104/106 不允许,
186 声明跳过(日志 + 报告 recovered_seq=None;审计由备份 + RepairReport 承担);
187 其余 PyHError(PERS-202 落盘失败)→ 上抛(文件已修,声明可下次 repair 补,幂等
188 保证不重复动作)。返回 recovered 事件 seq。
189 """
190 if not fixed: # 无修复动作:不声明(偏离 3)
191 return None
192 payload: dict[str, Any] = {"fixed": fixed}
193 if backup is not None:
194 payload["backup"] = str(backup)
195 if lost:
196 payload["lost"] = [int(x) for x in lost]
197 from pyharness.core.session import open_session # 惰性:避免装配期硬依赖
198 store = open_store( # 修复先于 open(CLI 序)→ 自开
199 sid, dir=_sessions_dir(ctx),
200 **flush_kwargs_of(getattr(ctx, "config", None)
201 or getattr(ctx, "settings", None)))
202 primary = None
203 try:
204 log_ = await open_session(sid, store)
205 log_._bus = _DirectBus(store)
206 env = await log_.append("session.recovered", payload,
207 actor="system", sync=True)
208 except PyHError as e:
209 primary = e
210 if e.code in ("EVT-104", "EVT-106"): # 终态/未 created:声明不适用
211 primary = None # suppressed contract result must not suppress cleanup errors
212 log.warning("repair: recovered 声明跳过(session=%s %s,审计以备份+"
213 "报告为准) fixed=%s", sid, e.code, fixed)
214 return None
215 raise # PERS-202 等:上抛(声明可下次补)
216 except BaseException as exc:
217 primary = exc
218 raise
219 finally:
220 try: store.close()
221 except BaseException as exc:
222 if primary is None: raise
223 primary.add_note(f"repair close failed: {type(exc).__name__}")
224 log.info("repair: session.recovered 声明 sid=%s seq=%d fixed=%s lost=%s",
225 sid, env.seq, fixed, lost or [])
226 return env.seq

241def _dropped_seqs(path: Path, offset: Optional[int]) -> list[int]:
242 """尾部半行含未完成事件候选 seq(0/1 个):信封序列化 seq 字段最先落,半行内
243 常可探测;探测不到则取"最后完整 seq + 1"(append 恒 max+1 分配的事实口径)。
244 offset 为 None/文件空/剩余为空 → []。
245 """
246 if offset is None:
247 return []
248 try:
249 data = path.read_bytes()[offset:]
250 except OSError:
251 return []
252 if not data.strip():
253 return []
254 probe = data[:_TAIL_SEQ_PROBE]
255 m = re.search(rb'"seq"\s*:\s*(\d+)', probe)
256 if m:
257 return [int(m.group(1))]
258 last = _log_last_seq(path) # 整文件半行且无 seq 可探:无完整事件
259 return [last + 1] if last else []
