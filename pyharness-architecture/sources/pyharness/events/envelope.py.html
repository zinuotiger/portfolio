27def _utc_now_ts() -> str:
28 """框架统一打点:UTC 微秒 ISO8601,以 Z 结尾(禁外部自报)。"""

33class Envelope(BaseModel):
34 """事件信封:十字段强校验(PRD §3.2 / EVENT-SCHEMA §2.1)。
35
36 冻结信封(extra="forbid" + frozen=True):拒多余字段、禁事后改字段;
37 payload 按 type 二次强校验在校验链第 3 步完成。
38 """
39
40 model_config = ConfigDict(extra="forbid", frozen=True)
41
42 seq: int = Field(ge=1) # 会话内单调 +1,框架分配
43 ts: str = Field(pattern=_TS_PATTERN) # ISO8601 UTC 微秒,框架打
44 type: str # 词表枚举;未注册 → EVT-102
45 session_id: str = Field(min_length=8) # 形如 s-abc12345;fork 新 id
46 actor: Literal["user", "agent", "llm", "tool", "system", "plugin"]
47 origin: Optional[str] = None # 插件/能力 id(如 plugin:echo)
48 task_id: Optional[str] = None # 所属任务段;无任务上下文为空
49 payload: dict = Field(default_factory=dict)
50 trace: Optional[dict] = None # 语义父关联,建议 {"parent_seq": int}
51 # 租户归属(GAP-10;可选字段,默认 None = 单租户/未声明)。
52 #
53 # 为什么落在**信封**而不是各 payload 模型:租户是**所有**事件共有的横切归属,
54 # 不是某一类事件特有的业务字段。放进 77 个 payload 模型需要改 77 处 schema
55 # (H-3 级改动且必然漂移),放进信封只需一处,且与 session_id/actor 同级——
56 # 它们同样是"每条事件都有的元信息"。旧日志缺该字段 ⇒ 反序列化为 None,
57 # 向后兼容、可 replay(INV-02 不变)。
58 #
59 # 纪律:租户由**框架侧**注入(SessionLog 携带,见 core/session.py),
60 # 不由 LLM/工具/插件自报——与 seq/ts 同款单点分配纪律。
61 tenant_id: Optional[str] = None # 租户归属;None = 未声明(单租户)

63 @field_validator("ts")
64 @classmethod
65 def _ts_utc(cls, v: str) -> str:
66 """框架统一打(禁外部自报):ts 必须 UTC 且以 Z 结尾。"""
67 if not v.endswith("Z"):

72def validate_envelope(raw: dict, seq_state: "SeqState",
73 session_open: bool = True) -> Envelope:
74 """五步校验链唯一实现,顺序不可交换;任一失败拒写(EVENT-SCHEMA §2.2)。
75
76 1 信封字段(类型/枚举/正则)→ EVT-100;2 类型注册 → EVT-102;
77 3 payload 强校验 → EVT-100;4 seq 连续性 → EVT-101;
78 5 会话状态:未 created 即写(且非 session.created)→ EVT-106。
79 (终态写违规 EVT-104 由 session.append 状态机侧把关,INV-01)
80 """
81 try:
82 env = Envelope.model_validate(raw)
83 except ValidationError as e:
84 raise_code("EVT-100", detail=summarize_validation(e))
85 # 2 类型必须已注册(EVT-102)
86 if not is_registered(env.type):
87 raise_code("EVT-102", type_=env.type)
88 # 3 payload 二次强校验 → 规范化(冻结信封:model_copy 换 payload,不原地改)
89 canonical = validate_payload(env.type, env.payload)
90 env = env.model_copy(update={"payload": canonical})
91 # 4 seq 必须 = _next_seq + 1(EVT-101)
92 expected = seq_state.next_seq(env.session_id)
93 if env.seq != expected:
94 raise_code("EVT-101", session_id=env.session_id,
95 expected=expected, got=env.seq)
96 # 5 会话状态:未 created 且本事件不是 opening 的 session.created → EVT-106
97 if not session_open and env.type != "session.created":
98 raise_code("EVT-106", type_=env.type, hint="首事件必须是 session.created(seq=1)")
99 return env

102def make_envelope(session_id: str, type_: str, actor: str, payload: dict, *,
103 origin: Optional[str] = None, task_id: Optional[str] = None,
104 trace: Optional[dict] = None,
105 tenant_id: Optional[str] = None,
106 seq_state: Optional["SeqState"] = None) -> Envelope:
107 """框架打点构造器:seq/ts 由框架统一分配(§3.1-5),再走 validate_envelope 全链。
108
109 调用方(会话层)在 append 成功落盘后调用 seq_state.commit(env) 记账;
110 seq_state.open 指示会话"已 created 且未 finished"。
111
112 ``tenant_id``(GAP-10):由**会话层**注入(``SessionLog.tenant_id``),不由调用
113 点逐个传——保证同一会话的每条事件归属一致,且新增调用点无需记得带上它。
114 """
115 if seq_state is None:
116 raise_code("EVT-100", detail="make_envelope 缺 seq_state:seq 只能由框架分配")
117 raw: dict[str, Any] = {
118 "seq": seq_state.next_seq(session_id), # 框架唯一打点(防伪造乱序)
119 "ts": _utc_now_ts(),
120 "type": type_,
121 "session_id": session_id,
122 "actor": actor,
123 "origin": origin,
124 "task_id": task_id,
125 "payload": payload,
126 "trace": trace,
127 "tenant_id": tenant_id,
128 }
129 return validate_envelope(raw, seq_state,
130 session_open=session_id in seq_state.open)

133class SeqState:
134 """per-session seq 分配器与空洞口径(§3.4)。
135
136 内存 _next 从日志 max 重建(rebuild);append 恒 max+1;compaction/fork/repair
137 空洞只解释不回填。open = 已 created 且未 finished(由会话层经 rebuild 开闸)。
138 """
139
140 def __init__(self) -> None:
141 self._next: dict[str, int] = {} # session_id → 下个可用 seq
142 self.open: set[str] = set() # 已 created 且未 finished
143
144 def rebuild(self, session_id: str, max_seq: int) -> None:
145 """启动/repair 后从日志重建:max_seq 为已落盘最大 seq,并开闸会话。"""
146 self._next[session_id] = max_seq + 1
147 self.open.add(session_id)
148
149 def next_seq(self, session_id: str) -> int:
150 """期望的下一个 seq;无记录 → 1(session.created 首事件)。"""
151 return self._next.get(session_id, 1)
152
153 def commit(self, env: Envelope) -> None:
154 """validate 通过并落盘成功后记账:_next = env.seq + 1。"""
155 self._next[env.session_id] = env.seq + 1

158def check_seq_gap(events: list[int], declared: list[tuple[int, int]]) -> list[int]:
159 """回放空洞自检:无 compacted/recovered 声明覆盖的 seq 空洞列表(供 warn_hole)。
160
161 declared = 合法空洞闭区间声明(如 context.compacted.ranges);返回未声明空洞,
162 升序;不中断回放。
163 """
164 declared_holes: set[int] = set()
165 for lo, hi in declared:
166 declared_holes.update(range(lo, hi + 1))
167 actual = set(events)
168 expected = set(range(1, max(events) + 1)) if events else set()
169 return sorted(expected - actual - declared_holes)

176def call_id_of(env: Any) -> str:
177 """事件的 ``call_id`` —— **唯一读取点**:payload 优先,回落 ``trace``。
178
179 存放位置按事件族不同(历史成因,见 LIMITATIONS **L-17**):``tool.call`` /
180 ``tool.result`` 在 **payload**;``guard.*`` / ``decision.issued`` /
181 ``receipt.emitted`` / ``approval.requested`` 在 **trace**。读取方**不得只认
182 一处**(R24 收口:全库读取点统一走本函数,避免未来新增读取方漏判一侧)。
183 取不到 → ``""``(不抛:审计读取面永不因缺字段中断)。
184 """
185 payload = getattr(env, "payload", None) or {}
186 cid = payload.get("call_id") if isinstance(payload, dict) else None
187 if not cid:
188 trace = getattr(env, "trace", None) or {}
189 cid = trace.get("call_id") if isinstance(trace, dict) else None
190 return str(cid or "")

193def declared_ranges(env: Any) -> list[tuple[int, int]]:
194 """声明事件 → **合法空洞闭区间**列表(空洞合法化的**唯一判据**)。
195
196 - ``context.compacted.ranges`` —— 折叠闭区间 ``[[lo, hi], …]``;
197 - ``session.recovered`` —— ``lost`` 的逐号单点区间,外加 ``fixed`` 中
198 ``"seq-holes:[…]"`` 词条(repair 对"未声明空洞"的声明;重复 repair 不重复告警)。
199
200 非声明事件 → ``[]``。**所有**空洞合法性判定都必须走本函数:
201 ``SessionLog`` 回放告警 / ``SessionStore`` 合法性 / ``repair`` 扫描 / 治理对账。
202
203 2026-09-21 R14-9:此前四处各自实现,其中 ``SessionLog._absorb`` **只认 compacted**
204 ⇒ repair 已用 ``seq-holes:[…]`` 声明的空洞,在下次 ``open_session`` 回放时又被报成
205 "未声明空洞(疑丢事件)"(实测:同一次合法修复被告警两次,属**假 F031 线索** + 日志
206 噪声)。判据多源必漂移。
207 """
208 t = getattr(env, "type", None)
209 payload = getattr(env, "payload", None) or {}
210 out: list[tuple[int, int]] = []
211 if t == "context.compacted":
212 for pair in (payload.get("ranges") or []):
213 try:
214 lo, hi = int(pair[0]), int(pair[1])
215 except (TypeError, ValueError, IndexError, KeyError):
216 continue
217 if lo <= hi:
218 out.append((lo, hi))
219 elif t == "session.recovered":
220 for s in (payload.get("lost") or []):
221 try:
222 out.append((int(s), int(s)))
223 except (TypeError, ValueError):
224 continue
225 for item in (payload.get("fixed") or []):
226 m = re.match(r"^seq-holes:\[(.*)\]$", str(item))
227 if m and m.group(1).strip():
228 for x in m.group(1).split(","):
229 if x.strip().isdigit():
230 out.append((int(x), int(x)))
231 return out
