121def _bump_counter(ctx: Any, key: str, delta: int) -> None:
122 """已读量入计数器(防循环烧预算):支持 dict / bump(key, n) / 属性三形态。
123
124 ctx.counters 未接线时静默跳过(log debug)——读数不受影响,仅防循环闸
125 少一个数据源(预算/轮数闸另有兜底)。
126 """
127 c = getattr(ctx, "counters", None)
128 if c is None:
129 return
130 if isinstance(c, dict): # dict 形态(单测/装配可直挂)
131 c[key] = c.get(key, 0) + delta
132 return
133 bump = getattr(c, "bump", None)
134 if callable(bump):
135 try:
136 bump(key, delta)
137 return
138 except Exception: # noqa: BLE001 降级到属性形态
139 pass

143 log.debug("spill 计数器形态不支持 bump,key=%s", key)
