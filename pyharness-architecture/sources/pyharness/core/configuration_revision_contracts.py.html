23def encoded(value):
24 try:
25 raw = json.dumps(value, ensure_ascii=False, sort_keys=True,
26 separators=(',', ':'), allow_nan=False).encode('utf-8')
27 except (ValueError, TypeError, RecursionError, UnicodeError):

51def public_tree(value, known_secrets=()):
52 encoded(value)
53 pending = [(value, 0)]
54 count = 0
55 while pending:
56 item, depth = pending.pop()
57 count += 1
58 if count > 2048 or depth > 12:
59 reject('configuration_metadata_limit')
60 if type(item) is str:
61 if (_PRIVATE_TEXT.search(item) or any(secret and secret in item for secret in known_secrets)
62 or any(ord(c) < 32 and c not in '\n\r\t' for c in item)):
63 reject('configuration_private_content')
64 elif type(item) is dict:
65 if any(type(k) is not str for k in item):
66 reject('configuration_invalid_metadata')
67 pending.extend((k, depth+1) for k in item)
68 pending.extend((v, depth+1) for v in item.values())
69 elif type(item) is list:
70 pending.extend((v, depth+1) for v in item)
71 elif item is not None and type(item) not in (bool, int, float):
72 reject('configuration_invalid_metadata')
73 return copy.deepcopy(value)

76def base_fingerprint(value, known_secrets=()):
77 """Fingerprint base policy while omitting secret values, including raw keys.
78
79 Actual selected-model credentials have a separate original revision binding.
80 This finite masking is deliberately not a DLP or authentication mechanism.
81 """
82 encoded(value)
83 sensitive = re.compile(r'(?i)api.?key|secret|password|passphrase|authorization|cookie|(?:^|_)token$')
84 nodes = 0
85 def masked(item, depth=0):
86 nonlocal nodes
87 nodes += 1
88 if nodes > 2048 or depth > 12:
89 reject('configuration_metadata_limit')
90 if type(item) is dict:
91 if any(type(k) is not str for k in item):
92 reject('configuration_invalid_metadata')
93 return {k: '[secret-reference-omitted]' if sensitive.search(k)
94 else masked(v, depth+1) for k, v in item.items()}
95 if type(item) is list:
96 return [masked(v, depth+1) for v in item]
97 if type(item) is str and (_PRIVATE_TEXT.search(item)
98 or any(secret and secret in item for secret in known_secrets)):
99 return '[secret-reference-omitted]'
100 return item
101 return digest(masked(value))

199def recent_commands(history_, tenant, kind, resource):
200 result = []
201 for command_key, raw in history_['commands'].items():
202 value = checked_command(raw)
