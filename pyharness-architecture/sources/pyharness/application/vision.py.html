17def preview_image(raw: bytes, filename: str) -> dict:
18 info = validate_image(raw)
19 extensions = {'image/png': {'.png'}, 'image/jpeg': {'.jpg', '.jpeg'},
20 'image/gif': {'.gif'}, 'image/webp': {'.webp'}}
21 from pathlib import PurePosixPath
22 if PurePosixPath(filename).suffix.lower() not in extensions[info['mime']]:
23 reject_attachment('extension-mismatch')
24 if info['w'] * info['h'] > 40_000_000:
25 reject_attachment('pixel-limit')
26 # Reject malformed containers before invoking the bounded pixel decoder.
27 if info['mime'] == 'image/png':
28 offset, kinds = 8, []
29 while offset + 12 <= len(raw):
30 length = int.from_bytes(raw[offset:offset+4], 'big')
31 end = offset + 12 + length
32 if end > len(raw): reject_attachment('truncated-image')
33 kind = raw[offset+4:offset+8]
34 checksum = int.from_bytes(raw[end-4:end], 'big')
35 if zlib.crc32(raw[offset+4:end-4]) != checksum:
36 reject_attachment('image-checksum')
37 kinds.append(kind)
38 offset = end
39 if kind == b'IEND': break
40 if offset != len(raw) or not kinds or kinds[0] != b'IHDR' or kinds[-1] != b'IEND' or b'IDAT' not in kinds:
41 reject_attachment('truncated-image')
42 elif info['mime'] == 'image/jpeg' and not raw.endswith(b'\xff\xd9'):
43 reject_attachment('truncated-image')
44 elif info['mime'] == 'image/gif' and not raw.endswith(b';'):
45 reject_attachment('truncated-image')
46 elif info['mime'] == 'image/webp' and int.from_bytes(raw[4:8], 'little') + 8 != len(raw):
47 reject_attachment('truncated-image')
48 try:
49 with warnings.catch_warnings():
50 warnings.simplefilter('error', Image.DecompressionBombWarning)
51 with Image.open(io.BytesIO(raw)) as decoded:
52 if decoded.size != (info['w'], info['h']):

54 # Bound total decoded work across animation frames, not only
55 # the first frame or the compressed upload size.
56 frames = getattr(decoded, 'n_frames', 1)
57 if frames > 100 or frames * info['w'] * info['h'] > 40_000_000:
58 reject_attachment('decoded-pixel-limit')
59 for frame in range(frames):
60 decoded.seek(frame)
61 decoded.load()
62 except (UnidentifiedImageError, OSError, ValueError, SyntaxError,
63 Image.DecompressionBombWarning, Image.DecompressionBombError):

68def require_connection(platform, connection):
69 state = platform.service.tenant_state() # public metadata, never secret resolution
70 ident = connection or state['active']
71 profile = next((p for p in state['profiles'] if p['id'] == ident), None)

79async def authorized_images(platform, sid, ids):
80 images = []
81 for ident in ids:
82 record, raw = await platform.artifact_content(ident)
83 if record['session_id'] != sid:
