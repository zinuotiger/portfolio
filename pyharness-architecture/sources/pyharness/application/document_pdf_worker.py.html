37def parse(path):
38 from pypdf import PdfReader
39 reader = PdfReader(path, strict=True)
40 if reader.is_encrypted:

47 if hasattr(value, 'get_object'):
48 value = value.get_object()
49 # Internal destinations and URI links are never executed or rendered here.
50 return isinstance(value, dict) and value.get('/S') not in {'/GoTo', '/URI'}
51 if unsafe_action(root.get('/OpenAction')) or root.get('/AA') or names.get('/JavaScript') or names.get('/EmbeddedFiles'):
52 return {'rejected': 'document_active_content'}
53 if not 1 <= len(reader.pages) <= 200:

62 return {'rejected': 'document_active_content'}
63 content = page.get_contents()
64 # ContentStream is a dictionary and may be false even with decoded data.
65 length = len(content.get_data()) if content is not None else 0
66 decoded += length
67 if length > 2 * 1024**2 or decoded > 8 * 1024**2:

69 text = page.extract_text() or ''

72 return {'rejected': 'document_text_limit'}
73 pages.append(text)
74 return {'capability_reason': 'text' if any(x.strip() for x in pages) else 'no_text_layer', 'pages': pages}
