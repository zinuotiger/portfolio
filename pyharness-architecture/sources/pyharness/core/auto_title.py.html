26async def auto_title(ctx: Any) -> Optional[str]:
27 """为会话生成标题并落盘(session.renamed);失败/已命名 → None(幂等)。"""
28 if any(e.type == "session.renamed" for e in ctx.session.events_after(0)):
29 return None
30 first = _first_user_text(ctx)
31 if not first:
32 return None
33 prompt = (f"给这段会话起一个简洁中文标题,≤24 字,只输出标题本身,"
34 f"不要引号/句号/解释:\n用户第一条消息: {first}")
35 try:
36 from pyharness.core.llm_diagnostics import model_call
37 with model_call("title_generation"):
38 text = await ctx.llm.mini(prompt, ctx=ctx)
39 except PyHError as e: # LLM 失败:标题留空(下次再试)
40 from pyharness.core.llm_diagnostics import public_diagnostics
41 d = public_diagnostics(e.ctx.get("diagnostics"), code=e.code, role="title_generation")
42 await ctx.session.append("system.error", {"code": e.code, "hint": d["summary"],
43 "ctx": {"role": "title_generation", "fatal": False, "diagnostics": d}}, actor="system")
44 log.info("auto_title skipped code=%s", e.code)
45 return None
46 title = text.strip().strip('"“”\' ').splitlines()[0:1]
47 title = (title[0] if title else "").strip()[:TITLE_MAX]
48 if not title:
49 return None
50 await ctx.session.append("session.renamed",
51 {"new_title": title, "by": "auto"},
52 actor="system")
53 return title
