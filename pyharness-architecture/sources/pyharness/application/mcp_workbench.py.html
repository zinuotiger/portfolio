44def _safe_schema(value):
45 """Finite private-content refusal, without rewriting schema property names."""
46 raw = _encoded(value)
47 if len(raw) > _MAX_BYTES:
48 fail('diagnostic_schema_limit')
49 pending = [value]
50 count = 0
51 while pending:
52 item = pending.pop()
53 count += 1
54 if count > 2048:
55 fail('diagnostic_schema_limit')
56 if type(item) is str and _PRIVATE_TEXT.search(item):
57 fail('diagnostic_private_content')
58 if type(item) is dict:
59 pending.extend(item.keys())
60 pending.extend(item.values())
61 elif type(item) is list:
62 pending.extend(item)
63 return value
