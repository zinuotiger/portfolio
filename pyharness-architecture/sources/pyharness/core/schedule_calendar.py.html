11 def _parse_spec(self, kind: str, expr: str, *,
12 require_future: bool = True) -> Union[CronSpec, IntervalSpec, AtSpec]:
13 """按 kind 解析表达式(require_future=False 供回放重建 at 已过期场景)。"""
14 if kind == "cron": # F048 核心:5 字段 cron
15 if expr != expr.strip() or "\t" in expr:
16 raise_code("CFG-601", hint=f"cron 表达式含首尾空白/制表符: {expr!r}",
17 advice="格式:分 时 日 月 周(如 '0 2 * * *'),禁首尾空白")
18 parts = expr.split()
19 if len(parts) != 5: # 段数错误
20 raise_code("CFG-601", hint=f"cron 须 5 字段: {expr}",
21 advice="格式:分 时 日 月 周(如 '0 2 * * *')")
22 self._check_cron_fields(parts)
23 return CronSpec(fields=tuple(parts))
24 if kind == "interval": # 固定间隔(秒)
25 secs = self._as_int(expr)
26 if secs is None or secs < MIN_INTERVAL_SECONDS:
27 raise_code("CFG-601", hint=f"interval 须 ≥{MIN_INTERVAL_SECONDS} 秒: {expr}",
28 advice="F048 最小粒度 1 分钟")
29 return IntervalSpec(seconds=secs)
30 if kind == "at": # 绝对时间(一次性)
31 dt = parse_iso(expr)
32 if dt is None or (require_future and dt <= self._now()):
33 raise_code("CFG-601", hint="at 时间须为未来的 ISO8601",
34 advice="示例:2026-09-08T10:30:00")
35 return AtSpec(fire_at=dt)
36 raise_code("EVT-100", hint=f"kind 非法: {kind}",
37 advice="kind ∈ cron/interval/at")

48 def _check_cron_fields(self, parts: list[str]) -> None:
49 """5 字段逐一校验:段语法(数值/区间/列表/步长)与值域(越界 → CFG-601)。"""
50 for field_, (lo, hi) in zip(parts, _CRON_BOUNDS):
51 for seg in field_.split(","):
52 if not seg or not _SEG_RE.match(seg):
53 raise_code("CFG-601", hint=f"cron 段非法: {field_!r}",
54 advice="段 ∈ 数值/a-b/a,b/*/*/n/a-b/n")
55 m = _SEG_RE.match(seg)
56 if m.group(1) == "*":
57 step = int(m.group(3) or 1)
58 if step < 1:
59 raise_code("CFG-601", hint=f"cron 步长非法: {seg}")
60 continue
61 a = int(m.group(1))
62 b = int(m.group(2)) if m.group(2) else a
63 step = int(m.group(3) or 1)
64 if a < lo or b > hi or a > b or step < 1: # 值域/序/步长
65 raise_code("CFG-601", hint=f"cron 值域越界: {seg}"
66 f"(字段 {lo}-{hi})")
67
68 @staticmethod
69 def _field_match(field_: str, v: int) -> bool:
70 """单字段匹配:*,a,a-b,列表,*/n,a-b/n 任一命中即真(注册期已校验语法)。"""
71 for seg in field_.split(","):
72 m = _SEG_RE.match(seg)
73 if not m:
74 continue # 防御:注册期已拒,绝不 True 误放
75 step = int(m.group(3) or 1)
76 if m.group(1) == "*": # '*' 或 '*/n'
77 if v % step == 0:
78 return True
79 continue
80 a = int(m.group(1))
81 b = int(m.group(2)) if m.group(2) else a
82 if a <= v <= b and (v - a) % step == 0:
83 return True
84 return False

86 def _cron_match(self, spec: CronSpec, dt: datetime) -> bool:
87 """cron 分钟匹配:五字段逐一命中(本地时钟;周 0=周日,数据表权威)。"""
88 vals = (dt.minute, dt.hour, dt.day, dt.month, (dt.weekday() + 1) % 7)
89 for field_, v in zip(spec.fields, vals):
90 if not self._field_match(field_, v):
91 return False
92 return True

100 def next_fire(self, job: ScheduleJob, now_dt: datetime) -> Optional[datetime]:
101 """按 kind 计算下次触发(cron 按日扫描;interval 按 last_fired 推进;
102 at 一次性:fire_at 已过 → None,耗尽)。
103
104 cron 无未来触发点(如 2/30)→ CFG-601 显式失败:注册期即拒,运行期不再
105 产生"已注册 + next_fire_at=None + 永不触发"的静默状态(BUG-1 修复)。
106 """
107 spec = self._spec_of(job)
108 if job.kind == "cron":
109 return self._cron_next_fire(spec, job.expr, now_dt)
110 if job.kind == "interval": # 固定间隔推进(≥60s)
111 base = job.last_fired_at or now_dt
112 return base + timedelta(seconds=spec.seconds)
113 return spec.fire_at if spec.fire_at > now_dt else None # at:已过即 None

115 def _cron_next_fire(self, spec: CronSpec, expr: str,
116 now_dt: datetime) -> datetime:
117 """cron 下次触发:按日推进,命中日取当日最早的匹配 (时,分)。
118
119 **不再逐分钟扫描**——旧实现上限 1440 分钟(24h),使"下次触发在 24h 之外"
120 的合法表达式(工作日/每周/每月)一律得到 None → 永不触发(BUG-1)。此处
121 按日推进把上限提到完整格里高利周期(CRON_MAX_SCAN_DAYS),整月不匹配时
122 直接跳到下月 1 日,故 400 年上限的实际代价极小。
123
124 界内无匹配 ⇒ 该表达式永远无触发点 → CFG-601 显式失败(不返回 None)。
125 """
126 minute_f, hour_f, dom_f, month_f, dow_f = spec.fields
127 hours = [h for h in range(24) if self._field_match(hour_f, h)]
128 minutes = [mi for mi in range(60) if self._field_match(minute_f, mi)]
129 start = now_dt.replace(second=0, microsecond=0) + timedelta(minutes=1)
130 cursor = start.replace(hour=0, minute=0)
131 for _ in range(CRON_MAX_SCAN_DAYS):
132 if not self._field_match(month_f, cursor.month):
133 # 整月不匹配:跳到下月 1 日,不逐日空转
134 cursor = (cursor.replace(day=1)
135 + timedelta(days=32)).replace(day=1)
136 continue
137 if (self._field_match(dom_f, cursor.day)
138 and self._field_match(dow_f, (cursor.weekday() + 1) % 7)):
139 for h in hours: # 升序遍历 → 首个 ≥start 即当日最早
140 for mi in minutes:
141 cand = cursor.replace(hour=h, minute=mi)
142 if cand >= start:
143 return cand
144 cursor += timedelta(days=1)
145 raise_code("CFG-601", expr=expr,
146 hint=f"cron 无未来触发点(400 年内无匹配): {expr}",
147 advice="检查日/月组合是否存在(如 2 月 30 日不存在)")

156 def _due(self, job: ScheduleJob, now_dt: datetime) -> bool:
157 """触发窗判定:next_fire_at 已到(cron 额外要求当前分钟命中表达式)。"""
158 if job.next_fire_at is None:
159 return False

162 if job.kind == "cron": # 分钟对齐:当前分钟须命中
