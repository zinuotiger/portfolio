27def usage_metrics(journals):
28 usages=[];complete=True;requests=0
29 for events in journals:
30 req=Counter(e.payload['model'] for e in events if e.type=='llm.request')
31 values=[e.payload for e in events if e.type=='llm.usage'];usages.extend(values)
32 observed=Counter(v['model'] for v in values)
33 requests+=sum(req.values())
34 if req!=observed:complete=False
35 complete=complete and bool(usages)
36 input_known=complete and all(not v.get('input_unknown',False) for v in usages)
37 price_known=input_known and all(v.get('cost_est') is not None for v in usages)
38 return dict(request_records=requests,usage_records=len(usages),usage_complete=complete,
39 input_tokens=sum(v['in_tokens'] for v in usages) if input_known else None,
40 output_tokens=sum(v['out_tokens'] for v in usages) if complete else None,
41 cost_est=sum(v['cost_est'] for v in usages) if price_known else None,
42 usage_precision=('estimated' if any(v.get('estimated',False) for v in usages) else 'recorded') if complete else 'unavailable',
43 price_status='recorded_estimate' if price_known else 'unavailable')
