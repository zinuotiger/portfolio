41 def _authorize(self, *, write=False, grant_write=False):
42 self.service._ensure_open()
43 # Location/runtime owner only; every handoff still checks the actual
44 # requester, fixed resource grants and current original output bytes.
45 self.store._delivery_service=self.service
46 actor = current_actor.get()
47 if actor is not None and actor.get('tenant') != self.service.tenant_id:
48 raise_code('GRD-403', reason='project_tenant_mismatch')
49 if Accounts.manages_tenant(self.service._storage_root(), self.service.tenant_id):
50 roles = Accounts(self.service._storage_root()).teams(actor['user_id']) if actor else []
51 role = next((row['role'] for row in roles if row['id'] == self.service.tenant_id), None)
52 if role is None or (write and role not in {'owner', 'member'}) or (grant_write and role != 'owner'):
53 raise_code('GRD-403', reason='project_forbidden')
54 return actor['user_id'] if actor else 'local-operator'
