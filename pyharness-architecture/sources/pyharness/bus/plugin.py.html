1"""pyharness/bus/plugin.py — 插件宿主与五态生命周期 (specs/bus.py.md PluginManager)
2
3PluginManager:声明式 manifest(id/version/requires[]/api_version)装载——
4api_version 校验 → 依赖拓扑(缺失/未激活/循环依赖 → BUS-003)→ 状态收束 →
5registry 注册插件记录 → ctx.session.append 写 plugin.installed(F004/F006)。
6
7五态状态机唯一迁移闸 _set_state(spec F006 迁移表):
8 absent → installed → activating → active → deactivating → inactive
9 ↘(安装即卸)uninstalled ↖(重激活) ↘ uninstalled
10非法迁移 → BUS-003;落定 uninstalled 后插件管理记录摘除(支持热插拔重装,F004)。
11
12偏离点(契约内自洽修正,spec 伪码存在自相矛盾处):
131. spec 伪码 install 未把插件写入 Registry(kind=plugin),但 deactivate/uninstall
14 均要求 registry.lookup("plugin", pid)——故 install 落定后注册 PluginRecord;
152. spec 迁移表缺 absent 起始边(否则首次 install 必 BUS-003)与 installed→uninstalled
16 (装后未激活即卸载须可执行),_LEGAL 补两边;
173. 宿主能力契约(host.enter/detach,spec 指向 DIS-SEAM 未随文件展开):能力对象
18 须含 id(注册 capability 键)/tool_keys+tool(注册 tool 双键)/subscriptions
19 [(pattern, handler[, when])](以 owner=plugin_id 装载订阅);detach 逆序摘净;
204. api_version 常量取 "1"(EVENT-SCHEMA §3.5.6 示例值);
215. uninstall/投递中拦截用专类 BusyUninstall(BUSY 字面量语义,码表无 BUSY 条目,
22 spec 异常表明示非 NNN 码,故不走 raise_code)。
23
24错误全经 raise_code(F019);能力 enter 半载自回滚,不留脏注册表(TLB-801 纪律)。
25"""

71def _coerce_cap(cap: Any) -> Capability:
72 """dict/鸭子对象 → Capability;缺 id 一律 TLB-801 拒(非法注册)。"""
73 if isinstance(cap, Capability):
74 return cap
75 if isinstance(cap, dict):
76 cid = cap.get("id")
77 if not cid:
78 raise_code("TLB-801", kind="capability", detail="能力声明缺 id 字段")
79 return Capability(id=cid,
80 tool_keys=list(cap.get("tool_keys", ())),
81 tool=cap.get("tool"),
82 subscriptions=list(cap.get("subscriptions", ())))
83 if not getattr(cap, "id", None):
84 raise_code("TLB-801", kind="capability", detail="能力声明缺 id 字段")
85 return cap

88class PluginHost:
89 """能力宿主:enter = 注册 capability/tool 双键 + 装载订阅(F003 双键+F002);
90 detach = 摘订阅(owner 精确)+ 注销工具/能力(逆序,后装先卸,F004)。"""
91
92 def __init__(self, registry: Registry, bus: "EventBus", owner: str) -> None:
93 self.registry = registry
94 self._bus = bus
95 self.owner = owner # 插件 id:订阅 owner,uninstall 按此摘除

97 async def enter(self, cap: Any, ctx: Any) -> None:
98 """能力装载:announce 注册 + 订阅装载;半载异常自回滚,无脏数据。"""
99 cap = _coerce_cap(cap)
100 registered_tools: list[str] = []
101 try:
102 self.registry.register("capability", cap.id, cap)
103 for name in cap.tool_keys: # tool/capability 双键
104 self.registry.register("tool", name,
105 cap.tool if cap.tool is not None else cap)
106 registered_tools.append(name)
107 for item in cap.subscriptions: # 订阅装载(owner = 插件 id)
108 pattern, handler, *rest = item
109 self._bus.subscribe(pattern, handler,
110 owner=self.owner,
111 when=rest[0] if rest else None)
112 except Exception:
113 # 半载回滚:摘已注册键 + 本插件订阅(禁留脏注册表)
114 for name in registered_tools:
115 try:
116 self.registry.unregister("tool", name)
117 except Exception as exc: # noqa: BLE001 回滚尽力而为
118 log.warning("host.enter 回滚摘工具失败 tool=%s: %s", name, exc)
119 try:
120 self.registry.unregister("capability", cap.id)
121 except Exception as exc: # noqa: BLE001
122 log.warning("host.enter 回滚摘能力失败 cap=%s: %s", cap.id, exc)
123 self._bus.unsubscribe_all(self.owner)
124 raise

126 async def detach(self, cap: Any, ctx: Any) -> None:
127 """能力摘除:摘订阅 → 注销工具 → 注销能力(detach 内含 unsubscribe_all)。
128
129 2026-09-21 补:随工具一起**注销其声明的插件 guard 钩子**。此前只有
130 ``register_guard_hook`` 没有注销面 ⇒ ①卸载后陈旧 Guard 仍在表里,新声明
131 仍会解析到它;②含钩子的插件重装时撞 TLB-801"名全库唯一",与
132 ``PluginManager.uninstall`` docstring 的"支持重装(F004)"矛盾。
133 钩子名取自工具定义的 ``guard_hooks``(须在注销工具**之前**读)。
134 """
135 cap = _coerce_cap(cap)
136 self._bus.unsubscribe_all(self.owner)
137 hooks: list[str] = []
138 for name in cap.tool_keys:
139 try:
140 defn = self.registry.lookup("tool", name)
141 except Exception: # noqa: BLE001 已摘/不存在:跳过
142 defn = None
143 if defn is not None:
144 for h in tuple(getattr(defn, "guard_hooks", ()) or ()):
145 if str(h) not in hooks:
146 hooks.append(str(h))
147 self.registry.unregister("tool", name)
148 self.registry.unregister("capability", cap.id)
149 if hooks:
150 try:
151 from pyharness.core.tools_guard import unregister_guard_hook
152 except Exception: # noqa: BLE001 依赖缺失不阻断摘除
153 unregister_guard_hook = None
154 for h in hooks:
155 try:
156 if unregister_guard_hook is not None:
157 unregister_guard_hook(h)
158 except Exception as exc: # noqa: BLE001 摘钩失败不阻断
159 log.warning("detach 注销 guard 钩子失败 hook=%s: %s", h, exc)

173class PluginManager:
174 """插件宿主:五态生命周期 + 热插拔(F004/F006,契约见模块 docstring)。"""
175
176 # F006 五态迁移表(absent→installed 为首次安装入口;installed→uninstalled
177 # 支持"装后未激活即卸载";uninstalled 落定后记录摘除 → 可重装)
178 _LEGAL: dict[str, tuple[str, ...]] = {
179 "absent": ("installed",),
180 "installed": ("activating", "uninstalled"),
181 "activating": ("active",),
182 "active": ("deactivating",),
183 "deactivating": ("inactive",),
184 "inactive": ("uninstalled", "activating"),
185 }
186
187 def __init__(self, registry: Optional[Registry] = None,
188 host_factory: Optional[Callable[[str], PluginHost]] = None,
189 api_version: str = API_VERSION) -> None:
190 """registry 缺省自建(EventBus+Registry 自洽实例);host_factory 缺省按
191 (registry, bus, pid) 构建默认 PluginHost(owner=pid)。"""
192 if registry is None:
193 from pyharness.bus.event_bus import EventBus
194 bus: EventBus = EventBus()
195 registry = Registry(bus)
196 self.registry = registry
197 self._bus: EventBus = registry._bus
198 self._host_factory = host_factory
199 self.api_version = api_version
200 self._state: dict[str, str] = {} # pid → 五态(absent 缺省)
201 self._manifest: dict[str, dict] = {} # pid → manifest(卸载重装事实源)
202
203 # ------------------------------------------------------------ 查询
204 def state(self, pid: str) -> str:
205 """当前状态;未安装 → absent。"""
206 return self._state.get(pid, "absent")
207
208 # ------------------------------------------------------------ 状态机闸
209 def _set_state(self, pid: str, target: str) -> None:
210 """五态状态机唯一迁移闸;非法迁移 → BUS-003 拒绝(F006)。"""
211 cur = self.state(pid)
212 if target not in self._LEGAL.get(cur, ()):
213 raise_code("BUS-003", pid=pid, detail=f"{cur}→{target} 非法(F006 迁移表)")
214 self._state[pid] = target
215
216 # ------------------------------------------------------------ 依赖拓扑
217 def _dep_closure(self, pid: str, requires: list) -> list[str]:
218 """传递依赖闭包(拓扑序,依赖在前);循环/缺失 → BUS-003(F006)。"""
219 closure: list[str] = []
220 seen: set[str] = set()
221
222 def visit(node: str, path: tuple) -> None:
223 """递归收集 node 的已装传递依赖;环与缺失在此检出。"""
224 for dep in self._manifest.get(node, {}).get("requires", ()):
225 if dep in path: # 循环依赖检出失败
226 raise_code("BUS-003", pid=pid,
227 detail=f"循环依赖:{'->'.join(path + (dep,))}")
228 if self.state(dep) == "absent":
229 raise_code("BUS-003", pid=pid,
230 detail=f"依赖 {dep} 缺失或未安装(先装依赖)")
231 visit(dep, path + (dep,))
232 if node not in seen:
233 seen.add(node)
234 closure.append(node)
235
236 for dep in requires: # 根 requires(含 pid 自环)
237 if dep in (pid,):
238 raise_code("BUS-003", pid=pid, detail=f"循环依赖:{pid}->{dep}(自依赖)")
239 if self.state(dep) == "absent":
240 raise_code("BUS-003", pid=pid,
241 detail=f"依赖 {dep} 缺失或未安装(先装依赖)")
242 visit(dep, (pid, dep))
243 return closure

245 # ------------------------------------------------------------ 安装
246 async def install(self, manifest: dict, ctx: Any) -> None:
247 """声明式装载:api_version → 依赖拓扑 → 状态 installed + registry 注册
248 + ctx.session.append("plugin.installed") 留痕(F006/F004)。"""
249 pid = manifest.get("id")
250 if not pid:
251 raise_code("TLB-801", kind="plugin", detail="manifest 缺 id(非法注册)")
252 if pid in self.registry.RESERVED:
253 raise_code("BUS-002", key=pid, hint="脊柱名/子系统名不可装为插件(原则 5)")
254 if manifest.get("api_version") != self.api_version:
255 raise_code("BUS-003", pid=pid,
256 detail=f"api_version={manifest.get('api_version')} 不匹配 "
257 f"(框架 API_VERSION={self.api_version})")
258 closure = self._dep_closure(pid, list(manifest.get("requires", ())))
259 for dep in closure: # 依拓扑序校验依赖已激活
260 if self.state(dep) != "active":
261 raise_code("BUS-003", pid=pid, detail=f"依赖 {dep} 未激活(先 activate)")
262 self._set_state(pid, "installed")
263 record = PluginRecord(id=pid, manifest=manifest, state="installed")
264 self._manifest[pid] = manifest
265 self.registry.register("plugin", pid, record)
266 try:
267 await ctx.session.append("plugin.installed",
268 {"plugin_id": pid,
269 "version": manifest["version"],
270 "api_version": manifest["api_version"]},
271 actor="plugin")
272 except Exception: # 留痕失败:回滚,不留半装态
273 self.registry.unregister("plugin", pid)
274 self._manifest.pop(pid, None)
275 self._state.pop(pid, None)
276 raise

296 async def activate(self, pid: str, ctx: Any) -> None:
297 """activate:导入模块 → 逐能力 enter(注册+订阅)→ active;失败回滚到
298 installed,不留半激活态(F006)。"""
299 self._set_state(pid, "activating") # absent/非法前驱 → BUS-003
300 record = self.registry.lookup("plugin", pid)
301 host = self._host_for(pid)
302 entered: list = []
303 try:
304 for cap in self._load_caps(record.manifest):
305 cap = _coerce_cap(cap)
306 await host.enter(cap, ctx) # announce:tool/capability 双键
307 entered.append(cap)
308 record.capabilities.append(cap)
309 for name in cap.tool_keys:
310 if name not in record.tool_keys:
311 record.tool_keys.append(name)
312 self._set_state(pid, "active")
313 record.state = "active"
314 except Exception:
315 # 失败回滚:状态归 installed,已 enter 能力逆序 detach 摘净
316 self._state[pid] = "installed"
317 record.state = "installed"
318 record.capabilities = []
319 record.tool_keys = []
320 for cap in reversed(entered):
321 try:
322 await host.detach(cap, ctx)
323 except Exception as exc: # noqa: BLE001 回滚尽力而为
324 log.warning("activate 回滚 detach 失败 cap=%s: %s",
325 getattr(cap, "id", "?"), exc)
326 raise

328 async def deactivate(self, pid: str, ctx: Any) -> None:
329 """deactivate:逆序 detach 能力(摘订阅+注销工具/能力),插件记录保留在
330 Registry(仅摘订阅不注销注册表,F004);投递中(BUSY)拒绝。"""
331 record = self.registry.lookup("plugin", pid)
332 if pid in self._bus._inflight:
333 raise BusyUninstall(pid) # 投递中不可卸(BUSY 语义)
334 self._set_state(pid, "deactivating") # 非 active 前驱 → BUS-003
335 host = self._host_for(pid)
336 try:
337 for cap in reversed(list(record.capabilities)): # 后装先卸
338 await host.detach(cap, ctx)
339 except Exception:
340 self._state[pid] = "active" # 失败回滚,不留半失活态
341 record.state = "active"
342 raise
343 record.capabilities = []
344 record.tool_keys = []
345 self._set_state(pid, "inactive")
346 record.state = "inactive"

348 async def uninstall(self, plugin_id: str, ctx: Any) -> None:
349 """热卸载:运行/迁移中拒绝(BusyUninstall,BUSY 语义)→ 摘剩余工具/能力
350 → registry 摘 plugin → plugin.uninstalled 留痕;落定后管理记录摘除,
351 支持重装(F004)。不回溯已发生事件。"""
352 record = self.registry.lookup("plugin", plugin_id) # 不存在 → 结构化错误
353 st = self.state(plugin_id)
354 if st in ("active", "activating", "deactivating"):
355 raise BusyUninstall(plugin_id) # running 拒绝(状态机前置)
356 host = self._host_for(plugin_id)
357 for cap in reversed(list(record.capabilities)): # 通常已在 deactivate 摘净
358 await host.detach(cap, ctx)
359 record.capabilities = []
360 record.tool_keys = []
361 self.registry.unregister("plugin", plugin_id)
362 self._set_state(plugin_id, "uninstalled")
363 record.state = "uninstalled"
364 self._manifest.pop(plugin_id, None) # 记录摘除 → state(pid) 回落 absent
365 self._state.pop(plugin_id, None)
366 try:
367 await ctx.session.append("plugin.uninstalled",
368 {"plugin_id": plugin_id}, actor="plugin")
369 except Exception as exc: # 卸载已完成,留痕失败告警不阻断
370 log.error("plugin.uninstalled 留痕失败 plugin_id=%s: %s",
371 plugin_id, exc)

373 # ------------------------------------------------------------ 插件发事件
374 async def emit_as(self, plugin_id: str, type_: str, payload: dict) -> dict:
375 """插件以自身身份发事件:自动补 origin=plugin:<id>;类型未注册 → EVT-102
376 (先 register_type;命名空间 plugin.<id>.<name>)。会话事件须经 session.append
377 打 seq,插件不可自报 seq(INV-01)。"""
378 if not self._bus._type_registered(type_):
379 raise_code("EVT-102", type_=type_,
380 hint="插件事件命名空间 plugin.<id>.<name>,先 register_type")
381 return await self._bus.emit(type_, {**payload, "origin": f"plugin:{plugin_id}"})
