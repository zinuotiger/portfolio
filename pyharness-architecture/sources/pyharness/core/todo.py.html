1"""pyharness/core/todo.py — 会话级待办 TodoManager(#32 清单项)。
2
3职责一句话:事件源派生的待办表(todo.updated 为唯一写事件,append-only 真源),
4LLM 经 todo.* 工具消费(ctx.todos),外壳/桌面经同一管理器读看板。
5
6设计约束:
7- 状态只由事件派生:每次变更 append 一条 todo.updated {task_id, todos[]}(整表
8 快照语义 = 简易事件溯源;EVENT-SCHEMA TodoUpdatedPayload 字段级权威);
9- 单任务会话语义:task_id 缺省 "main"(与 goal 关联任务同口径,1 会话 1 任务);
10- 每任务 ≤20 项(词表 max_length 同源);done 项保留在表内(compaction 清空);
11- append 强类型二次校验(词表层),payload 非法即 EVT-100 拒写——不落脏事件。
12"""
13from __future__ import annotations
14
15import logging
16from typing import Any, Optional
17
18from pyharness.errors import raise_code
19
20log = logging.getLogger("pyharness.todo")
21
22DEFAULT_TASK_ID = "main"
23MAX_ITEMS = 20
24"""每任务待办上限(与 TodoUpdatedPayload.todos max_length=20 同源)。"""
25
26_OPS = ("add", "list", "done", "remove", "clear_done")
27
28
29class TodoManager:
30 """会话级待办注册表:add/list/done/remove/clear_done + 事件重建。
31
32 session = SessionLog 或同型 async append 协议(写事件落点);缺失时写操作
33 EVT-100 拒(只读重建实例)。_todos 为 {task_id: [TodoItem 型 dict]} 内存态,
34 可由事件全量重建(INV-01 不持第二份事实源)。"""
35
36 def __init__(self, session: Any = None, *, max_items: int = MAX_ITEMS) -> None:
37 self.session = session
38 self._max_items = int(max_items)
39 self._todos: dict[str, list[dict]] = {}
40 self._id_seq: int = 0
41
42 # ============================================================ 只读看板
43 def list_items(self, task_id: str = DEFAULT_TASK_ID) -> list[dict]:
44 """某任务待办(副本,只读;按添加序)。"""
45 return [dict(x) for x in self._todos.get(task_id, [])]
46
47 def active_count(self, task_id: str = DEFAULT_TASK_ID) -> int:
48 """未完成项计数(看板/预算徽章数据源)。"""
49 return sum(1 for x in self._todos.get(task_id, []) if not x.get("done"))
50
51 # ============================================================ 写操作
52 async def apply_op(self, op: str, *, text: Optional[str] = None,
53 todo_id: Optional[int] = None,
54 task_id: Optional[str] = None) -> str:
55 """op 分发(LLM 工具面):返回用户可读 summary 文本(executor 关4 直用)。
56
57 op ∈ add/list/done/remove/clear_done;非法 op → EVT-100;id 不存在 →
58 EVT-101(与 goal 定位异常同族)。list 为只读(不 append)。"""
59 tid = task_id or DEFAULT_TASK_ID
60 if op not in _OPS:
61 raise_code("EVT-100", hint=f"未知 todo op: {op}",
62 advice=f"op ∈ {_OPS}")
63 if op == "list":
64 return self._render(tid)
65 self._require_session()
66 items = self._todos.setdefault(tid, [])
67 if op == "add":
68 text = str(text or "").strip()
69 if not text:
70 raise_code("EVT-100", hint="todo add 缺 text", field="text")
71 if len(items) >= self._max_items:
72 raise_code("EVT-100", hint=f"待办超上限({self._max_items})",
73 advice="先 done/remove 再添加")
74 self._id_seq += 1
75 items.append({"id": self._id_seq, "text": text, "done": False})
76 await self._emit(tid)
77 return f"已添加待办 [{self._id_seq}] {text}(共 {len(items)} 项)"
78 # 其余 op 先定位 id(不存在 → EVT-101)
79 if op == "done":
80 for x in items:
81 if x["id"] == int(todo_id or -1):
82 x["done"] = True
83 await self._emit(tid)
84 return f"待办 [{x['id']}] {x['text']} 已完成"
85 raise_code("EVT-101", todo_id=todo_id, hint="todo id 不存在")
86 if op == "remove":
87 for i, x in enumerate(items):
88 if x["id"] == int(todo_id or -1):
89 items.pop(i)
90 await self._emit(tid)
91 return f"已移除待办 [{x['id']}] {x['text']}"
92 raise_code("EVT-101", todo_id=todo_id, hint="todo id 不存在")
93 if op == "clear_done":
94 kept = [x for x in items if not x.get("done")]
95 dropped = len(items) - len(kept)
96 self._todos[tid] = kept
97 await self._emit(tid)
98 return f"已清理 {dropped} 项已完成待办(剩 {len(kept)} 项)"
99 raise_code("EVT-100", hint=f"未知 todo op: {op}") # pragma: no cover
100
101 # ============================================================ 事件落点
102 async def _emit(self, task_id: str) -> None:
103 """todo.updated 强类型落盘(词表二次校验;失败上抛不静默)。"""
104 await self._require_session().append(
105 "todo.updated",
106 {"task_id": task_id, "todos": list(self._todos.get(task_id, []))},
107 actor="tool")
108
109 def _require_session(self) -> Any:
110 if self.session is None:
111 raise_code("EVT-100", hint="TodoManager 未注入 session,无法写事件",
112 advice="构造时注入 SessionLog 或同型 append 协议对象")
113 return self.session
114
115 # ============================================================ 渲染
116 def _render(self, task_id: str = DEFAULT_TASK_ID) -> str:
117 items = self._todos.get(task_id, [])
118 if not items:
119 return "当前无待办"
120 lines = [f"- [{'x' if x['done'] else ' '}] ({x['id']}) {x['text']}"
121 for x in items]
122 return "\n".join(lines)
123
124 # ============================================================ 事件重建
125 @classmethod
126 def rebuild_from_events(cls, events: list[Any]) -> "TodoManager":
127 """从事件流重建(崩溃恢复/新 ctx 接旧会话):逐条 todo.updated 重放
128 (last-wins 整表覆盖);未注入 session(只读实例,写操作将拒)。"""
129 mgr = cls(session=None)
130 for ev in events:
131 if getattr(ev, "type", "") != "todo.updated":
132 continue
133 p = getattr(ev, "payload", {}) or {}
134 tid = p.get("task_id") or DEFAULT_TASK_ID
135 raw = p.get("todos") or []
136 mgr._todos[tid] = [dict(x) for x in raw]
137 for x in mgr._todos[tid]: # id 续号(单调不重号)
138 mgr._id_seq = max(mgr._id_seq, int(x.get("id") or 0))
139 return mgr
