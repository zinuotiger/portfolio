93async def _run_on_session(spine: Any, session: Any, scope: Any, intent: str, *,
94 task_id: Optional[str], origin: str,
95 owner: Optional[str], depth: int = 0,
96 approval: Any = None, ask: Any = None) -> tuple[Any, int]:
97 loop = AgentLoop(getattr(spine, "settings", None))
98 env = await session.append(
99 "user.message", {"content": intent}, actor="user",
100 origin=origin, task_id=task_id, sync=True)
101 ctx = _runtime_ctx(spine, session, scope, loop, task_id=task_id,
102 owner=owner, depth=depth)
103 if approval is not None:
104 ctx.approval = approval
105 if ask is not None:
106 ctx.ask = ask
107 child_runtime = None
108 if session is not getattr(spine,'session',None) and ctx.platform_runtime is not None:
109 child_runtime = ctx.platform_runtime.for_child(session.sid)
110 ctx.platform_runtime = child_runtime
111 # N2-b:子会话/编排会话拥有**自己的任务段** —— 证据工件需要段锚才能被
112 # ``collect_for_task`` 聚合。修复前本函数不开段 ⇒ 子会话实测 segment.start=0。
113 # 段关闭放在 finally:与 ``TaskQueue._run_task`` 同款保证(段一定配对关闭)。
114 seg_start: Optional[int] = None
115 if task_id:
116 seg_env = await session.append("segment.start", {"task_id": task_id},
117 actor="system", sync=True)
118 seg_start = int(getattr(seg_env, "seq", 0) or 0)
119 delegated_owner=getattr(child_runtime,'delegated_task_lifecycle',None)
120 try:
121 if callable(delegated_owner):
122 result=await delegated_owner(env,ctx,lambda:loop.wake(env,ctx=ctx))
123 elif ctx.platform_runtime is None:
124 result = await loop.wake(env, ctx=ctx)
125 else:
126 async def governed_turn():
127 await ctx.platform_runtime.prepare_evidence(env,ctx)
128 return await loop.wake(env,ctx=ctx)
129 result = await asyncio.wait_for(governed_turn(),ctx.platform_runtime.definition.timeout)
130 if child_runtime is not None and getattr(result,'reason',None)=='complete':
131 await child_runtime.finish(ctx)
132 finally:
133 if child_runtime is not None and not callable(delegated_owner):
134 await child_runtime.release(ctx)
135 if seg_start is not None:
136 await session.append(
137 "segment.end", {"task_id": task_id, "start_seq": seg_start},
138 actor="system")
139 return result, int(getattr(env, "seq", 0) or 0)

142class EngineIntentRunner:
143 """JobManager runner: execute one background intent on the shared session.
144
145 经会话 TaskQueue 单飞泵提交(与交互轮次共享同一 loop/泵),避免另起 AgentLoop
146 直驱父会话造成"同会话并发 LLM 驱动 + 共享 counters 预算串扰"(EngineIntentRunner
147 绕过单飞泵,2026-09-13 修复)。无队列装配(轻装配/单测)时兜底直驱。入队用队列
148 自动 task_id(勿传 job id:JobManager 已为该 job id 开段,复用会撞"段已打开")。
149 """
150
151 def __init__(self, spine: Any) -> None:
152 self.spine = spine
153 self.queue_tasks: dict[str, str] = {}
154
155 async def run(self, ctx: Any, *, intent: str, task_id: str,
156 budget: Any = None) -> Any:
157 tq = getattr(self.spine, "task_queue", None)
158 session = getattr(self.spine, "session", None)
159 if tq is not None and session is not None:
160 # 先落 user.message(泵 runner 按 enqueued_seq 定位),再入队 + 等终态
161 await session.append("user.message", {"content": intent}, actor="user",
162 origin="job", task_id=task_id, sync=True)
163 queued = await tq.submit(intent) # 队列自动 t-N(避开 job 段)
164 self.queue_tasks[task_id] = queued
165 try:
166 res = await tq.wait_for(queued)
167 except asyncio.CancelledError:
168 primary = None
169 try:
170 await tq.cancel(queued)
171 except BaseException as exc:
172 primary = exc
173 # This is cancellation of work, distinct from a waiter's timeout.
174 terminal = asyncio.create_task(tq.wait_for(queued))
175 while True:
176 try:
177 res = await asyncio.shield(terminal)
178 break
179 except asyncio.CancelledError:
180 if terminal.done(): raise
181 # Repeated cancellation cannot abandon the work owner.
182 continue
183 if primary is not None:
184 raise primary
185 if not res.ok:
186 raise
187 # Completion won the race; preserve its actual success.
188 finally:
189 self.queue_tasks.pop(task_id, None)
190 ok = bool(getattr(res, "ok", False))
191 reason = "complete" if ok else str(getattr(res, "code", "") or "error")
192 return SimpleNamespace(reason=reason) # JobManager 读 .reason 归一
193 result, _ = await _run_on_session(
194 self.spine, session, getattr(self.spine, "scope", None),
195 intent, task_id=task_id, origin="job",
196 owner=getattr(ctx, "owner", None))
197 return result

215 def child_session(self, sub_id: str) -> Any:
216 """Create a persisted child SessionLog with an isolated bus.
217
218 The child must not share the parent bus: the parent engine persists all
219 events seen on its bus, so child events would otherwise be duplicated
220 into the parent JSONL.
221 """
222 from pyharness.bus import EventBus, bus_kwargs_of
223 from pyharness.core.session import SessionLog
224 from pyharness.events.vocab import EVENT_TYPES, SYNC_TYPES
225 from pyharness.persistence import open_store
226
227 # N1:子会话的 flush 标量同源(装配层解析后注入)
228 store = open_store(sub_id, dir=self.sessions_dir,
229 **_flush_kwargs_of(getattr(self.spine, "settings", None)))
230 child_bus = EventBus(**bus_kwargs_of( # F005 背压阈值同源注入
231 getattr(self.spine, "settings", None)))
232
233 async def _record(type_: str, payload: Any) -> None:
234 await store.append(payload, sync=type_ in SYNC_TYPES)
235
236 self._stores[sub_id] = store
237 self._child_buses[sub_id] = child_bus
238 try:
239 for type_ in EVENT_TYPES:
240 child_bus.subscribe(type_, _record, owner=f"subagent-store:{sub_id}")
241 # GAP-10:子会话**继承父会话租户**——租户是会话树的横切归属,
242 # 子 Agent 的事件同样必须可归属到同一租户。
243 log_ = SessionLog(sid=sub_id, persistence=store, bus=child_bus,
244 tenant_id=getattr(self.spine.session, "tenant_id", None))
245 # N2-b:子会话与父会话**同产治理证据** —— 触发点同为段关闭(``segment.end``),
246 # 只是换了"事件落点"(child log)与"事件汇点"(child bus)。生产者工厂由装配层
247 # 注入,本模块不反向 import engine(INV-08)。
248 if self._evidence_producer is not None:
249 holder = SimpleNamespace(
250 session=log_, governance=getattr(self.spine, "governance", None))
251 child_bus.subscribe(
252 "segment.end", self._evidence_producer(holder),
253 owner=f"subagent-evidence:{sub_id}")
254 return log_
255 except BaseException as primary:
256 try: self.close_child(sub_id)
257 except BaseException as secondary: primary.add_note(f'child rollback failed: {secondary!r}')
258 raise

260 def close_child(self, sub_id):
261 errors = []
262 child_bus = self._child_buses.pop(sub_id, None)
263 if child_bus is not None:
264 for owner in (f'subagent-store:{sub_id}', f'subagent-evidence:{sub_id}'):
265 try: child_bus.unsubscribe_all(owner)
266 except BaseException as exc: errors.append(exc)
267 store = self._stores.pop(sub_id, None)
268 if store is not None:
269 try: store.close()
270 except BaseException as exc: errors.append(exc)
271 if errors: raise BaseExceptionGroup('child cleanup failed', errors)

273 def completed_result(self, sub_id):
274 """Read a closed child's terminal facts; caller must establish ownership."""
275 import re
276 from pyharness.persistence import SessionReader
277 if not re.fullmatch(r'[A-Za-z0-9_-]{1,100}', sub_id):
278 return None
279 result = None
280 for event in SessionReader(sub_id, dir=self.sessions_dir).replay():
281 payload = event.payload
282 if event.type == 'subagent.joined' and payload.get('sub_id') == sub_id:
283 result = ('done', payload['summary'], None)
284 elif event.type == 'subagent.failed' and payload.get('sub_id') == sub_id:
285 match = re.match(r'\[([A-Z]+-\d+)\]', payload.get('summary',''))
286 result = ('failed', None, match[1] if match else 'CYC-999')
287 elif event.type == 'system.cancelled' and payload.get('what') == 'subagent:'+sub_id:
288 result = ('cancelled', None, None)
289 return result

291 async def run_child(self, env: Any) -> str:
292 from pyharness.core.approval import ApprovalProvider
293 from pyharness.core.user_ask import AskProvider
294 # Local event sequences are only unique within a session. Sharing a
295 # provider would collide two children with the same approval sequence.
296 # Children await their own loop directly, so they must not pause/resume
297 # another child's or the parent's TaskQueue.
298 approval = ApprovalProvider(session=env.session,
299 bus=self._child_buses.get(env.sub_id), config=self.spine.settings,
300 channel=getattr(self.spine, 'channel', None))
301 ask = AskProvider(session=env.session, bus=self._child_buses.get(env.sub_id))
302 self.approval_providers[env.sub_id] = approval
303 self.ask_providers[env.sub_id] = ask
304 try:
305 if env.scope is None:
306 from pyharness.errors import PyHError
307 raise PyHError('GRD-403', ctx={'reason':'child_scope_unavailable'})
308 result, start_seq = await _run_on_session(
309 self.spine, env.session, env.scope,
310 env.spec.task, task_id=f"subagent:{env.sub_id}",
311 origin=f"subagent:{env.sub_id}", owner="system",
312 depth=int(getattr(env.spec, "depth", 0)), approval=approval, ask=ask)
313 # Returning from the loop only means it stopped. A public plan or
314 # earlier response cannot turn a failed/budget-limited child into a
315 # successful join. Preserve the same explicit-success contract as
316 # TaskQueue, including cancellation propagation to the work owner.
317 reason = getattr(result, 'reason', None)
318 if reason == 'cancelled':
319 raise asyncio.CancelledError
320 if reason != 'complete':
321 from pyharness.errors import PyHError
322 code = getattr(result, 'error_code', None) or (
323 'LLM-305' if reason == 'budget' else 'CYC-999')
324 raise PyHError(code)
325 text = _last_text(env.session, start_seq)
326 text = text or str(getattr(result, "reason", "") or "")
327 # notify=False suppresses parent notifications, not durable child
328 # outcome facts. Reopening a completed child must not report a crash.
329 from pyharness.core.subagent import summarize
330 await env.session.append('subagent.joined',
331 {'sub_id':env.sub_id,'summary':summarize(text,2000)},actor='agent',sync=True)
332 return text
333 except BaseException as exc:
334 if isinstance(exc, asyncio.CancelledError):
335 await env.session.append('system.cancelled',
336 {'what':'subagent:'+env.sub_id,'reason':'cancelled'},actor='system',sync=True)
337 else:
338 await env.session.append('subagent.failed',
339 {'sub_id':env.sub_id,'summary':'['+str(getattr(exc,'code','CYC-999'))+']'},
340 actor='agent',sync=True)
341 raise
342 finally:
343 try:
344 from pyharness.core import proc
345 await asyncio.to_thread(proc.close_session, env.sub_id)
346 finally:
347 try:
348 await approval.aclose()
349 finally:
350 self.approval_providers.pop(env.sub_id, None)
351 self.ask_providers.pop(env.sub_id, None)
352 if getattr(env, 'sub_id', '') in self._stores:
353 self.close_child(env.sub_id)
