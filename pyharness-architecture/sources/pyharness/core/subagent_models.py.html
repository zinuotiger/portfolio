112def summarize(out: Any, max_chars: int = SUMMARY_MAX_CHARS) -> str:
113 """子结果摘要化(≤max_chars,截断补 …;恒非空——joined payload summary 必填)。
114
115 摘要由框架从执行输出截断生成,不由子 LLM 直接回写(防注入操纵摘要,SECURITY
116 §5 同 executor.summarize 语义);空输出给占位记号,保证 joined 事件合法。
117 """
118 text = render_result_text(out)
119 if not text:
120 return "…" # 空输出占位(非空摘要契约)
121 if len(text) <= max_chars:
122 return text
123 return text[: max_chars - 1] + "…"

125def _scale_limits(limits: Any, ratio: float) -> Any:
126 """父 BudgetLimits × ratio → 子预算(逐字段缩放,下限 1;None → None)。
127
128 鸭子构造:优先实例化真实 BudgetLimits(lazy import,装配期才需要);不可导入时
129 退回 SimpleNamespace 同字段(单测/隔离环境仍可断言属性)。
130 """
131 if limits is None:
132 return None
133 try:
134 from pyharness.core.scope import BudgetLimits # lazy:INV-08 单向装配
135 return BudgetLimits(
136 max_in_tokens=max(1, int(limits.max_in_tokens * ratio)),
137 max_out_tokens=max(1, int(limits.max_out_tokens * ratio)),
138 max_cost_yuan=float(limits.max_cost_yuan) * ratio,
139 warn_ratio=getattr(limits, "warn_ratio", 0.8))
140 except Exception: # noqa: BLE001 装配期 scope 缺席 → 鸭子回落
141 return type("ScaledLimits", (), {
142 "max_in_tokens": max(1, int(limits.max_in_tokens * ratio)),
143 "max_out_tokens": max(1, int(limits.max_out_tokens * ratio)),
144 "max_cost_yuan": float(limits.max_cost_yuan) * ratio,
145 "warn_ratio": getattr(limits, "warn_ratio", 0.8)})()
