72 @model_validator(mode='after')
73 def bounded_allocation(self):
74 if len({n.node_id for n in self.nodes})!=len(self.nodes):raise ValueError('duplicate funded node')
75 for key in TaskGraphLimits.model_fields:
76 if sum(getattr(n.allocated_budget,key) for n in self.nodes)!=getattr(self.allocated_budget,key):
77 raise ValueError('funded allocation sum')
78 if getattr(self.allocated_budget,key)>getattr(self.limits,key):raise ValueError('funded graph limit')
79 return self

93 @model_validator(mode='after')
94 def bound_grant(self):
95 if self.experiment_run_id!=self.selection.experiment_run_id:raise ValueError('funded run identity')
96 if self.graph_command.confirmation_digest!=self.graph_plan.confirmation_digest:raise ValueError('funded graph plan')
97 if self.graph_command.confirm_host_boundary!=self.selection.confirm_host_boundary:raise ValueError('funded boundary')
98 for key in TaskGraphLimits.model_fields:
99 if not getattr(self.graph_plan.allocated_budget,key)<=getattr(self.selection.budget_grant,key)<=getattr(self.graph_plan.limits,key):
100 raise ValueError('explicit grant must cover allocation within graph limits')
101 if any(n.boundary=='host_unrestricted' for n in self.graph_plan.nodes) and not self.selection.confirm_host_boundary:
102 raise ValueError('funded host boundary unconfirmed')
103 return self
