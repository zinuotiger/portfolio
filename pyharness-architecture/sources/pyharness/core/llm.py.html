145 def _egress_guard(self, method: str, ctx: Any) -> None:
146 """LLM 出网治理闸——**统一边界 + 显式 Negative Space**(GAP-1)。
147
148 ``_chat_any`` 是 LLM 出网的**唯一汇点**:``chat`` / ``chat_stream`` /
149 ``mini`` / ``summarize`` / ``json_chat`` 五个公开出口全部经它,故治理只
150 需在此**一处**加闸——不需要在 8 个出口复制 ``authorize()``(那会造成
151 架构碎片化,且多份副本迟早漂移)。
152
153 **为什么边界不是 ``governance.authorize()``**
154
155 ``governance`` 包是**工具调用**授权层:g1–g7 按**工具名**前缀匹配
156 (``fs.*`` / ``web.*`` / ``exec.*``),``authorize()`` 收 ``ToolCall``,产出的
157 ``decision.issued`` / ``receipt.emitted`` 经 ``call_id`` 串联 ``tool.call`` /
158 ``tool.result``。**LLM 请求不是工具调用**:无工具名、无参数集合、无
159 Provider 副作用。把它塞进 ``authorize()`` 只能靠伪造 ``ToolCall``,会污染
160 工具治理链语义。故**不这么做**(P1:决策权与执行权分离,不因凑指标而破坏)。
161
162 **LLM 出网实际适用的治理面 = 会话级预算硬闸**,本闸把它下沉到唯一汇点:
163 ``ctx.scope.check_budget()``(超限抛 ``BudgetExhausted``,与 ``agent_loop``
164 轮起点**同源同信号**)。修复前该闸只在轮起点调用,``mini``(自动标题)/
165 ``summarize``(压缩)/``json_chat``(计划)三条**轮内**系统出口可绕过。
166
167 **三态语义**
168
169 1. scope 在位且预算 ok ⇒ 放行(正常落 ``llm.request`` / usage);
170 2. scope 在位且 paused/exhausted ⇒ **拒**(抛 ``BudgetExhausted``,
171 **零出网**:不产生 ``llm.request``);
172 3. scope 缺失(纯内存/轻装配/单测替身)⇒ **降级放行**。理由:预算闸是
173 **配额面**而非**授权面**,缺配额不等于越权;而事件落点/守卫缺失会让
174 审计不成立,那些才必须 fail-closed(见 executor ``_require_wiring``)。
175
176 **Negative Space(显式声明哪些 LLM 出口不经本闸,及理由)**
177
178 - ``OpenAICompatAdapter.chat`` / ``.chat_stream``:适配器层,位于本类之后。
179 已核验**全库无生产代码直接调用**(CI 静态闸扫此),唯一路径 = 本类
180 ``_chat_any`` ⇒ 已被本闸覆盖。
181 - ``OpenAICompatAdapter.ping`` / 降级链健康探针:``ping`` 不产生 tokens、不落
182 F029 计量、不进 ``llm.request``;让其受会话预算约束会使"探测可用性"被预算
183 拒,语义颠倒。**明确不治理**,并登记于 ``LIMITATIONS.md``。
184 - 出网**之后**的审计面(``llm.request`` / ``llm.response`` / ``llm.usage``)
185 不属本闸:那是**观测**面,由适配器落事件(唯一真源仍是 append-only 日志)。
186 """
187 scope = getattr(ctx, "scope", None) if ctx is not None else None
188 if scope is None:
189 return # 三态 ③:未装配配额面
190 check = getattr(scope, "check_budget", None)
191 if callable(check):
192 check() # 三态 ②:超限即抛,零出网

249def _extract_json_text(text: str) -> Any:
250 """裁剪 ```json 围栏后抽取首个完整 JSON 对象/数组。"""
251 cleaned = re.sub(r"^```(?:json)?\s*", "", text.strip(), flags=re.I)
252 cleaned = re.sub(r"\s*```$", "", cleaned).strip()
253 for open_ch, close_ch in (("{", "}"), ("[", "]")):
254 lo, hi = cleaned.find(open_ch), cleaned.rfind(close_ch)
255 if lo != -1 and hi > lo:
256 return json.loads(cleaned[lo:hi + 1])
257 raise ValueError("模型输出不含 JSON 对象/数组")

275 async def aclose(self):
276 self.closed = True
277 task = getattr(self, '_close_task', None)
278 if task is None or (task.done() and (task.cancelled() or task.exception() is not None)):
279 task = self._close_task = asyncio.create_task(self._close_owned())
280 await asyncio.shield(task)

283 errors = []
284 for adapter in {id(a): a for a in self.registry.values()}.values():

287 await adapter.aclose()
288 except BaseException as exc:
289 errors.append(exc)

291 raise BaseExceptionGroup('model transport cleanup failed', errors)

342 detail="适配器缺端点三元组:构造 OpenAICompatAdapter 需 "
343 "triple 或注入 transport")
344 self._transport = build_client(self._triple, self.timeout)
345 return self._transport

365def resolve_secret_ref(ref: str) -> str:
366 """F016 单口最小落地:env/file/tenant 引用。缺失 → CRED-701。
367
368 偏离 6:credentials 模块(真 F016)未在本阶段实现,此为默认 resolver;凭据模块落地后
369 由装配层以 resolver 注入 build_client 替换。禁字面量密钥(配置层已拒载,CFG §7.1)。
370 """
371 val: Optional[str] = None
372 if ref.startswith("tenant:"):
373 from pyharness.core.tenant_settings import resolve_tenant_secret
374 return resolve_tenant_secret(ref)
375 if ref.startswith("env:"):
376 val = os.environ.get(ref[4:].strip())
377 elif ref.startswith("file:"):
378 path = os.path.expanduser(ref[5:].strip())
379 # CRED-702(2026-09-21 R27 接线):凭据文件**权限过宽** ⇒ 拒载,不静默读。
380 # 该码此前**已登记但全库从不抛出**(TS-06 实测:43 码中唯一无测试者)——
381 # 文档承诺的"权限过宽(>600)→ 启动拒载"在实现里根本不存在。POSIX 查 mode
382 # 的 group/other 位;Windows 的 mode 位是合成的、无安全含义(仓库既有口径:
383 # ACL 尽力而为),故 Windows 上跳过该检查。
384 if os.name != "nt":
385 try:
386 mode = os.stat(path).st_mode & 0o777
387 except OSError:
388 mode = None
389 if mode is not None and mode & 0o077:
390 raise_code("CRED-702", ref=ref, path=path, mode=oct(mode),
391 hint="凭据文件权限过宽(>600):chmod 600 后重试")
392 try:
393 with open(path, "r", encoding="utf-8") as fh:
394 val = (fh.readline().strip() or None) # 首行;空文件 = 缺失
395 except OSError as e:
396 raise_code("CRED-701", ref=ref, reason=type(e).__name__,
397 hint="凭据缺失:检查 file:PATH 与权限(600)")
398 else:
399 raise_code("CRED-701", ref=ref, reason="secret_ref 语法非法",
400 hint="凭据只接受 env:NAME / file:PATH / tenant:租户:档案 引用")
401 if not val:
402 raise_code("CRED-701", ref=ref,
403 hint="凭据缺失:配置 env/file/tenant 引用后重试")
404 return val

407def build_client(triple: AdapterTriple, limits: TimeoutLimits, *,
408 resolver: Any = None) -> _OpenAICompatHTTPTransport:
409 """客户端构建(ADI §1.1):按三元组建传输客户端;key 经 resolver(F016 单口)缺失 → CRED-701。
410
411 偏离 2:spec 返回 openai.AsyncOpenAI——openai SDK 非本阶段运行依赖(真实调用留 e2e),
412 此处返回 httpx OpenAI 兼容传输(complete/stream/ping,语义与 create 对齐),可整体注入。
413 """
414 if triple.credential_store is not None:
415 resolver = lambda ref: triple.credential_store.resolve_ref(ref, binding=triple.credential_binding)
416 else:
417 resolver = resolver or resolve_secret_ref
418 key = resolver(triple.api_key_ref) # 缺失 → CRED-701(绝不空串/None 续跑)
419 if not key:
420 raise_code("CRED-701", ref=triple.api_key_ref,
421 hint="凭据解析为空,拒绝构建客户端(绝不空串续跑)")
422 base = triple.base_url.rstrip("/") # 裸域或 /v1 结尾;禁写 /chat/completions
423 timeout = httpx.Timeout(limits.connect_s, # 连接/写/池 档
424 read=limits.first_token_s, # 首 token 档(60s)
425 write=limits.connect_s, pool=limits.connect_s)
426 transport = _OpenAICompatHTTPTransport(base_url=base, api_key=key, timeout=timeout)
427 if triple.credential_store is not None:
428 transport._validate = lambda: triple.credential_store.validate_binding(
429 triple.api_key_ref, triple.credential_binding)
430 return transport

437def model_configuration_admission(actual,expected,registry,model):
438 from pyharness.core.model_connection_identity import connection_registry_key
439 actual_keys=[actual.model,*(actual.fallback_models or [])]
440 wanted=[expected.model,*(expected.fallback_models or [])]
441 if len(actual_keys)!=len(wanted):return False,False
442 limits=TimeoutLimits.from_cfg(expected.timeout)
443 for key,name in zip(actual_keys,wanted):
444 if key==name:continue # old explicit borrowed-adapter configuration
445 adapter=registry.get(key)
446 if type(adapter) is not OpenAICompatAdapter or adapter._closed:return False,False
447 triple=adapter._triple
448 if (triple is None or adapter.model!=name or triple.model!=name or
449 triple.api_key_ref!=expected.api_key or triple.base_url!=expected.base_url or
450 adapter.timeout!=limits):return False,False
451 store=triple.credential_store;binding=triple.credential_binding
452 if store is not None:
453 # Original store validates the exact generation; no secret is read.
454 store.validate_binding(triple.api_key_ref,binding)
455 derived=connection_registry_key(expected.api_key,expected.base_url,name,limits,binding,
456 str(store.root.resolve()) if store is not None else '')
457 if key!=derived:return False,False
458 value=actual.model_dump();value['model']=expected.model;value['fallback_models']=expected.fallback_models
459 return value==expected.model_dump(),model in [*actual_keys,*wanted]
