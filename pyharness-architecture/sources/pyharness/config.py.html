156def env_subset() -> dict:
157 """L3 层:白名单内 PH_* 收集并按映射展开为嵌套键;名单外 PH_* → CFG-607 警告不中止。"""
158 out: dict = {}
159 for name, dotted in ENV_WHITELIST.items():
160 raw = os.environ.get(name)
161 if raw is None:
162 continue
163 if name == "PH_CFG_PATH":
164 continue # cfg_path 已在 collect_config_files 消费
165 val = parse_env_value(raw)
166 if val is not None:
167 set_dotted(out, dotted, val) # 点分键写入嵌套 dict
168 for name in os.environ: # 名单外:警告不中止(CFG §4.5 禁止面除外)
169 if name.startswith("PH_") and name not in ENV_WHITELIST:
170 log.warning("CFG-607 unknown env key: %s (白名单外,忽略)", name)
171 return out

194def load_settings(*cli_overrides: tuple[str, Any]) -> "Settings":
195 """F021 主管线:merge(L1←L2←L3←L4)→ Settings.model_validate → validate_rules。
196
197 非法 → CFG-601 列字段,拒绝启动(启动第 1 步);未知键 → CFG-607 警告不中止。
198 cli_overrides: (点分键, 值) 序列,来自 CLI --key value。
199 """
200 raw = _merge_layers(None, cli_overrides)
201 try:
202 settings = Settings.model_validate(raw) # 类型/范围/枚举(pydantic 逐键)
203 except ValidationError as e:
204 raise_code("CFG-601", reason="type_range",
205 fields=_validation_fields(e),
206 detail="类型/范围/枚举越界,修四层配置对应字段")
207 validate_rules(settings) # 结构/秘密/安全单调(§5.1 ③-⑤)
208 # 未知键收集:raw 平面键 − 模型键 → CFG-607 警告(只报键名,不含值)
209 known = set(flatten(settings.model_dump()))
210 for key in sorted(set(flatten(raw)) - known):
211 log.warning("CFG-607 unknown config key: %s", key)
212 log.info("config loaded: %d keys", len(flatten(settings.model_dump())))
213 return settings

221def validate_rules(settings: "Settings") -> None:
222 """pydantic 之外的五类规则(CFG §5.1 ③-⑤):超时档序、guard disabled、秘密
223 语法/字面量拒载、安全单调越权。任一失败 → CFG-601 启动中止。"""
224 t = settings.llm.timeout
225 if not (t.connect_s < t.first_token_s < t.total_s): # ③ 时间档序
226 raise_code("CFG-601", reason="structural", fields=["llm.timeout.*"],
227 detail="超时须 connect_s < first_token_s < total_s")
228 dis = set(settings.security.guards.disabled)
229 if dis and (dis == BUILTIN_GUARDS or not dis <= BUILTIN_GUARDS): # 单调只紧不松
230 raise_code("CFG-601", reason="越权", fields=["security.guards.disabled"],
231 detail="禁全关/未注册 guard 名;只许关单个内置(真子集)")
232 for tool, lvl in settings.security.tool_danger_extra.items(): # ⑤ danger 只许调高
233 if _DANGER_RANK[lvl] < _DANGER_RANK[baseline_danger(tool)]:
234 raise_code("CFG-601", reason="越权",
235 fields=[f"security.tool_danger_extra.{tool}"],
236 detail=f"danger 只许调高收紧,{tool} 基线={baseline_danger(tool)}")
237 # 非回环绑定必须已配 token 引用(CFG §3.8);空串 token 视为未配置
238 host = settings.shell.web.host
239 token = settings.shell.web.token

242 detail="绑非回环必须配 token(env:/file: 引用)")
243 for loc, ref in secret_fields(settings): # ④ 秘密语法

245 raise_code("CFG-601", reason="secret_literal", fields=[loc],
246 detail="字面量密钥拒载,改 env:/file: 引用")

260def secret_ref_ok(value: Any) -> bool:
261 """secret-ref 语法校验:仅 env:NAME / file:PATH;字面量匹配 sk- 前缀或 ≥32 位
262 疑似密钥 → False(空串/None = 未配置,允许;使用时由凭据模块 CRED-701 拒)。"""

302def looks_like_secret(key: str, value: Any) -> bool:
303 """疑似秘密值判定(render_show 出口脱敏用):键名提示 或 sk- 前缀 或 ≥32 位无分隔串。"""
304 if not isinstance(value, str):

307 return False # 引用本身不是秘密值

343def hot_update(holder: SettingsHolder, key: str, value: Any, by: str = "system") -> None:
344 """热更唯一入口:key ∈ RUNTIME_MUTABLE 且值过全量校验 → 生效并写 config.updated
345 事件;否则 CFG-608 拒绝(不静默、不部分生效)。"""
346 if key not in RUNTIME_MUTABLE: # 策略/预算/安全类 → 拒绝
347 raise_code("CFG-608", key=key, hint="该键需重启生效;热更仅观测键")
348 old = get_dotted(holder.settings.model_dump(), key)
349 candidate = deep_merge(holder.settings.model_dump(), _set_dotted_new(key, value))
350 try:
351 new_settings = Settings.model_validate(candidate)
352 except ValidationError:
353 raise_code("CFG-601", reason="type_range", fields=[key],
354 detail="热更值类型/范围非法")
355 validate_rules(new_settings) # 热更值同过全量校验
356 holder.settings = new_settings # 整体替换快照(不可变语义保持)

363def render_show(settings: "Settings") -> dict:
364 """config show 后端:生效值扁平化,秘密只回显 env:NAME/file:PATH 引用(--json 支持)。"""
365 flat = flatten(settings.model_dump())
366 out: dict = {}
367 for k, v in flat.items():
368 if isinstance(v, str) and (v.startswith("env:") or v.startswith("file:")):
369 out[k] = v # 秘密回显引用不显示值(无 --show-secrets)
370 elif looks_like_secret(k, v):
371 out[k] = redact(str(v))
372 else:
373 out[k] = v
374 return out

377def _config_error_text(e: ConfigError) -> str:
378 """config validate 错误文案:码+处置+字段明细(不吐 env 值/完整文件)。"""
379 fields = e.ctx.get("fields")
380 suffix = ""
381 if fields:
382 suffix = " 字段:" + ",".join(str(f) for f in fields)
383 tip = e.ctx.get("detail")
384 if tip:
385 suffix += f" 明细:{tip}"
386 return f"{e.code}:{e.spec.advice}{suffix}"
