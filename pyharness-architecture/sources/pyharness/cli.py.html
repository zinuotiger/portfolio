108async def _attach_engine(sh: ShellCtx) -> None:
109 """CLI 真引擎装配:把轻量门面换成 spine + runner + 真实 TaskQueue。
110
111 bootstrap_shell 只做会话/权限门面;chat/run/plan 真正干活前必须把
112 engine 装配进来(原设计留了 ctx.handlers/runner 注入点但从未接线)。
113 """
114 ctx = getattr(sh, "ctx", None)
115 if ctx is None:
116 raise_code("CYC-999", hint="ShellCtx 缺 ctx,无法装配引擎")
117 if getattr(ctx, "engine_spine", None) is not None:
118 return
119 from pyharness import engine as _eng
120 log_ = getattr(ctx, "session", None)
121 if log_ is None:
122 raise_code("CYC-999", hint="引擎装配前无会话(chat/run/plan 应先开/建会话)")
123 channel = "cli" if not getattr(sh, "headless", True) else None
124 # The bootstrap provider owns verdict subscriptions on this same bus.
125 # Close it before replacing it so only the runtime receives its verdicts.
126 bootstrap_approval = getattr(ctx, "approval", None)
127 if bootstrap_approval is not None:
128 await bootstrap_approval.aclose()
129 await _eng.attach_engine_to_ctx(
130 ctx, ctx.settings, log_=log_,
131 sessions_dir=Path(getattr(getattr(ctx, "storage", None),
132 "sessions_dir", ".")).expanduser(),
133 bus=ctx.bus,
134 store=getattr(log_, "_persistence", None),
135 channel=channel)

138def _attach_log_persistence(bus: Any, log_: Any, store: Any) -> None:
139 """总线 → 存储订阅(**委托唯一实现** ``persistence.session_recorder``)。
140
141 2026-09-21 R9:此前这里是**第三份**各自实现 —— 属主固定 ``"persistence"``(多会话
142 同总线时"按属主摘除"会互相误伤)且**无 sid 过滤**(总线上他会话事件会被写进本会话
143 store,同 R8-1 在 engine 侧修掉的缺陷)。现统一:属主 ``persistence:{sid}`` +
144 过滤在 ``session_recorder`` 单点。
145
146 SessionLog 构造时 bus=None(open_session 面),装配后回填 _bus 并先订后写。
147 """
148 from pyharness.events import EVENT_TYPES
149 from pyharness.persistence import session_recorder
150
151 sid = str(getattr(log_, "sid", "") or "")
152 record = session_recorder(store, sid)
153 for t in EVENT_TYPES: # 词表逐类型订阅(精确命中)
154 bus.subscribe(t, record, owner=f"persistence:{sid}")
155 log_._bus = bus # 回填总线:append 即分发即落盘

235def main(argv: Optional[list[str]] = None) -> int:
236 """进程入口(console_scripts `pyharness`):事件循环只在这里创建;任何未捕获
237 异常 → CYC-999 结构化到 stderr、退出 1。返回进程退出码。"""
238 try:
239 parsed = parse_args(argv if argv is not None else sys.argv[1:])
240 return asyncio.run(cli_main(parsed))
241 except KeyboardInterrupt:
242 print("\n[interrupted] 已退出", file=sys.stderr)
243 return EXIT_INTERRUPT
244 except _CancelledRound: # 信号→循环机制层透传的受控路径(偏离 10)
245 print("\n[已取消当前轮,再按 Ctrl-C 退出]", file=sys.stderr)
246 return EXIT_INTERRUPT
247 except PyHError as e: # 引擎错误:带码退出(1)
248 _emit_error(e, json_mode=_args_json())
249 return 1
250 except SystemExit: # argparse 用法错误(2)/--help(0)透传
251 raise
252 except Exception as e: # 未预期兜底,禁现场造码(CYC-999)
253 print(f"CYC-999 {e!r}", file=sys.stderr)
254 log.debug("cli uncaught", exc_info=True)
255 return 1

265 return offline_cmd(r.cmd, r.positional, r.flags)

275 hint="用法:pyharness run \"任务文本\" 或管道喂入")

280 if r.cmd == "desktop":

282 return await run_desktop(sh.ctx)

293 return await serve(sh.ctx, client_id="acp:cli")

342 # 其余命令(repair/desktop/acp/引擎单发)不预开会话,由各自处理器按需装配。
343 if sid is None and cmd in {"chat", "run"}:
344 bad = await _scan_unhealthy(sessions_dir, cfg=cfg) # repair 模块未装配时跳过
345 if bad is not None:
346 sid = bad # 命中损坏会话 → 修复并续跑
347 if sid is not None and cmd in {"chat", "run", "plan"}:
348 path = sessions_dir / f"{sid}.jsonl"
349 if not path.exists():
350 raise_code("EVT-106", sid=sid,
351 hint="会话不存在(日志缺失);先 `session list` 确认 sid,"
352 "或省略 --session 开启新会话")
353 await _ensure_repair(ctx, sid) # 损坏先修再 open(F060)
354 ctx.session = await _open_log(cfg, sid, ctx.bus)
355 _wire_queue(ctx)
356 elif sid is None and cmd in {"chat", "run"}:
357 # 新建会话:run/chat 是会话产生者(引擎后续只在已开会话上跑任务)
358 ctx.session = await _new_session(cfg, ctx.bus)
359 _wire_queue(ctx)

361 interactive_cmd = cmd in INTERACTIVE_CMDS and not r.flags.get("once")
362 headless = not (sys.stdin.isatty() and interactive_cmd)
363 ctx.channel = None if headless else "cli" # R8:无通道即安全默认
364 return ShellCtx(ctx=ctx, loop=asyncio.get_running_loop(),
365 channel=ctx.channel, headless=headless, flags=r.flags)

368def install_two_stage_sigint(sh: ShellCtx) -> dict:
369 """两段式 SIGINT:第 1 次置位并 raise _CancelledRound(取消当前轮);第 2 次
370 SystemExit(130)。Windows 同语义(禁 SIGKILL);返回 state 供测试取 handler。"""
371 state: dict[str, Any] = {"armed": False}
372
373 def handler(signum: int, frame: Any) -> None:
374 if not state["armed"]:
375 state["armed"] = True
376 print("\n[Ctrl-C] 再按一次退出(取消当前轮)", file=sys.stderr)
377 raise _CancelledRound()
378 raise SystemExit(EXIT_INTERRUPT)
379
380 try:
381 signal.signal(signal.SIGINT, handler)
382 except (ValueError, OSError): # 非主线程/平台限制:降级不装
383 log.warning("SIGINT handler 安装失败(非主线程?),Ctrl-C 语义降级", exc_info=True)
384 state["handler"] = handler
385 return state

395async def interactive_loop(sh: ShellCtx, *, once: Any = False) -> int:
396 """chat 交互主循环:多行输入(`\\` 续行)、斜杠命令交 handle_slash(零 LLM)、
397 普通文本交 task_queue.submit 并流式渲染、审批交互弹提示;Ctrl-C 一次取消当前
398 轮、两次退出(130);once 真值 = 单轮即返(--once 非交互注入,见偏离 6)。"""
399 state = install_two_stage_sigint(sh)
400 once_round = bool(once)
401 while True:
402 try:
403 raw = await read_user_input(sh, once=once_round)
404 if raw is None: # EOF(Ctrl-D)→ 退出 0
405 return 0
406 if raw.startswith("/"): # 斜杠:不给 LLM 的控制通道(F041)
407 cont = await commands.handle_slash(raw, sh.ctx)
408 code = _shell_exit_code(sh)
409 if code is not None: # /exit 经 ctx.shell 置位
410 return code
411 if not cont: # 兼容旧面 false=退出
412 return 0
413 if once_round:
414 return 0
415 continue
416 await _submit_wait_report(sh, raw) # 文本意图:入队+渲染+终态
417 if once_round:
418 return 0 # 单轮模式:处理完一轮即返
419 except _CancelledRound:
420 state["armed"] = False # 重布防:下一轮再按 = 取消该轮
421 print("\n[已取消当前轮,再按 Ctrl-C 退出]", file=sys.stderr)
422 continue # 回循环顶(等价伪码递归,见偏离 5)
423 except PyHError as e: # 引擎错误回显后回循环(交互韧性)
424 _emit_error(e, json_mode=bool(sh.flags.get("json")))
425 continue
426 except KeyboardInterrupt: # 兜底(信号未达 handler 的路径)
427 return EXIT_INTERRUPT

430async def _submit_wait_report(sh: ShellCtx, raw: str) -> None:
431 """一轮文本意图:submit → 订阅渲染 → wait_for 终态;失败摘要回显(引擎装配后
432 事件由渲染器呈现,此处只兜终态失败)。"""
433 q = getattr(sh.ctx, "task_queue", None)
434 if q is None:
435 raise_code("CYC-999", hint="任务队列未装配(runner 注入点待引擎装配)")
436 meta = {"channel": sh.ctx.channel or "headless"}
437 session = getattr(sh.ctx, "session", None)
438 if session is not None:
439 await session.append(
440 "user.message", {"content": raw}, actor="user", origin="cli",
441 sync=True)
442 task_id = await q.submit(raw, meta=meta)
443 res = await _wait_and_render(sh, task_id)
444 if res is not None and not _result_ok(res):
445 _emit_plain(f"[任务失败] {_result_code(res) or 'CYC-999'}"
446 f" {_result_summary(res)}", err=True)

449async def read_user_input(sh: ShellCtx, *, once: bool = False) -> Optional[str]:
450 """多行读取 + once 直通:once → --once 文本或管道全文;stdin 非 tty → 整读
451 一次性消费(headless run/chat 语义);tty → 逐行,行尾 `\\` 续行,空输入重提示;
452 EOF(Ctrl-D) → None。"""
453 if once:
454 txt = sh.flags.get("once") or sys.stdin.read()
455 return (txt or "").strip() or None
456 if not sys.stdin.isatty():
457 data = sys.stdin.read()
458 return data.strip() or None # headless 一次性消费
459 lines: list[str] = []
460 json_mode = bool(sh.flags.get("json"))
461 while True:
462 _write_prompt(sh, json_mode=json_mode)
463 try:
464 line = await asyncio.to_thread(input, "")
465 except EOFError:
466 return None # Ctrl-D:净退
467 except KeyboardInterrupt:
468 raise # 两段式向上冒泡(130)
469 if not line:
470 if not lines:
471 continue # 空输入重提示
472 break # 续行中空行 = 提交
473 if line.endswith("\\"):
474 lines.append(line[:-1]) # 续行:去尾 \ 拼接
475 continue
476 lines.append(line)
477 break
478 return "\n".join(lines).strip() or None

489async def run_intent(sh: ShellCtx, text: str, *, headless: bool) -> int:
490 """run 单发任务:无人值守单轮 submit → 阻塞至终态 → 报告结果/拒绝清单;
491 headless 下 channel=None 使审批自动拒(APR-501)与 critical 直拒(GRD-401)
492 天然生效(R8)。返回 0/1。"""
493 if not text:
494 raise_code("EVT-100", advice="run 需要任务文本(stdin 或参数)",
495 hint="用法:pyharness run \"任务文本\" 或管道喂入")
496 sh.ctx.channel = None if headless else "cli" # 无审批通道即安全默认(R8)
497 q = getattr(sh.ctx, "task_queue", None)
498 if q is None:
499 raise_code("CYC-999", hint="任务队列未装配(runner 注入点待引擎装配)")
500 session = getattr(sh.ctx, "session", None)
501 if session is not None:
502 await session.append(
503 "user.message", {"content": text}, actor="user", origin="cli",
504 sync=True)
505 task_id = await q.submit(text, meta={"channel": sh.ctx.channel})
506 res = await _wait_and_render(sh, task_id)
507 rejected = _rejected_list(res) # guard.rejected 拒绝明细(G4)
508 if sh.flags.get("json"):
509 _json_line({"task_id": task_id,
510 "result": {"summary": _result_summary(res),
511 "code": _result_code(res) or None,
512 "reason": _result_reason(res) or None},
513 "rejected": [str(r) for r in rejected]})
514 else:
515 print(_result_summary(res))
516 for r in rejected: # 拒绝清单逐条(含 headless 拒)
517 print(f"[拒绝] {r}")
518 reason = _result_reason(res)
519 if reason in {"complete", "cancelled_by_user"}:
520 return 0
521 if _result_ok(res):
522 return 0
523 return 1 # 引擎错误终态(reason=error)
