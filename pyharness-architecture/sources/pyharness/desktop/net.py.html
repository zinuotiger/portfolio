22def pick_free_port(host: str = HOST) -> int:
23 """127.0.0.1 随机空闲端口(bind 0 取系统分配 → 释放交 uvicorn;竞窗极小,F065 边界)。"""
24 with socket.socket(socket.AF_INET6 if ':' in host else socket.AF_INET, socket.SOCK_STREAM) as s:
25 s.bind((host, 0))
26 return int(s.getsockname()[1])

33def resolve_bind(cfg: Any) -> tuple:
34 """解析实际监听 ``(host, port)`` —— `shell.web.host`/`shell.web.port` 的**唯一读取点**。
35
36 R24 接线:此前该二键**零读取者**(L-23:绑模块常量 + ``pick_free_port()``,运维设了
37 不生效)。接线取舍(**安全姿态优先**):
38
39 - **host 只允许 loopback**(127.0.0.1 / localhost / ::1);配成其他地址 → ``CFG-601``
40 **拒绝启动**。理由:"服务仅绑 127.0.0.1"是本项目的安全基线,把 host 做成可指
41 ``0.0.0.0`` 的旋钮等于让一次手滑就对外开端口 ⇒ 这里 **fail-closed**。
42 - **port**:``0`` 或**被占用** → 回落 ``pick_free_port()``(保持"总能起来"的既有体验,
43 不因端口冲突让整个桌面壳启动失败);否则用配置值。
44 """
45 web = getattr(getattr(cfg, "shell", None), "web", None)
46 host = str(getattr(web, "host", None) or HOST).strip()
47 if host not in _LOOPBACK_HOSTS:
48 raise_code("CFG-601", field="shell.web.host", value=host,
49 advice="Web 壳只允许绑 loopback(127.0.0.1/localhost/::1);"
50 "对外暴露面不在本项目的安全姿态内")
51 if host == 'localhost':
52 host = HOST # bind and probe the same concrete loopback address
53 try:
54 want = int(getattr(web, "port", 0) or 0)
55 except (TypeError, ValueError):
56 want = 0
57 if want <= 0:
58 return host, pick_free_port(host)
59 with socket.socket(socket.AF_INET6 if ':' in host else socket.AF_INET, socket.SOCK_STREAM) as s:
60 s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
61 try:
62 s.bind((host, want))
63 except OSError:
64 fallback = pick_free_port(host)
65 log.warning("shell.web.port=%d 不可用(被占用/无权限),回落空闲口 %d",
66 want, fallback)
67 return host, fallback
68 return host, want

71def run_uvicorn(app: "DesktopApp", port: int, host: str = HOST) -> None:
72 """后台线程 serve:单 worker、禁 reload(seq 单调前提 ADR-005);阻塞至 should_exit。
73
74 偏离 5:手工持 loop(server.serve 在自建 loop 上运行)并把 app.loop 暴露给
75 DesktopBridge._run 跨线程投递;线程收尾复位 app.server/app.loop。
76 PyInstaller --windowed 打包:进程无 console 句柄 → sys.stderr 为 None,
77 uvicorn 默认 formatter 调 isatty() 崩(AttributeError)→ 无 stderr 时
78 log_config=None 跳过 dictConfig(用标准 logging,日志进兜底 handler)。
79 """
80 kw: dict = dict(log_level="warning", workers=1)
81 if sys.stderr is None: # windowed/frozen 无 stderr:禁用 uvicorn 自配日志
82 kw["log_config"] = None
83 config = uvicorn.Config(app.api, host=host, port=port, **kw)
84 server = uvicorn.Server(config)
85 app.server = server
86 loop = asyncio.new_event_loop()
87 asyncio.set_event_loop(loop)
88 app.loop = loop
89 try:
90 loop.run_until_complete(server.serve(sockets=None))
91 except Exception: # noqa: BLE001 服务线程异常仅日志(域外 CYC-999)
92 log.exception("uvicorn serve 异常退出 port=%s", port)
93 finally:
94 app.loop = None
95 app.server = None
96 asyncio.set_event_loop(None)
97 loop.close()

100def _probe_port(port: int, host: str = HOST) -> bool:
101 """127.0.0.1:port 是否可连(就绪探测;失败静默返回 False)。"""
102 try:
103 with socket.create_connection((host, port), timeout=0.3):
104 return True
105 except OSError:
106 return False

109def wait_until_listening(port: int, timeout: float = LISTEN_TIMEOUT_S, *, host: str = HOST, app=None) -> bool:
110 """同步就绪探测(webview 开窗前阻塞;失败 → 调用方退 1 不弹空窗)。"""
111 deadline = time.monotonic() + timeout
112 while time.monotonic() < deadline:
113 if (app is None or getattr(getattr(app, 'server', None), 'started', False)) and _probe_port(port, host=host):
114 return True
115 time.sleep(0.05)
116 return False
