53def _write_line(out: Optional[TextIO], obj: dict) -> None:
54 """协议行写 stdout:每次写+flush(双工实时);buffer 优先(UTF-8 字节)。"""
55 stream = out if out is not None else sys.stdout
56 line = json.dumps(obj, ensure_ascii=False) + "\n"
57 buf = getattr(stream, "buffer", None)
58 if buf is not None: # 真实 stdout:UTF-8 字节,免控制台编码坑
59 buf.write(line.encode("utf-8"))
60 buf.flush()
61 else: # StringIO 等注入替身:文本直写
62 stream.write(line)
63 stream.flush()

87def _redact(ctx: dict) -> dict:
88 """ctx 脱敏(INV-09):疑似敏感键打码,值截断 200 字符,嵌套只留类型标记。"""
89 out: dict[str, Any] = {}
90 for k, v in (ctx or {}).items():
91 if any(hint in str(k).lower() for hint in _SENSITIVE_KEY_HINTS):
92 out[str(k)] = "[redacted]"
93 elif isinstance(v, (str, int, float, bool)) or v is None:
94 out[str(k)] = str(v)[:200] if isinstance(v, str) else v
95 else:
96 out[str(k)] = f"<{type(v).__name__}:{len(v)}>" if hasattr(
97 v, "__len__") else f"<{type(v).__name__}>"
98 return out

100def engine_error(rid: Optional[Union[int, str]], e: PyHError) -> dict:
101 """引擎错 → -32603 壳 + data={code,message,advice}(ADR-011 远端只回 code+advice)。
102
103 热路径码(GRD-401/403/LLM-310/PERS-202)不加 detail;其余码的 ctx 经 _redact
104 后才进 detail——远端绝不见 ctx 敏感值/密钥原文。
105 """
106 body: dict[str, Any] = {"code": e.code, "message": e.message,
107 "advice": e.spec.advice}
108 if e.code not in _HOT_CODES:
109 body["detail"] = _redact(e.ctx or {})
110 return error_resp(rid, -32603, e.message, body)

116def parse_request(line: str,
117 *, write: Optional[TextIO] = None) -> Optional[JsonRpcRequest]:
118 """行解析(协议错 → 标准码响应就地写出并返回 None,调用方 continue)。
119
120 -32700:非法 JSON;-32600:信封非法(非对象/jsonrpc 版本/method 类型/
121 id 类型);-32602:params 非对象。id 缺失/None = 通知(id=None)。
122 """
123 try:
124 raw = json.loads(line)
125 except json.JSONDecodeError as e:
126 _write_line(write, error_resp(None, -32700, "Parse error",
127 {"detail": str(e)[:300]}))
128 return None
129 if (not isinstance(raw, dict)
130 or raw.get("jsonrpc") != "2.0"
131 or not isinstance(raw.get("method"), str)
132 or not raw.get("method")
133 or ("id" in raw and raw["id"] is not None
134 and not _is_id_type(raw["id"]))):
135 rid = raw.get("id") if isinstance(raw, dict) and _is_id_type(
136 raw.get("id")) else None
137 _write_line(write, error_resp(rid, -32600, "Invalid Request"))
138 return None
139 params = raw.get("params")
140 if params is not None and not isinstance(params, dict):
141 _write_line(write, error_resp(raw.get("id"), -32602,
142 "Invalid params"))
143 return None
144 return JsonRpcRequest(jsonrpc="2.0", id=raw.get("id"),
145 method=raw["method"], params=params or {})
