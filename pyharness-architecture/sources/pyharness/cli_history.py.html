41async def _cmd_search(sh: ShellCtx, positional: list[str], flags: dict) -> int:
42 """search <q>:FTS 全文检索(派生索引只读,不触真源)。"""
43 from pyharness.core.session_query import SessionQueryIndex
44 from pyharness.persistence import SessionReader, session_log_paths
45 q = " ".join(positional).strip()
46 if not q:
47 raise_code("EVT-100", advice="search 需要检索词",
48 hint="用法:pyharness search <检索词>")
49 sessions_dir = _sessions_dir_ctx(sh)
50 db = Path(str(sh.ctx.settings.storage.db_path)).expanduser()
51 sources: dict = {}
52 # 唯一枚举点(R14-3):**辅助档**(轮转段/修复备份/隔离档)不是独立会话 —— 此前
53 # 按 `stem.startswith("s-")` 判定会把它们当会话注册,而备份/段与主文件**同 seq**
54 # ⇒ rebuild 撞 fts_rows 主键冲突 ⇒ search 直接 PERS-202 崩(实测)。
55 #
56 # 回放源用 **SessionReader**(R14-5):``open_store`` 会取 INV-07 **写**锁并持有到
57 # 查询结束 ⇒ 一条只读命令把**所有**会话锁住,期间其他进程开会话撞 PERS-202
58 # (实测:并发 open 交替 OPEN_OK / BLOCKED)。只读命令不该持有写锁。
59 for f in session_log_paths(sessions_dir):
60 sources[f.stem] = SessionReader(f.stem, dir=sessions_dir)
61 idx = SessionQueryIndex(db_path=db, sources=sources)
62 try:
63 await idx.enter(ctx=SimpleNamespace(config=sh.ctx.settings))
64 # 逐会话重建(R14-4):只动**本次确实读到源**的会话行。此前用全量
65 # ``rebuild()``(DELETE 整表再重插),而打不开的会话(如被活进程持锁 ⇒
66 # open_store 抛 PERS-202 ⇒ 上面 `continue` 未登记源)**索引行被一并抹掉且不再
67 # 重建** —— 实测:被锁会话的索引水位 2 → 0,静默不可搜。sources 为空时更是
68 # 直接清空整个索引。
69 # 残余(如实标注):日志被**手工**移除的会话,其陈旧索引行不再被顺带清除
70 # (正规删除走 ``delete_session`` 级联)。刻意不"按目录反推清理"——索引库是
71 # **全局**的而会话目录按租户分,反推会误删他租户的行。
72 for sid in sources:
73 await idx.rebuild(session_id=sid)
74 res = await idx.query(q)
75 hits = [{"session_id": h.session_id, "seq": h.seq, "type": h.type,
76 "ts": h.ts, "snippet": h.snippet, "rank": h.rank}
77 for h in res.hits]
78 finally:
79 try:
80 await idx.detach()
81 except Exception: # noqa: BLE001
82 pass
83 if flags.get("json"):
84 _json_line({"query": q, "timed_out": res.timed_out, "hits": hits})
85 return 0
86 if not hits:
87 print("无结果")
88 return 0
89 for h in hits:
90 print(f"[{h['session_id']}#{h['seq']} {h['type']}] {h['snippet']}")
91 return 0

93async def _cmd_session(sh: ShellCtx, positional: list[str], flags: dict) -> int:
94 """session list / session show <sid>:只读会话清单与回放展示。
95
96 ``2026-09-21 R14-6``:``show`` 改用**只读回放源** ``SessionReader``(与 ``search`` 同款)
97 —— 此前借 ``open_store`` 读,而它会取 INV-07 **写**锁 ⇒ **会话正被使用时**
98 ``session show`` 自身撞 PERS-202(实测:无锁 rc=0 / 有锁 rc≠0),即"看不了正在跑的
99 会话"。只读命令不该持写锁。
100 """
101 from pyharness.core.session import open_session
102 from pyharness.persistence import SessionReader
103 sessions_dir = _sessions_dir_ctx(sh)
104 act = (positional or ["list"])[0]
105 if act == "list":
106 from pyharness.desktop.sessions import DesktopSessionManager
107 mgr = DesktopSessionManager(dir=sessions_dir, bus=sh.ctx.bus,
108 config=sh.ctx.settings)
109 rows = mgr.list()
110 if flags.get("json"):
111 _json_line({"sessions": [dict(r) for r in rows]})
112 return 0
113 if not rows:
114 print("0 个会话")
115 return 0
116 for r in rows:
117 name = (r.get("title") or r.get("preview") or "新会话")
118 state = "已终态" if r.get("finished") else "活跃"
119 print(f"{r['sid']} {name} {r.get('lines', 0)} 事件 {state}")
120 return 0
121 if act == "show":
122 sid = positional[1] if len(positional) > 1 else flags.get("session")
123 if not sid:
124 raise_code("EVT-100", advice="session show 需要 sid",
125 hint="用法:pyharness session show <sid>")
126 from pyharness.desktop.sessions import validate_session_id
127 sid = validate_session_id(str(sid))
128 if not (sessions_dir / f"{sid}.jsonl").exists():
129 raise_code("EVT-106", session_id=sid,
130 hint="会话不存在(先 session list 确认 sid)")
131 # 只读回放源(R14-6):不取会话写锁 ⇒ 会话正被他进程使用时也能查看
132 log_ = await open_session(sid, SessionReader(sid, dir=sessions_dir))
133 evs = list(log_.events_after(0))
134 msgs = log_.derive_messages()
135 if flags.get("json"):
136 _json_line({"sid": sid, "messages": msgs,
137 "events": [e.model_dump(exclude_none=True)
138 for e in evs]})
139 return 0
140 print(f"会话 {sid} 事件 {len(evs)} 条:")
141 for m in msgs:
142 role = m.get("role", "?")
143 content = str(m.get("content") or "").replace("\n", " ")[:200]
144 if not content and m.get("tool_calls"):
145 content = "工具调用"
146 print(f" [{role}] {content}")
147 return 0
148 raise_code("EVT-100", advice=f"未知 session 子命令:{act}",
149 hint="可用:session list | session show <sid>")

151async def _cmd_fork(sh: ShellCtx, positional: list[str], flags: dict) -> int:
152 """fork <sid>:物理复制事件流到新会话 + 父会话落 fork.created 声明。"""
153 from pyharness.core.session import open_session
154 from pyharness.persistence import flush_kwargs_of, open_store
155 sessions_dir = _sessions_dir_ctx(sh)
156 if not positional:
157 raise_code("EVT-100", advice="fork 需要源会话 sid",
158 hint="用法:pyharness fork <sid>")
159 from pyharness.desktop.sessions import validate_session_id
160 sid = validate_session_id(str(positional[0]))
161 if not (sessions_dir / f"{sid}.jsonl").exists():
162 raise_code("EVT-106", session_id=sid, hint="源会话不存在")
163 src = open_store(sid, dir=sessions_dir,
164 **flush_kwargs_of(sh.ctx.settings))
165 try:
166 events = list(src.replay())
167 finally:
168 src.close()
169 if not events:
170 raise_code("EVT-106", session_id=sid, hint="源会话为空(缺 session.created)")
171 new_sid = f"s-fork-{uuid.uuid4().hex[:8]}"
172 tgt = open_store(new_sid, dir=sessions_dir,
173 **flush_kwargs_of(sh.ctx.settings))
174 try:
175 for e in events:
176 if e.type == "session.finished":
177 continue
178 await tgt.append(e.model_copy(update={"session_id": new_sid}),
179 sync=True)
180 await tgt.flush()
181 finally:
182 tgt.close()
183 store2 = open_store(sid, dir=sessions_dir,
184 **flush_kwargs_of(sh.ctx.settings))
185 log_ = await open_session(sid, store2)
186 try:
187 base_seq = max((e.seq for e in events), default=0)
188 await log_.append("fork.created",
189 {"new_session_id": new_sid, "base_seq": base_seq,
190 "reason": "cli fork"},
191 actor="system", sync=True)
192 finally:
193 store2.close()
194 if flags.get("json"):
195 _json_line({"source_sid": sid, "new_session_id": new_sid,
196 "base_seq": base_seq, "events": len(events)})
197 else:
198 print(f"已分叉 {sid} → {new_sid}({len(events)} 条事件,base_seq={base_seq})")
199 return 0
