30class ExperimentRevision(RegressionRecord):
31 experiment_id:ExperimentId
32 selection:ConfirmExperimentPreparation
33 preparation_ref:PreparationReference
34 graph_ref:TaskGraphReference
35 case_nodes:list[ExperimentCaseNodes]=Field(min_length=1,max_length=10)

38 @model_validator(mode='after')
39 def exact_cases(self):
40 if self.experiment_id!=self.selection.experiment_id or [v.case_ref for v in self.case_nodes]!=self.selection.case_refs:
41 raise ValueError('experiment fixed cases differ')
42 if self.dataset_total<len(self.case_nodes):raise ValueError('dataset count differs')
43 if len({v.task_ref.product_task_id for v in self.case_nodes})!=len(self.case_nodes):
44 raise ValueError('distinct prepared case tasks required')
45 names=[name for v in self.case_nodes for name in (v.left_node_id,v.right_node_id)]
46 if len(set(names))!=len(names):raise ValueError('duplicate comparison node')
47 return self
