187 # ============================================================ 崩溃恢复
188 async def recover(self) -> int:
189 """崩溃遗留 job 的**事件级**恢复(R24;此前只有读侧如实显示、日志里无终态)。
190
191 进程被杀 ⇒ ``_run`` 的 ``finally`` 未执行 ⇒ 日志里只有 ``job.started`` 没有
192 终态事件。本方法重放本会话事件,对"有 started 无终态"的 job **补写
193 ``job.failed(reason="crash")``**,使审计/派生视图与读侧判定一致
194 (``cli._cmd_job`` 已在"无宿主持锁"时报 failed)。
195
196 - **幂等**:补写后再重放不会重复(已见终态即跳过);
197 - **不新增事件写者**:仍走本类 ``self._session.append``(与 ``_run``/``cancel`` 同源);
198 - **失败不阻断**:事件词表/终态(EVT-104)等异常只记日志(恢复是收尾,不是主链)。
199 返回补写的条数。
200 """
201 started: set = set()
202 done: set = set()
203 for ev in self._session.events_after(0):
204 t = getattr(ev, "type", "")
205 p = getattr(ev, "payload", None) or {}
206 jid = str(p.get("job_id") or "")
207 if not jid:
208 continue
209 if t == "job.started":
210 started.add(jid)
211 elif t in ("job.completed", "job.failed"):
212 done.add(jid)
213 orphan = sorted(started - done)
214 for jid in orphan:
215 try:
216 await self._session.append(
217 "job.failed", {"job_id": jid, "reason": "crash"},
218 actor="system", task_id=f"job:{jid}", sync=True)
219 except PyHError as e: # EVT-104 终态会话等:记日志不阻断
220 log.warning("崩溃 job 补写终态失败 code=%s job=%s", e.code, jid)
221 continue
222 if orphan:
223 log.warning("崩溃遗留 job 已补写终态(reason=crash):%s", orphan)
224 return len(orphan)

304 父会话 closed/finished → job.started 落盘被 EVT-104 拒(事件失败 = 没启动,
305 回滚登记)。owner/owner_channel 取自 ctx(通道上下文注入,见偏离 11)。
306 返回 job_id(j-N,顺序可猜非机密);父会话 running 态不受影响,可继续对话。
307 """
308 if not intent or not intent.strip(): # 空意图拒提交(白名单前置)
309 raise_code("EVT-100", hint="空 job 意图,start 拒绝")
310 if len(intent) > MAX_INTENT_CHARS: # 超长拒提交
311 raise_code("EVT-100", hint=f"job 意图超长(>{MAX_INTENT_CHARS} 字符)")
312 owner = getattr(ctx, "owner", None)
313 if not owner or not str(owner).strip(): # 身份缺失 → 孤儿 job(偏离 11)
314 raise_code("EVT-100", hint="ctx.owner 缺失:提交方身份必须由通道上下文"
315 "注入(cli:alice/acp:<client>/web:<session>),非 LLM 自报")
316 if self._active_count() >= self._limit: # 并发 ≤4:满拒(偏离 2/3)
317 raise PyHError("JOB-001", ctx={
318 "running": self._active_count(), "limit": self._limit,
319 "advice": "先等若干 job 终态或取消后再提交"})
320 n = next(self._ids)
321 j = Job(id=f"j-{n}", owner=owner, intent=intent, state="queued",
322 task_id=f"job:j-{n}")
323 j.notify_to = (meta or {}).get("notify_to") \
324 if isinstance(meta, dict) else None
325 if not j.notify_to:
326 j.notify_to = getattr(ctx, "owner_channel", None)
327 self._running[j.id] = j # 先登记(事件失败可回滚)
328 try:
329 env = await self._session.append( # task_id 走信封(偏离 4)
330 "job.started", {"job_id": j.id}, actor="system", task_id=j.task_id)
331 except PyHError:
332 self._running.pop(j.id, None) # 事件没落成 = 没启动
333 raise
334 j.created_seq = env.seq
335 j.started_ts = _now_iso()
336 j.state = "running" # 落盘成功 → 转 running
337 # 协程并发(原则 5):不占父会话 running 态,父可继续对话
338 j.task = asyncio.create_task(self._run(j, ctx))
339 # 取消竞态:task.cancel() 落在任务尚未开始执行前会使协程体永不运行,
340 # finally 的终态事件全丢(注册表永挂 running 占并发)→ 统一以
341 # _cancel_pending 旗标表达取消,_run 入口自查转 CancelledError(finally
342 # 仍落终态);任务已开始(_begun)则由 cancel 直接 task.cancel() 传播。
343 log.info("job started id=%s owner=%s intent_len=%d",
344 j.id, j.owner, len(intent))
345 return j.id

347 # ============================================================ 执行体
348 async def _run(self, j: Job, ctx: Any) -> None:
349 """job 执行体(段围栏 + 终态归一):开段 → 跑 runner → 各终态分支落事件。
350
351 成功 → job.completed;PyHError(含预算/LLM-310)→ job.failed(reason=码);
352 CancelledError → job.failed(reason=cancelled|session-closed)(F025,re-raise
353 不吞);未预期 → job.failed(CYC-999)兜底;任何异常不得逃逸出 finally 之外
354 (终态事件 append/关段失败仅记日志,见偏离 6/7)。终态事件恒先于
355 session.finished(钩子序 child-first,同 subagent 纪律)。
356 """
357 j._begun = True # 执行起点标记(cancel 分发判定)
358 t0 = time.monotonic()
359 reason = "ok"
360 try:
361 if j._cancel_pending: # 早到取消(任务创建前):自查转取消
362 j._cancel_pending = False
363 raise asyncio.CancelledError() # finally 仍会落 failed(cancelled)
364 # F044 段锚:先落盘再执行,崩溃后段不悬空(task_queue 强同步开段)
365 j.started_seq = await self._tq.open_segment(j.task_id)
366 result = await self._run_runner(j, ctx)
367 rr = getattr(result, "reason", None)
368 if rr in _OK_REASONS: # 成功(含 runner 无 reason 返回)
369 j.state, reason = "completed", "ok"
370 else: # RunResult 式非成功(reason 归一)
371 j.state, reason = "failed", rr # budget/max_turns/error/stall/…
372 except PyHError as e: # 结构化失败(含预算/LLM-310)
373 j.state, reason = "failed", e.code
374 except asyncio.CancelledError: # F025 取消:不吞,re-raise
375 j.state = "failed"
376 reason = j._cancel_reason or "cancelled" # session-closed vs user-cancel
377 raise
378 except Exception: # 兜底:CYC-999,堆栈仅本地
379 log.exception("job crashed id=%s", j.id)
380 j.state, reason = "failed", "CYC-999"
381 finally:
382 # 终态记账:耗时 + 7 天过期锚 + 终态事件 + 关段(顺序不可换)
383 j.elapsed_ms = int((time.monotonic() - t0) * 1000)
384 j.expires_at = time.time() + self._retention_days * 86400
385 if j.state != "completed":
386 j.reason = reason # completed 无原因字段(偏离 4)
387 ev = "job.completed" if j.state == "completed" else "job.failed"
388 try:
389 payload: dict = {"job_id": j.id, "elapsed_ms": j.elapsed_ms}
390 if ev == "job.failed":
391 payload["reason"] = reason # job.failed 载荷模型含 reason
392 env = await self._session.append(ev, payload, actor="system",
393 task_id=j.task_id)
394 j.finished_seq = env.seq
395 except PyHError as e: # EVT-104/PERS-202:终态仅记日志
396 log.error("job terminal append failed code=%s job=%s",
397 e.code, j.id)
398 if j.started_seq: # 段已开出才配对关闭(偏离 7)
399 try:
400 await self._tq.close_segment(j.task_id, j.started_seq)
401 except PyHError as e:
402 log.error("job close segment failed code=%s job=%s",
403 e.code, j.id)
404 await self._notify(j) # 完成通知(终态后广播/注入)
405
406 async def _run_runner(self, j: Job, ctx: Any) -> Any:
407 """执行器调用(注入 seam,见偏离 1):runner.run(ctx, intent=…, task_id=…)。
408
409 未装配 runner → CYC-999 快速失败(防伪造已执行);同步/异步返回兼容。
410 """
411 runner = self._runner
412 if runner is None:
413 raise_code("CYC-999", job_id=j.id,
414 hint="JobManager 未装配执行器(runner):job 拒绝执行,"
415 "防伪造已执行(task_queue 同款纪律)")
416 fn = getattr(runner, "run", runner)
417 if not callable(fn):
418 raise_code("CYC-999", job_id=j.id,
419 hint=f"执行器不可调用: {type(runner).__name__}")
420 budget = getattr(ctx, "budget", None)
421 res = fn(ctx, intent=j.intent, task_id=j.task_id, budget=budget)
422 if inspect.isawaitable(res):
423 return await res
424 return res

427 async def status(self, job_id: str, *, by: str) -> JobStatus:
428 """状态/进度查询(owner 授权):state + 派生进度 + 最近日志尾。
429
430 进度从段内 todo.updated 清单推导(done/总数);无 todo → None、0/0 项;
431 job 不存在 → EVT-101;by 非 owner 且非 system → GRD-401(知道 id ≠ 有权)。
432 """
433 j = self._get(job_id)
434 self._authorize(j, by)
435 todos = await self._segment_todos(j)
436 done, progress = 0, None
437 if todos: # 进度 = 完成项占比(F040 派生)
438 done = sum(1 for t in todos if t.get("done"))
439 progress = round(done / len(todos), 3)
440 j.progress = progress # 查询结果镜像(非权威,INV-01)
441 tail = await self._log_tail(j, n=10)
442 return JobStatus(job_id=j.id, state=j.state, owner=j.owner,
443 progress=progress,
444 todo_summary=f"{done}/{len(todos)} 项",
445 elapsed_ms=j.elapsed_ms, log_tail=tail,
446 error=j.reason)

459 # ============================================================ 取消(F025)
460 async def cancel(self, job_id: str, *, by: str) -> bool:
461 """取消后台 job(owner 授权):传播 CancelledError → 归一 job.failed。
462
463 在途 → 先落 system.cancelled 审计再取消任务(沿 await 链传播,F025);
464 已终态 → 幂等 False;不存在 → EVT-101;job_id 猜中不可取消他人 job
465 (GRD-401)。任务创建前到达的取消以 _cancel_pending 旗标补投(防竞态丢取消)。
466 """
467 j = self._get(job_id)
468 self._authorize(j, by)
469 if j.terminal:
470 return False # 终态幂等(防重放)
471 j._cancel_pending = True # 早到取消统一旗标(入口自查兜底)
472 if j.task is not None and not j.task.done() and j._begun:
473 j.task.cancel() # 任务已开始执行 → 沿 await 链传播
474 primary = None
475 try:
476 await self._session.append('system.cancelled',
477 {'what': j.task_id, 'reason':'user-cancel'}, actor='system', task_id=j.task_id)
478 except BaseException as exc:
479 primary = exc
480 try:
481 if j.task is not None:
482 results = await asyncio.shield(asyncio.gather(j.task, return_exceptions=True))
483 for error in results:
484 if isinstance(error, BaseException) and not isinstance(error, asyncio.CancelledError): raise error
485 except BaseException as exc:
486 if primary is None: raise
487 primary.add_note(f'job cleanup failed: {type(exc).__name__}')
488 if primary is not None: raise primary
489 return True

500 async def reap_expired(self) -> int:
501 """结果 7 天清理:摘除全部终态超期句柄(事件永留 JSONL,审计/重放不受影响)。
502
503 由管理器定时器/会话空闲期调用;纯内存清理,返回清理数。
504 """
505 now = time.time()
506 reaped: list[str] = []
507 for jid, j in list(self._running.items()):
508 if j.terminal and j.expires_at is not None and j.expires_at < now:
509 reaped.append(jid)

511 if reaped:
512 log.info("reaped expired job handles=%s", reaped)
513 return len(reaped)

515 # ============================================================ 会话关闭钩子
516 async def _on_session_closing(self, type_: Any, payload: Any) -> None:
517 """child-first 清理钩子(agent.close 在 append session.finished **前**回调)。
518
519 取消全部在途 job → await 各自落 failed 终态(先落终态再 finished,否则
520 finished 后回写撞 EVT-104);取消原因落 session-closed(偏离 8)。
521 """
522 live = [j for j in self._running.values() if j.state in _ACTIVE_STATES]
523 if not live:
524 return
525 errors = []
526 for j in live:
527 j._cancel_reason = "session-closed"
528 j._cancel_pending = True
529 if j.task is not None and not j.task.done() and j._begun:
530 j.task.cancel()
531 for j in live:
532 try:
533 await self._session.append("system.cancelled",
534 {"what": j.task_id, "reason": "session-closed"},
535 actor="system", task_id=j.task_id)
536 except BaseException as exc:
537 errors.append(exc)
538 tasks = [j.task for j in live if j.task is not None]
539 if tasks:
540 results = await asyncio.gather(*tasks, return_exceptions=True)
541 errors.extend(e for e in results if isinstance(e, BaseException)
542 and not isinstance(e, asyncio.CancelledError))
543 if errors: raise BaseExceptionGroup('job shutdown failed', errors)

545 # ============================================================ 完成通知
546 async def _notify(self, j: Job) -> None:
547 """完成通知:总线广播终态 + (completed 且 notify_to)注入可见摘要。
548
549 目标会话已 finished → 跳过补写仅记日志(通知非强同步事实,EVT-104 兜底)。
550 """
551 if self._bus is not None:
552 ev = "job.completed" if j.state == "completed" else "job.failed"
553 try:
554 await self._bus.emit(ev, {"job_id": j.id,
555 "elapsed_ms": j.elapsed_ms,
556 "reason": j.reason})
557 except PyHError as e: # 广播失败不影响终态事实
558 log.error("job notify bus failed code=%s job=%s", e.code, j.id)
559 if not (j.notify_to and j.state == "completed"):
560 return
561 if self._session_closed():
562 return # 目标已终态:不补写(仅日志)
563 try:
564 await self._session.append(
565 "agent.message",
566 {"content": f"[后台任务 {j.id} 完成] {j.intent[:120]}"},
567 actor="system")
568 except PyHError as e: # EVT-104 竞态:跳过
569 log.error("job notify append failed code=%s job=%s", e.code, j.id)
