36 except Exception: # noqa: BLE001 假窗/旧版无事件面
37 log.debug("window.events.closing 不可用,关窗停服由 bridge.on_close 承担")

39 def notify(*args):
40 try:
41 window.evaluate_js("window.dispatchEvent(new CustomEvent('pyharness-host-visibility',{detail:{hidden:"
42 + ('true' if hidden else 'false') + "}}))")
43 except Exception:
44 log.debug('Appearance visibility notification unavailable')
45 return notify
46 for name, hidden in [('minimized',True),('hidden',True),('restored',False),('shown',False)]:

63def main(argv: Optional[list[str]] = None) -> int:
64 """进程入口(pyharness-desktop):装配 ctx → uvicorn 后台线程 → webview 开窗 → 关窗优雅停服。
65
66 返回 int 退出码(0 正常 / 1 启动失败:CFG-601 配置错、端口/WebView2 不可用、就绪超时)。
67 argv 未用(桌面无 CLI 参数,保留签名对称)。非主线程调 webview 会 RuntimeError(文档注明)。
68 """
69 try:
70 ctx = _bootstrap_desktop(None)
71 except PyHError as e: # CFG-601 等装配错
72 advice = e.ctx.get("advice") or e.ctx.get("hint") or e.spec.advice
73 print(f"{e.code}:{advice}", file=sys.stderr)
74 return 1
75 app = DesktopApp(ctx=ctx)
76 try:
77 host, port = resolve_bind(getattr(ctx, "settings", None)
78 or getattr(ctx, "config", None))
79 except OSError:
80 log.exception("pick_free_port 失败")
81 return 1
82 tok_path = app.publish_operator_token() # R31-2:随机令牌须可查,否则浏览器入口关闭
83 if tok_path:
84 log.info("操作者令牌已落盘(浏览器访问用 ?token=<该文件内容>):%s", tok_path)
85 app.url = app.bootstrap_url(host, port) # R31-2:带操作者令牌
86 threading.Thread(target=run_uvicorn, args=(app, port, host),
87 name="desktop-uvicorn", daemon=True).start()
88 if not wait_until_listening(port, host=host, app=app):
89 log.error("uvicorn 就绪超时 port=%s(不弹空窗,退出 1)", port)
90 app.shutdown_gracefully()
91 return 1
92 wv = _load_webview()
93 if wv is None: # WebView2/环境缺失:退 1
94 print("pywebview 不可用:请安装 Microsoft Edge WebView2 Runtime 后重试",
95 file=sys.stderr)
96 app.shutdown_gracefully()
97 return 1
98 bridge = DesktopBridge(app)
99 app.bridge = bridge
100 try:
101 _create_window(wv, app, bridge)
102 wv.start(debug=False) # 阻塞至全部窗口关闭
103 except Exception: # noqa: BLE001 窗口生命周期未预期
104 log.exception("webview.start 未预期退出")
105 app.shutdown_gracefully()
106 return 1
107 app.shutdown_gracefully() # 关窗 = 优雅停服
108 return 0

111async def run_desktop(ctx: Any) -> int:
112 """CLI desktop 子命令桥(cli.py 调用):复用已装配 ctx 走同一 DesktopApp 生命周期。
113
114 等价入口:`uv run pyharness desktop`;桌面必须主线程启动(RuntimeError 域外)。
115 """
116 app = DesktopApp(ctx=ctx)
117 host, port = resolve_bind(getattr(ctx, "settings", None)
118 or getattr(ctx, "config", None))
119 tok_path = app.publish_operator_token() # R31-2:随机令牌须可查,否则浏览器入口关闭
120 if tok_path:
121 log.info("操作者令牌已落盘(浏览器访问用 ?token=<该文件内容>):%s", tok_path)
122 app.url = app.bootstrap_url(host, port) # R31-2:带操作者令牌
123 threading.Thread(target=run_uvicorn, args=(app, port, host),
124 name="desktop-uvicorn", daemon=True).start()
125 try:
126 await wait_listening_async(app, port, host=host)
127 wv = _load_webview()
128 if wv is None:
129 raise_code('CYC-999', hint='pywebview 不可用(需 WebView2 Runtime)')
130 bridge = DesktopBridge(app)
131 app.bridge = bridge
132 _create_window(wv, app, bridge)
133 wv.start(debug=False)
134 finally:
135 app.shutdown_gracefully()
136 return 0
