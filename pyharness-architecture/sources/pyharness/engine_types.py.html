112 async def start_flush_ticker(self) -> bool:
113 """幂等启动攒批落盘定时器(间隔 = ``store.flush_interval_s``;GAP-13)。
114
115 语义与边界:
116
117 - **幂等**:已在跑(未 done)→ 返回 False,不重复起任务;
118 - **无 store / 间隔 ≤0** → 不起(返回 False),不是错误;
119 - **单次失败不杀定时器**:``store.flush()`` 抛 ``PERS-202`` 时只记日志,
120 落盘故障状态由 ``SessionStore._fail_streak`` / ``_suspended`` 自行记账
121 (拒新不丢旧),定时器继续按周期间隔重试 —— 崩溃窗口因此**有上界**;
122 - **不吞取消**:收到 ``CancelledError`` 立即 re-raise(取消协议);
123 - 停止方式 = ``close()`` 置位 ``_flush_stop`` 并 await,超时则 cancel。
124
125 该定时器**不是第二写者**:它只调 store 自己的 ``flush()``,与强同步路径
126 共用同一把攒批缓冲与 seq 校验(INV-07 单写者不变)。
127 """
128 if self._flush_task is not None and not self._flush_task.done():
129 return False
130 store = self.persistence
131 if store is None:
132 return False
133 try:
134 interval = float(getattr(store, "flush_interval_s", 0.5) or 0.5)
135 except (TypeError, ValueError):
136 interval = 0.5
137 if interval <= 0:
138 return False
139 stop = asyncio.Event()
140 self._flush_stop = stop
141
142 async def _loop() -> None:
143 while True:
144 try:
145 await asyncio.wait_for(stop.wait(), timeout=interval)
146 return # 收到停止信号:正常退出
147 except asyncio.TimeoutError:
148 pass
149 except asyncio.CancelledError:
150 raise
151 try:
152 r = store.flush()
153 if inspect.isawaitable(r):
154 await r
155 except asyncio.CancelledError:
156 raise
157 except Exception: # noqa: BLE001 单次失败不杀定时器
158 log.warning("flush ticker 落盘失败(store 自行记账 PERS-202)",
159 exc_info=True)
160
161 self._flush_task = asyncio.create_task(
162 _loop(), name=f"flush-ticker:{getattr(self.session, 'sid', '?')}")
163 return True

165 async def close(self) -> None:
166 """Stop all writers before releasing subscriptions, stores and connections."""
167 task = getattr(self, '_close_task', None)
168 if task is None:
169 self._closing = True
170 task = self._close_task = asyncio.create_task(self._close_owned())
171 await asyncio.shield(task)
172
173 async def _close_owned(self):
174 errors = []
175 async def attempt(fn, *args, **kwargs):
176 try:
177 value = fn(*args, **kwargs)
178 if inspect.isawaitable(value):
179 await value
180 except BaseException as exc:
181 errors.append(exc)
182 if self.schedule is not None:
183 await attempt(getattr(self.schedule, 'aclose', None) or self.schedule.stop)
184 if self.jobs is not None:
185 await attempt(self.jobs._on_session_closing, 'session.closing', None)
186 if self.subagent is not None:
187 # Stop child admissions and await child cleanup while dependencies live.
188 # Parent providers must also drain before their tools are unregistered.
189 await attempt(self.subagent.close_children, reason='parent-detach')
190 if self.task_queue is not None:
191 await attempt(self.task_queue.shutdown, reason='session-close')
192 if self.subagent is not None:
193 await attempt(self.subagent.detach)
194 if self.approval is not None:
195 await attempt(getattr(self.approval, 'aclose', None) or self.approval.detach)
196 for name in ('_flush_task', 'llm_probe'):
197 task = getattr(self, name, None)
198 if task is not None:
199 task.cancel()
200 result = await asyncio.gather(task, return_exceptions=True)
201 errors.extend(e for e in result if isinstance(e, BaseException) and not isinstance(e, asyncio.CancelledError))
202 setattr(self, name, None)
203 self._flush_stop = None
204 from pyharness.core import proc as proc_mod
205 await attempt(asyncio.to_thread, proc_mod.close_session, str(getattr(self.session, 'sid', '')))
206 while self._mcp_clients:
207 await attempt(self._mcp_clients.pop().close)
208 if self.fts is not None:
209 await attempt(self.fts.detach, None)
210 self.fts = None
211 store = self.persistence
212 if store is not None and getattr(self, 'owns_persistence', True):
213 flush = getattr(store, 'flush', None) or getattr(store, 'flush_sync', None)
214 if callable(flush): await attempt(flush)
215 close = getattr(store, 'close', None)
216 if callable(close): await attempt(close)
217 if self.bus is not None:
218 for owner in self.bus_owners:
219 await attempt(self.bus.unsubscribe_all, owner)
220 self.bus_owners = []
221 if getattr(self, 'owns_llm_runtime', False):
222 await attempt(self.llm_runtime.aclose)
223 if errors:
224 raise BaseExceptionGroup('engine cleanup failed', errors)
