14def _literal_annotation(values: list[Any]) -> Any:
15 """动态 Literal 构造(eval 受限命名空间;值不可哈希 → TypeError 上抛)。
16
17 Python 3.13 的 typing.Literal 为特殊形式,无法经 __class_getitem__ 动态
18 展开,故把值逐一 repr 成 Python 字面量后在仅含 Literal 的命名空间 eval——
19 值来自 JSON-Schema dict,repr 文本不可执行任意代码(白名单命名空间)。
20 """
21 for v in values:
22 hash(v) # 不可哈希(如 dict)→ TypeError → TLB-803
23 expr = "Literal[" + ", ".join(repr(v) for v in values) + "]"
24 return eval(expr, {"Literal": Literal}) # noqa: S307 受限命名空间,见上

26def _annotation_from(prop: dict[str, Any], model_name: str, key: str) -> Any:
27 """单个属性 schema → typing 注解;不支持形态抛 ValueError(TLB-803 上游收)。
28
29 顺序:const/enum(不依赖 type)→ 标量 → array → object → null → anyOf/oneOf
30 (仅“单实型 + null”组合 → Optional);组合型/allOf/$ref → ValueError 早失败,
31 不静默降级成宽松类型(坏 schema 注册期即暴露,而非调用期神秘失败)。
32 """
33 if not isinstance(prop, dict):
34 raise ValueError(f"属性 {key!r} 的 schema 须为 dict,"
35 f"实为 {type(prop).__name__}")
36
37 # ---- const:字面量单值
38 if "const" in prop:
39 v = prop["const"]
40 if v is None:
41 return Optional[Any]
42 return _literal_annotation([v])
43
44 # ---- enum:字面量集合(允许 null 成员 → Optional)
45 if "enum" in prop:
46 vals = prop["enum"]
47 if not isinstance(vals, list) or not vals:
48 raise ValueError(f"属性 {key!r} 的 enum 须为非空 list")
49 if any(v is None for v in vals):
50 non_null = [v for v in vals if v is not None]
51 if not non_null:
52 return Optional[Any]
53 return Optional[_literal_annotation(non_null)]
54 return _literal_annotation(vals)
55
56 # ---- anyOf/oneOf:仅“单实型 + null”(OpenAI nullable 惯用形态)
57 for comb in ("anyOf", "oneOf"):
58 if comb in prop:
59 parts = prop[comb]
60 if not isinstance(parts, list) or not parts:
61 raise ValueError(f"属性 {key!r} 的 {comb} 须为非空 list")
62 real = [p for p in parts
63 if not (isinstance(p, dict) and p.get("type") == "null")]
64 if len(real) != len(parts) - 1 or len(real) != 1:
65 raise ValueError(f"属性 {key!r} 的 {comb} 仅支持单实型+null 组合")
66 return Optional[_annotation_from(real[0], model_name, key)]
67 if "$ref" in prop or "allOf" in prop:
68 raise ValueError(f"属性 {key!r} 含 $ref/allOf,本注册表编译器不支持"
69 "(能力包须内联展开)")
70
71 t = prop.get("type")
72 if t is None:
73 raise ValueError(f"属性 {key!r} 缺 type(且无 enum/const/组合型)")
74
75 if t == "string":
76 return str
77 if t == "integer":
78 return int
79 if t == "number":
80 return float
81 if t == "boolean":
82 return bool
83 if t == "null":
84 return type(None)
85
86 if t == "array":
87 items = prop.get("items")
88 if items is None:
89 return list[Any] # 合法 JSON-Schema:元素任意
90 if isinstance(items, dict):
91 return list[_annotation_from(items, model_name, f"{key}[]")]
92 raise ValueError(f"属性 {key!r} 的 items 仅支持单一 schema dict")
93
94 if t == "object":
95 props = prop.get("properties", {})
96 if not isinstance(props, dict):
97 raise ValueError(f"属性 {key!r} 的 properties 须为 dict")
98 return _object_model(f"{model_name}_{_sanitize(key)}", props,
99 prop.get("required"))
100
101 raise ValueError(f"属性 {key!r} 的 type {t!r} 不支持")

103def _field_spec(prop: dict[str, Any], ann: Any, *,
104 required: bool) -> tuple[Any, Any]:
105 """属性约束装配 → (annotation, default);default 为 ... / 具体默认值 / Field。
106
107 映射:str maxLength/minLength → max_length/min_length、pattern;数值
108 minimum/maximum/exclusiveMinimum/exclusiveMaximum → ge/le/gt/lt;array
109 minItems/maxItems → min_length/max_length。Literal 化字段(enum/const/组合)
110 忽略标量约束(Literal 已锁值域)。约束值非法(int()/re.compile 失败)→
111 ValueError → TLB-803 早失败,不静默忽略。
112 可选字段(非 required)一律 Optional 化 + 缺省 None(schema default 优先):
113 LLM 省略可选参数不误伤,显式传 null 亦放行。
114 """
115 kw: dict[str, Any] = {}
116 t = prop.get("type")
117 literalized = any(k in prop for k in ("enum", "const", "anyOf", "oneOf"))
118 if not literalized:
119 if t == "string":
120 if "maxLength" in prop:
121 kw["max_length"] = int(prop["maxLength"])
122 if "minLength" in prop:
123 kw["min_length"] = int(prop["minLength"])
124 if "pattern" in prop:
125 re.compile(str(prop["pattern"])) # 无效正则早失败(TLB-803)
126 kw["pattern"] = str(prop["pattern"])
127 elif t in ("integer", "number"):
128 for s_key, f_key in (("minimum", "ge"), ("maximum", "le"),
129 ("exclusiveMinimum", "gt"),
130 ("exclusiveMaximum", "lt")):
131 if s_key in prop:
132 kw[f_key] = prop[s_key]
133 elif t == "array":
134 if "minItems" in prop:
135 kw["min_length"] = int(prop["minItems"])
136 if "maxItems" in prop:
137 kw["max_length"] = int(prop["maxItems"])
138
139 if not required:
140 if ann is not type(None):
141 ann = Optional[ann]
142 kw["default"] = prop.get("default") # 可选字段恒有默认(缺省 None)
143 return ann, Field(**kw)
144 # 必填字段
145 if "default" in prop:
146 kw["default"] = prop["default"]
147 return ann, Field(**kw)
148 if kw:
149 return ann, Field(**kw)
150 return ann, ...

152def _object_model(model_name: str, props: dict[str, Any],
153 required: Any) -> type[BaseModel]:
154 """properties dict → pydantic 模型(extra=forbid,恒 populate_by_name)。
155
156 属性名可为任意字符串(非标识符亦可,create_model 经 **dict 展开,实测可用;
157 model_dump 输出原名 == schema 属性名,INV-06 可逐字段比对)。required 引用
158 未知属性 / 非 list → ValueError(TLB-803 上游收)。
159 """
160 if not isinstance(props, dict):
161 raise ValueError("object 的 properties 须为 dict")
162 req = required or []
163 if not isinstance(req, list) or not all(isinstance(r, str) for r in req):
164 raise ValueError("required 须为 str 元素 list")
165 missing = [r for r in req if r not in props]
166 if missing:
167 raise ValueError(f"required 引用未知属性: {missing}")
168
169 fields: dict[str, Any] = {}
170 for key, prop in props.items():
171 ann = _annotation_from(prop, model_name, key)
172 fields[key] = _field_spec(prop, ann, required=key in req)
173 return create_model(model_name,
174 __config__=ConfigDict(extra="forbid",
175 populate_by_name=True),
176 **fields)

178def _compile_model(tool_name: str, schema: Any) -> type[BaseModel]:
179 """工具注册时的 schema 编译入口(F026 早失败;失败 ValueError → TLB-803)。"""
180 if isinstance(schema, BaseModel):
181 raise ValueError("禁止 pydantic 模型对象混入 schema(须 JSON-Schema dict)")
182 if not isinstance(schema, dict):
183 raise ValueError(f"schema 须为 JSON-Schema dict,实为 {type(schema).__name__}")
184 t = schema.get("type")
185 if t not in (None, "object"):
186 raise ValueError(f"顶层 type 须为 object(或省略),实为 {t!r}")
187 if t is None and "properties" not in schema and "required" not in schema:
188 # 顶层既无 type 也无 properties/required(如 {"path": …} 裸键)→ 非
189 # object 参数形态,拒(空参数工具应显式写 {"type": "object"})
190 raise ValueError("顶层缺 type/properties,非 object 参数形态")
191 props = schema.get("properties", {})
192 return _object_model(f"ToolArgs_{_sanitize(tool_name)}", props,
193 schema.get("required"))
