1"""pyharness/core/budget.py — 预算门面(只读快照面,BudgetGate)。
2
3一句话职责:把**已在链上**的预算数据源(Scope 预算硬闸的 BudgetLimits + 会话级
4UsageCounters,均来自 F032 装配)聚合为一个只读门面,供外壳的预算仪表盘与
5`/api/budget` 查询面消费——不引入第二套预算状态,不新增写路径。
6
7设计纪律(原则 1:单一真源):
8- 计数真源 = events 日志的 llm.usage 事件(UsageCounters 是它的投影,可整体重建);
9 BudgetGate 只读 `counters` 与 `limits`,自身零状态、零写入、零事件。
10- 与 Scope.budget_state() **同源同判定**:同一个 counters 实例 + 同一份 limits
11 (engine 装配时注入同一对象),因此门面读数与 agent-loop 的强制终态判据一致,
12 不会出现"仪表盘说没超、循环说超了"的双口径。
13- 硬闸/终态归 agent-loop + Scope(见 scope.py/L4);本模块**不拦不抛**。
14
15历史缺口(2026-09-12 修复):装配层从未赋值 ctx.budget,导致
16`ApplicationService.budget_dashboard` 恒回 {"disabled": true, "reason":
17"budget-gate-unassembled"},桌面预算仪表盘只有 token 报表、没有成本闸位。
18"""

56class BudgetGate:
57 """会话预算门面(只读):counters + limits → snapshot / state。
58
59 用法(外壳只读消费):
60 snap = gate.snapshot(sid); snap.limit_cny, snap.used_cny, snap.ratio
61 `scope` 可选注入:注入时 state 直接取 Scope.budget_state()(含 warn/exhausted
62 迁移语义与 budget.paused 事件副作用),否则由本门面按同一判据纯计算(不落事件)。

65 def __init__(self, *, limits: Any = None, counters: Any = None,
66 scope: Optional[Any] = None, session_id: str = "") -> None:
67 self.limits = limits if limits is not None else BudgetLimits()
68 self.counters = counters
69 self.scope = scope
70 self.session_id = session_id
71
72 # ------------------------------------------------------------ 只读读数
73 def _usage(self) -> tuple[int, int, float, int]:
74 """(in, out, cost, requests) 现读;counters 缺失 → 全零(不猜)。"""
75 c = getattr(self.scope, '_counters', None) or self.counters
76 if c is None:
77 return 0, 0, 0.0, 0
78 used = c.task_total() if callable(getattr(c, 'task_total', None)) else c
79 return (int(getattr(used, "in_tokens", 0) or 0),
80 int(getattr(used, "out_tokens", 0) or 0),
81 float(getattr(used, "cost_est", 0.0) or 0.0),
82 int(getattr(c, "requests", 0) or 0))
83
84 def state(self) -> str:
85 """与 Scope.budget_state() 同判定:注入 scope 时直接复用其状态机。"""
86 if self.scope is not None:
87 try:
88 return str(self.scope.budget_state())
89 except Exception: # noqa: BLE001 只读降级
90 pass
91 return self._compute_state()
92
93 def _compute_state(self) -> str:
94 used_in, used_out, used_cny, _ = self._usage()
95 lim = self.limits
96 if (used_out >= getattr(lim, "max_out_tokens", 0)
97 or used_in >= getattr(lim, "max_in_tokens", 0)
98 or used_cny >= float(getattr(lim, "max_cost_yuan", 0.0))):
99 return "exhausted"
100 warn_ratio = float(getattr(lim, "warn_ratio", 0.8))
101 if (used_out >= warn_ratio * getattr(lim, "max_out_tokens", 0)
102 or used_cny >= warn_ratio * float(getattr(lim, "max_cost_yuan", 0.0))):
103 return "warn"
104 return "ok"
105
106 def snapshot(self, sid: Optional[str] = None) -> BudgetSnapshot:
107 """只读快照(幂等;不写日志、不落事件、不改任何状态)。"""
108 used_in, used_out, used_cny, requests = self._usage()
109 lim = self.limits
110 limit_cny = float(getattr(lim, "max_cost_yuan", 0.0) or 0.0)
111 warn_ratio = float(getattr(lim, "warn_ratio", 0.8))
112 ratio = (used_cny / limit_cny) if limit_cny else 0.0
113 return BudgetSnapshot(
114 sid=str(sid or self.session_id or ""),
115 state=self.state(),
116 limit_cny=limit_cny, used_cny=round(used_cny, 6),
117 ratio=round(ratio, 4), warn_ratio=warn_ratio,
118 warned=ratio >= warn_ratio,
119 limit_in_tokens=int(getattr(lim, "max_in_tokens", 0) or 0),
120 limit_out_tokens=int(getattr(lim, "max_out_tokens", 0) or 0),
121 used_in_tokens=used_in, used_out_tokens=used_out,
122 requests=requests)
