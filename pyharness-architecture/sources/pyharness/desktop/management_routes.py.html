107 @router.get('/schedules')
108 async def schedules():
109 from pyharness.core.schedule import Scheduler
110 from dataclasses import asdict
111 rows=[];schedules=[]
112 for session in (await desktop.service.list_sessions())['sessions'][:200]:
113 # Do not start runtimes/credentials while rendering an overview.
114 log=await desktop.service.require_session(session['sid'])
115 scheduler=Scheduler.rebuild_for_session(log)
116 for job in await scheduler.list_jobs():
117 record=asdict(job)
118 record['session_id']=session['sid']
119 record['intent']=scheduler._jobs[job.name].template['intent']
120 for field in ('next_fire_at','last_fired_at'):
121 record[field]=record[field].astimezone().isoformat() if record[field] else None
122 schedules.append(record)
123 for event in log.events_after(0):
124 if event.type.startswith('schedule.'):
125 rows.append({'session_id':session['sid'],'type':event.type,'timestamp':event.ts,'payload':event.payload})
126 return {'events':rows[-200:],'schedules':schedules}
