50def public_manual_text(value):
51 from pyharness.application.source_snapshot import check_content
52 from pyharness.core.context_observation import public_request
53 from pyharness.errors import PyHError
54 try:check_content(value.encode('utf-8'))
55 except (UnicodeError,PyHError):raise ValueError('private or invalid regression text')
56 view=public_request(dict(model='manual-regression-text',messages=[dict(role='user',content=value)]),layer='adapter_input')
57 if view['withheld'] or view['truncated'] or '[withheld:' in value:
58 raise ValueError('regression text requires an explicit usable public value')
59 return value

86def current_regression_content(command,*,reading=False):
87 from pyharness.errors import raise_code
88 value=command.model_dump() if hasattr(command,'model_dump') else command
89 values=[value['title']]
90 if 'input_text' in value:
91 values.append(value['input_text'])
92 for item in value['assertions']:
93 if type(item['expected']) is str:values.append(item['expected'])
94 if item.get('pointer') is not None:values.append(item['pointer'])
95 try:
96 for text in values:public_manual_text(text)
97 except ValueError:
98 raise_code('GRD-403' if reading else 'EVT-100',reason='regression_content_not_public' if reading else 'regression_content_rejected')
