16def _owned_task_message(log_: Any, task: Any) -> Optional[Any]:
17 """严格归属窗口:取 (上一个 task.enqueued, 本任务 enqueued] 内最后一条 user.message。
18
19 修复(P1-1):旧实现取"enqueued 之前最后一条 user.message",会把 plan 步骤的
20 /plan 旧消息顶给步骤执行(plan_mode 提交纯意图,不落 user.message),并发下还
21 会让 schedule 任务拿到前台消息。窗口为空(纯意图任务)→ None,由调用方按
22 task.intent 补写信号消息。只读遍历事件源,零副作用。
23 """
24 mark = int(getattr(task, "enqueued_seq", 0) or 0)
25 if mark <= 0:
26 return None
27 prev_enq = 0
28 env = None
29 for e in log_.events_after(0):
30 seq = int(getattr(e, "seq", 0) or 0)
31 if seq > mark:
32 break
33 if e.type == "task.enqueued":
34 if seq < mark:
35 prev_enq = seq
36 env = None # 新窗口:清空旧归属(前窗口消息不复用)
37 continue # 本任务自身 enqueued:非消息
38 if e.type == "user.message" and seq > prev_enq:
39 env = e
40 return env

54 # 段是同步函数、且此时可能尚无运行中的事件循环——本 seam 是 chat/run 的
55 # 唯一生产驱动点,循环必已就绪。

58 except Exception as e: # noqa: BLE001 定时器起不来不阻断任务
59 log.warning("flush ticker 启动失败: %s", type(e).__name__)
60 # 懒预载:created 已落后首跑前装载示例插件(util.now;EVT-106 首事件
61 # 约束 → 不可在 open_session 后立即 append,故延迟到 submit 时刻)
62 if not getattr(spine, "_plugins_ready", False):
63 try:
64 await preload_plugins(spine)
65 except Exception as e: # noqa: BLE001 预载失败不阻断
66 log.debug("plugin lazy preload failed: %s", type(e).__name__)
67 spine._plugins_ready = True
68 # 装配期策略留痕(F-01):**必须在 session.created 之后**发射——装配函数
69 # (_build_governance/activate_orchestration)运行时 seq=0,直接 append 会被
70 # EVT-106 拒("事件须在 session.created 之后")并让整个装配抛出。本处是
71 # 生产唯一驱动点(chat/run 均经 task_queue → 本 seam),此时会话已开启。

78 except Exception as e: # noqa: BLE001 治理留痕失败不阻断任务
79 log.warning("policy.updated 装配留痕失败: %s", type(e).__name__)
80 spine._policy_announced = True

82 # 放在本 seam(而非装配期):装配期 seq=0,append 会被 EVT-106 拒;此处是生产
83 # 唯一驱动点且会话已开启 —— 与上方策略留痕同款。幂等闸保证每 spine 恰一次;

87 rec = getattr(jobs, "recover", None)
88 if callable(rec):
89 try:
90 await rec()
91 except Exception as e: # noqa: BLE001 恢复失败不阻断
92 log.warning("job 崩溃恢复失败: %s", type(e).__name__)
93 spine._jobs_recovered = True
94 intent = str(getattr(task, "intent", "") or "").strip()
95 env = _owned_task_message(log_, task) # 严格归属窗口(P1-1)
96 if env is None and intent:

101 actor="system",
102 origin=f"task:{getattr(task, 'id', '')}",
103 task_id=getattr(task, "id", None),
104 sync=True)
105 if env is None:
106 raise PyHError("CYC-999", ctx={"hint": "runner 找不到对应 user.message 事件"})
107 ag_ctx = await _agent_ctx_of(spine, env)
108 # 当前任务关联(F044 弱耦合字符串引用;2026-09-21 R11-6 修):
109 # goal/todo 工具与 llm.chunk 出口经 ``ctx.task_id`` 取"当前任务",而此前
110 # **全库无任何写入点** ⇒ 恒 None ⇒ ``goal`` 的"所关联任务失败 → 目标状态
111 # 推导"(goal.py:357)与"关联任务"渲染**从未生效**。任务在同一会话内串行
112 # (单 running),故按任务边界设置/清除安全;每个会话(含子会话)各自持 ctx,
113 # 不跨会话串。异常/取消路径经 finally 还原。

116 # KF-C(A4/R24):把**框架侧**"本轮来源"提到 ctx —— 到点触发那一轮此前**无任何
117 # 通道**告知模型"这是定时触发" ⇒ 模型只会复述任务状态("提醒已设好"),而不是
118 # 产出提醒本身。取值只取 `task.meta.source`(**受控**:由 schedule._fire 写入),
119 # 不取任何模型/工具文本(注入防御);由 system_prompt 的受控段渲染。

139 log.debug("auto_title failed: %s", type(e).__name__)
140 # 证据归档**不在此处**:触发点已移到段生命周期(segment.end 订阅,见

153async def archive_task_evidence(spine: EngineSpine, task_id: str,
154 ctx: Any = None) -> Optional[str]:
155 """任务段治理证据归档——``EvidenceCollector`` 的**生产生产者**(GAP-8)。
156
157 修复前:``EvidenceCollector`` 已订阅真总线(4 类事件),但**唯一写点**
158 ``archive()`` 在全库**零调用** ⇒ ``evidence.archived`` 永不产生、索引恒空、
159 ``collect_for_task()`` 恒返回 ``()``。
160
161 **产生规则(冻结)**——这是一个**设计裁定**,不是"为了有调用点而调用":
162
163 1. **单位 = 任务段**(``task_id``)。段是回放/复算/预算审计的最小单位(F044),
164 也正是 ``collect_for_task`` 的聚合键;证据按段组织才有查询语义。
165 2. **触发条件 = 段内至少一条 ``decision.issued``**。治理证据回答的是"这次
166 授权/拒绝发生在什么上下文、依据什么、能否复核",没有治理决策的段**不归档**
167 ——"不发生"与"失败"是两件事,不为空段造工件。
168 3. **只存引用,不复制事件内容**(INV-G4/E1):refs 指向段锚 + 段内每条决策
169 (``decision_id``)与凭证(``receipt_id``)。事件日志仍是唯一真源。
170 4. **幂等**:同一 ``task_id`` 只归档一次——判据取自**日志重放**(已存在指向该
171 段锚的 ``evidence.archived`` 即跳过),故**重启后重放同样成立**,不依赖内存。
172 5. **降级**:治理层未接线(纯内存/单测)、无段锚、引用构造失败 ⇒ 静默跳过,
173 绝不阻断任务(证据是派生视图,不是主链前提)。
174
175 返回 ``evidence_id``;未归档(不满足条件)返回 ``None``。
176 """
177 log_ = getattr(spine, "session", None)
178 coll = getattr(getattr(spine, "governance", None), "evidence", None)
179 if log_ is None or coll is None:
180 return None
181 try:
182 events = list(log_.events_after(0))
183 seg_locator = f"{task_id}:seg"
184 # ① 幂等闸:日志重放判定(重启后仍成立)
185 for e in events:
186 if getattr(e, "type", None) != "evidence.archived":
187 continue
188 for r in (getattr(e, "payload", None) or {}).get("refs") or []:
189 if (isinstance(r, dict) and r.get("kind") == "segment"
190 and str(r.get("locator")) == seg_locator):
191 return None
192 # ② 段窗口:本任务 segment.start 之后的全部事件属该段
193 start = next((int(e.seq) for e in events
194 if getattr(e, "type", None) == "segment.start"
195 and str((getattr(e, "payload", None) or {})
196 .get("task_id") or "") == str(task_id)), None)
197 if start is None:
198 return None
199 seg = [e for e in events if int(getattr(e, "seq", 0) or 0) >= start]
200 decisions = [e for e in seg
201 if getattr(e, "type", None) == "decision.issued"]
202 if not decisions:
203 return None # 无治理事实:不归档(非失败)
204 receipts = [e for e in seg
205 if getattr(e, "type", None) == "receipt.emitted"]
206 denied = [d for d in decisions
207 if str((getattr(d, "payload", None) or {})
208 .get("verdict") or "") == "reject"]
209 refs: list[dict] = [{"kind": "segment", "locator": seg_locator}]
210 refs += [{"kind": "decision_id",
211 "locator": str((getattr(d, "payload", None) or {})
212 .get("decision_id") or "")}
213 for d in decisions]
214 refs += [{"kind": "receipt_id",
215 "locator": str((getattr(r, "payload", None) or {})
216 .get("receipt_id") or "")}
217 for r in receipts]
218 refs = [r for r in refs if r["locator"]] # 空 locator 不合法,剔除
219 # 终态标量(N3/M5-3):``task.completed`` / ``task.failed`` **先于**
220 # ``segment.end`` 落盘,故此刻可读。以**标量并入 claim** —— 不新增 payload
221 # 字段(INV-G4「只存引用」;payload schema 变更属 H-3,不在本轮范围)。
222 term = _terminal_state_of(seg, str(task_id))
223 ev = await coll.archive(
224 claim=f"task {task_id} 的治理事实可追溯(终态={term})",
225 refs=refs, ctx=ctx or SimpleNamespace(session=log_),
226 summary=(f"终态={term};决策 {len(decisions)} 条(拒绝 {len(denied)})、"
227 f"凭证 {len(receipts)} 条"))
228 return getattr(ev, "evidence_id", None)
229 except Exception: # noqa: BLE001 派生视图:绝不阻断主链
230 log.warning("证据归档跳过 task=%s", task_id, exc_info=True)
231 return None

233def _terminal_state_of(seg_events: list, task_id: str) -> str:
234 """段内终态标量(供 claim 标注;N3/M5-3)。
235
236 ``task.completed`` → ``success``;``task.failed`` → ``cancelled`` 或
237 ``error:<code>``;两者皆无(段被外力关闭)→ ``unknown``。
238 """
239 for e in reversed(seg_events):
240 t = getattr(e, "type", None)
241 if t == "task.completed":
242 return "success"
243 if t == "task.failed":
244 p = getattr(e, "payload", None) or {}
245 if str(p.get("reason") or "") == "cancelled":
246 return "cancelled"
247 return f"error:{p.get('error') or 'unknown'}"
248 return "unknown"

250def _evidence_producer(spine: EngineSpine):
251 """段关闭时的证据生产者回调(**N3/M5-3 的触发点**)。
252
253 **为什么不放在 ``run_for_task`` 的函数体里**:该函数跑在独立子任务中
254 (``task_queue._run_task`` 的 ``create_task``),五种终态里有三种(异常/预算/取消)
255 会让它**不执行到收尾**⇒ 段内有治理决策却零证据(实测 3/5)。而 ``segment.end``
256 由 ``_run_task`` 自身 ``finally`` 中的 ``close_segment`` 落盘,``_run_task``
257 **不被取消**(取消只作用于子任务)⇒ 实测 **5/5** 落盘。挂到该事件即继承此保证,
258 并天然消除 N3(生产者不再早于段关闭)。
259
260 幂等由 ``archive_task_evidence`` 内的日志重放闸保证;失败在该函数内吞掉
261 (派生视图不阻断主链),故本回调不会影响段关闭。
262 """
263 async def _on_segment_end(type_: str, env: Any) -> None:
264 if getattr(env, "type", None) != "segment.end":
265 return
266 task_id = str((getattr(env, "payload", None) or {}).get("task_id") or "")
267 if not task_id:
268 return
269 await archive_task_evidence(spine, task_id)
270 return _on_segment_end
