201 # ============================================================ 前置查权
202 def can_use(self, tool_name: str) -> bool:
203 """guard 单调链 scope 前置(F014/DIS §6.3.1):deny/strict 域外/critical
204 任一命中 → False;调用方在 False 时写 guard.rejected(GRD-401,scope-hidden)
205 强同步,Provider 零执行(INV-05)。high 危险此处放行,转审批(F015)。"""
206 self._ensure_bound()
207 p = self.policy
208 if p.allowed_tools is not None and tool_name not in p.allowed_tools:
209 return False
210 if tool_name in p.deny_tools: # 显式禁止(只增,单调)
211 return False
212 domain = tool_name.split(".", 1)[0] if "." in tool_name else tool_name
213 if domain in SELF_DOMAINS: # 会话内自管理(目标/待办等)
214 return True # 零外部副作用,不随沙箱收紧
215 if (p.sandbox_level == "strict"
216 and not self._inside_workspace_domain(tool_name)):
217 return False # strict:仅 workspace 域 + allowlist 网络
218 if self._danger_mark(tool_name) == "critical":
219 return False # critical 不可审批,本层即终局
220 return True

223 def tighten(self, deny: set[str], *, reason: str) -> None:
224 """策略只紧不松:追加 deny,写 scope.updated 事件(留痕可审计)。
225
226 幂等:deny ⊆ 现有 → 直接返回不产生事件(防刷日志);只追加(单调),无
227 relax/un_tighten API——放宽在结构上不可能(GWT-S6-01:hasattr 断言
228 AttributeError)。reason 必填(sandbox strict/guard 拒绝等触发源)。
229 """
230 self._ensure_bound()
231 if not deny or deny <= self.policy.deny_tools:
232 return # 无新增:幂等返回,不落事件
233 self.policy.deny_tools |= deny # 只追加(单调)
234 self._record("scope.updated", {"op": "tighten", "added": sorted(deny),
235 "reason": reason})

237 def note_tighten(self, deny: Iterable[str], *, reason: str) -> None:
238 """装配期收紧留痕(幂等落 scope.updated,不改策略读面)。
239
240 用于**同步装配段**(engine.build_runner_components 无法 await);策略本身
241 已由装配方直接写入 policy.deny_tools,本方法只补审计事件——不这样做时
242 预设档位切换(locked/readonly)在日志里无痕,违反"动作以事件留痕"。
243 deny 与现有集合无新增时静默返回(与 tighten 同幂等语义)。
244 """
245 added = set(deny)
246 if not added:
247 return
248 self._record("scope.updated", {"op": "tighten", "added": sorted(added),
249 "reason": reason})

257 def budget_state(self) -> BudgetState:
258 """预算硬闸(F032/DIS §6.3.2):读计数器判 ok/warn/exhausted。
259
260 硬闸只用 token 数(out/in/cost 任一 ≥ 上限即 exhausted,F032 判定以
261 token 为主);warn=80%(只提醒不拦)。状态迁移事件化:首达 exhausted →
262 budget.paused 事件落日志;首达 warn → bus budget.warn(尽力而为)。
263 agent-loop._must_stop 对 paused/exhausted 强制终态(reason=budget)。
264 计数器 None(未注入)→ 按零读数返回 ok(事件降级,偏离 3)。
265 """
266 self._ensure_bound()
267 used = self._usage()
268 lim = self.limits
269 ancestors = getattr(self._counters, 'ancestors', lambda: ())()
270 parent_exhausted = any(
271 account._limits is not None and (
272 account.task_total().out_tokens >= account._limits.max_out_tokens or
273 account.task_total().in_tokens >= account._limits.max_in_tokens or
274 account.task_total().cost_est >= account._limits.max_cost_yuan)
275 for account in ancestors)
276 if (parent_exhausted or used.out_tokens >= lim.max_out_tokens
277 or used.in_tokens >= lim.max_in_tokens
278 or used.cost_est >= lim.max_cost_yuan):
279 if self._budget_state != "exhausted": # 首达才落事件(防刷)
280 self._record("budget.paused",
281 {"state": "exhausted", "used": self._used_dict(used)})
282 self._budget_state = "exhausted"
283 return "exhausted"
284 if used.out_tokens >= lim.warn_ratio * lim.max_out_tokens:
285 if self._budget_state != "warn":
286 self._emit_bus("budget.warn", {"used_out": used.out_tokens})
287 self._budget_state = "warn"
288 return "warn"
289 self._budget_state = "ok"
290 return "ok"

297 paused 为防御性同判;ok/warn 原样返回不拦。

384 def _danger_mark(self, tool_name: str) -> str:
385 """危险分级查表(fnmatch 前缀规则,命中即返;表与 F023 guard 同源加载)。

395 def _record(self, type_: str, payload: dict) -> None:
396 """策略/预算事件落日志(scope.updated/budget.paused 留痕,审计用)。
397
398 SessionLog.append 为 async、scope 判定面为同步(偏离 4):append 返回
399 awaitable 且有运行中事件循环 → create_task 排程投递(fire-and-forget,
400 失败只记日志);append 同步(FakeSession)→ 直接记录;未接线 → 日志降级。
401 """
402 sess = self._session
403 if sess is None:
404 log.debug("scope=%s 未接线事件出口,%s 未记录", self.session_id, type_)
405 return
406 try:
407 r = sess.append(type_, payload, actor="system")
408 except Exception as exc: # noqa: BLE001 事件失败不阻断策略主路径
409 log.warning("scope append %s 失败: %s", type_, exc)
410 return
411 if inspect.isawaitable(r):
412 try:
413 asyncio.get_running_loop()
414 except RuntimeError: # 无运行中事件循环:同步/未装配上下文
415 log.warning("scope=%s 无运行中事件循环,%s 未投递",
416 self.session_id, type_)
417 return
418 task = asyncio.create_task(self._safe_append(r, type_))
419 self._tasks.add(task)
420 task.add_done_callback(self._tasks.discard)
421
422 async def _safe_append(self, coro: Any, type_: str) -> None:
423 """异步落点包装:append 失败只记日志,不向调用方传播(尽力而为)。"""
424 try:
425 await coro
426 except Exception as exc: # noqa: BLE001 EVT-102(词表外)等
427 log.warning("scope append %s 异步失败: %s", type_, exc)
428
429 def _emit_bus(self, type_: str, payload: dict) -> None:
430 """bus 尽力而为出口(budget.warn 告警;未接线/失败 → 日志降级,不阻断)。"""
431 bus = self._bus
432 if bus is None:
433 return
434 from pyharness.bus import schedule_emit
435 try:
436 schedule_emit(bus, type_, payload)
437 except Exception as exc: # noqa: BLE001 未注册类型等
438 log.warning("scope emit %s 失败: %s", type_, exc)

442def _validate_minimal(p: ScopePolicy) -> None:
443 """最小权限编译校验(F021;越权配置 → CFG-601 列字段,拒启动会话)。
444
445 两种越权(见偏离 2):
446 1) 空 deny 且无任何 critical 禁项 = scope 无防线(全放行)→ 拒;
447 2) allowed_domains 含通配 "*" = allow-all/过宽(域名 allowlist 只收字面
448 域名,URL 级匹配由 g5 做)→ 拒。
449 """
450 if not p.deny_tools and not any(
451 _DANGER_RANK.get(str(m.get("level")), 0) >= _DANGER_RANK["critical"]
452 for m in p.danger_marks):
453 raise_code("CFG-601", reason="越权",
454 fields=["security.policy.deny_tools_extra"],
455 detail="deny 空且无 critical 禁项=scope 无防线(最小权限须留"
456 "至少一道拒绝面;默认禁高危组)")
457 for d in sorted(p.allowed_domains):
458 if "*" in d:
459 raise_code("CFG-601", reason="越权",
460 fields=["security.network.allowed_domains"],
461 detail=f"域名通配过宽:{d}(allowlist 只收字面域名,禁 "
462 "allow-all/通配)")

465def build_scope(cfg: Any, session_id: str, *,
466 session: Any = None, counters: Any = None,
467 bus: Any = None) -> Scope:
468 """作用域创建与绑定(F021/DIS §6.3.3):编译默认最小权限策略并注册到会话。
469
470 cfg = Settings(或含 security/budget/loop/storage 域的分层配置对象);
471 每会话恰一 scope:session_id 已在册 → BUSY 拒(注册隔离;仅 release 解绑)。
472 越权配置(空 deny 无防线/域名通配过宽)→ CFG-601 列非法字段,拒绝启动。
473 session/counters/bus 为注入式装配可选接线(偏离 3)。
474
475 异常:PyHError(BUSY,重复 build)/ConfigError(CFG-601,配置越权)。
476 """
477 if session_id in _scopes:
478 raise PyHError("BUSY", ctx={
479 "session_id": session_id,
480 "advice": "该会话已有活动 scope;重复 create active scope 被拒"
481 "(注册隔离,每会话恰一 scope)"})
482 sandbox = getattr(getattr(cfg, "security", None), "sandbox", None)
483 p = ScopePolicy(
484 role=DEFAULT_ROLE, # config 无 scope.role 键(偏离 1)
485 deny_tools=set(getattr(
486 getattr(getattr(cfg, "security", None), "policy", None),
487 "deny_tools_extra", None) or []), # 基集外只增(安全默认)
488 danger_marks=_load_danger_marks(cfg), # 内置分级表 + extra 上调
489 allowed_domains=set(getattr(
490 getattr(getattr(cfg, "security", None), "network", None),
491 "allowed_domains", None) or []),
492 workspace_root=session_workspace(
493 getattr(getattr(cfg, "storage", None), "workspaces_dir",
494 "~/.pyharness/workspaces"),
495 session_id), # F055 fresh workspace
496 sandbox_level=getattr(sandbox, "level", "strict") or "strict", # 默认 strict
497 read_extra_dirs=[str(d) for d in (getattr(
498 getattr(getattr(cfg, "security", None), "policy", None),
499 "read_extra_dirs", None) or [])]) # A2:workspace 外只读例外
500 _validate_minimal(p) # 越权拒绝(CFG-601)
501 window = getattr(getattr(cfg, "loop", None), "max_context_tokens", None)
502 ratio = getattr(getattr(getattr(cfg, "loop", None), "compact", None),
503 "trigger_ratio", None)
504 s = Scope(p, BudgetLimits.from_cfg(cfg), session_id,
505 window_tokens=int(window or DEFAULT_WINDOW_TOKENS),
506 window_ratio=float(ratio or DEFAULT_WINDOW_RATIO),
507 session=session, counters=counters, bus=bus)
508 _scopes[session_id] = s # 注册(绑定;scope.updated 留痕)
509 return s

512def from_snapshot(snap: ScopeSnapshot, session_id: str, *,
513 session: Any = None, counters: Any = None,
514 bus: Any = None) -> Scope:
515 """遮蔽层创建(F059 fork 继承):从父快照拷贝策略独立演进。
516
517 子层 = 父策略快照 + 独立注册;子层 tighten 只追加本层 deny,不透写父层;
518 父会话后续 tighten 不影响子会话(遮蔽,无透写,COW 起点)。
519 重复创建 → BUSY(与 build_scope 同注册隔离)。
520 """
521 if session_id in _scopes:
522 raise PyHError("BUSY", ctx={
523 "session_id": session_id,
524 "advice": "子会话已有活动 scope;重复 fork 被拒(查重复 fork 路径)"})
525 p = ScopePolicy(**snap.policy.model_dump()) # 深拷贝策略(COW 起点)
526 s = Scope(p, BudgetLimits(**snap.limits.model_dump()),
527 session_id,
528 window_tokens=snap.window_tokens,
529 window_ratio=snap.window_ratio,
530 counters=counters, session=session, bus=bus,
531 parent_snap=snap) # 遮蔽层记父快照(F059)
532 _scopes[session_id] = s # 独立注册(隔离于父表)
533 return s
