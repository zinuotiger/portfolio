75def _count_text_tokens(text: str) -> int:
76 """纯文本 token 粗估(确定性;与 session._estimate_tokens 同口径):
77 中文逐字 1 token,西文词按 4 字符 ≈ 1 token。仅供相对比较与声明留痕。"""
78 cjk = sum(1 for ch in text if ord(ch) in _CJK_RANGE)
79 words = sum(max(1, (len(w) + 3) // 4) for w in _ASCII_WORD.findall(text))
80 return cjk + words

82def estimate_tokens(value: Union[str, dict, list, tuple, None]) -> int:
83 """token 估算入口:str = 文本;dict = 消息(取 content,含结构开销);
84 list/tuple = 逐元素累加(消息列表每条 +_MSG_OVERHEAD)。纯确定性,无 IO。"""
85 if value is None:
86 return 0
87 if isinstance(value, str):
88 return _count_text_tokens(value) + _MSG_OVERHEAD
89 if isinstance(value, dict):
90 content = value.get("content", "") if value.get("role") else value
91 if isinstance(content, (list, tuple)):
92 import json
93 content = json.dumps(list(content), ensure_ascii=False)
94 return _count_text_tokens(str(content)) + _MSG_OVERHEAD
95 return sum(estimate_tokens(item) for item in value)

103def _projection(events: list[Envelope]) -> list[dict]:
104 """区间事件 → 派生消息投影(会计/前缀规划用;语义 = session reducer 对区间的
105 映射,EVENT-SCHEMA §4.2:user→user、llm.response(有 content)→assistant、
106 空 content response + tool.result → tool 配对、context.compacted → system
107 摘要消息)。纯函数,不建历史,只用于 token 差值口径与前缀估算。
108
109 偏离:user.message_edited 的跨区间改写(target 可能在本区间外)不追,会计
110 近似(声明留痕用途,误差仅影响 tokens_after 数值精度,不影响任何决策)。"""
111 msgs: list[dict] = []
112 pending: Optional[int] = None # 待配对 tool 的空 content response seq
113 for ev in events:
114 t = ev.type
115 if t == "user.message":
116 msgs.append({"role": "user", "content": ev.payload["content"]})
117 elif t == "context.compacted":
118 msgs.append({"role": "system",
119 "content": f"[已压缩 {ev.payload['ranges']}] "
120 f"{ev.payload.get('summary', '')}"})
121 elif t == "llm.response":
122 c = ev.payload.get("content") or ""
123 if c:
124 msgs.append({"role": "assistant", "content": c})
125 else:
126 pending = ev.seq
127 elif t == "tool.result" and pending is not None:
128 msgs.append({"role": "tool", "content": ev.payload["summary"],
129 "name": ev.payload["name"]})
130 pending = None
131 # 其余事件(guard.*/approval.*/plan.*/session.*/tool.call…)不进 LLM 投影
132 return msgs

134def render_segment(events: list[Envelope], max_chars: int = 6000) -> str:
135 """任务段 → 可摘要文本(决策轮文本化;tool 结果截断,安全审计细节不入文本)。
136
137 SECURITY.md §3.1:guard.*/approval.* 细节一律不进入摘要输入——模型摘要
138 不可能编造其未见内容(结构上保证 guard 拒绝记录不进摘要)。"""
139 lines: list[str] = []
140 for ev in events:
141 t = ev.type
142 if t in ("guard.rejected", "guard.evaluated", "approval.requested",
143 "approval.granted", "approval.denied", "approval.timeout",
144 "tool.call"):
145 continue # 安全审计/审批细节不进摘要(§3.1)
146 if t == "user.message":
147 lines.append(f"用户: {ev.payload['content']}")
148 elif t == "agent.message":
149 lines.append(f"助手: {ev.payload['content']}")
150 elif t == "llm.response":
151 c = ev.payload.get("content") or ""
152 if c:
153 lines.append(f"模型: {c}")
154 elif t == "tool.result":
155 summ = (ev.payload.get("summary") or "")[:200] # 工具原始结果截断
156 lines.append(f"工具[{ev.payload.get('name', '?')}]: {summ}")
157 elif t == "plan.proposed":
158 lines.append(f"计划目标: {ev.payload.get('goal', '')}")
159 elif t == "goal.updated":
160 gid = ev.payload.get("goal_id", "")
161 st = ev.payload.get("status", "")
162 lines.append(f"目标[{gid}]: {st} "
163 f"{ev.payload.get('note') or ev.payload.get('desc') or ''}")
164 elif t == "todo.updated":
165 todos = ev.payload.get("todos") or []
166 # payload 已 model_dump 归一:todo 项为 dict(兼容对象形态)
167 done_n = sum(1 for it in todos
168 if (it.get("done") if isinstance(it, dict)
169 else getattr(it, "done", False)))
170 lines.append(f"待办更新: {done_n}/{len(todos)} 项完成")
171 elif t == "context.compacted":
172 lines.append(f"(此前折叠: {ev.payload['ranges']})")
173 text = "\n".join(x for x in lines if x).strip()
174 return text[:max_chars]

176def render_compacted_message(env: Envelope) -> str:
177 """compacted 声明的人类可读渲染(EVENT-SCHEMA §4.2 摘要代区间锚文本)。
178
179 说明:会话 reducer 对 context.compacted 的实际插桩格式由 session.
180 _fold_history 决定(`[已压缩 {ranges}] {summary}`,ranges 为 Python 列表字面
181 量);本函数按 compaction 规格提供 [lo-hi] 紧凑形态,供压缩报告与
182 kv_prefix_plan 前缀内容估算使用(与 reducer 无耦合,纯只读工具)。"""
183 p = env.payload
184 rng = ",".join(f"[{lo}-{hi}]" for lo, hi in p.get("ranges", []))
185 return f"[已压缩 {rng}] {p.get('summary', '')}"
