20def _sanitize_tool_name(name: str) -> str:
21 """DeepSeek 工具名正则 ^[a-zA-Z0-9_-]+$(禁点号);fs.read_file → fs_read_file。
22
23 真链实测:带点工具名 → HTTP 400 'Invalid tools[0].function.name' →
24 被归一为 LLM-304。发送前清洗、响应 tool_calls 名还原(见 chat)。
25 """
26 return re.sub(r"[^a-zA-Z0-9_-]", "_", name)

28def _sanitize_tools(tools: Optional[list]) -> tuple:
29 """下发 tools 清洗 + 反向映射表。返回 (clean_tools, {sanitized: original})。"""
30 if not tools:
31 return tools, {}
32 rev: dict[str, str] = {}
33 clean: list[dict] = []
34 for t in tools:
35 fn = (t or {}).get("function") or {}
36 orig = fn.get("name", "")
37 if orig and orig != _sanitize_tool_name(orig):
38 rev[_sanitize_tool_name(orig)] = orig
39 clean.append(t)
40 if not rev:
41 return tools, {}
42 # 需重建带清洗名的 tools(嵌套 dict 深拷贝改 name)
43 import copy as _copy
44 out = []
45 for t in clean:
46 t2 = _copy.deepcopy(t)
47 fn = t2.get("function")
48 if fn and fn.get("name"):
49 fn["name"] = _sanitize_tool_name(str(fn["name"]))
50 out.append(t2)
51 return out, rev

62def _load_arguments(raw_json: Optional[str], call_id: Optional[str]) -> dict:
63 """arguments 二次 json.loads(空串/纯空白容错为 {});语法错/非对象 → ToolCallSyntaxError。"""
64 raw = (raw_json or "").strip()
65 if not raw:
66 return {} # 空串/纯空白 = 无参数调用(容错)
67 try:
68 args = json.loads(raw)
69 except json.JSONDecodeError as e:
70 raise ToolCallSyntaxError(call_id, e) from e
71 if not isinstance(args, dict):
72 raise ToolCallSyntaxError(call_id, "arguments 非 JSON 对象(零猜原则,拒本轮)")
73 return args

75def parse_tool_calls(message: Any) -> list[ToolCall]:
76 """tool_calls wire 拆解(ADI §1.4):function.arguments 是字符串内嵌 JSON,二次 loads。
77
78 非 function 类型跳过(留 raw 审计);同响应 id 重复 → ToolCallSyntaxError(配对必乱);
79 返回按数组序升序 = 执行序。message = 传输层 choices[0].message(鸭子对象)。
80 """
81 seen: set = set()
82 out: list[ToolCall] = []
83 for idx, tc in enumerate(getattr(message, "tool_calls", None) or []):
84 if getattr(tc, "type", None) not in (None, "function"):
85 continue # 非 function 不进管道,留 raw 审计
86 tc_id = tc.id
87 if tc_id in seen:
88 raise ToolCallSyntaxError(tc_id, "同响应 id 重复,拒本轮")
89 seen.add(tc_id)
90 raw_json = (tc.function.arguments or "") if tc.function else ""
91 out.append(ToolCall(id=tc_id, index=idx, name=tc.function.name,
92 raw_args=_load_arguments(raw_json, tc_id),
93 raw_json=raw_json)) # 原文留审计(INV-06)
94 return out

119async def _emit_chunk(ctx: Any, delta: str) -> None:
120 """llm.chunk 上总线(瞬时事件,禁 append 入日志;EVENT-SCHEMA §1.3)。"""
121 bus = getattr(ctx, "bus", None)
122 if bus is None:
123 return # 未装配总线:纯内存/单测模式
124 try:
125 payload = {"delta": delta, "session_id": str(getattr(
126 getattr(ctx, "session", None), "sid", "") or "")}
127 task_id = getattr(ctx, "task_id", None)
128 if task_id:
129 payload["task_id"] = str(task_id)
130 r = bus.emit("llm.chunk", payload)
131 if asyncio.iscoroutine(r):
132 await r
133 except Exception as exc: # noqa: BLE001 UI 通道尽力而为
134 log.debug("llm.chunk emit failed: %s", exc)

143async def accumulate_stream(stream: AsyncIterator[Any], ctx: Any, *, progress=None) -> tuple:
144 """SSE 增量聚合(ADI §5.3):content 拼接、tool_calls 按 index 分片合并(追加不覆盖)。
145
146 返回 (text, calls, finish_reason, usage);chunk 只经总线给 UI、不进日志;usage 只出现在
147 include_usage 末块(choices=[] 仅该块);聚合结果与逐 chunk 拼接逐字节一致(INV 比对)。
148 """
149 parts: list[str] = []
150 slots: dict[int, dict] = {}
151 finish: Optional[str] = None
152 usage: Any = None
153 if progress is not None:
154 progress.update(parts=parts, slots=slots, usage=None)
155 async for chunk in stream:
156 choices = getattr(chunk, "choices", None) or []
157 if not choices: # choices=[] + usage:仅流末块
158 u = getattr(chunk, "usage", None)
159 if u is not None:
160 usage = u
161 if progress is not None:
162 progress['usage'] = u
163 continue
164 d = choices[0].delta
165 content = getattr(d, "content", None)
166 if content: # 增量文本:拼 + 上总线(不留日志)
167 parts.append(content)
168 await _emit_chunk(ctx, content)
169 for dt in (getattr(d, "tool_calls", None) or []): # id/name 只在该 index 首块
170 slot = slots.setdefault(dt.index, {"id": None, "name": "", "arguments": ""})
171 if dt.id:
172 slot["id"] = dt.id
173 fn = getattr(dt, "function", None)
174 if fn is not None:
175 if fn.name:
176 slot["name"] = fn.name
177 if fn.arguments:
178 slot["arguments"] += fn.arguments # 跨块追加,绝不覆盖
179 fr = choices[0].finish_reason
180 if fr:
181 finish = fr
182 calls = [_slot_to_call(i, slots[i]) for i in sorted(slots)]
183 return "".join(parts), calls, finish, usage

185def _est_tokens(text: str) -> int:
186 """输出 token 粗估(流式计量盲区兜底;与 session._estimate_tokens 同启发式)。
187
188 CJK 逐字 1 token;ASCII 词按 4 字符 ≈ 1;+4 结构开销。仅用于 usage 缺失时
189 让 F029 账目与预算闸至少计入 output 量级,不构成权威计数。
190 """
191 if not text:
192 return 0
193 cjk = sum(1 for ch in text if "一" <= ch <= "鿿")
194 words = sum(max(1, (len(w) + 3) // 4)
195 for w in re.findall(r"[A-Za-z0-9_./\\@:+-]+", text))
196 return cjk + words + 4

198def _http_status_code(status: int) -> str:
199 """HTTP 状态 → 码(401/403→302;429/≥500→303;其余 4xx→304;只按状态不按文本)。"""
200 if status in (401, 403):
201 return "LLM-302" # 认证/权限:不重试直接降级
202 if status == 429 or status >= 500:
203 return "LLM-303" # 限流/上游 5xx:退避可重试
204 return "LLM-304" # 其余 4xx 业务错:不重试不降级

214def normalize_exc(e: Exception) -> str:
215 """异常类别/HTTP 状态 → LLM-3xx 唯一映射(ADI §7);绝不匹配厂商错误文本(ADR-011)。
216
217 非 API 异常原样上抛(不上 LLM 码——内部缺陷 → CYC-999 兜底,禁伪装模型域错误)。
218 """
219 if isinstance(e, asyncio.TimeoutError):
220 return "LLM-301" # 编排总闸(180s)
221 if isinstance(e, ProviderStatusError) or (httpx is not None
222 and isinstance(e, httpx.HTTPStatusError)):
223 return _http_status_code(int(getattr(e, "status_code", getattr(getattr(e, "response", None), "status_code", 0))))
224 if isinstance(e, ProviderProtocolError):
225 return "LLM-304" # 响应结构非法(内容过滤/坏 JSON)
226 oa = _openai_module()
227 if oa is not None: # openai SDK 异常类(映射矩阵 ADI §7.2)
228 if isinstance(e, (oa.AuthenticationError, oa.PermissionDeniedError)):
229 return "LLM-302"
230 if isinstance(e, (oa.RateLimitError, oa.APIConnectionError,
231 oa.APITimeoutError, oa.InternalServerError)):
232 return "LLM-303"
233 if isinstance(e, (oa.BadRequestError, oa.NotFoundError,
234 oa.UnprocessableEntityError, oa.APIStatusError, oa.APIError)):
235 return "LLM-304"
236 if httpx is not None and isinstance(e, httpx.TransportError):
237 return "LLM-303" # 断网/连接/读超时(httpx 传输族)
238 raise e # 非 API 异常不上 LLM 码(→CYC-999)
