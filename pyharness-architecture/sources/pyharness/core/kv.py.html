1"""pyharness/core/kv.py — 会话级 KV 存储(storage.kv,#53 存储域最小落地)。
2
3一句话职责:会话私有的小数据键值存取(get/set/delete/keys),JSON 文件落盘
4(<sid>.kv.json,随会话目录;无 SQLite 依赖——通用小数据用文件足够,PRD 语义)。
5
6设计:
7- storage 域工具(strict 下 workspace 面可见,engine 已放行 storage.*);
8- 单文件整体读写(值 ≤ 64KB 上限,防巨型值撑爆文件);set 覆盖幂等;
9- 文件缺失 = 空表(首访自建);损坏 JSON 报 PERS-221(repair 语义,不静默清空);
10- 事件溯源纪律:KV 是插件存储域(非会话日志),不进事件词表(与 DSH storage 域
11 定位一致:通用小数据存取,审计走 fs 面)。

32class SessionKV:
33 """会话级 KV(路径注入;lazy 载入,每次写后整体落盘)。"""
34
35 def __init__(self, path: Path, session_id: str = "") -> None:
36 self._path = Path(path)
37 self._sid = session_id
38 self._data: dict[str, str] = {}
39 self._loaded = False
40 self._lock = threading.RLock()
41
42 # ------------------------------------------------------------ 读写
43 def _load(self) -> None:
44 if self._loaded:
45 return
46 if not self._path.exists():
47 self._loaded = True
48 return
49 try:
50 raw = json.loads(self._path.read_text(encoding="utf-8"))
51 if not isinstance(raw, dict):
52 raise ValueError("KV 顶层必须是 JSON object")
53 self._data = {k: str(v) for k, v in raw.items()}
54 self._loaded = True
55 except (OSError, ValueError) as e:
56 raise_code("PERS-221", module="kv", sid=self._sid,
57 hint=f"KV 文件损坏/不可读:{type(e).__name__};"
58 "用 repair 定位或删除该会话 kv 文件重建")
59
60 def _save(self) -> None:
61 self._path.parent.mkdir(parents=True, exist_ok=True)
62 # 原子写:同目录临时文件 + os.replace(防半写)
63 fd, tmp = tempfile.mkstemp(dir=str(self._path.parent),
64 prefix=".kv-", suffix=".tmp")
65 try:
66 with os.fdopen(fd, "w", encoding="utf-8") as fh:
67 json.dump(self._data, fh, ensure_ascii=False, sort_keys=True)
68 os.replace(tmp, self._path)
69 except OSError as e:
70 raise_code("PERS-221", module="kv", sid=self._sid,
71 reason=type(e).__name__)
72 finally:
73 if os.path.exists(tmp):
74 try:
75 os.unlink(tmp)
76 except OSError:
77 pass
78
79 # ------------------------------------------------------------ 工具 op 面
80 def apply_op(self, op: str, *, key: Optional[str] = None,
81 value: Optional[str] = None) -> str:
82 """op ∈ get/set/delete/keys;返回用户可读文本(executor 关4 直用)。"""
83 with self._lock:
84 return self._apply_op_unlocked(op, key=key, value=value)
85
86 def _apply_op_unlocked(self, op: str, *, key: Optional[str] = None,
87 value: Optional[str] = None) -> str:
88 """锁内事务体:provider 在线程池并发调用时防丢失更新。"""
89 if op not in _OPS:
90 raise_code("EVT-100", hint=f"未知 storage.kv op: {op}",
91 advice=f"op ∈ {_OPS}")
92 self._load()
93 if op == "keys":
94 ks = sorted(self._data)
95 return "、".join(ks) if ks else "(空)"
96 key = str(key or "").strip()
97 if not key:
98 raise_code("EVT-100", hint="storage.kv 缺 key 字段")
99 if op == "get":
100 v = self._data.get(key)
101 return v if v is not None else f"(无此键: {key})"
102 if op == "delete":
103 existed = self._data.pop(key, None) is not None
104 if existed:
105 self._save()
106 return f"已删除 {key}" if existed else f"(无此键: {key})"
107 # set
108 value = "" if value is None else str(value)
109 if len(value) > MAX_VALUE_CHARS:
110 raise_code("EVT-100", hint=f"KV 值超上限({MAX_VALUE_CHARS} 字符)")
111 self._data[key] = value
112 self._save()
113 return f"已写入 {key}({len(value)} 字符)"

116def register(registry: Any) -> list[str]:
117 """五要素登记 + Provider 绑定(engine 装配面调用;Provider 经 ctx.storage.kv)。"""
118 from pyharness.core.tools_registry import ToolDefinition
119 registry.register_tool(ToolDefinition(
120 name=TOOL_NAME, danger="none",
121 description="会话级键值存取:set(key,value)/get(key)/delete(key)/keys;"
122 "跨轮持久(JSON 落盘),适合记偏好/中间结论等小数据",
123 schema={"type": "object",
124 "properties": {
125 "op": {"type": "string", "enum": list(_OPS)},
126 "key": {"type": "string", "maxLength": 256},
127 "value": {"type": "string", "maxLength": MAX_VALUE_CHARS}},
128 "required": ["op"], "additionalProperties": False},
129 timeout_s=10, owner="builtin", ctx_path="storage.kv", version="1.0.0"),
130 provider=_KVHandle())
131 return [TOOL_NAME]

134class _KVHandle:
135 """Provider handle:op 分派到 ctx.storage.kv(SessionKV)。"""
136
137 def handle(self, args: dict, ctx: Any) -> str:
138 kv = getattr(getattr(ctx, "storage", None), "kv", None)
139 if kv is None:
140 raise_code("CYC-999", module="storage.kv",
141 hint="ctx.storage.kv 未装配(引擎未激活 SessionKV),不可用")
