12 args: dict, *, prior: Any,

14 """关2.5 审批编排(request → granted 重入治理授权 → 放行/拒绝裁决)。
15
16 返回 ``ApprovalRoundResult``(S3-2-2:取代混合字符串契约)。语义:
17
18 - ``denied`` / ``timeout``:不产生 D2、不再次 authorize、不执行 provider
19 ——返回 ``decision=prior(D1)``, ``approved=False``, ``executed=False``;
20 - ``APR-501``(headless 无通道):同上,并记 ``_rejected``;
21 - ``granted``:重入治理授权 ``authorize(prior=D1)`` 产生 **D2**;
22 重入 reject → GRD-403(批准作废,单调性高于人类即时意志);
23 重入仍 approval → 须过绑定校验(本调用参数指纹 == 审批提供器为该
24 call_id 记录的绑定)才放行,不一致 = 批准针对异参数 → 拒(偏离 3 收紧);
25 allow/校验通过 → 放行至关3。
26
27 **R1**:``denied``/``timeout`` 的语义归 approval transport + executor,
28 不向 ``DecisionEngine`` 传 ``approval_verdict``。
29 """
30 ap = getattr(ctx, "approval", None)
31 if ap is None:
32 raise_code("CYC-999", module="tools_executor",
33 hint="ctx.approval 未接线:guard 判 need_approval 但审批"
34 "服务缺失;拒绝执行(fail-closed)")
35 # binding:由 execute() 传入的授权↔执行绑定指纹(与 Decision.inputs_digest
36 # 同源,全流程只算一次;不再在本函数重复计算)。
37 try:
38 verdict = await ap.request(call, summarize(args), ctx, binding=binding)
39 except PyHError as e:
40 if e.code == "APR-501": # headless 无通道:直接拒
41 self._rejected.add(call.call_id)
42 # ADR-022 D-7(a):本路径**不发** approval.requested(零事件零等待),
43 # 故派生层**没有任何事件可供反查 call_id** ⇒ 必须经**既有**
44 # tool.error 通道补配对,否则 assistant.tool_calls 悬空 → 端点 400。
45 # 仅更换**事件出口**(副作用即本调用目的),判定语义与返回类型不变;
46 # 未新增事件类型、未触碰 guard.rejected/approval.denied 的产生逻辑。
47 await self._on_error(ctx, call, "APR-501",
48 "审批不可用(APR-501),未执行")
49 return ApprovalRoundResult(decision=prior, approved=False,
50 executed=False,
51 message="审批不可用(APR-501),未执行")
52 raise # 其余错误:装配/落盘故障上抛
53 if verdict != "granted": # denied/timeout = 不执行
54 return ApprovalRoundResult(decision=prior, approved=False,
55 executed=False,
56 message=f"审批{verdict},未执行")
57 # granted ≠ 放行:重入治理授权(第二次求值 → D2;supersedes=D1)
58 # inputs_digest 与首次一致(同 call/同 args → 同绑定指纹)。approval_ref =
59 # 本次 approval lifecycle 的**真实 identity**(= approval.requested 的 seq),
60 # 由 approval 侧只读访问器提供;D1 无此值(它在请求产生**之前**已发射),
61 # 属正确语义——见 INV-APPROVAL-REF(S4-P1-2 冻结口径)。
62 get_ref = getattr(ap, "approval_ref_of", None)
63 approval_ref = get_ref(call.call_id) if get_ref is not None else None
64 d2 = await self._authorize_governance(ctx, call, prior=prior,
65 inputs_digest=binding,
66 approval_ref=approval_ref)
67 if d2 == "reject": # 批准时策略收紧 → 作废
68 self._rejected.add(call.call_id)
69 return ApprovalRoundResult(decision=d2, approved=True, executed=False,
70 message="审批后 guard 重入拒绝(GRD-403 语义)")
71 if d2 == "approval": # 重入仍要求审批(同参已批)
72 check = getattr(ap, "grant_binding", None)
73 if check is not None and check(call.call_id) != binding:
74 # 绑定不一致 = 该 grant 不是针对本次参数的授权(篡改/复用)
75 self._rejected.add(call.call_id)
76 return ApprovalRoundResult(
77 decision=d2, approved=True, executed=False,
78 message="审批绑定校验失败(GRD-403):批准与本次调用参数不一致,"
79 "拒绝执行")
80 return ApprovalRoundResult(decision=d2, approved=True, executed=True)
