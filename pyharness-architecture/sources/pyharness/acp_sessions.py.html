15async def _create_session(ctx: Any) -> str:
16 """新建会话(initialize 无 sessionId 路径):委托 ctx.session.create()。
17
18 门面缺 create(如 ctx.session 即已绑定日志)→ EVT-106 明确报错(偏离 1:
19 绝不静默自建,防伪造会话状态)。
20 """
21 mgr = getattr(ctx, "session", None)
22 fn = getattr(mgr, "create", None)
23 if not callable(fn):
24 raise_code("EVT-106", hint="会话门面未装配 create(桥无法新建会话);"
25 "显式传 sessionId 绑定既有会话")
26 sid = await _maybe_await(fn())
27 if not isinstance(sid, str) or not sid.strip():
28 raise_code("EVT-106", session_id=sid,
29 hint="会话门面 create 返回非法 sid(须非空字符串)")

32async def _open_session(ctx: Any, sid: str) -> Any:
33 """绑定既有会话(重放重建):委托 ctx.session.open_session(sid)。
34
35 两种注入形状(偏离 1):(a) 门面 open_session → 委托;(b) ctx.session 即已绑定
36 会话日志且 sid 匹配 → 直用;均不可达 → EVT-106(先 initialize 或注入门面)。
37 """
38 mgr = getattr(ctx, "session", None)
39 if mgr is None:
40 raise_code("EVT-106", session_id=sid,
41 hint="会话未装配(ctx.session=None);先 initialize")
42 fn = getattr(mgr, "open_session", None)
43 if callable(fn):
44 return await _maybe_await(fn(sid))
45 if getattr(mgr, "sid", None) == sid: # ctx.session 即当前会话日志
46 return mgr
47 raise_code("EVT-106", session_id=sid,
48 hint="会话门面未装配 open_session 且 ctx.session 未绑定该 sid;"
49 "先 initialize 或注入门面")

51async def _ensure_bound(ctx: Any, st: AcpState,

53 """解析当前操作会话:同柄复用,异柄重开;未初始化/无参数 → EVT-106。"""
54 sid = sid_param if sid_param is not None else st.session_id

56 raise_code("EVT-106", hint="会话未就绪:先 initialize"
