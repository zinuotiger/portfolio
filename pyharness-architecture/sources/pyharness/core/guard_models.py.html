33class Decision(StrEnum):
34 """guard 决策三值(原则 3;无 bypass 值)。
35
36 StrEnum:Decision.REJECT == "reject" 成立(executor 伪码 `d == "reject"`
37 直接比较可用,无需 .value)。语义:
38 ALLOW 放行,继续下一 guard 或执行(仅"本 guard 不反对")
39 REJECT 终局拒绝(guard.rejected 强同步;零副作用;无续跑 API)
40 APPROVAL 请求人类裁决(仅 danger=high 且有通道;critical 强制转 reject)
41 """

44 REJECT = "reject"
45 APPROVAL = "approval"

47def danger_default_policy(danger: str, *, has_channel: bool) -> Decision:
48 """danger 分级默认策略机器表达(SECURITY §4.2;headless=R8)。
49
50 none/low → allow;high → (has_channel ? approval : reject);critical → reject
51 (不可审批,无审批通道是刻意设计——人类真想删须下调沙箱留事件而非弹窗点
52 同意,POL-DGR-1)。供 g-danger/装配层/测试消费;危险级非法值按 none 处理。
53 """
54 if danger == "high":
55 return Decision.APPROVAL if has_channel else Decision.REJECT
56 if danger == "critical":
57 return Decision.REJECT
58 return Decision.ALLOW # none/low/未知 → 放行(分级信息由 match 层过滤)
