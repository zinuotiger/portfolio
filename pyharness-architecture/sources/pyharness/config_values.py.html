33def deep_merge(base: dict, override: dict) -> dict:
34 """键级深合并(CFG §1.2):标量后层覆盖、字典键级并集、列表整体替换、null=删键回落 L1。"""
35 out = dict(base)
36 for k, v in override.items():
37 if v is None: # 显式 null = 删键,回落低层默认
38 out.pop(k, None)
39 continue
40 if isinstance(v, dict) and isinstance(out.get(k), dict):
41 out[k] = deep_merge(out[k], v) # 字典:键级深合并,两键并集
42 else:
43 out[k] = v # 标量覆盖;列表整体替换(不逐元素合并)
44 return out

46def flatten(d: dict, prefix: str = "") -> dict:
47 """嵌套 dict → 点分键平面(env/CLI/hot-update 寻址用)。"""
48 out: dict = {}
49 for k, v in d.items():
50 key = f"{prefix}.{k}" if prefix else k
51 if isinstance(v, dict):
52 out.update(flatten(v, key)) # 递归展开
53 else:
54 out[key] = v # 叶子收进平面
55 return out

58 """点分键写入嵌套 dict(逐层建 dict)。"""

80def yaml_load_file(path: str) -> dict:
81 """YAML 安全加载单个文件;文件缺失 → 回落 L1 记 info;语法错 → CFG-601(file_parse)中止。"""
82 p = Path(path).expanduser()
83 if not p.exists():
84 log.info("config file missing, fallback L1: %s", path)
85 return {}
86 try:
87 data = yaml.safe_load(p.read_text(encoding="utf-8")) or {}
88 if not isinstance(data, dict):
89 raise ValueError("root 必须是映射")
90 return data
91 except Exception as e: # 语法/权限/类型错一律启动中止(半套比没配更危险)
92 raise_code("CFG-601", reason="file_parse", file=str(p),
93 detail=f"{type(e).__name__}:{e}")

118def parse_env_value(raw: str) -> Any:
119 """env 值严格解析:bool/int/float/列表(逗号或 JSON 数组);空串 = 未设置。
120
121 偏离点:白名单含字符串键(llm.model/base_url/storage 路径/log.file 等),
122 全部按伪码在 int/float 失败后 CFG-601 会让这些键永不可用——故末尾回退原串,
123 交由 pydantic schema 收口(字符串落数字字段仍会 CFG-601 type_range,语义不变)。
124 """
125 if raw == "":
126 return None # 空串 = 未设置
127 low = raw.strip().lower()
128 if low in {"1", "true", "yes", "on"}:
129 return True
130 if low in {"0", "false", "no", "off"}:
131 return False
132 if "," in raw or raw.startswith("["): # 列表:逗号或 JSON 数组
133 try:
134 return json.loads(raw) if raw.startswith("[") else [s.strip() for s in raw.split(",")]
135 except json.JSONDecodeError as e:
136 raise_code("CFG-601", reason="env_parse", detail=str(e))
137 try:
138 return int(raw) # int 优先
139 except ValueError:
140 try:
141 return float(raw) # float
142 except ValueError:
143 return raw # 原串回退(字符串键;数字键由 pydantic 拒)
