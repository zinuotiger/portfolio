50class EventStreamHub:
51 """总线 → SSE fan-out:写锁保证 seq 顺序广播(EVT-103 由总线隔离订阅者异常)。
52
53 forward(type_, payload) 作为 EventBus 订阅者被调用:Envelope(日志分发,payload=信封)
54 按会话过滤投递并推进游标;裸 dict(瞬时 llm.chunk 直推)广播给全部客户端(流式碎片刻
55 画在哪个会话由前端按自身 task 上下文过滤)。瞬时事件天然不入日志 → 时间线只取持久。
56 """
57
58 def __init__(self) -> None:
59 self.clients: set[StreamClient] = set()
60 self._seq_locks: dict[str, asyncio.Lock] = {} # per-session 写锁(含 "_all")
61
62 def register(self, client: StreamClient) -> _HubSession:
63 """注册订阅游标(SSE 连接建立时);退出上下文即注销。"""
64 return _HubSession(self, client)

66 def _lock_for(self, sid: str) -> asyncio.Lock:
67 """懒建会话写锁(forward 恒在事件循环内执行,Lock 可安全创建)。"""
68 return self._seq_locks.setdefault(sid or "_all", asyncio.Lock())

70 async def forward(self, type_: str, payload: Any) -> None:
71 """总线订阅者:按 seq 序把事件扇出到匹配客户端队列(队列无界不阻塞总线)。"""
72 if not _has_envelope_scope(payload):
73 sid, seq = None, None # 无会话作用域:兼容旧广播
74 else:
75 sid = _env_session_id(payload) # 持久/带作用域瞬时事件按会话过滤
76 seq = getattr(payload, "seq", None)
77 from pyharness.application.product_execution_sessions import is_product_session
78 # Legacy SSE has tenant-level subscriptions, not product/source authority.
79 # Product views poll their guarded exact execution endpoint instead.
80 if is_product_session(sid):return
81 text = _serialize_event(type_, payload)
82 # 租户归属:优先**落盘信封**(GAP-10,免疫重启/跨租户 sid 撞名),回落进程内映射
83 event_tenant = event_tenant_of(payload, session_id=sid or "")
84 async with self._lock_for(sid):
85 for c in tuple(self.clients):
86 if c.dead:
87 continue
88 if not event_tenant_allowed(c.tenant, event_tenant):
89 continue # 租户过滤:禁止跨租户事件泄漏(单点判据)
90 if c.sid and sid and c.sid != sid:
91 continue # 会话过滤:只投本会话客户端
92 c.queue.put_nowait((type_, text))
93 if seq is not None and isinstance(seq, int) and seq > c.last_seq:
94 c.last_seq = seq # 游标只进不退(断线续拉用)
95
96 def close_all(self) -> None:
97 """优雅停服:全部客户端置 dead 并投 None 哨兵 → SSE 生成器净退。"""
98 for c in tuple(self.clients):
99 c.dead = True
100 try:
101 c.queue.put_nowait(None)
102 except Exception: # noqa: BLE001 队列已关等本地异常
103 log.debug("SSE client queue already closed", exc_info=True)
104 self.clients.clear()

124def _serialize_event(type_: str, payload: Any) -> str:
125 """事件 → SSE data 行 JSON(ensure_ascii=False,紧凑;中文原样直读)。"""
126 if callable(getattr(payload, "model_dump", None)):
127 body = _env_dict(payload)
128 elif isinstance(payload, dict):
129 if "seq" in payload or "type" in payload:
130 body = dict(payload)
131 body.setdefault("type", type_)
132 else:
133 meta = {k: payload[k] for k in ("session_id", "task_id")
134 if payload.get(k) is not None}
135 inner = {k: v for k, v in payload.items() if k not in meta}
136 body = {"type": type_, **meta, "payload": inner}
137 else:
138 body = {"type": type_, "payload": dict(payload or {})}
139 if type_ == 'llm.error' or (type_ == 'system.error' and
140 str(body.get('payload', {}).get('code', '')).startswith('LLM-')):
141 from pyharness.core.llm_diagnostics import public_diagnostics
142 p = body.get('payload', {})
143 context = p.get('ctx') or {}
144 d = public_diagnostics(p.get('diagnostics') or context.get('diagnostics'),
145 code=p.get('code'), role=context.get('role', 'other'))
146 body['payload'] = {'code': d['code'], 'message': d['summary'], 'diagnostics': d}
147 return json.dumps(body, ensure_ascii=False, separators=(",", ":"))
