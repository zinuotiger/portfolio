1"""pyharness/core/user_ask.py — agent 反问用户(带选项,#38 Provider 面)。
2
3一句话职责:工具轮内 agent 抛问题 → 事件落盘(user.question,ask_id=seq)→
4外壳(桌面/CLI)轮询 pending → 人答复(user.answer)→ 本 Provider settle 并
5把答复文本回喂 LLM。语义与 approval 同构(请求事件即事实,答复事件强同步),
6但无权限语义(反问不拦执行,只是要信息)——所以独立成面,不并入 guard。
7
8阻塞口径:request() 等待答复至多 ttl_s(默认 120s,事件载荷可下调);超时返回
9占位提示(不抛错——对话继续,LLM 据"用户未答复"改口或换路径)。
10"""

24class AskProvider:
25 """会话级反问宿主:request(挂 pending+落事件+等待)/answer(裁决落事件)。"""

27 def __init__(self, *, session: Any = None, bus: Any = None) -> None:
28 self._session = session
29 self._bus = bus
30 self._pending: dict[int, asyncio.Event] = {}
31 self._answers: dict[int, dict] = {}
32 self._bg_tasks: set[asyncio.Task] = set() # 持有后台答复任务引用(防 GC 丢)

34 # ------------------------------------------------------------ 请求面
35 async def request(self, question: str, options: Optional[list] = None,
36 *, ttl_s: int = DEFAULT_TTL_S) -> str:
37 """抛问题给用户:落 user.question → 等待答复 → 返回答复文本。
38
39 options 为拍板候选(≤10 项);答复可能不在候选中(自由文本)。
40 会话未接线(EVT-100)或等待超时(占位文本,不抛)两种收敛。"""
41 sess = self._session
42 if sess is None:
43 raise_code("EVT-100", hint="AskProvider 未注入 session(user.ask 不可用)")
44 env = await sess.append(
45 "user.question",
46 {"question": str(question)[:1000],
47 "options": [str(o)[:200] for o in (options or [])][:10],
48 "ttl_s": max(5, min(3600, int(ttl_s)))},
49 actor="agent", sync=True)
50 ask_id = env.seq
51 waiter = asyncio.Event()
52 self._pending[ask_id] = waiter
53 try:
54 try:
55 await asyncio.wait_for(waiter.wait(), timeout=max(5, int(ttl_s)))
56 except asyncio.TimeoutError:
57 log.info("user.ask timeout ask_id=%s", ask_id)
58 return "(用户未在时限内答复,如需继续请说明或自行决策)"
59 ans = self._answers.pop(ask_id, {})
60 choice = ans.get("choice")
61 if choice is not None and choice:
62 return f"用户选择: {choice}"
63 return f"用户答复: {ans.get('text') or '(空)'}"
64 finally:
65 self._pending.pop(ask_id, None)

67 # ------------------------------------------------------------ 答复面
68 def answer(self, ask_id: int, *, choice: Optional[str] = None,
69 text: Optional[str] = None, by: str = "user") -> bool:
70 """桌面/CLI 裁决桥:答复落盘(user.answer 强同步)+ settle 等待者。
71
72 未知 ask_id(已答/超时/过期)→ 返回 False(幂等忽略,防重放)。"""
73 aid = int(ask_id)
74 waiter = self._pending.get(aid)
75 if waiter is None:
76 return False
77 choice_v = (str(choice or "").strip())[:500] or None
78 text_v = (str(text or "").strip())[:2000] or None
79 if not choice_v and not text_v:
80 raise_code("EVT-100", field="answer",
81 hint="答复须给 choice 或 text 至少其一")
82 self._pending.pop(aid)
83 if self._session is not None:
84 coro = self._write_answer_and_settle(
85 aid, waiter, choice_v, text_v, by)
86 try:
87 task = asyncio.get_running_loop().create_task(coro)
88 self._bg_tasks.add(task) # 持引用:防 GC 中途回收
89 task.add_done_callback(self._bg_tasks.discard)
90 except RuntimeError:
91 coro.close()
92 self._pending[aid] = waiter
93 return False
94 else:
95 self._finish_answer(aid, waiter, choice_v, text_v)
96 return True

98 async def answer_async(self, ask_id: int, *, choice: Optional[str] = None,
99 text: Optional[str] = None, by: str = "user") -> bool:
100 """异步答复:先强同步落 user.answer,再唤醒等待者并返回成功。"""
101 aid = int(ask_id)
102 waiter = self._pending.get(aid)
103 if waiter is None:
104 return False
105 choice_v = (str(choice or "").strip())[:500] or None
106 text_v = (str(text or "").strip())[:2000] or None
107 if not choice_v and not text_v:
108 raise_code("EVT-100", field="answer",
109 hint="答复须给 choice 或 text 至少其一")
110 self._pending.pop(aid)
111 if self._session is not None:
112 try:
113 await self._session.append(
114 "user.answer",
115 {"ask_id": aid, "choice": choice_v, "text": text_v,
116 "by": str(by or "user")[:64]},
117 actor="user", sync=True)
118 except Exception:
119 self._pending[aid] = waiter # 落盘失败:恢复可重试状态
120 raise
121 self._finish_answer(aid, waiter, choice_v, text_v)
122 return True

124 async def _write_answer_and_settle(self, aid: int, waiter: asyncio.Event,
125 choice: Optional[str], text: Optional[str],
126 by: str) -> None:
127 """同步 answer 的后台强同步路径:失败时恢复 waiter 供再次答复。"""
128 try:
129 await self._session.append(
130 "user.answer",
131 {"ask_id": aid, "choice": choice, "text": text,
132 "by": str(by or "user")[:64]},
133 actor="user", sync=True)
134 except Exception as exc: # noqa: BLE001 异步后台入口
135 self._pending[aid] = waiter
136 log.warning("user.answer 留痕失败: %s", exc)
137 return
138 self._finish_answer(aid, waiter, choice, text)
139
140 def _finish_answer(self, aid: int, waiter: asyncio.Event,
141 choice: Optional[str], text: Optional[str]) -> None:
142 self._answers[aid] = {"choice": choice, "text": text}
143 if not waiter.is_set():
144 waiter.set()

146 # ------------------------------------------------------------ 只读
147 def pending_list(self) -> list[dict]:
148 """未决反问清单(桌面/CLI 轮询源;排序稳定按 ask_id)。"""
149 out = []
150 sess = self._session
151 if sess is None:
152 return out
153 try:
154 evs = list(sess.events_after(0))
155 except Exception: # noqa: BLE001
156 return out
157 for ev in evs:
158 if ev.type != "user.question":
159 continue
160 p = ev.payload or {}
161 if int(ev.seq) in self._pending:
162 out.append({"ask_id": ev.seq,
163 "question": p.get("question", ""),
164 "options": list(p.get("options") or []),
165 "ttl_s": p.get("ttl_s", DEFAULT_TTL_S)})
166 return out
