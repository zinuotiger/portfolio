7class _WorkspaceCommittedPayload(_PayloadBase):
8 """Control fact: typed project revision in an immutable digest-addressed blob."""

12 revision: int = Field(ge=1)

20 @field_validator('schema_version', mode='before')
21 @classmethod
22 def strict_version(cls, value):
23 if type(value) is not int or value != 1:
24 raise ValueError('unsupported control version')
25 return value

27 @model_validator(mode='after')
28 def coherent_reference(self):
29 if self.blob_ref != self.digest or self.revision != self.parent_revision + 1:
30 raise ValueError('incoherent resource reference')
31 return self
