42def _norm_value(v: Any) -> Any:
43 """canonical_json 值归一(键排序/类型归一:bool→int、整值 float→int、嵌套递归)。"""
44 if isinstance(v, bool):
45 return int(v)
46 if isinstance(v, dict):
47 return {str(k): _norm_value(x) for k, x in v.items()}
48 if isinstance(v, (list, tuple)):
49 return [_norm_value(x) for x in v]
50 if isinstance(v, float) and v.is_integer():
51 return int(v)
