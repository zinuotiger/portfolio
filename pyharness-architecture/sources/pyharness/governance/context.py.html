1"""pyharness/governance/context.py — 治理层单实例上下文(S2-1 骨架;S3-2-1 加 authorize)。
2
3一句话职责:治理层的**唯一挂载面**——``ctx.governance``(ADR-018 契约冻结:
4单实例挂载,**禁止**向 ``Ctx`` 追加治理散字段)。
5
6形状(全部已接线):
7 policy : PolicyEngine ← S2-1
8 decisions : DecisionEngine ← S3-2-1
9 receipts : ReceiptStore ← S4/M4
10 evidence : EvidenceCollector ← S5/M6
11 audit : AuditSystem ← S5/M7(replay-only,不订阅)
12
13``authorize()``(S3-2-1,B4):``tool_executor`` 关 2 的**唯一治理入口**——
14``GuardChain.evaluate_detailed() → EvaluationResult → DecisionEngine.decide()
15→ Decision``。它**只做编排**,不重新求值、不重新判定 verdict。
16
17决策语义所有权(S3-2-1 R1 裁定):``critical→reject`` 归 ``GuardChain``;
18``high+无通道→reject`` 与 ``denied/timeout→不执行`` 归 ``approval.py``/executor。
19故本方法**不得**向 ``DecisionEngine.decide()`` 传 ``approval_available`` /
20``approval_verdict``——有测试守卫钉死这一点。
21
22四字段先以 ``None`` 占位而非省略,是为了**冻结形状**(ADR-018 §目录冻结):
23装配层与测试可稳定断言 ``ctx.governance.decisions is None``("尚未接线"),
24而不是 ``AttributeError``——缺失可见,不静默。
25"""

59 @staticmethod
60 def principal_of(ctx: Any) -> Principal:
61 """正式主体派生(M5/S4-P1-3):取自运行时**真实 caller/channel identity**。
62
63 ``ctx.channel`` 由各外壳在装配时**框架侧**写入(非客户端自报),经
64 ``EngineSpine.channel`` → ``create_agent`` / ``_runtime_ctx`` 落到 agent
65 ctx(GAP-11 修复:此前该跳缺失,agent ctx 无 ``channel`` 属性,致每次授权
66 都静默降级为 SYSTEM):
67
68 - CLI:``"cli"``;headless → ``None``(``cli.py``)
69 - ACP:``"acp:<client_id>"``(``acp.py``;显式忽略客户端自报 ``by``)
70 - Desktop:``"desktop"``(``application/service.py`` / ``desktop/app.py``)
71
72 经**既有** ``Principal.from_legacy_by`` 统一解析(HUMAN + 通道 + id)。
73
74 **五态语义(N5 冻结;规范表见 ``core/channel.py`` 模块 docstring)**——
75 本层因 ADR-018:308 不得 import ``pyharness.core.*``,故**消费**本包已有的
76 ``Principal.from_legacy_by`` 分类,并由 ``tests/invariants`` 的真值表断言
77 与 ``core.channel.resolve_channel`` **逐状态一致**(沿用 ``HUMAN_CHANNELS``
78 的"受守卫的重复"先例):
79
80 1. ``channel="cli|web|acp:<id>|desktop"`` ⇒ HUMAN 主体(前缀匹配,``acp:<id>``
81 **保留 ACP 身份**,不降级);
82 2. ``channel`` **显式声明为 ``None``** ⇒ headless / 无人类通道 ⇒
83 **明确的 SYSTEM 主体**(沿用 approval 侧 ``by="system"`` 既有语义,
84 **绝不伪装** HUMAN/AGENT);
85 3. ``channel`` **属性缺失(从未声明)** ⇒ **``APR-503`` fail-closed**
86 ——"没人声明过这是什么通道"与"已声明无通道"是两件事;
87 4. ``channel`` 为**非白名单字符串**(如 ``"hacker"``)⇒ ``APR-503``;
88 5. ``channel`` 为**空串/非字符串**(如 ``""``)⇒ ``APR-503``
89 ——**修复点**:此前 ``ch or _FRAMEWORK_BY`` 的 falsy 短路会把 ``""``
90 **静默降级为 system**,与第 3/4 条的 fail-closed 精神相悖。
91 """
92 if not hasattr(ctx, "channel"):
93 raise_code("APR-503",
94 why="ctx.channel 未声明:通道身份缺失,禁止静默降级为 "
95 "system(fail-closed)。外壳须显式声明 "
96 "channel='cli'|'acp:<id>'|'desktop',或显式声明 "
97 "headless(channel=None)。")
98 ch = getattr(ctx, "channel", None)
99 if ch is None: # 显式 headless:唯一合法的无通道声明
100 return Principal.from_legacy_by(_FRAMEWORK_BY)
101 # 其余一律交既有分类:白名单/前缀合法 ⇒ HUMAN;空串/非白名单 ⇒ APR-503。
102 # 不再做 falsy 合并 —— 声明就是声明,不得被 `or` 改写。
103 return Principal.from_legacy_by(ch)

105 async def authorize(self, call: Any, ctx: Any, *,
106 principal: Optional[Principal] = None,
107 inputs_digest: str = "",
108 prior: Optional[Decision] = None,
109 approval_ref: Optional[int] = None) -> Decision:
110 """唯一治理入口(S3-2-1 建立;S3-2-2 起发射 ``decision.issued``)。
111
112 调用链:``GuardChain.evaluate_detailed()`` → ``EvaluationResult`` →
113 ``DecisionEngine.decide()`` → ``Decision`` → ``session.append``。
114
115 **scope 前置的唯一运行时所有者是 ``GuardChain._evaluate_full``**——
116 本方法不自行 ``scope.can_use``,避免与规则链重复计算同一条件(职责上提,
117 非重复计算)。
118
119 **不传** ``approval_available`` / ``approval_verdict``(R1:该语义归
120 ``approval.py``/executor)。
121
122 异常:治理层未接线(``decisions``/``chain`` 为 None)→ ``CYC-999``,
123 fail-closed(绝不返回"看似授权的"决策)。
124 """
125 if self.decisions is None:
126 raise_code("CYC-999", module="governance.context", field="decisions",
127 why="治理层未接线:decisions 为空,禁止授权(fail-closed)")
128 chain = self.policy.chain
129 if chain is None:
130 raise_code("CYC-999", module="governance.context", field="chain",
131 why="治理层未接线:guard 链为空,无法求值(fail-closed)")
132 scope = getattr(ctx, "scope", None)
133 # 1) 求值(唯一真源;verdict 与 refs 来自同一次实际求值)
134 # session=getattr(ctx,"session",None)(ADR-021):guard.evaluated /
135 # guard.rejected 必须与同 call_id 的 tool.call/tool.result 落在**发起
136 # 该调用**的会话——缺此一跳,子 Agent 的守卫事件会落到装配期会话
137 # (父),使其日志内 INV-04 不可满足(AuditSystem 报 NO-GUARD-EVENT)。
138 # 链仍唯一,只换汇点。用 getattr 缺省 None = 沿用装配期会话,与本方法
139 # 对 ctx 其余属性(scope/channel)的防御式取法一致,且对未装配 ctx.session
140 # 的调用方(轻装配/单测替身)保持既有语义不变。
141 evaluation = await chain.evaluate_detailed(
142 call, scope, session=getattr(ctx, "session", None))
143 # 2) 装配决策(纯;策略取自引擎当前策略——指纹随内容变)
144 decision = await self.decisions.decide(
145 evaluation, principal=principal or self.principal_of(ctx), call=call,
146 policy=self.policy.current(scope),
147 inputs_digest=inputs_digest, prior=prior, approval_ref=approval_ref)
148 # 3) 治理证据事件(强同步;一次 authorize 恰一条)
149 await self._emit_decision_issued(ctx, decision, call)
150 # 4) 决策凭证(M4;强同步):按 receipt_kind_of 产生规则发射——
151 # ALLOW/REJECT → kind="decision";审批后重新授权(D2)→ kind="approval";
152 # D1(审批请求决策)→ 不产生(R-2)。未接线(纯内存/单测)时降级为不发射。
153 if self.receipts is not None:
154 await self.receipts.emit(ctx, decision,
155 call_id=getattr(call, "call_id", "") or "")
156 return decision

158 @staticmethod
159 async def _emit_decision_issued(ctx: Any, decision: Decision,
160 call: Any) -> None:
161 """发射 ``decision.issued``(一次 authorize 恰一条)。
162
163 I/O 边界:仅 ``session.append``(``sync=True``)——治理层"对下只写事件"
164 (ADR-013);``call_id`` 走 ``Envelope.trace``(不入载荷)。
165 """
166 sess = getattr(ctx, "session", None)
167 if sess is None:
168 return # 未接线(纯内存/单测):降级
169 p = decision.principal
170 payload = {
171 "decision_id": decision.decision_id,
172 "verdict": str(decision.verdict),
173 "tool": decision.tool,
174 "guard_ids": list(decision.guard_ids),
175 "policy_refs": list(decision.policy_refs),
176 "policy_fingerprint": decision.policy_fingerprint,
177 "inputs_digest": decision.inputs_digest,
178 "principal_kind": str(p.kind) if p is not None else "system",
179 "principal_id": p.id if p is not None else _FRAMEWORK_BY,
180 "principal_channel": p.channel if p is not None else None,
181 "ts": decision.ts,
182 "approval_ref": decision.approval_ref,
183 "supersedes": decision.supersedes,
184 }
185 r = sess.append("decision.issued", payload, actor="system", sync=True,
186 trace={"call_id": getattr(call, "call_id", "")})
187 if hasattr(r, "__await__"):
188 await r
