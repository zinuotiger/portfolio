81def _canonical(value):

93def _sign(claims, key):
94 body = _encode(_canonical(claims))
95 return body + '.' + _encode(hmac.digest(key, body.encode('ascii'), 'sha256'))

98def page_events(events, query: AuditQuery, *, tenant: str, subject: str, key: bytes):
99 """Build one bounded public page from an already validated complete log."""
100 # Keyed, domain-separated identity bindings resist offline tenant/username
101 # dictionary checks even though the signed token is not encrypted.
102 binding = {'v': 1, 'tenant': hmac.digest(key, _canonical(['tenant', tenant]), 'sha256').hex(),
103 'subject': hmac.digest(key, _canonical(['subject', subject]), 'sha256').hex(),
104 'project_id': query.project_id, 'limit': query.limit}
105 watermark = len(events)
106 after = 0
107 issued = int(time.time())
108 if query.cursor:
109 try:
110 body, signature = query.cursor.split('.')
111 if not hmac.compare_digest(_encode(hmac.digest(key, body.encode('ascii'), 'sha256')), signature):
112 raise ValueError('signature')
113 data = base64.b64decode(body + '=' * (-len(body) % 4), altchars=b'-_', validate=True)
114 claims = CursorClaims.model_validate_json(data).model_dump()
115 if _encode(_canonical(claims)) != body:
116 raise ValueError('canonical')
117 if any(claims.get(k) != value for k, value in binding.items()):
118 raise ValueError('binding')
119 watermark, after, issued = claims['watermark'], claims['after'], claims['issued']
120 if issued > int(time.time()) or int(time.time()) - issued > 3600:
121 raise_code('EVT-100', reason='audit_cursor_expired')
122 if not (0 < after < watermark <= len(events)):
123 raise ValueError('range')
124 if claims['digest'] != _digest(events[:watermark]):
125 raise ValueError('watermark')
126 except (ValueError, KeyError, TypeError, UnicodeError, binascii.Error):
127 raise_code('EVT-100', reason='audit_cursor_invalid')
128 frozen = events[:watermark]
129 digest = _digest(frozen)
130 matching = [event for event in frozen if query.project_id is None or
131 event['payload'].get('resource_id') == query.project_id]
132 if query.cursor and after not in {event['seq'] for event in matching[:-1]}:
133 raise_code('EVT-100', reason='audit_cursor_invalid')
134 selected = [event for event in matching if event['seq'] > after][:query.limit]
135 has_more = bool(selected and selected[-1]['seq'] < matching[-1]['seq'])
136 cursor = _sign({**binding, 'watermark': watermark, 'digest': digest,
137 'after': selected[-1]['seq'], 'issued': issued}, key) if has_more else None
138 public = [AuditEvent(seq=e['seq'], type=e['type'], ts=e['ts'],
139 **{k: e['payload'][k] for k in ('resource_kind', 'resource_id', 'revision', 'digest')
140 if k in e['payload']}) for e in selected]
141 return AuditPage(events=public, total=len(matching), page_count=len(public), has_more=has_more,
142 next_cursor=cursor, source_watermark=Watermark(seq=watermark, digest=digest)).model_dump()
