161class GuardChain:
162 """guard 单调拒绝链(原则 3 落地;L3 调用裁决层)。
163
164 结构不可变纪律:chain 注册序 = 求值序,只增不改(插件追加链尾);disabled
165 仅经 disable()(config 显式声明 + guard.disabled 留痕)写入;运行期无任何
166 API 可移除/重排/回翻一次拒绝(INV-04,测试钉死方法面)。唯一状态 = 链装配
167 (chain/disabled/_snapshot_seq),求值本身无状态残留(取消/重入语义 = 重新
168 evaluate 一次,以新策略为准,GRD-403)。
169 """
170
171 # 无 bypass/放行/执行类方法面(测试断言 AttributeError;executor 才持有
172 # Provider 执行权——guard 永远只是"拒绝或请示"的裁决器)
173 _FORBIDDEN_API: tuple[str, ...] = (
174 "override", "bypass", "force_allow", "set_allow", "allow", "execute",
175 "run", "mark_allowed", "clear_reject", "release_decision",
176 )
177
178 def __init__(self, chain: Optional[list[Guard]] = None, *,
179 session: Any = None, disabled: Optional[Iterable[str]] = None,
180 credential_paths: Optional[Iterable[str]] = None,
181 validator: Optional[Callable[..., dict]] = None,
182 approval_channel: Optional[bool] = None,
183 path_exists: Optional[Callable[[str], bool]] = None,
184 link_resolver: Optional[Callable[[str], str]] = None,
185 bus: Any = None) -> None:
186 """构造链:chain 缺省 = 内置 g1-g7(固定序);session/bus 注入式装配。
187
188 session = 事件落点(SessionLog.append 同型:append(type, payload, *,
189 actor=…, sync=…, trace=…));None = 未接线(evaluated/rejected 日志降级,
190 但 rejected 强同步语义要求装配层注入真实 session——单测以替身覆盖)。
191 approval_channel = 审批通道在位与否(偏离 3;None 视同有通道)。
192 """
193 built = chain if chain is not None else build_builtin_chain(
194 credential_paths=credential_paths, validator=validator,
195 approval_channel=approval_channel, path_exists=path_exists,
196 link_resolver=link_resolver)
197 ids = [g.id for g in built]
198 if len(set(ids)) != len(ids):
199 raise_code("TLB-801", guard="chain",
200 advice="guard id 全链唯一(重复装配被拒)")
201 self.chain: list[Guard] = list(built) # 注册序 = 求值序(只增不改)
202 self.disabled: set[str] = set(disabled or ())
203 bad = self.disabled - set(ids)
204 if bad or self.disabled & FORCED_GUARDS:
205 raise_code("CFG-601", reason="越权",
206 fields=[f"guards.disabled:{sorted(bad or self.disabled)}"],
207 detail="仅五内置可关且须 config 声明;g-schema/g-danger "
208 "强制恒在")
209 self._session: Any = session # 事件落点(注入)
210 self._bus: Any = bus # registry.updated 尽力出口
211 self._approval_channel: Optional[bool] = approval_channel
212 self._snapshot_seq: int = 1 # 装配版本(审批重入复核)
213 self._tasks: set[asyncio.Task] = set() # fire-and-forget 任务登记

215 # ------------------------------------------------------ 单调链求值主函数
216 async def _evaluate_full(self, call: Any, scope: Any, *,
217 session: Any = None
218 ) -> tuple[Decision, tuple[str, ...], tuple[str, ...]]:
219 """**唯一** waterfall 求值实现(S3-2-1;B1)。
220
221 返回 ``(verdict, guard_ids, policy_refs)``——三者来自**同一次**实际求值:
222 verdict 与命中面在同一循环迭代内产生并同时返回,**不重跑、不缓存、不读
223 历史事件**(S3-2-1 明令)。
224
225 规则(spec 伪代码权威):
226 - scope 前置:can_use 不可见 → 终局 REJECT(guard_id=scope-hidden,
227 policy_ref=GRD-401;executor 关2a 已先行拦截,此处兜底直调消费方);
228 - 命中 guard 逐个 check;首个非 allow 即短路返回(reject/approval 都到
229 此为止,后续 guard 不再求值 = 单调拒绝的求值面保证);
230 - approval 降级两闸(统一在 evaluate 层,插件 approval 也覆盖):
231 danger=critical → 强制转 reject(POL-DGR-1,不可审批,F014);
232 无审批通道 → 转 reject(APR-501,headless=R8,偏离 3);
233 - 每次求值恰好一条 guard.evaluated(INV-04:执行前必有;缺 = 非法执行);
234 reject 再强同步 guard.rejected(sync=True,崩溃不丢拦截事实,INV-05);
235 - 全链 allow 才返回 ALLOW——guard 无放行权,执行权归 executor 关3。
236 异常:append 强同步失败按 PERS-202/EVT-1xx 语义上抛(拒绝事实必须落地,
237 调用方 fail-closed:缺 evaluated/rejected 不进入 Provider)。
238
239 **事件汇点按调用解析(ADR-021)**:``session`` 缺省(None)= 沿用装配期
240 ``self._session``(主路径**逐字不变**);治理层经 ``authorize()`` 显式传入
241 ``ctx.session`` ⇒ 子 Agent 的 ``guard.evaluated``/``guard.rejected`` 与
242 同 ``call_id`` 的 ``tool.call``/``tool.result`` **落同一会话**。链对象仍
243 **唯一**(``spine.guard is policy.chain``),变的是汇点而非链。
244 """
245 sess = session if session is not None else self._session
246 can_use = getattr(scope, "can_use", None)
247 if can_use is None:
248 raise_code("GRD-401", hint="evaluate 需要会话 Scope(can_use 前置 "
249 "不可缺);直接传 ScopePolicy 违反调用契约")
250 if not can_use(call.name): # scope 前置:不可见 → 终局
251 await self._audit(call, Decision.REJECT, "scope-hidden", "GRD-401",
252 session=sess)
253 await self._append_rejected(call, "scope-hidden", "GRD-401",
254 session=sess)
255 return Decision.REJECT, ("scope-hidden",), ("GRD-401",)
256 enabled = self.enabled_guard_ids()
257 for g in self.chain: # 按注册序 waterfall
258 if g.id in self.disabled:
259 continue # 仅 config 显式关闭(留过事件)
260 if not g.match(call):
261 continue # 非本 guard 管辖 → 看下一个
262 d_str, policy = await g.check(call, scope) # (allow/reject/approval)
263 if d_str not in _DECISIONS:
264 raise_code("CYC-999", guard=g.id, decision=d_str,
265 hint="guard.check 只能返回 allow/reject/approval"
266 "(插件契约违规=bug 证据)")
267 d = Decision(d_str)
268 if d is Decision.APPROVAL: # approval 降级两闸(统一兜底)
269 if _danger_of(call) == "critical":
270 d, policy = Decision.REJECT, "POL-DGR-1" # 不可审批(F014)
271 elif self._approval_channel is False:
272 d, policy = Decision.REJECT, "APR-501" # 无通道(R8)
273 if d is not Decision.ALLOW: # 首个非 allow 即短路(waterfall)
274 await self._audit(call, d, g.id, policy, session=sess)
275 if d is Decision.REJECT: # 强同步:拦了且没执行(INV-05)
276 await self._append_rejected(call, g.id, policy,
277 session=sess)
278 return d, (g.id,), ((policy,) if policy else ()) # reject/approval 到此为止
279 await self._audit(call, Decision.ALLOW, enabled, None, session=sess)
280 return Decision.ALLOW, tuple(enabled), () # 全链 allow → executor 关3

282 async def evaluate(self, call: Any, scope: Any, *,
283 session: Any = None) -> Decision:
284 """兼容 wrapper(位置参数/返回类型/求值语义不变;仅新增可选 kwarg)。
285
286 返回 ``Decision``(StrEnum;executor 伪码 ``d == "reject"`` 直接可比较)。
287 实现是 ``_evaluate_full`` 的**视图**,不是第二套求值。
288 ``session``(ADR-021)同 ``_evaluate_full``:缺省 None = 沿用装配期会话。
289 """
290 return (await self._evaluate_full(call, scope, session=session))[0]

292 async def evaluate_detailed(self, call: Any, scope: Any, *,
293 session: Any = None) -> Any:
294 """rich result API(S3-2-1;B1):同一次求值的结构化投影。
295
296 返回 ``governance.decision.EvaluationResult``(verdict + guard_ids +
297 policy_refs),供治理层装配 ``Decision``。**不重跑求值、不缓存上次结果、
298 不读历史事件**——直接复用 ``_evaluate_full`` 的返回值。
299
300 ``session``(ADR-021):治理层经 ``authorize()`` 传 ``ctx.session``,使
301 ``guard.evaluated``/``guard.rejected`` 落在**发起该调用**的会话;缺省
302 None = 沿用装配期会话(主路径不变)。
303
304 ``EvaluationResult`` 经**函数内**延迟 import 取得,避免给 ``core`` 增加
305 模块级治理依赖(import 图不变)。
306 """
307 from pyharness.governance.decision import EvaluationResult
308 d, guard_ids, policy_refs = await self._evaluate_full(
309 call, scope, session=session)

415 async def _append(self, type_: str, payload: dict, *, actor: str,
416 sync: bool = False,
417 trace: Optional[dict] = None,
418 session: Any = None) -> None:
419 """append 统一封装:兼容同步替身/异步 SessionLog(await 实际返回)。
420
421 **唯一解析点**(ADR-021):``session is None`` → 沿用装配期 ``self._session``
422 (主路径逐字不变);显式传入 → 该事件落调用方指定会话。
423 """
424 sess = session if session is not None else self._session
425 if sess is None:
426 log.debug("guard 未接线事件出口,%s 未记录(sync=%s)", type_, sync)
427 return
428 r = sess.append(type_, payload, actor=actor, sync=sync, trace=trace)
429 if inspect.isawaitable(r):
430 await r

432 def _record(self, type_: str, payload: dict) -> None:
433 """词表外事件(guard.disabled)尽力而为留痕(偏离 1;scope.py 同款)。
434
435 真实 SessionLog 现会 EVT-102 拒写(词表 57 锁定,后续阶段门扩展);
436 未接线/失败只记日志,不阻断策略主路径(与 scope._record 一致)。
437 """
438 sess = self._session
439 if sess is None:
440 log.debug("guard 未接线事件出口,%s 未记录", type_)
441 return
442 try:
443 r = sess.append(type_, payload, actor="system")
444 except Exception as exc: # noqa: BLE001 尽力而为
445 log.warning("guard append %s 失败: %s", type_, exc)
446 return
447 if inspect.isawaitable(r):
448 try:
449 asyncio.get_running_loop()
450 except RuntimeError:
451 return
452 task = asyncio.create_task(self._safe_record(r, type_))
453 self._tasks.add(task)
454 task.add_done_callback(self._tasks.discard)
455
456 async def _safe_record(self, coro: Any, type_: str) -> None:
457 try:
458 await coro
459 except Exception as exc: # noqa: BLE001 EVT-102 等
460 log.warning("guard append %s 异步失败: %s", type_, exc)

462 def _emit_bus(self, type_: str, payload: dict) -> None:
463 """总线尽力而为出口(registry.updated 瞬时事件,仅总线无载荷模型)。"""
464 bus = self._bus
465 if bus is None:
466 return
467 from pyharness.bus import schedule_emit
468 try:
469 schedule_emit(bus, type_, payload)
470 except Exception as exc: # noqa: BLE001 未注册类型等
471 log.warning("guard emit %s 失败: %s", type_, exc)

475# 插件 guard 钩子注册表(hook 名 → Guard 实例;capability 经 Definition
476# guard_hooks 声明后由 match_guard_by_hook 解析,announce 时挂链尾,F023)
477_PLUGIN_HOOKS: dict[str, Guard] = {}
478
479
480def register_guard_hook(hook: str, guard: Guard) -> None:
481 """登记插件 guard 钩子名(能力包启动期调用;重名/非 Guard → TLB-801)。"""
482 if not isinstance(guard, Guard):
483 raise_code("TLB-801", hook=hook, advice="插件 guard 须继承 Guard")
484 if hook in _PLUGIN_HOOKS:
485 raise_code("TLB-801", hook=hook, advice="guard hook 名全库唯一")
486 _PLUGIN_HOOKS[hook] = guard
487
488
489def unregister_guard_hook(hook: str) -> bool:
490 """注销插件 guard 钩子名(幂等;返回是否真的摘除)。
491
492 2026-09-21 补:此前**只有注册没有注销**,而 ``PluginManager.uninstall`` 的
493 docstring 明写"支持重装(F004)" —— 含 guard 钩子的插件重装时会在
494 ``register_guard_hook`` 处撞 TLB-801"名全库唯一",即**该声明对这类插件为假**;
495 且卸载后陈旧 Guard 仍留在表里,``match_guard_by_hook`` 仍会把它挂到新声明上。
496 """
497 return _PLUGIN_HOOKS.pop(str(hook), None) is not None
498
499
500def match_guard_by_hook(defn: Any) -> list[Guard]:
501 """按 Definition.guard_hooks 名字解析出 Guard 实例(announce 时挂载用)。
502
503 guard_hooks 中未登记的钩子 = 声明了不存在的 guard(非法注册证据)→
504 TLB-801 拒(能力须先 register_guard_hook 再声明)。返回实例按 hooks 声明序。
505 """
506 hooks = tuple(getattr(defn, "guard_hooks", ()) or ())
507 out: list[Guard] = []
508 for name in hooks:
509 g = _PLUGIN_HOOKS.get(str(name))
510 if g is None:
511 raise_code("TLB-801", hook=str(name),
512 advice="guard_hooks 声明了未登记的 guard:先 "
513 "register_guard_hook 再挂载")
514 out.append(g)
515 return out

520 bus: Any = None,
521 approval_channel: Optional[bool] = None,
522 validator: Optional[Callable[..., dict]] = None,
523 path_exists: Optional[Callable[[str], bool]] = None,
524 link_resolver: Optional[Callable[[str], str]] = None) -> GuardChain:
