105class DesktopSessionManager:
106 """形状 (c):真 persistence 会话门面(desktop 自装配)——多会话磁盘真源管理。
107
108 create:新建 sid → open_store + open_session(session.created seq=1,EVT-106 引导);
109 open_session:文件不存在 → EVT-106;存在 → repair 鸭子前置(F060 未装配跳过) →

111 list:扫描目录 *.jsonl,头行(session.created)取标题,尾行取终态(不整文件重放)。

145 def _attach_persistence(self, log_: SessionLog, store: Any, sid: str) -> None:
146 """总线 → 存储订阅(事件一入内存即入真源队列;强同步即写即刷 F011)。
147
148 sid 过滤:桌面总线全局共享(多会话同 bus),订阅者必须只落本会话事件
149 ——否则后开会话的事件会串写进先开会话文件(实测:两个 session.created
150 同文件、回放错乱)。store.append 侧无 sid 校验,过滤责任在订阅者。
151 2026-09-21 R8:该订阅**改为委托唯一实现** ``persistence.session_recorder``
152 (engine 侧长期缺同一过滤 ⇒ 同一关注点两处实现、漏一处;现单点)。
153 """
154 if self.bus is None:
155 return
156 # 属主**含租户**(2026-09-21 R12):会话目录按租户分目录 ⇒ 同名 sid 跨租户合法,
157 # 若属主只含 sid,则两租户同属主 → 关闭/删除其一即摘掉**另一租户**的落盘订阅
158 # (其会话从此静默不落盘)。同 R8-4(原生壳固定属主)的同类:属主必须含**全部**
159 # 隔离维度。
160 owner = f"persistence:{getattr(self, 'tenant_id', None) or ''}:{sid}:{id(self)}"
161 record = session_recorder(store, sid)
162
163 try:
164 for t in EVENT_TYPES:
165 self.bus.subscribe(t, record, owner=owner)
166 except BaseException as primary:
167 try:
168 self.bus.unsubscribe_all(owner)
169 except BaseException as secondary:
170 primary.add_note(f'subscription rollback failed: {type(secondary).__name__}')
171 raise
172 log_._bus = self.bus # append → 分发 → 落盘闭环
173 self._owners[sid] = owner

176 async def create(self) -> str:
177 """新建会话:sid → store → open_session → 总线落盘订阅 → session.created 首事件。"""
178 sid = f"s-{uuid.uuid4().hex[:12]}" # Envelope.session_id min_length=8
179 store = await self._open_store(sid)
180 try:
181 log_ = await open_session(sid, store, tenant_id=self.tenant_id)
182 self._attach_persistence(log_, store, sid)
183 await log_.append("session.created", {"title": "", "model": self._default_model()},
184 actor="system", sync=True)
185 except BaseException as primary:
186 self._rollback_open(sid, store, primary)
187 raise
188 self._logs[sid] = log_
189 self._stores[sid] = store
190 return sid

234 async def open_session(self, sid: str) -> SessionLog:
235 """打开既有会话(重放重建 + 落盘订阅);文件不存在 → EVT-106。"""
236 sid = validate_session_id(sid)
237 async with self._lock_for(sid):
238 if sid in self._logs: # 同柄复用(INV-03 缓存一致防双实例漂移)
239 return self._logs[sid]
240 if not self._path(sid).exists():
241 raise_code("EVT-106", session_id=sid,
242 hint="会话不存在(日志缺失);先 `session list` 确认 sid 或新建会话")
243 await self._ensure_repaired(sid) # 损坏先修再 open(F060;未装配跳过)
244 store = await self._open_store(sid)
245 try:
246 log_ = await open_session(sid, store, tenant_id=self.tenant_id)
247 self._attach_persistence(log_, store, sid)
248 except BaseException as primary:
249 self._rollback_open(sid, store, primary)
250 raise
251 self._logs[sid] = log_
252 self._stores[sid] = store
253 return log_

276 def _prune_archives(self, days: int) -> list[str]:
277 """清理超过保留期的归档目录(只读判定 + 递归删除;失败只告警)。"""
278 import shutil as _sh
279
280 root = self.archive_root()
281 if not root.is_dir():
282 return []
283 cutoff = time.time() - days * 86400
284 gone: list[str] = []
285 for entry in sorted(root.iterdir()):
286 try:
287 if entry.is_dir() and entry.stat().st_mtime < cutoff:

294 async def delete_session(self, sid: str) -> dict:
295 """删除会话及轮转/备份文件;先摘订阅并关闭句柄。
296
297 ``storage.archive_days``(R24 接线,默认 30 天;``0`` = 不归档立即删)——
298 此前该键**零读取者**且删除**直接 unlink**(L-21:误删不可恢复,而文档承诺
299 30 天窗口)。现:>0 时把该会话的**全部相关文件**移入同域归档区
300 (``{sid}-{ts}/``),再按保留期清理过期归档。
301 """
302 sid = validate_session_id(sid)
303 async with self._lock_for(sid):
304 root = self.dir.resolve()
305 candidates = {root / f"{sid}.jsonl"}
306 candidates.update(root.glob(f"{sid}.*.jsonl"))
307 candidates.update(root.glob(f"{sid}.jsonl.*"))
308 if sid not in self._logs and not any(p.is_file() for p in candidates):
309 raise_code("EVT-106", session_id=sid, hint="会话不存在,无法删除")
310 owner = self._owners.pop(sid, None)
311 if owner and self.bus is not None:
312 try:
313 self.bus.unsubscribe_all(owner)
314 except Exception: # noqa: BLE001
315 log.warning("desktop delete unsubscribe %s failed", sid,
316 exc_info=True)
317 store = self._stores.pop(sid, None)
318 if store is not None:
319 await _await(store.flush())
320 store.close()
321 self._logs.pop(sid, None)
322 days = self._archive_days()
323 moved: list[str] = []
324 removed: list[str] = []
325 dest: Optional[Path] = None
326 if days > 0:
327 dest = self.archive_root() / f"{sid}-{now_ts()}"
328 try:
329 dest.mkdir(parents=True, exist_ok=True, mode=0o700)
330 except OSError as e: # noqa: BLE001 归档不可用 → 拒绝删
331 raise_code("PERS-202", op="archive", path=str(dest), why=str(e),
332 advice="归档目录不可创建;删除**已中止**(不静默永久删)")
333 for path in sorted(candidates):
334 try:
335 resolved = path.resolve()
336 resolved.relative_to(root)
337 if not resolved.is_file():
338 continue
339 if dest is not None:
340 resolved.replace(dest / resolved.name) # 移动(可恢复)
341 moved.append(resolved.name)
342 else:
343 resolved.unlink() # archive_days=0:立即删
344 removed.append(resolved.name)
345 except FileNotFoundError:
346 continue
347 pruned = self._prune_archives(days) if days > 0 else []
348 # 每会话串行锁随会话释放(2026-09-21 R8:此前 _locks 只增不减 ⇒
349 # 见过的每个 sid 都留一个 asyncio.Lock;此处补 create→…→release 的最后一环)
350 self._locks.pop(sid, None)
351 return {"sid": sid, "removed": removed + moved, "count": len(removed) + len(moved),
352 "archived": str(dest) if dest is not None else None,
353 "archive_days": days, "pruned": pruned}

370 def list(self) -> list[dict]:
371 """会话摘要清单:只读头部若干行 + 尾行 + 行数,不整文件重放。
372
373 排序:按最后活动时间(st_mtime)倒序——最新使用的会话排最上(桌面
374 会话列表时间序)。preview = 首条 user.message 内容前 10 字(列表显示
375 用;title 为空时它就是会话的"名字")。
376 """
377 out: list[dict] = []
378 self.list_scanned_sessions = 0
379 if not self.dir.exists():
380 return out
381 # 唯一枚举点(R14-3):此前用 ``stem.startswith("s-")`` 声称"跳过轮转段/备份",
382 # 但 ``{sid}.1.jsonl`` / ``{sid}.corrupt-*`` 的 stem **同样**以 ``s-`` 开头
383 # ⇒ 该守卫从未生效,列表混入"备份伪装成的会话"(实测:1 会话 → 列出 3 条)。
384 for path in session_log_paths(self.dir):
385 sid = path.stem
386 summary: dict[str, Any] = {"sid": sid, "title": "", "preview": "",
387 "finished": False}
388 try:
389 stat = path.stat()
390 stamp = (stat.st_mtime_ns, stat.st_size, stat.st_ino)
391 cached = self._summaries.get(sid)
392 if cached and cached[0] == stamp:
393 out.append(dict(cached[1]))
394 continue
395 self.list_scanned_sessions += 1
396 summary["updated"] = stat.st_mtime # 使用时间(最后写入)
397 summary["size_bytes"] = stat.st_size
398 lines: list[bytes] = []
399 count = 0
400 last: bytes | None = None
401 with open(path, "rb") as f:
402 for raw in f:
403 count += 1
404 if len(lines) < 60:
405 lines.append(raw)
406 last = raw
407 summary["lines"] = count
408 if lines:
409 head = json.loads(lines[0])
410 if head.get("type") == "session.created":
411 summary["title"] = (head.get("payload") or {}).get("title", "")
412 tail = json.loads(last or lines[-1])
413 summary['seq'] = tail.get('seq', 0)
414 summary["finished"] = tail.get("type") == "session.finished"
415 # 首条 user.message 内容前 10 字(会话"名字";只扫头部防 O(n))
416 for ln in lines[:60]:
417 try:
418 e = json.loads(ln)
419 except Exception: # noqa: BLE001 坏行跳过
420 continue
421 if e.get("type") == "user.message":
422 c = (e.get("payload") or {}).get("content") or ""
423 summary["preview"] = c.strip()[:10]
424 break
425 self._summaries[sid] = (stamp, dict(summary))
426 except Exception as e: # noqa: BLE001 文件坏/竞态 → 摘要留最小
427 # 一行告警(不打整段 traceback):UI 每次轮询会话列表都读此路径,
428 # 单个不可读文件若打 traceback 会持续刷屏;只报类型名即可定位。
429 log.warning("desktop list 读 %s 失败(摘要降级):%s",
430 path.name, type(e).__name__)
431 out.append(summary)
432 out.sort(key=lambda s: s.get("updated", 0), reverse=True) # 时间倒序
433 present = {row['sid'] for row in out}
434 for sid in set(self._summaries)-present:
435 self._summaries.pop(sid,None)
436 return out

442 async def shutdown_all(self) -> None:
443 """收尾:全量 flush + 关句柄 + 摘落盘订阅(幂等;正常路径强同步事件已落盘)。"""
444 errors = []
445 for sid, store in list(self._stores.items()):
446 try: await _await(store.flush())
447 except BaseException as exc: errors.append(exc)
448 try: store.close()
449 except BaseException as exc: errors.append(exc)
450 if self.bus is not None:
451 for owner in set(self._owners.values()):
452 try: self.bus.unsubscribe_all(owner)
453 except BaseException as exc: errors.append(exc)
454 self._stores.clear()
455 self._owners.clear()
456 self._logs.clear()
457 if errors: raise BaseExceptionGroup('session manager cleanup failed', errors)
